def need(value, message):

def remember(error):

        FIRST[0] = error
        FIRST_TRACEBACK[0] = BaseException.__dict__['__traceback__'].__get__(error,BaseException)
    elif error is not FIRST[0]:

def canonical(value):
    return (json.dumps(value,sort_keys=True,separators=(',',':'),ensure_ascii=True,
                       allow_nan=False)+'\n').encode('ascii')

def pairs(items):
    result = {}
    for key,value in items:
        need(key not in result,'duplicate JSON key')
        result[key] = value
    return result

def decode(raw):
    def no_number(unused):
        raise ValueError('noninteger JSON number')
    return json.loads(raw.decode('utf-8'),object_pairs_hook=pairs,
                      parse_float=no_number,parse_constant=no_number)

def identity(st):
    return (st.st_dev,st.st_ino,st.st_mode,st.st_uid,st.st_nlink,
            st.st_size,st.st_mtime_ns,st.st_ctime_ns)

def acquire(path, flags, mode=0o600):
    need(not UNKNOWN[0] and FIRST[0] is None,'blocked custody')
    slot = next(x for x in SLOTS if x[1] in ('FREE','CLOSED'))
    slot[:] = [None,'OPEN_PENDING']
    try:
        slot[0] = os.open(path, flags|os.O_NOFOLLOW|os.O_CLOEXEC,mode)
        slot[1] = 'OWNED'
    except BaseException as error:
        UNKNOWN[0] = True
        raise remember(error)
    return slot

def close(slot):
    need(slot[1] == 'OWNED','known close once')
    slot[1] = 'CLOSE_ENTERED'
    try:
        os.close(slot[0])
        slot[1] = 'CLOSED'
    except BaseException as error:
        slot[1],UNKNOWN[0] = 'CLOSE_UNKNOWN',True
        raise remember(error)

def read_pin(pin, cap=8*1024*1024):
    need(type(pin) is dict and set(pin)=={'path','bytes','mode','sha256'},'exact pin')
    path = Path(pin['path'])
    need(path.is_absolute() and path.resolve(strict=True)==path,'canonical input')
    before = path.lstat()
    need(stat.S_ISREG(before.st_mode) and before.st_nlink==1 and
         before.st_uid in (0,os.getuid()) and stat.S_IMODE(before.st_mode)==pin['mode'] and
         type(pin['bytes']) is int and 0<=pin['bytes']<=cap and before.st_size==pin['bytes'],'input inode/mode/size')
    slot = acquire(path,os.O_RDONLY|os.O_NONBLOCK)
    parts=[]; count=0
    try:
        need(identity(os.fstat(slot[0]))==identity(before),'same opened input')
        while True:
            part=os.read(slot[0],min(65536,cap+1-count))
            if not part: break
            count+=len(part); need(count<=cap,'bounded full input'); parts.append(part)
        raw=b''.join(parts)
        need(identity(os.fstat(slot[0]))==identity(path.lstat())==identity(before) and
             count==pin['bytes'] and hashlib.sha256(raw).hexdigest()==pin['sha256'],'complete stable selected bytes')
    except BaseException as error:
        remember(error)
    finally:
        if slot[1]=='OWNED':
            try: close(slot)
            except BaseException: pass
    if FIRST[0] is not None: raise FIRST[0]
    return raw

def publish(path, raw, expected=None):

    need(expected is None or expected==wanted,'exact canonical returned publication')
    slot=acquire(path,os.O_WRONLY|os.O_CREAT|os.O_EXCL)
    try:
        at=0
        while at<len(raw):

        os.fsync(slot[0]); os.fchmod(slot[0],0o444); os.fsync(slot[0])
    except BaseException as error:

    directory=acquire(path.parent,os.O_RDONLY|os.O_DIRECTORY)

def deadline_read(fd, size, deadline):
    remaining=(deadline-time.monotonic_ns())/1_000_000_000
    need(remaining>0,'original entry deadline before read')
    ready,unused,errors=select.select([fd],[],[fd],remaining)
    need(bool(ready) and not errors,'original entry deadline/read readiness')
    return os.read(fd,size)

def capture_readback_known(path, slot, deadline):
    """Full independent cleanup read through a reader acquired before birth.

    A working error does not suppress either stream. No new native descriptor
    or authority is acquired. Only proven descendant emptiness authorises this
    final stable read; unknown live writers remain explicitly pending.
    """
    need(slot[1]=='OWNED','actual retained prebirth capture reader')
    before=path.lstat()
    need(stat.S_ISREG(before.st_mode) and before.st_uid==os.getuid() and
         before.st_nlink==1 and stat.S_IMODE(before.st_mode)==0o600,
         'actual original capture')
    need(identity(os.fstat(slot[0]))==identity(before),'same preowned capture inode')
    h=hashlib.sha256(); size=0
    while True:
        need(time.monotonic_ns()<deadline,'original whole deadline during capture readback')
        block=os.read(slot[0],65536)
        if not block: break
        size+=len(block)
        need(size<=8*1024*1024,'unchanged complete capture qualification bound')
        h.update(block)
    need(size==before.st_size and identity(os.fstat(slot[0]))==identity(path.lstat())==identity(before),
         'full independent stable capture with actual EOF')
    return {'path':str(path),'bytes':size,'sha256':h.hexdigest(),'full_read':True,
            'actual_eof':True,'reader_acquired_before_birth':True}


def request_stop(signum, unused_frame):
    # Never throw through an ownership transition or destroy the surviving S.
    if STOP_SIGNAL[0] is None:
        STOP_SIGNAL[0]=signum


def install_stop_handlers():
    for signum in (signal.SIGINT,signal.SIGTERM,signal.SIGHUP):
        ticket=[signum,signal.getsignal(signum),'INSTALL_PENDING']
        SIGNAL_TICKETS.append(ticket)
        signal.signal(signum,request_stop)
        ticket[2]='INSTALLED'


def restore_stop_handlers():
    # Called only after proven native domain and holder retirement.
    for ticket in SIGNAL_TICKETS:
        if ticket[2]!='INSTALLED':
            continue
        ticket[2]='RESTORE_ENTERED'
        try:
            signal.signal(ticket[0],ticket[1]); ticket[2]='RESTORED'
        except BaseException as error:
            ticket[2]='RESTORE_UNKNOWN'; UNKNOWN[0]=True; remember(error)


def working_checkpoint(deadline):
    need(STOP_SIGNAL[0] is None,'original catchable signal requests owned retirement')
    need(time.monotonic_ns()<deadline,'original whole reservation before work')


def retain_pending(owner, reason):
    """Keep the real surviving native owner, not merely an UNKNOWN report.

    No extra retirement entry, attempt, acquisition, timer, or grant is made.
    This blocked operation remains unresolved and its continuing elapsed cost
    is not zero or clipped. The original independent terminal owner must not
    admit physical completion, successful CLOSE, or a replacement attempt.

    PENDING_CUSTODY[0]=owner
    PENDING_CUSTODY[1]=reason
    PENDING_CUSTODY[2]=CAPTURE_OBSERVATIONS
    CHILD['phase']='ACTUAL_SURVIVING_CUSTODY_PENDING'
    while True:
        try:
            signal.pause()
        except BaseException as error:
            remember(error)

def close_pair(pair,index):
    fd=pair[index]
    if fd is None: return
    pair[index]=None
    pair[2]='CLOSE_ENTERED'
    try:
        os.close(fd)
        pair[2]='OWNED' if pair[1-index] is not None else 'CLOSED'
    except BaseException as error:
        pair[2]='CLOSE_UNKNOWN'; UNKNOWN[0]=True
        raise remember(error)
