def need(value, message):
    if value is not True:
        raise RuntimeError(message)

def remember(error):
    if FIRST[0] is None:
        FIRST[0] = error
    elif error is not FIRST[0]:
        for i,item in enumerate(SECONDARY):
            if item is error:
                break
            if item is None:
                SECONDARY[i] = error
                break
        else:
            UNKNOWN[0] = True
    return FIRST[0]

def canonical(value):
    return (json.dumps(value,sort_keys=True,separators=(',',':'),ensure_ascii=True,
                       allow_nan=False)+'\n').encode('ascii')

def pairs(items):
    result = {}
    for key,value in items:
        need(key not in result,'duplicate JSON key')
        result[key] = value
    return result

def decode(raw):
    def no_number(unused):
        raise ValueError('noninteger JSON number')
    return json.loads(raw.decode('utf-8'),object_pairs_hook=pairs,
                      parse_float=no_number,parse_constant=no_number)

def identity(st):
    return (st.st_dev,st.st_ino,st.st_mode,st.st_uid,st.st_nlink,
            st.st_size,st.st_mtime_ns,st.st_ctime_ns)

def acquire(path, flags, mode=0o600):
    need(not UNKNOWN[0] and FIRST[0] is None,'blocked custody')
    slot = next(x for x in SLOTS if x[1] in ('FREE','CLOSED'))
    slot[:] = [None,'OPEN_PENDING']
    try:
        slot[0] = os.open(path, flags|os.O_NOFOLLOW|os.O_CLOEXEC,mode)
        slot[1] = 'OWNED'
    except BaseException as error:
        UNKNOWN[0] = True
        raise remember(error)
    return slot

def close(slot):
    need(slot[1] == 'OWNED','known close once')
    slot[1] = 'CLOSE_ENTERED'
    try:
        os.close(slot[0])
        slot[1] = 'CLOSED'
    except BaseException as error:
        slot[1],UNKNOWN[0] = 'CLOSE_UNKNOWN',True
        raise remember(error)

def read_pin(pin, cap=8*1024*1024):
    need(type(pin) is dict and set(pin)=={'path','bytes','mode','sha256'},'exact pin')
    path = Path(pin['path'])
    need(path.is_absolute() and path.resolve(strict=True)==path,'canonical input')
    before = path.lstat()
    need(stat.S_ISREG(before.st_mode) and before.st_nlink==1 and
         before.st_uid in (0,os.getuid()) and stat.S_IMODE(before.st_mode)==pin['mode'] and
         type(pin['bytes']) is int and 0<=pin['bytes']<=cap and before.st_size==pin['bytes'],'input inode/mode/size')
    slot = acquire(path,os.O_RDONLY|os.O_NONBLOCK)
    parts=[]; count=0
    try:
        need(identity(os.fstat(slot[0]))==identity(before),'same opened input')
        while True:
            part=os.read(slot[0],min(65536,cap+1-count))
            if not part: break
            count+=len(part); need(count<=cap,'bounded full input'); parts.append(part)
        raw=b''.join(parts)
        need(identity(os.fstat(slot[0]))==identity(path.lstat())==identity(before) and
             count==pin['bytes'] and hashlib.sha256(raw).hexdigest()==pin['sha256'],'complete stable selected bytes')
    except BaseException as error:
        remember(error)
    finally:
        if slot[1]=='OWNED':
            try: close(slot)
            except BaseException: pass
    if FIRST[0] is not None: raise FIRST[0]
    return raw

def deadline_read(fd, size, deadline):
    remaining=(deadline-time.monotonic_ns())/1_000_000_000
    need(remaining>0,'original entry deadline before read')
    ready,unused,errors=select.select([fd],[],[fd],remaining)
    need(bool(ready) and not errors,'original entry deadline/read readiness')
    return os.read(fd,size)

def capture_readback(path):

    need(stat.S_ISREG(before.st_mode) and before.st_uid==os.getuid() and
         before.st_nlink==1 and stat.S_IMODE(before.st_mode)==0o600,'actual original capture')
    # Complete capture bytes are independently read, not inferred from a digest.
    slot=acquire(path,os.O_RDONLY|os.O_NONBLOCK)
    h=hashlib.sha256(); size=0
    try:
        need(identity(os.fstat(slot[0]))==identity(before),'same capture inode')
        while True:
            block=os.read(slot[0],65536)
            if not block: break
            size+=len(block); need(size<=8*1024*1024,'capture qualification bound'); h.update(block)
        need(size==before.st_size and identity(os.fstat(slot[0]))==identity(path.lstat())==identity(before),'full independent stable capture')
    except BaseException as error:
        remember(error)
    finally:

            try: close(slot)
            except BaseException: pass
    if FIRST[0] is not None: raise FIRST[0]
    return {'path':str(path),'bytes':size,'sha256':h.hexdigest(),'full_read':True}

def close_pair(pair,index):
    fd=pair[index]
    if fd is None: return
    pair[index]=None
    pair[2]='CLOSE_ENTERED'
    try:
        os.close(fd)
        pair[2]='OWNED' if pair[1-index] is not None else 'CLOSED'
    except BaseException as error:
        pair[2]='CLOSE_UNKNOWN'; UNKNOWN[0]=True
        raise remember(error)
