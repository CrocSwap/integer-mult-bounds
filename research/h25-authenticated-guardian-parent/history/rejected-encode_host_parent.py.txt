def need(value, reason):
    if value is not True:
        raise ValueError(reason)


def plain(value, depth=0, nodes=None):

    nodes[0] += 1

    if type(value) is dict:
        need(len(value) <= 1024 and all(type(k) is str for k in value), 'DATA keys')

    elif type(value) is list:
        need(len(value) <= 8192, 'DATA list')
        for item in value:
            plain(item, depth + 1, nodes)
    elif type(value) is str:
        need(len(value) <= 131072, 'DATA string')
    elif type(value) is int:
        need(-(1 << 53) < value < (1 << 53), 'exact JavaScript DATA integer')
    else:
        need(value is None or type(value) is bool, 'plain scalar DATA only')


def canonical(value):

                      ensure_ascii=True, allow_nan=False)
