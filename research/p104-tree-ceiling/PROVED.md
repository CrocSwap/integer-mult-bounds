## CongestionTreeCore.TreeData (the tree data `T`; every field is a hypothesis of the theorems)
```lean
structure TreeData (σ ρ μ : Type) [DecidableEq σ] [DecidableEq ρ] [DecidableEq μ]
    (rsA rsB : List Row) (rootA rootB : Row) (nB : ℕ) where
  J : Finset (σ × ρ)
  S : Finset σ
  Bset : Finset σ
  Mv : Finset μ
  Mreg : Finset μ
  Mun : Finset μ
  dA : μ → ℕ
  dB : μ → ℕ
  nm : μ → ℕ
  ms : σ → List μ
  R : σ → Finset ρ
  x : σ → ℕ
  hJS : ∀ p ∈ J, p.1 ∈ S
  hScard : S.card ≤ 104
  hfib : ∀ s ∈ S, (J.filter fun p => p.1 = s).card ≤ 48
  hJcard : 4962 ≤ J.card
  hB : Bset ⊆ S
  hBcard : nB ≤ Bset.card
  hcell : ∀ s ∈ S, (J.filter fun p => p.1 = s).card ≤ x s + (R s).card
  hxk : ∀ s ∈ S, x s ≤ (J.filter fun p => p.1 = s).card
  htreeA : ∀ s ∈ S, s ∉ Bset → Tr rsA nm dA dB rootA (R s) (ms s)
  htreeB : ∀ s ∈ Bset, Tr rsB nm dA dB rootB (R s) (ms s)
  hnd : ∀ s ∈ S, (ms s).Nodup
  hMv : ∀ s ∈ S, ∀ e ∈ ms s, e ∈ Mv
  hcap : ∀ e ∈ Mv, (S.filter fun s => e ∈ ms s).card ≤ nm e
  hMsub : Mv ⊆ Mreg ∪ Mun
  hdim : ∀ e ∈ Mreg ∪ Mun, 1 ≤ dA e ∧ dA e < dB e ∧ dB e ≤ 8
```

## CongestionTreeCore.Tr (out-trees of charts; its constructor premises are hypotheses on each tree)

```lean
inductive Tr (rs : List Row) (nm dA dB : μ → ℕ) : Row → Finset ρ → List μ → Prop
  | node (r : Row) (R : Finset ρ) (ch : List (Row × Finset ρ × μ × List μ))
```

## CTreeCore.Row and CongestionTreeCore.lineOK

```lean
structure Row where
  i : Nat
  dim : Nat
  n : Nat
  m : Nat
  mask : Nat
  g : List Nat
  ge : List Nat

def Row.ok (r r' : Row) : Bool :=
  decide (r.dim < r'.dim) && decide (r.n ≤ r'.n) && decide (r'.m ≤ r.m) && ((r.mask >>> r'.i) % 2 == 1)

def lineOK (rA rB : Row) (capI lam gA gB : ℕ) : Bool :=
  (List.range 49).all (fun x => (List.range 49).all (fun r =>
    decide (lam * min 48 (x + r) ≤ gv rA r + x * capI + gA) && decide (lam * min 48 (x + r) ≤ gv rB r + x * capI + gB)))
  && decide (gB ≤ gA)
```

### Invoice, caller and Finv
Namespaces: `CallerCost` (`Hist`, `Inv`, `scale`, `invoice`, `armT`, `armP`, `callerT`, `callerP`, `caller`), `JointCeiling` (`Fh`, `Finv`).

```lean
abbrev Hist := List (ℕ × ℕ)
structure Inv where
  T : Hist
  P : Hist
def scale (L : ℕ) (l : Hist) : Hist := l.map fun p => (p.1, L * p.2)
def invoice (c n : Inv) (L : ℕ) : Inv := ⟨c.T ++ scale L n.T, c.P ++ scale L n.P⟩
def armT (n c : ℕ) : Hist := (List.range n).map fun i => (8 * i + 9, c)
def armP (m c : ℕ) : Hist := (List.range m).map fun i => (8 * i + 16, c)
def callerT (a b : ℕ) : Hist := armT (a + 2) 208 ++ armT (b + 2) 208 ++ [(8 * (a + b + 4) + 1, 208)]
def callerP (a b : ℕ) : Hist :=
  armP a 208 ++ armP b 208 ++ [(8, 384), (1, 384), (8, 16 * (a + b + 4)), (9, 32)]
def caller (a b : ℕ) : Inv := ⟨callerT a b, callerP a b⟩
noncomputable def Fh (s : ℝ) (l : Hist) : ℝ := (l.map fun p => (p.2 : ℝ) * (p.1 : ℝ) ^ s).sum
noncomputable def Finv (s : ℝ) (i : Inv) : ℝ := Fh s i.T - Fh s i.P
```

### Owned-surplus and cap definitions (`CongestionNative.lean`), price and register definitions
```lean
noncomputable def Omega (ks : List ℕ) (mr : ℕ) (s : ℝ) : ℝ := dprice ks s + mr - (6 * (9 : ℝ) ^ s + 10)
noncomputable def capC (s : ℝ) : ℝ := ((9 : ℝ) ^ s - (8 : ℝ) ^ s) / 48
noncomputable def corrC (cstF : ℝ → ℝ) (s : ℝ) : ℝ := 8 * cstF (1 - s)
noncomputable def dprice (ks : List ℕ) (s : ℝ) : ℝ := (ks.map fun k : ℕ => ((k : ℕ) : ℝ) ^ s).sum

def ex (g : ℝ → ℝ) (d : ℕ → ℝ) (n : ℕ) : ℝ := gsum g d n - g (d n - d 0)
```

### `CF`, `Fc`, `Fup`, `dLo` (`NumCertK.lean`)
```lean
abbrev CF := List (ℕ × ℤ)
noncomputable def Fc (cf : CF) (s : ℝ) : ℝ := (cf.map fun p => (p.2 : ℝ) * (p.1 : ℝ) ^ s).sum
/-- upper bound of F on s in [a,b] -/
def Fup (cf : CF) (a b : ℚ) : ℚ :=
  (cf.map fun p => if 0 ≤ p.2 then (p.2 : ℚ) * pHi p.1 b else (p.2 : ℚ) * pLo p.1 a).sum
/-- lower bound of F' on s in [a,b] -/
def dLo (cf : CF) (a b : ℚ) : ℚ :=
  (cf.map fun p => if 0 ≤ p.2 then (p.2 : ℚ) * (lnLo p.1 * pLo p.1 a)
    else (p.2 : ℚ) * (lnHi p.1 * pHi p.1 b)).sum

```
