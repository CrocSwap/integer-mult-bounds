def _vsame(left, right):
    """Bounded exact native tree equality; no coercion or foreign callbacks.

    The existing producer/receiver JSON domain is integer-only: floats, including
    finite floats, remain rejected. Bytes are supported only as exact byte leaves.
    Dictionary field sets are unordered; list and tuple order and kind are exact.
    The caller cannot supply the private budget or a comparison callback.
    """
    budget = [0, 0, 0]  # paired nodes, both-side text, both-side bytes

    def walk(a, b, depth):
        budget[0] += 1
        if depth > 24 or budget[0] > 65536:
            return False
        kind = type(a)
        if kind is not type(b):
            return False
        if a is None:
            return True
        if kind is bool:
            return a is b
        if kind is int:
            if not -(1 << 64) <= a < (1 << 64) or not -(1 << 64) <= b < (1 << 64):
                return False
            return a == b
        if kind is str:
            if len(a) > 8192 or len(b) > 8192:
                return False
            budget[1] += len(a) + len(b)
            if budget[1] > 1048576:
                return False
            return a == b
        if kind is bytes:
            if len(a) > 8388608 or len(b) > 8388608:
                return False
            budget[2] += len(a) + len(b)
            if budget[2] > 33554432:
                return False
            return a == b
        if kind is list or kind is tuple:
            if len(a) > 1024 or len(b) > 1024 or len(a) != len(b):
                return False
            index = 0
            while index < len(a):
                if not walk(a[index], b[index], depth + 1):
                    return False
                index += 1
            return True
        if kind is dict:
            if len(a) > 1024 or len(b) > 1024 or len(a) != len(b):
                return False
            # Complete both key-domain walks before membership or lookup. Exact
            # str keys use native hashing/equality; no unknown key gets hashed.
            for key in a:
                if type(key) is not str or len(key) > 8192:
                    return False
                budget[1] += len(key)
                if budget[1] > 1048576:
                    return False
            for key in b:
                if type(key) is not str or len(key) > 8192:
                    return False
                budget[1] += len(key)
                if budget[1] > 1048576:
                    return False
            for key in a:
                if key not in b:
                    return False
                if not walk(a[key], b[key], depth + 1):
                    return False
            return True
        # Includes float, all subclasses, user objects and callable values.
        return False

    return walk(left, right, 0)


def _vplain(value, depth=0, budget=None):
    # Exact native values only. No repr/str of unknown objects or error objects.
    if budget is None:
        budget = [0, 0]
    budget[0] += 1
    if depth > 24 or budget[0] > 65536:
        raise RuntimeError('Variant raw object bound exceeded')
    if value is None or type(value) is bool:
        return value
    if type(value) is int:
        if not -(1 << 64) <= value < (1 << 64):
            raise RuntimeError('Variant integer bound exceeded')
        return value
    if type(value) is str:
        budget[1] += len(value)
        if len(value) > 8192 or budget[1] > 524288:
            raise RuntimeError('Variant string bound exceeded')
        return value
    if type(value) in (list, tuple):
        if len(value) > 1024:
            raise RuntimeError('Variant sequence bound exceeded')
        return [_vplain(item, depth + 1, budget) for item in value]
    if type(value) is dict:
        if len(value) > 1024:
            raise RuntimeError('Variant mapping bound exceeded')
        for key in value:
            if type(key) is not str:
                raise RuntimeError('Variant foreign mapping key')
            _vplain(key, depth + 1, budget)
        return {key: _vplain(value[key], depth + 1, budget) for key in value}
    raise RuntimeError('Variant foreign raw value')


def _vencode(value):
    # Before cold capture no imports are added. Exact strings are escaped as ASCII JSON.
    value = _vplain(value)
    def emit(item):
        if item is None:
            return 'null'
        if type(item) is bool:
            return 'true' if item else 'false'
        if type(item) is int:
            return str(item)
        if type(item) is str:
            parts = ['"']
            for c in item:
                if c == '"':
                    parts.append('\\"')
                elif c == '\\':
                    parts.append('\\\\')
                elif ord(c) < 32 or ord(c) >= 127:
                    code = ord(c)
                    if code <= 65535:
                        parts.append('\\u%04x' % code)
                    else:
                        code -= 65536
                        parts.append('\\u%04x\\u%04x' % (55296 + (code >> 10), 56320 + (code & 1023)))
                else:
                    parts.append(c)
            parts.append('"')
            return ''.join(parts)
        if type(item) is list:
            return '[' + ','.join(emit(x) for x in item) + ']'
        return '{' + ','.join(emit(k) + ':' + emit(item[k]) for k in sorted(item)) + '}'
    result = (emit(value) + '\n').encode('ascii')
    if len(result) > _VARIANT_WIRE_LIMIT:
        raise RuntimeError('Variant wire bound exceeded')
    return result


def _vwrite(path, raw):
    # Pending-open quarantine covers an unreturned possible descriptor. It does
    # not prove release; parent process teardown must close an uncertain FD.
    state = _VARIANT_IO
    if state[2] or state[3] or state[4] is not None:
        raise RuntimeError('Variant I/O quarantined')
    if type(raw) is not bytes or len(raw) > _VARIANT_WIRE_LIMIT:
        raise RuntimeError('Variant write bound differs')
    ops = _vops()
    open0, write0, sync0, close0 = ops[0], ops[4], ops[5], ops[3]
    flags = _VARIANT_WRITE_FLAGS
    primary = secondary = None
    try:
        state[3] = True
        state[4] = open0(path, flags, 0o600)
        state[3] = False
        offset = calls = 0
        while offset < len(raw):
            if calls >= 4096:
                raise RuntimeError('Variant write call bound exceeded')
            count = write0(state[4], raw[offset:offset + 65536])
            if type(count) is not int or count <= 0:
                raise RuntimeError('Variant write made no progress')
            offset += count
            calls += 1
        sync0(state[4])
    except BaseException as error:
        # Inline first bind before any allocating helper/call/formatting.
        primary = error
        state[0] = error
        state[2] = True
    finally:
        if state[4] is not None:
            descriptor = state[4]
            state[4] = None
            try:
                close0(descriptor)
            except BaseException as error:
                secondary = error
                state[1] = error
                state[2] = True
    if primary is not None:
        raise primary
    if secondary is not None:
        raise secondary


def _vraw(path, value):
    # Caller roots the returned object before entering this allocating encoder.
    _vwrite(path, _vencode(value))


def _vread(path, limit):
    state = _VARIANT_IO
    if state[2] or state[3] or state[4] is not None:
        raise RuntimeError('Variant I/O quarantined')
    if type(limit) is not int or not 0 < limit <= 67108864:
        raise RuntimeError('Variant read bound differs')
    ops = _vops()
    open0, read0, stat0, close0 = ops[0], ops[1], ops[2], ops[3]
    primary = secondary = None
    result = before = after = None
    try:
        state[3] = True
        state[4] = open0(path, _VARIANT_READ_FLAGS)
        state[3] = False
        before = stat0(state[4])
        if before.st_mode & 0o170000 != 0o100000 or before.st_size > limit:
            raise RuntimeError('Variant file type or size differs')
        blocks, total, calls = [], 0, 0
        while True:
            if calls >= 4096:
                raise RuntimeError('Variant read call bound exceeded')
            block = read0(state[4], min(65536, limit + 1 - total))
            calls += 1
            if not block:
                break
            blocks.append(block)
            total += len(block)
            if total > limit:
                raise RuntimeError('Variant file grew past bound')
        after = stat0(state[4])
        if ((before.st_dev, before.st_ino, before.st_mode, before.st_size,
             before.st_mtime_ns, before.st_ctime_ns) !=
            (after.st_dev, after.st_ino, after.st_mode, after.st_size,
             after.st_mtime_ns, after.st_ctime_ns) or total != after.st_size):
            raise RuntimeError('Variant file changed during read')
        result = b''.join(blocks)
    except BaseException as error:
        primary = error
        state[0] = error
        state[2] = True
    finally:
        if state[4] is not None:
            descriptor = state[4]
            state[4] = None
            try:
                close0(descriptor)
            except BaseException as error:
                secondary = error
                state[1] = error
                state[2] = True
    if primary is not None:
        raise primary
    if secondary is not None:
        raise secondary
    return result, after


def _vstdout(raw):
    # Descriptor 1 is borrowed from the owning parent and is never closed here.
    # receipt.json already exists before this terminal mirror is attempted.
    state = _VARIANT_IO
    if state[2] or state[3] or state[4] is not None:
        raise RuntimeError('Variant stdout after I/O quarantine')
    if type(raw) is not bytes or len(raw) > _VARIANT_WIRE_LIMIT:
        raise RuntimeError('Variant stdout bound differs')
    write0 = _vops()[4]
    try:
        offset = calls = 0
        while offset < len(raw):
            if calls >= 4096:
                raise RuntimeError('Variant stdout call bound exceeded')
            count = write0(1, raw[offset:offset + 65536])
            if type(count) is not int or count <= 0:
                raise RuntimeError('Variant stdout made no progress')
            offset += count
            calls += 1
    except BaseException as error:
        state[0] = error
        state[2] = True
        raise

def _vfile(path, bound):
    raw, info = _vread(path, bound)
    row = {'sha256': hashlib.sha256(raw).hexdigest(), 'bytes': len(raw),
           'mode': info.st_mode & 0o7777,
           'identity': {'device': info.st_dev, 'inode': info.st_ino,
             'mode': info.st_mode, 'links': info.st_nlink, 'uid': info.st_uid,
             'gid': info.st_gid, 'size': info.st_size,
             'mtime_ns': info.st_mtime_ns, 'ctime_ns': info.st_ctime_ns}}
    return raw, row


def _vcustody(plan):
    # All expected files are independently frozen source/header/compiler/link
    # closure, including full system header/search/macro/specs witness inputs.
    # This loop observes bytes; it cannot invent closure completeness.
    rows, total = [], 0
    for entry in plan['source_files']:
        raw, current = _vfile(entry['path'], 67108864)
        total += len(raw)
        if total > 268435456:
            raise RuntimeError('Variant custody aggregate byte bound exceeded')
        rows.append({'path': entry['path'], **current})
    return rows


def _vcheck_custody(observed, expected):

                (not _vsame(row['bytes'], fixed['bytes'])) or (not _vsame(row['mode'], fixed['mode']))):
