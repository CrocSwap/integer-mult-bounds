def _file(path):
    path = Path(path)
    need(path.is_file() and not path.is_symlink(), 'Expected regular native source or binary')
    before = path.stat()
    digest, size = hashlib.sha256(), 0
    with path.open('rb') as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b''):
            size += len(block)
            need(size <= 64 * 1024 * 1024, 'Native build input/output exceeds source-reviewed file bound')
            digest.update(block)
    after = path.stat()
    identity = lambda value: (value.st_dev, value.st_ino, value.st_mode, value.st_size,
                              value.st_mtime_ns, value.st_ctime_ns)
    need(identity(before) == identity(after) and size == before.st_size, 'Native file changed during hash')
    return {'sha256': digest.hexdigest(), 'bytes': size, 'mode': stat.S_IMODE(after.st_mode)}


def _source(root, relative, custody):
    need(relative in custody, 'Native source absent from immutable stage custody')
    row = _file(root / relative)
    need({key: row[key] for key in ('sha256', 'mode')} == custody[relative],
         'Native source differs from immutable stage')
    return row


def _fresh(packet):
    need(not packet['late_started'], 'Native qualification is closed after late audit control')
    api = packet['bootstrap_api']
    need(api[0]() == BOOTSTRAP_ROOT_SHA256, 'Actual bootstrap source changed')
    return api[1](packet['bootstrap_capability'], PROBE_PROVIDER_SHA256)


def _preflight(packet):

def _dependencies():
    need(type(BUILD_CONTRACT) is dict and
         set(BUILD_CONTRACT) == {'compiler', 'dependencies', 'extension_suffix', 'include', 'timeout_seconds'},
         'Cold compiler/header/link contract is not configured')
    need(type(BUILD_CONTRACT['compiler']) is tuple and BUILD_CONTRACT['compiler'] and
         all(type(arg) is str for arg in BUILD_CONTRACT['compiler']) and
         Path(BUILD_CONTRACT['compiler'][0]).is_absolute(), 'Pinned compiler command required')
    need(type(BUILD_CONTRACT['dependencies']) is tuple and BUILD_CONTRACT['dependencies'],
         'Pinned compiler/link dependency closure required')
    rows = {}
    for path, expected_sha, expected_size in BUILD_CONTRACT['dependencies']:
        need(type(path) is str and Path(path).is_absolute(), 'Compiler dependency path must be absolute')
        row = _file(path)
        need(row['sha256'] == expected_sha and row['bytes'] == expected_size,
             'Cold compiler/link dependency identity differs')
        rows[path] = row
    need(BUILD_CONTRACT['compiler'][0] in rows, 'Compiler executable not in authenticated build closure')
    return rows


def prepare(root, output, custody, bootstrap_root, bootstrap_capability, bootstrap_api):
    """Build three cold modules before any original or new control is run."""
    root, output = Path(root), Path(output)
    packet = {'bootstrap_root': bootstrap_root, 'bootstrap_capability': bootstrap_capability,
              'bootstrap_api': bootstrap_api, 'builds': {}, 'controls': {},
              'late_started': False, 'late_completed': False, 'direct_close_stop': None}
    _fresh(packet)
    _source(root, 'integration/cold_build_evidence.py', custody)
    before = _dependencies()
    need(sysconfig.get_config_var('EXT_SUFFIX') == BUILD_CONTRACT['extension_suffix'] and
         sysconfig.get_paths()['include'] == BUILD_CONTRACT['include'],
         'Native build Python suffix/header path differs')
    need(type(BUILD_CONTRACT['timeout_seconds']) is int and
         1 <= BUILD_CONTRACT['timeout_seconds'] <= 90, 'Cold compiler timeout outside reviewed bound')
    directory = output / 'cold-native-metadata'
    need(not directory.exists(), 'Refusing any prior native metadata build directory')
    directory.mkdir()
    packet['preflight_before_build'] = _preflight(packet)
    source_root = root / 'areas/cache-validity/probe-native'
    sources_before = {filename: _source(root, 'areas/cache-validity/probe-native/' + filename, custody)
                      for _, filename, _ in _SOURCES}
    for key, filename, module_name in _SOURCES:
        target = directory / (module_name + BUILD_CONTRACT['extension_suffix'])
        command = list(BUILD_CONTRACT['compiler']) + [
            '-shared', '-fPIC', '-O3', '-std=c11', '-Wall', '-Wextra', '-Werror',
            '-I' + BUILD_CONTRACT['include'], str(source_root / filename), '-o', str(target)]
        context = {'source': sources_before[filename], 'source_name': filename,
                   'no_cached_binary': True, 'module_name': module_name, 'target': str(target)}
        attempt = cold_build_evidence.run(command, directory, key,
                                          BUILD_CONTRACT['timeout_seconds'], context)
        receipt = {'command': command, 'returncode': attempt['returncode'],
                   'stdout': attempt['logs']['stdout'], 'stderr': attempt['logs']['stderr'],
                   'attempt_path': attempt['attempt_path'], 'attempt_journal_path': attempt['journal_path'],
                   'process_policy': attempt['process_group_policy'],
                   'source': sources_before[filename], 'source_name': filename,
                   'no_cached_binary': True, 'module_name': module_name, 'path': str(target),
                   'status': 'EXIT0_LOGS_STABLE_PENDING_FINAL_SOURCE_CUSTODY'}
        (directory / (key + '-build.json')).write_text(json.dumps(receipt, sort_keys=True, indent=2) + '\n')
        need(attempt['returncode'] == 0, 'Cold metadata compilation failed; no retry')
        receipt['binary'] = _file(target)
        packet['builds'][key] = receipt
        (directory / (key + '-build.json')).write_text(json.dumps(receipt, sort_keys=True, indent=2) + '\n')
    after = {filename: _source(root, 'areas/cache-validity/probe-native/' + filename, custody)
             for _, filename, _ in _SOURCES}
    need(sources_before == after and _dependencies() == before, 'Cold build source/compiler custody changed')
    for key, receipt in packet['builds'].items():
        receipt['status'] = 'COLD_NATIVE_BUILD_SOURCE_CUSTODY_PASS'
        (directory / (key + '-build.json')).write_text(json.dumps(receipt, sort_keys=True, indent=2) + '\n')
    packet['compiler_dependencies'] = before
    packet['preflight_after_build'] = _preflight(packet)
    for key, filename in (('direct_controls', 'direct_controls.py'),
                          ('bootstrap_controls', 'bootstrap_controls_v2.py'),
                          ('metadata_controls', 'metadata_controls_v3.py')):
        relative = 'areas/cache-validity/probe-native/' + filename
        _source(root, relative, custody)
        packet[key] = adapters.load(root / relative, '_h25_v2_' + key)
    production = packet['builds']['production']
    packet['extension_path'] = production['path']
    packet['extension_sha256'] = production['binary']['sha256']
    return packet

def run_controls(packet, phases):
    """No scientific authority is shared with these distinct direct fixtures."""
    pre = phases.call('bootstrap_preload_controls', packet['bootstrap_controls'].run_preload_controls,
                      *_arguments(packet))
    need(pre.get('schema') == 'h25-bootstrap-preload-controls-v2' and
         pre.get('expected_records') == 21 and len(pre.get('records', [])) == 21 and
         pre.get('scientific_calls') == 0, 'Bootstrap preload control census differs')
    pre['status'] = CONTROL_STATUS['bootstrap_preload']
    packet['controls']['bootstrap_preload'] = pre
    metadata = phases.call('native_metadata_controls', packet['metadata_controls'].run_metadata_controls,
                           packet['bootstrap_root'], packet['bootstrap_capability'],
                           packet['bootstrap_api'], PROBE_PROVIDER_SHA256)
    need(metadata.get('schema') == 'h25-native-metadata-controls-v3' and
         metadata.get('status') == 'PASS' and metadata.get('expected_records') == 16 and
         metadata.get('observed_record_count') == len(metadata.get('records', [])) == 16 and
         metadata.get('native_lifecycle_before') == metadata.get('native_lifecycle_after') and
         metadata.get('scientific_calls') == 0 and metadata.get('native_creations') == 0 and
         metadata.get('native_checks') == 0 and metadata.get('native_inspections') == 0,
         'Native metadata control census or actual lifecycle continuity differs')
    packet['controls']['native_metadata'] = metadata
    direct = packet['direct_controls']
    disarmed = phases.call('native_disarmed_controls', direct.run_disarmed_controls, _raw_control(packet, 'production'))
    bounds = phases.call('native_bounds_controls', direct.run_bounds_controls, _raw_control(packet, 'bounds'))
    watcher = _raw_control(packet, 'watcher')
    provider = _fresh(packet)
    binding = packet['bootstrap_api'][2](packet['bootstrap_capability'], packet['extension_path'], packet['extension_sha256'])
    captured_close = binding.close
    try:
        result = phases.call('native_direct_controls', direct.run_controls, provider, binding, watcher)
    except BaseException:
        try:
            captured_close()
        except BaseException as close_error:
            packet['direct_close_stop'] = {'type': type(close_error).__name__, 'message': str(close_error)}
        raise
    need(result.get('schema') == 'h25-native-direct-control-results-v1' and
         result.get('inspection_calls') == 58 and len(result.get('records', [])) == 60 and
         result.get('scientific_imports') == 0 and result.get('scientific_arithmetic_calls') == 0 and
         len(disarmed) == 1 and len(bounds) == 1, 'Native direct/fixture control census differs')
    packet['controls']['native_probe'] = {
        'status': CONTROL_STATUS['native_probe'], 'direct': result,
        'disarmed': disarmed, 'bounds': bounds, 'records': len(result['records']) + len(disarmed) + len(bounds)}
    return packet['controls']

    packet['final_native_diagnostics_before_late'] = packet['bootstrap_api'][3](packet['bootstrap_capability'])

    result['status'] = CONTROL_STATUS['bootstrap_late']
    packet['controls']['bootstrap_late'] = result
    packet['late_completed'] = True
    return result

        'builds', 'compiler_dependencies', 'preflight_before_build', 'preflight_after_build',
        'final_native_diagnostics_before_late', 'final_preflight_before_late',
        'late_started', 'late_completed', 'direct_close_stop')}
