def need(condition, message):
    if not condition:
        raise RuntimeError(message)


def _relative(name):
    need(type(name) is str, 'Stage path must be exact text')
    path = PurePosixPath(name)
    need(bool(path.parts) and not path.is_absolute() and '..' not in path.parts and
         '\\' not in name and path.as_posix() == name, 'Noncanonical stage path: ' + name)
    return name


def _row(path):
    need(path.is_file() and not path.is_symlink(), 'Missing or symlinked data source: ' + str(path))
    hasher = hashlib.sha256()
    with path.open('rb') as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b''):
            hasher.update(block)
    return {'sha256': hasher.hexdigest(), 'mode': stat.S_IMODE(path.stat().st_mode)}


def _manifest(root):
    path = root / 'STAGE.json'
    need(path.is_file() and not path.is_symlink(), 'Missing regular data stage manifest')
    manifest = json.loads(path.read_bytes())
    need(manifest.get('schema') == 'h25-data-handling-immutable-stage-v1' and
         type(manifest.get('files')) is dict, 'Wrong immutable data stage contract')
    for name in manifest['files']:
        _relative(name)
    return manifest


def custody(root):
    """Fresh complete new helper/control/integration source census and hashes."""
    manifest = _manifest(root)
    selected = {name for name in manifest['files']
                if (name.startswith(('areas/', 'integration/')) or '/' not in name)
                and Path(name).suffix in ('.py', '.c')}
    actual = set()
    for path in root.iterdir():
        need(not path.is_symlink(), 'Symlink in stage root: ' + str(path))
        if path.is_file() and path.suffix in ('.py', '.c'):
            actual.add(path.name)
    for directory in (root / 'areas', root / 'integration'):
        need(directory.is_dir() and not directory.is_symlink(), 'Missing regular new source directory')
        for path in directory.rglob('*'):
            need(not path.is_symlink(), 'Symlink in new data source tree: ' + str(path))
            if path.is_file() and path.suffix in ('.py', '.c'):
                actual.add(path.relative_to(root).as_posix())
    need(actual == selected, 'Complete new data source census changed')
    need({name for _, name, _ in _MODULES} | {'integration/full_driver.py', 'integration/scientific_driver.py', 'integration/data_adapters.py',
         'integration/native_packet.py', 'integration/probe_binding_v3.py',
         'integration/guarded_adapters_v4.py', 'areas/cache-validity/cold_authority_v7.py'} <= selected,
         'Required data-handling source missing from stage')
    result = {}
    for name in sorted(selected):
        observed = _row(root / name)
        need(observed == manifest['files'][name], 'New data source custody failed: ' + name)
        result[name] = observed
    return result


def _load(root, relative, name, source_custody):
    need(relative in source_custody and _row(root / relative) == source_custody[relative],
         'Fresh data import custody failed: ' + relative)
    return adapters.load(root / relative, name)

def _runtime_images(writer):
    """Fresh bounded hash reads of actual interpreter/JSON runtime file paths.

    This records file custody, not a claim about arbitrary process memory. The
    writer separately checks live module/class/function bindings at entry/exit.
    """
    paths = {'python_executable': Path(sys.executable).resolve()}
    modules = {}
    for owner, _, _ in writer._h25_serialization_bindings:
        if type(owner) is ModuleType and owner.__name__ in ('json', 'json.encoder', '_json'):
            modules[owner.__name__] = owner
    need(set(modules) == {'json', 'json.encoder', '_json'}, 'Actual JSON runtime module census changed')
    for name, module in modules.items():
        need(type(module.__file__) is str, 'JSON runtime path missing')
        paths[name] = Path(module.__file__).resolve()
    result = {}
    for name, path in sorted(paths.items()):
        need(path.is_file() and not path.is_symlink(), 'JSON runtime image missing')
        before = path.stat()
        hasher, count = hashlib.sha256(), 0
        with path.open('rb') as stream:
            for block in iter(lambda: stream.read(1024 * 1024), b''):
                count += len(block)
                hasher.update(block)
        after = path.stat()
        identity = lambda value: (value.st_dev, value.st_ino, value.st_mode, value.st_size,
                                  value.st_mtime_ns, value.st_ctime_ns)
        need(identity(before) == identity(after) and count == before.st_size, 'JSON runtime image changed while read')
        result[name] = {'path': str(path), 'sha256': hasher.hexdigest(), 'bytes': count,
                        'mode': stat.S_IMODE(after.st_mode)}
    return result

def transport(root, baseline, output):
    """Original bounded historical artifact transport; no scientific imports."""

    data_before = custody(root)

        need(observed == manifest['files'][name], 'Historical transport stage custody failed: ' + name)
        before[name] = observed

    finally:
        sys.argv = old_argv

        need(_row(root / name) == expected, 'Historical transport closure changed: ' + name)
    need(adapters.custody(root, 'all') == original_before and custody(root) == data_before,
         'Transport integration source custody changed')
    need(_manifest(root) == manifest, 'Transport stage manifest changed')
    output.mkdir(parents=True)
    receipt = {'schema': 'h25-data-handling-transport-v1', 'status': 'H25_DATA_HANDLING_TRANSPORT_PASS',

def teardown_cache(state):
    """Exactly one terminal attempt; retain actual owner's diagnostic on failure."""
    if state['cache_binding'] is None:
        return None
    need(not state['native_packet']['late_started'], 'Cache teardown forbidden after late audit control')
    need(not state['cache_teardown_attempted'], 'Cache teardown already attempted')
    state['cache_teardown_attempted'] = True
    owner = state['cache_binding']['owner']
    try:
        receipt = owner.teardown()
    except BaseException:
        # Copy only the JSON-ready diagnostic; never manufacture missing counters.
        try:
            state['cache_teardown_stop'] = owner.teardown_stop
        except BaseException as diagnostic_error:
            state['cleanup_errors'].append({'stage': 'read-owner-teardown-stop',
                                            'type': type(diagnostic_error).__name__})
        raise
    state['cache_teardown_receipt'] = receipt
    return receipt


def cleanup_owned(state):
    """Failure-path cleanup, preserving primary exceptions and every STOP record."""
    if state is None or state['native_packet']['late_started']:
        return []
    errors = []
    for owned in reversed(state['owned_states']):
        if owned['cache_binding'] is not None and not owned['cache_teardown_attempted']:
            try:
                teardown_cache(owned)
            except BaseException as error:
                item = {'type': type(error).__name__, 'message': str(error),
                        'arm': owned['arm'], 'cache_teardown_stop': owned['cache_teardown_stop']}
                owned['cleanup_errors'].append(item)
                errors.append(item)
    return errors

def epoch_retire(state,reason,primary=None,secondary=None,restoration_error=None,error_channel=None,cleanup_scope=None):
    if state is None:return None
    record=state.get('epoch_acquisition')
    if record is not None and primary is not None and record['primary'] is None:record['primary']=primary
    if record is not None and restoration_error is not None and restoration_error is not primary:
        if record['secondary'] is None:record['secondary']=restoration_error
        record['uncertain']=True
    if record is not None and secondary is not None and secondary is not primary:
        if record['secondary'] is None:record['secondary']=secondary
        record['uncertain']=True
    suite=state.get('epoch_suite')
    if suite is None:return None
    if primary is not None and suite.primary_error is None:suite.primary_error=primary
    if restoration_error is not None and restoration_error is not primary:
        if suite.secondary_error is None:suite.secondary_error=restoration_error
        suite.uncertain=True
    if secondary is not None and secondary is not primary:
        if suite.secondary_error is None:suite.secondary_error=secondary
        suite.uncertain=True
        try:suite.secondary.append(secondary)
        except BaseException:pass
    # Retirement is source-only plain metadata disposal, not a late native check.
    cleanup=state.get('epoch_cli_cleanup')
    scope=('outer' if error_channel is None else 'inner') if cleanup_scope is None else cleanup_scope
    need(type(scope) is str and scope in ('inner','outer'),'Epoch exact retirement scope')
    outer=scope=='outer'
    if error_channel is not None:channel=error_channel
    elif cleanup is not None:channel=cleanup['outer_retirement_channel' if outer else 'retirement_channel']
    else:channel=suite.outer_retirement_channel if outer else suite.inner_retirement_channel
    try:return suite.retire_preserving(reason,primary,channel)
    finally:
        # Every declared attempt has its own channel and fixed R/J handoff.
        # No ACKed channel is cleared, reused or inferred from journal growth.
        if outer:
            suite.outer_retirement_error=channel['error'];suite.outer_journal_error=channel['journal_error']
        else:
            suite.inner_retirement_error=channel['error'];suite.inner_journal_error=channel['journal_error']
        if cleanup is not None and channel['error'] is not None:
            name='outer_retirement_error' if outer else 'retirement_error'
            if cleanup[name] is None:cleanup[name]=channel['error']
            cleanup['uncertain']=True
        channel['acknowledged']=True

def epoch_cleanup_owned(state):
    """Pending/acquired owners remain on the original parent cleanup path."""
    if state is None or state['native_packet']['late_started']:return []
    errors=[]
    for owned in reversed(state['owned_states']):
        record=owned.get('epoch_acquisition')
        if record is None or owned['cache_binding'] is None or owned['cache_teardown_attempted']:continue
        record['close_attempted']=True;record['phase']='PARENT_CLOSE_ATTEMPTING'
        try:
            teardown_cache(owned)
        except BaseException as secondary:
            if record['secondary'] is None:record['secondary']=secondary
            record['uncertain']=True;record['phase']='PARENT_CLOSE_FAILED'
            # Retain the raw error before any evidence formatting/allocation.
            item={'type':type(secondary).__name__,'arm':owned['arm'],'cache_teardown_stop':owned['cache_teardown_stop']}
            owned['cleanup_errors'].append(item);errors.append(item)
        else:
            record['close_result']=owned['cache_teardown_receipt'];record['phase']='PARENT_TEARDOWN_RETURNED'
    errors.extend(cleanup_owned(state))  # legacy owners; attempted epoch owner is skipped
    return errors
