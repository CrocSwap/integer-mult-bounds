## JunkLawA.LawAModel (structure; its fields are the premises hspan, hdiag, hodd, hpart)
```lean
structure LawAModel (K : Type*) [Field K] (P D : ℕ) where
  M : Matrix (Fin P) (Fin P) K
  arr : Fin D → (Fin P → K)
  /-- every row `v_t` is a combination of arrivals: one arrival vector per owned descending move -/
  hspan : ∀ t, M t ∈ Submodule.span K (Set.range arr)
  /-- Y_t odd spec: the diagonal coefficient `c_t ≠ 0` -/
  hdiag : ∀ t, M t t ≠ 0
  /-- `Ev t s`: `s` is an even partner of `t` -/
  Ev : Fin P → Fin P → Prop
  /-- odd off-diagonals vanish -/
  hodd : ∀ t s, s ≠ t → M t s ≠ 0 → Ev t s
  /-- 48 even partners -/
  hpart : ∀ t, (univ.filter fun s => s ≠ t ∧ Ev t s).card ≤ 48
```

```lean
abbrev Hist := List (ℕ × ℕ)
structure Inv where
  T : Hist
  P : Hist
def scale (L : ℕ) (l : Hist) : Hist := l.map fun p => (p.1, L * p.2)
def invoice (c n : Inv) (L : ℕ) : Inv := ⟨c.T ++ scale L n.T, c.P ++ scale L n.P⟩
def armT (n c : ℕ) : Hist := (List.range n).map fun i => (8 * i + 9, c)
def armP (m c : ℕ) : Hist := (List.range m).map fun i => (8 * i + 16, c)
def callerT (a b : ℕ) : Hist := armT (a + 2) 208 ++ armT (b + 2) 208 ++ [(8 * (a + b + 4) + 1, 208)]
def callerP (a b : ℕ) : Hist :=
  armP a 208 ++ armP b 208 ++ [(8, 384), (1, 384), (8, 16 * (a + b + 4)), (9, 32)]
def caller (a b : ℕ) : Inv := ⟨callerT a b, callerP a b⟩
noncomputable def Fh (s : ℝ) (l : Hist) : ℝ := (l.map fun p => (p.2 : ℝ) * (p.1 : ℝ) ^ s).sum
noncomputable def Finv (s : ℝ) (i : Inv) : ℝ := Fh s i.T - Fh s i.P
```

```lean
noncomputable def Omega (ks : List ℕ) (mr : ℕ) (s : ℝ) : ℝ := dprice ks s + mr - (6 * (9 : ℝ) ^ s + 10)

noncomputable def dprice (ks : List ℕ) (s : ℝ) : ℝ := (ks.map fun k : ℕ => ((k : ℕ) : ℝ) ^ s).sum
```

### Junk sets (`JunkLawA.lean`)
```lean
noncomputable def outN (M : Matrix (Fin P) (Fin P) K) (t : Fin P) : Finset (Fin P) :=
  univ.filter fun s => s ≠ t ∧ M t s ≠ 0
noncomputable def jc (M : Matrix (Fin P) (Fin P) K) (t : Fin P) : ℕ := (outN M t).card
-- inside namespace LawAModel, with {D : ℕ} (m : LawAModel K P D):
noncomputable def j (t : Fin P) : ℕ := jc m.M t
noncomputable def J : ℕ := ∑ t, m.j t
noncomputable def junkSet (M : Matrix (Fin P) (Fin P) K) : Finset (ℕ × ℕ) :=
  (univ.filter fun p : Fin P × Fin P => p.1 ≠ p.2 ∧ M p.1 p.2 ≠ 0).image fun p => ((p.1 : ℕ), (p.2 : ℕ))
```
