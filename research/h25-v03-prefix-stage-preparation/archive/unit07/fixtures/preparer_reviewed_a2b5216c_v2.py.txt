def plain(value):

        need(kind is list or kind is dict,'exact plain value required')
        need(len(item)<=MAX_ROWS,'container bound')

    except BaseException as error:primary=owner.fail(error);raise primary
    finally:
        _close_slot(owner,slot);owner.busy=False
        if primary is None and owner.primary is not None:raise owner.primary


def install_file(owner,parent_path,name,raw,mode=0o444,copy=False):
    owner.ready();need(type(name) is str and name not in ('','.','..') and '/' not in name,'single file basename')
    need(type(raw) is bytes and len(raw)<=FILE_CAP,'bounded exact installed bytes');integer(mode,0,0o7777)
    need(parent_path in owner.directories,'retained actual destination directory')
    need(len(owner.writes)<8192,'installation row bound')
    if copy:
        need(owner.copy_attempts<41,'private copy attempt bound');owner.copy_attempts+=1
    path=parent_path+'/'+name;path_text(path);parent=owner.fd_slots[owner.directories[parent_path]]
    record={'path':path,'raw':raw,'mode':mode,'write_calls':0,'bytes_written':0,'attempted':True,'created':False,'observed':None}
    owner.writes.append(record);owner.fd_slots.append(None);slot=len(owner.fd_slots)-1;primary=None;owner.busy=True
    try:
        owner.pending=True
        owner.fd_slots[slot]=_io(owner,'open',name,os.O_RDWR|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW|os.O_CLOEXEC,0o600,dir_fd=parent)
        owner.pending=False;record['created']=True;owner.created.append({'path':path,'kind':'file','created':True})
        cap=4096 if len(raw)<=RECORD_CAP else 8192
        while record['bytes_written']<len(raw):
            need(record['write_calls']<cap,'write call bound');record['write_calls']+=1;owner.write_calls+=1
            offset=record['bytes_written'];count=_io(owner,'write',owner.fd_slots[slot],raw[offset:])
            need(type(count) is int and 0<count<=len(raw)-offset,'write progress');record['bytes_written']+=count
        _io(owner,'fchmod',owner.fd_slots[slot],mode);_io(owner,'fsync',owner.fd_slots[slot])
        record['final_identity']=identity(_io(owner,'fstat',owner.fd_slots[slot]))
        need(record['final_identity']['links']==1,'new file alias appeared')
    except BaseException as error:primary=owner.fail(error);raise primary
    finally:
        _close_slot(owner,slot);owner.busy=False
        if primary is None and owner.primary is not None:raise owner.primary
    pin,observed=_owned_read(owner,parent_path,name,path,FILE_CAP)
    need(observed==raw and pin['identity']==record['final_identity'],'installed bytes/final chmod identity differ')
    record['observed']=pin;owner.outputs[path]=pin;return pin


def install_record(owner,parent_path,name,value):
    raw=canonical(value);owner.generated_records.append({'path':parent_path+'/'+name,'raw':raw});owner.generated_bytes+=len(raw);owner.total_bytes+=len(raw)
    need(owner.total_bytes<=TOTAL_CAP,'generated plus retained aggregate bytes')
    return install_file(owner,parent_path,name,raw)


def verify_source(owner,pin,expected=None,cap=FILE_CAP):
    validate_blob(pin,cap)

    return _source_read(pin['path'],pin['sha256'],cap,owner,pin)


def source_row(pin,category):
    return {'category':category,'path':pin['path'],'sha256':pin['sha256'],'bytes':pin['bytes'],'mode':pin['identity']['mode']&0o7777}


def validate_source_rows(rows):
    need(type(rows) is list and 1<=len(rows)<=8191,'parent inner row bound')
    paths=set();categories=set();total=0
    for row in rows:
        exact(row,('category','path','sha256','bytes','mode'));need(type(row['category']) is str and row['category'] in SOURCE_CATEGORIES,'actual source category')
        path_text(row['path']);digest(row['sha256']);integer(row['bytes'],1,FILE_CAP);integer(row['mode'],0,0o7777)
        need(row['path'] not in paths,'duplicate source path');paths.add(row['path']);categories.add(row['category']);total+=row['bytes']
    need(categories==SOURCE_CATEGORIES and total<=TOTAL_CAP,'complete category set/declared aggregate bound')


def reconcile_rows(base,derived):
    rows={row['path']:dict(row) for row in base}
    for row in derived:
        if row['path'] in rows:
            prior=rows[row['path']];need(all(prior[k]==row[k] for k in ('path','sha256','bytes','mode')),'same path inconsistent metadata')
        else:rows[row['path']]=dict(row)
    result=list(rows.values());validate_source_rows(result);return result


def validate_expectation(value):
    exact(value,('outcome','exception_type','message','terminal_function','v03_discriminator','compiler_launches','candidate_image_loads','complete_raw_and_owned_terminal_required'))
    need(type(value['v03_discriminator']) is bool and type(value['compiler_launches']) is int and value['compiler_launches']==0 and type(value['candidate_image_loads']) is int and value['candidate_image_loads']==0 and value['complete_raw_and_owned_terminal_required'] is True,'prefix expectation scope')
    if value['outcome']=='PREFIX_RETURN':need(value['exception_type'] is None and value['message'] is None and value['terminal_function'] is None,'positive expectation');return
    need(value['outcome']=='EXACT_SOURCE_REFUSAL' and value['exception_type']=='RuntimeError','source refusal outcome')
    if value['terminal_function']=='_vcheck_literal_sources':need(value['message'] in ('Variant source role path differs','Variant reviewed source/header differs','Variant role file omitted from closure'),'literal refusal site');return
    if value['terminal_function']=='_vcheck_custody':need(value['message']=='Variant current source closure differs','custody refusal site');return
    need(value['terminal_function']=='_v03_original_input_prefix' and (value['message']=='Variant literal configuration differs' or (value['message']=='Variant original preflight differs' and value['v03_discriminator'] is False)),'original preflight refusal site')

    simple(request['queue'],queue=True)
    need(type(request['mode']) is str and request['mode'] in ('normal','optimized'),'exact mode')
    need(type(request['lanes']) is str and 1<=len(request['lanes'])<=8192 and type(request['finite_only']) is bool,'lanes/finite-only')
    integer(request['session_seconds'],1,600);digest(request['source_closure_admission_sha256'])
    exact(request['outer_policy'],('shared_seconds','root_seconds','root_address_space_bytes','root_rss_sum_bytes'))

    for name,expected in (('entry_source',ENTRY_PIN),('supervisor_source',SUPERVISOR_PIN),('native_tcb_source',TCB_PIN),('runner_source',RUNNER_PIN),('original_variant',(VARIANT_SHA,41337))):

    need(request['caller_source']['path'].endswith('/run_queue4_prefix_v2.py'),'new caller basename')
    exact(request['receiver_sources'],PROJECT_PINS)
    for name,pin in request['receiver_sources'].items():validate_blob(pin);need((pin['sha256'],pin['bytes'])==PROJECT_PINS[name],'exact retained graph pin')
    validate_source_rows(request['parent_source_rows'])
    exact(request['admissions'],GATE_FIELDS)
    for key,fields in GATE_FIELDS.items():

    child=request['child']
    exact(child,('stage_directory','stage_manifest','stage_directories','root_source','driver_source','gate_values','expectation','id','generation','launch_nonce','root_seconds','address_space_bytes','rss_bytes'))

    for key in ('stage_manifest','root_source','driver_source','gate_values'):validate_blob(child[key],RECORD_CAP)
    integration=child['stage_directory']+'/integration'
    need(child['stage_manifest']['path']==child['stage_directory']+'/STAGE.json','child manifest path')
    need(child['root_source']['path']==integration+'/full_driver.py' and (child['root_source']['sha256'],child['root_source']['bytes'])==(ROOT_SHA,38851),'genuine Root source')
    need(child['driver_source']['path']==integration+'/v03_control_driver_v1.py' and 0<child['driver_source']['bytes']<=262144,'configured driver source')
    need(type(child['stage_directories']) is list and len(child['stage_directories'])<=16384,'child directory inventory')
    for rel in child['stage_directories']:
        need(type(rel) is str and rel and not rel.startswith('/') and all(p not in ('','.','..') for p in rel.split('/')),'relative stage directory')
    need(child['stage_directories']==sorted(set(child['stage_directories'])),'sorted unique child directories')
    validate_expectation(child['expectation'])
    root=request['canonical_root'];need(request['runner_source']['path']==root+'/run_guarded_v3.py','canonical actual runner')
    paths={'Sb':root+'/stages/'+request['stage'],'packet':root+'/control/'+request['queue']+'-packet','K':root+'/control/'+request['queue']+'-private'}

    need(all(row['path'] not in excluded and not any(row['path']==p or row['path'].startswith(p+'/') for p in paths.values()) for row in request['parent_source_rows']),'base contains destination/backedge')

def census_child(owner,child,base_pins):
    manifest_raw=verify_source(owner,child['stage_manifest'],cap=RECORD_CAP)[1];declared=decode(manifest_raw)
    need(type(declared) is dict and type(declared.get('files')) is dict,'child declared file inventory')
    root=child['stage_directory'];stack=[('',root)];files=[];directories=[]
    while stack:
        relative,current=stack.pop();slot=open_existing_directory(owner,current);fd=owner.fd_slots[slot]
        before=identity(_io(owner,'fstat',fd))
        need(not before['mode']&0o222,'child immutable directory mode')
        iterator=None;primary=None;owner.pending=True
        iterator_record={'path':current,'handle':None,'close_attempted':False,'closed':False};owner.iterators.append(iterator_record)
        try:
            owner.scandir_calls+=1;need(owner.scandir_calls<=16384,'scandir bound');iterator=_io(owner,'scandir',fd);iterator_record['handle']=iterator;owner.pending=False
            while True:
                try:
                    owner.io_counts['iterator_next']=owner.io_counts.get('iterator_next',0)+1;entry=next(iterator)
                except StopIteration:break
                owner.entry_count+=1;need(owner.entry_count<=16384,'child census entry bound')
                name=entry.name;need(type(name) is str and name not in ('','.','..') and '/' not in name,'actual child basename')
                rel=name if not relative else relative+'/'+name
                owner.io_counts['entry_stat']=owner.io_counts.get('entry_stat',0)+1;owner.stat_calls+=1
                info=entry.stat(follow_symlinks=False)
                if stat.S_ISDIR(info.st_mode):directories.append(rel);stack.append((rel,current+'/'+name))
                elif stat.S_ISREG(info.st_mode):files.append(rel)
                else:raise Stop('child symlink/special entry')
        except BaseException as error:primary=owner.fail(error);raise primary
        finally:
            if iterator is not None:
                old=iterator;iterator=None;iterator_record['handle']=None;iterator_record['close_attempted']=True
                try:
                    owner.io_counts['iterator_close']=owner.io_counts.get('iterator_close',0)+1;old.close();iterator_record['closed']=True
                except BaseException as error:
                    owner.fail(error)
                    if primary is None:raise owner.primary
        after=identity(_io(owner,'fstat',fd))
        need(all(before[k]==after[k] for k in ('device','inode','mode','mtime_ns','ctime_ns')),'child directory drift')
    need(sorted(files)==sorted(list(declared['files'])+['STAGE.json']) and sorted(directories)==child['stage_directories'],'actual child census differs')
    for rel,metadata in declared['files'].items():
        need(type(rel) is str and not rel.startswith('/') and all(p not in ('','.','..') for p in rel.split('/')),'child relative file')
        pin=base_pins.get(root+'/'+rel);need(pin is not None,'child file omitted from genuine source rows')
        need(metadata=={'sha256':pin['sha256'],'mode':pin['identity']['mode']&0o7777},'child declared row differs')
    return declared

def _prepare(request,owner):

    base_pins={}
    for row in request['parent_source_rows']:
        pin,raw=_source_read(row['path'],row['sha256'],FILE_CAP,owner)
        need(pin['bytes']==row['bytes'] and pin['identity']['mode']&0o7777==row['mode'],'base source actual size/mode differs');base_pins[row['path']]=pin
    for key in ('root_source','driver_source','gate_values'):verify_source(owner,child[key],cap=RECORD_CAP)

    need(provider is not None and provider['sha256']==PROVIDER_SHA,'genuine provider omitted/differs')
    census_child(owner,child,base_pins)
    owner.phase='OWNING_DESTINATIONS'
    for existing in (root+'/stages',root+'/control'):open_existing_directory(owner,existing)
    sb=make_directory(owner,root+'/stages',request['stage']);packet=make_directory(owner,root+'/control',request['queue']+'-packet');private=make_directory(owner,root+'/control',request['queue']+'-private')

    for name in ('parent-queue.lock','science.lock'):

    private_locks=[]

    tcb=install_file(owner,control,'NATIVE-INITIAL-TCB-v1.json',raw_sources['native_tcb_source'],copy=True)

    original=install_file(owner,data,'variant_driver_v1.py.txt',raw_sources['original_variant'],copy=True)

    readiness=install_record(owner,packet,'SOURCE-READINESS.json',{'files':{'run_queue4_prefix_v2.py':{'sha256':caller['sha256'],'bytes':caller['bytes']},'supervisor.py':{'sha256':supervisor['sha256'],'bytes':supervisor['bytes']}}})
    policy=install_record(owner,data,'RECEIVER-POLICY.json',receiver_policy)
    session=root+'/runtime/runs/'+request['label']+'/science';output=session+'/producer-output'
    actual=[source_row(pin,'project') for pin in (entry,original,policy,caller,supervisor,readiness,tcb,*receiver_policy.values())]

    case.update(output=output,command=construct_command(child,request['mode'],output))

def prepare_prefix_stage(request):
    """One actual preparation attempt; failed owner remains in PREPARATIONS."""
    need(_PREPARATION_USED[0] is False,'one public preparation attempt per fresh process');_PREPARATION_USED[0]=True
    owner=make_preparation_owner();owner.burned=True;primary=None;result=None
    try:
        result=_prepare(request,owner)
    except BaseException as error:
        primary=owner.fail(error)
    finally:
        try:close_preparation(owner)
        except BaseException as error:owner.fail(error)
        owner.final_bill=operation_bill(owner)
        if result is not None:result['operation_bill']=owner.final_bill
    if owner.primary is not None:raise owner.primary
    owner.result=result
    return result
