/* Unexecuted actual-owner controls. All native acquisition calls reached by
 * these cases are mocked. No mount, namespace, statx or LSM probe occurs.
 * Compilation/execution requires independent GO and parent admission. */
#define syscall fact_mock_syscall
#define fstat fact_mock_fstat
#define main fact_unused_entry_main
#include "../proposal/science_view_entry.c"
#undef main
#undef syscall
#undef fstat
#include <stdarg.h>

static int failures, mock_errno, mock_mode, mock_calls;
int fact_mock_fstat(int fd,struct stat *st) {
    (void)fd;memset(st,0,sizeof *st);st->st_dev=makedev(8,1);st->st_ino=99;st->st_mode=S_IFREG|0600;return 0;
}
static void check(bool value,const char *label) {
    if(!value){fprintf(stderr,"control failure: %s\n",label);failures++;}
}
static struct sv_statmount_buffer mount_buffer(uint64_t id,uint64_t parent,const char *type) {
    struct sv_statmount_buffer b={0};
    b.h=(struct sv_statmount_head){.size=512+(uint32_t)strlen(type)+1,
        .mask=SV_STATMOUNT_MASK,.dev_major=8,.dev_minor=1,.magic=0xef53,
        .id=id,.parent=parent,.propagation=MS_PRIVATE,.ns=9};
    memcpy(b.str,type,strlen(type)+1);return b;
}
long fact_mock_syscall(long operation,...) {
    va_list ap;va_start(ap,operation);mock_calls++;
    if(mock_errno){va_end(ap);errno=mock_errno;return -1;}
    if(operation==SV_NR_LSM_LIST_MODULES) {
        uint64_t *ids=va_arg(ap,uint64_t *);uint32_t *bytes=va_arg(ap,uint32_t *);
        unsigned flags=va_arg(ap,unsigned);
        check(*bytes==16&&flags==0,"LSM request uses actual fixed capacity and flags");
        ids[0]=110;ids[1]=100;*bytes=16;va_end(ap);return mock_mode==1?1:2;
    }
    if(operation==SV_NR_LISTMOUNT) {
        struct sv_mnt_request *req=va_arg(ap,struct sv_mnt_request *);
        uint64_t *ids=va_arg(ap,uint64_t *);size_t slots=va_arg(ap,size_t);
        unsigned flags=va_arg(ap,unsigned);
        check(req->size==32&&!req->spare&&req->id==UINT64_MAX&&!req->param&&!req->ns&&
            slots==129&&flags==0,"listmount requests M plus one from current actual root");
        ids[0]=1;ids[1]=mock_mode==3?1:2;va_end(ap);return mock_mode==2?129:2;
    }
    if(operation==SV_NR_STATMOUNT) {
        struct sv_mnt_request *req=va_arg(ap,struct sv_mnt_request *);
        struct sv_statmount_buffer *b=va_arg(ap,struct sv_statmount_buffer *);
        size_t capacity=va_arg(ap,size_t);unsigned flags=va_arg(ap,unsigned);
        check(req->size==32&&!req->spare&&!req->ns&&req->param==0x63&&
            capacity==sizeof *b&&flags==0,"statmount requests only callback-free selected fields");
        *b=mount_buffer(req->id,1,"ext4");
        if(mock_mode==4&&req->id==2)b->h.ns=10;
        va_end(ap);return 0;
    }
    if(operation==SYS_statx) {
        int fd=va_arg(ap,int);const char *path=va_arg(ap,const char *);
        int flags=va_arg(ap,int);unsigned mask=va_arg(ap,unsigned);
        struct sv_statx *sx=va_arg(ap,struct sv_statx *);
        check(fd==77&&!*path&&flags==(AT_EMPTY_PATH|AT_SYMLINK_NOFOLLOW)&&mask==0x4103,
            "object join requests actual held FD and unique mount mask");
        memset(sx,0,sizeof *sx);sx->mask=mask;sx->mode=S_IFREG|0600;sx->ino=99;
        sx->dev_major=8;sx->dev_minor=1;sx->mount_id=1;
        if(mock_mode==5)sx->mask&=~0x4000U;
        if(mock_mode==6)sx->mount_id=999;
        if(mock_mode==7)sx->ino=100;
        va_end(ap);return 0;
    }
    va_end(ap);check(false,"unexpected native acquisition");errno=ENOSYS;return -1;
}
static void reset(void) {
    error_count=0;uncertain=error_overflow=lsm_checked=topology_checked=false;
    native_queries_started=native_queries_completed=0;mock_errno=mock_mode=mock_calls=0;
    memset(&mounts_before,0,sizeof mounts_before);memset(&mounts_after,0,sizeof mounts_after);
}
static struct sv_mount_row row(uint64_t id,uint64_t parent,uint64_t attr) {
    struct sv_statmount_buffer b=mount_buffer(id,parent,"ext4");struct sv_mount_row r;
    b.h.attr=attr;check(decode_mount(&b,id,&r),"fixture passes actual decoder");return r;
}
int main(void) {
    reset();check(read_lsm_stack()&&lsm_checked&&native_queries_started==1&&native_queries_completed==1,
        "real LSM owner commits complete exact stack only");
    reset();mock_mode=1;check(!read_lsm_stack()&&!lsm_checked&&error_count==1,
        "partial LSM count cannot witness selected stack");
    reset();mock_errno=E2BIG;check(!read_lsm_stack()&&!lsm_checked&&error_count==1&&errors[0].number==E2BIG&&
        native_queries_started==1&&native_queries_completed==0,"LSM raw capacity errno precedes failure projection");
    uint64_t duplicate[2]={100,100};check(!valid_lsm_result(2,16,duplicate),"duplicate module ID refused");
    uint64_t extra[2]={100,110};check(!valid_lsm_result(3,24,extra),"additional module cannot be truncated away");

    reset();check(census(&mounts_before)&&mounts_before.count==2&&native_queries_started==3,
        "actual census owner acquires every unsaturated row");
    check(graph_rooted(&mounts_before,1),"actual parent graph reaches held root");
    uint64_t joined=0;
    check(join_object(77,&mounts_before,SV_FS_LOCAL,&joined)&&joined==1,"actual object join uses returned unique mount ID");
    mock_mode=5;check(!join_object(77,&mounts_before,SV_FS_LOCAL,&joined),"missing unique-ID result bit refused");
    mock_mode=6;check(!join_object(77,&mounts_before,SV_FS_LOCAL,&joined),"uncatalogued returned unique ID refused");
    mock_mode=7;check(!join_object(77,&mounts_before,SV_FS_LOCAL,&joined),"different actual descriptor inode refused");
    reset();mock_mode=2;check(!census(&mounts_before)&&mock_calls==1,"M plus one saturation refuses before any row query");
    reset();mock_mode=3;check(!census(&mounts_before),"duplicate enumerated ID refused");
    reset();mock_mode=4;check(!census(&mounts_before),"mixed namespace rows refused");
    reset();mock_errno=EACCES;check(!census(&mounts_before)&&error_count==1&&errors[0].number==EACCES,
        "listmount actual primary errno retained");
    struct sv_statmount_buffer b=mount_buffer(1,1,"ext4");struct sv_mount_row decoded;
    reset();b.h.fs_type=64;check(!decode_mount(&b,1,&decoded),"type offset outside returned string region refused");
    b=mount_buffer(1,1,"ext4");b.h.mask&=~UINT64_C(0x40);check(!decode_mount(&b,1,&decoded),"missing namespace result bit refused");
    b=mount_buffer(1,1,"ext4");b.h.attr=SV_MOUNT_IDMAP;check(!decode_mount(&b,1,&decoded),"idmapped class refused");
    b=mount_buffer(1,1,"fuse");check(!decode_mount(&b,1,&decoded),"unanalysed delegated class refused");
    b=mount_buffer(1,1,"ext4");memset(b.str,'x',sizeof b.str);b.h.size=sizeof b;
    check(!decode_mount(&b,1,&decoded),"unterminated type refused within fixed returned extent");

    reset();mounts_before.count=1;mounts_before.row[0]=row(1,1,0);
    mounts_after.count=3;mounts_after.row[0]=row(1,1,3);
    mounts_after.row[1]=row(2,1,2);mounts_after.row[2]=row(3,1,2);
    check(exact_view_relation(1,1,1,2,3),"exact two attached clones and base readonly relation accepted");
    mounts_after.row[0].h.attr=2;check(!exact_view_relation(1,1,1,2,3),"writable base recurrence refused");
    mounts_after.row[0].h.attr=3;mounts_after.row[1].h.parent=3;
    check(!exact_view_relation(1,1,1,2,3),"clone attached under wrong mount refused");
    mounts_after.row[1].h.parent=1;mounts_after.row[2].h.dev_minor=2;
    check(!exact_view_relation(1,1,1,2,3),"different clone backing superblock refused");
    reset();native_queries_started=SV_MAX_QUERIES;
    check(!read_lsm_stack()&&mock_calls==0&&errors[0].number==E2BIG,"query capacity reserved before native call");
    if(failures)return 1;
    puts("unselected native fact control: finite operation oracles passed");return 0;
}
