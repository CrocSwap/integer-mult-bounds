class Model:
    """Independent identity rule: consuming wait releases q for unrelated U."""
    def __init__(self, code=0, failures=None, observations=None, rss=0, clock=0, size=0):
        self.code, self.failures = code, failures or {}
        self.observations = list(observations) if observations is not None else ['terminal']
        self.rss, self.clock, self.size = rss, clock, size
        self.events, self.signals = [], []
        self.anchor, self.group = False, 'ABSENT'
        self.pid = 741

    def fault(self, operation):
        if operation in self.failures:
            raise self.failures[operation]

    def spawn(self, *args, **kwargs):
        self.events.append('spawn')
        self.anchor, self.group = True, 'OWNED'
        self.fault('spawn')
        return self

    def consume(self):
        self.anchor, self.group = False, 'UNRELATED'
        return self.code

    def poll(self):
        self.events.append('consuming-poll')
        return self.consume()

    def wait(self):
        self.events.append('wait')
        self.fault('wait')
        return self.consume()

    def peek(self, kind, pid, flags):
        if kind != os.P_PID or pid != self.pid or flags != os.WEXITED | os.WNOHANG | os.WNOWAIT:
            raise AssertionError('non-consuming owned wait request')
        self.events.append('peek')
        self.fault('peek')
        item = self.observations.pop(0)
        if item is None or type(item) is not str:
            return item
        row = os.waitid_result((self.pid, 0, signal.SIGCHLD, self.code, os.CLD_EXITED))
        if row.si_pid != self.pid or row.si_code != os.CLD_EXITED:
            raise AssertionError('actual stdlib observation field layout')
        return row

    def kill(self, pid, sig):
        if pid != self.pid or sig != signal.SIGKILL:
            raise AssertionError('selected numeric signal')
        self.events.append('signal')
        self.signals.append(self.group)
        self.fault('signal')

    def group_rss(self, pid):
        if pid != self.pid:
            raise AssertionError('owned group sample')
        self.events.append('rss')
        self.fault('rss')
        return self.rss

    def sleep(self, delay):
        if delay != 0.05:
            raise AssertionError('original sampling delay')
        self.events.append('sleep')

    def ports(self):
        return dict(os=SimpleNamespace(waitid=self.peek, killpg=self.kill,
                    P_PID=os.P_PID, WEXITED=os.WEXITED, WNOHANG=os.WNOHANG,
                    WNOWAIT=os.WNOWAIT, waitid_result=os.waitid_result,
                    CLD_EXITED=os.CLD_EXITED, CLD_KILLED=os.CLD_KILLED, CLD_DUMPED=os.CLD_DUMPED),
                    subprocess=SimpleNamespace(Popen=self.spawn), signal=signal,
                    group_rss=self.group_rss, time=SimpleNamespace(monotonic=lambda:self.clock, sleep=self.sleep),
                    RSS_CAP=1024**3, OUTPUT_CAP=2*1024**2, TIMEOUT=180)

    def folder(self):
        model = self
        class Stream:
            def __init__(self, name): self.name = name
            def close(self):
                model.events.append('close-'+self.name)
                model.fault('close-'+self.name)
        class Leaf:
            def __init__(self, name): self.name = name
            def open(self, mode):
                if mode != 'wb': raise AssertionError('original capture mode')
                model.events.append('open-'+self.name)
                model.fault('open-'+self.name)
                return Stream(self.name)
            def stat(self): return SimpleNamespace(st_size=model.size)
        class Folder:
            def __truediv__(self, name): return Leaf(name)
        return Folder()


def actual_owner(model):
    tree = ast.parse(RUNNER.read_bytes())
    body = [n for n in tree.body if
            isinstance(n, ast.FunctionDef) and n.name == 'run_group_capture' or
            isinstance(n, ast.Assign) and any(isinstance(t, ast.Name) and t.id == '_GROUP_CUSTODY' for t in n.targets)]
    if len(body) != 2: raise AssertionError('one actual owner and preallocated custody cell')
    ns = model.ports()
    exec(compile(ast.Module(body=body, type_ignores=[]), str(RUNNER), 'exec'), ns)
    return ns


def invoke(model, ns=None):
    ns = actual_owner(model) if ns is None else ns
    result, error = None, None
    try:
        result = ns['run_group_capture'](['/inert/time'], '/inert/stage', {}, model.folder(), 0, 1000, 0)
    except BaseException as caught:
        error = caught
    return ns, result, error

        m = Model()
        tree = ast.parse((OLD/'implementation/source/run_guarded_v3.py').read_bytes())
        main = next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == 'main')
        nodes = [n for n in ast.walk(main) if isinstance(n, ast.With) and len(n.items) == 2 and
                 any(isinstance(x.context_expr, ast.Call) and isinstance(x.context_expr.func, ast.Attribute) and
                     x.context_expr.func.attr == 'open' for x in n.items)]
        self.assertEqual(len(nodes), 1)
        # The predecessor with statement is executed verbatim; only ports are inert.
        class ContextStream:
            def __enter__(self): return self
            def __exit__(self, *args): return False
        folder = SimpleNamespace()
        class OldFolder:
            def __truediv__(self, name): return SimpleNamespace(open=lambda mode:ContextStream())
        ns = dict(m.ports(), folder=OldFolder(), timed=['/inert/time'], stage='/inert/stage',
                  env={}, peak=0, stop=None, runner_start=0, cap=1000, spent=0)
        exec(compile(ast.Module(body=nodes, type_ignores=[]), '<actual predecessor block / inert identity model>', 'exec'), ns)
        self.assertEqual(m.signals, ['UNRELATED'])
        new = Model(); state, result, error = invoke(new)
        self.assertIsNone(error); self.assertEqual(result, (0, 0, None))
        self.assertEqual(new.signals, ['OWNED'])
        self.assertLess(new.events.index('signal'), new.events.index('wait'))
        self.assertEqual(new.events.count('wait'), 1)

    def test_normal_and_original_refusal_return_codes(self):
        for code in (0, 1, -9):

                self.assertIsNone(error); self.assertEqual(result, (code, 0, None))
                self.assertEqual(m.events[-2:], ['close-stderr.txt','close-stdout.txt'])

    def test_first_and_all_cleanup_errors_retain_actual_objects_and_tracebacks(self):
        failures = {name: RuntimeError(name) for name in
                    ('rss','signal','wait','close-stderr.txt','close-stdout.txt')}
        m = Model(failures=failures, observations=[None]); ns, result, error = invoke(m)
        self.assertIs(error, failures['rss']); self.assertIsNone(result)
        state = ns['_GROUP_CUSTODY'][0]
        self.assertEqual([r[0] for r in state['errors']], list(failures.values()))
        for caught, tb in state['errors']:
            self.assertIsNotNone(tb); self.assertIs(tb.tb_frame.f_code, ns['run_group_capture'].__code__)
            self.assertIsNotNone(tb.tb_next)
        for operation in ('signal','wait','close-stderr.txt','close-stdout.txt'):
            self.assertEqual(m.events.count(operation),1)
        self.assertTrue(state['unknown'])

    def test_clean_body_promotes_first_cleanup_error_without_skipping_later_cleanup(self):
        failures = {name: RuntimeError(name) for name in ('signal','wait','close-stderr.txt','close-stdout.txt')}
        m=Model(failures=failures); ns,result,error=invoke(m)
        self.assertIs(error,failures['signal']);self.assertIsNone(result)
        self.assertEqual([r[0] for r in ns['_GROUP_CUSTODY'][0]['errors'] if r[0] is not None],list(failures.values()))
        self.assertEqual(m.events[-4:],['signal','wait','close-stderr.txt','close-stdout.txt'])
