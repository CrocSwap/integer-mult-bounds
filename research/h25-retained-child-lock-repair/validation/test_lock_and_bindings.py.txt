def need(value,message):
    if value is not True:raise ValueError(message)


def actual_call(path,message):
    tree=ast.parse(path.read_bytes())
    calls=[n for n in ast.walk(tree) if isinstance(n,ast.Call) and
           any(isinstance(a,ast.Constant) and a.value==message for a in n.args)]
    if len(calls)!=1:raise AssertionError('one actual binding guard: '+message)
    return copy.deepcopy(calls[0])


def lock_projection(model,close_error=None,open_error=None,status='PASS',old=False):
    """Actual main owner/handlers/admin exit; working scientific body is one inert port."""
    path=(FIRST if old else NEW)/'implementation/source/run_guarded_v3.py'
    tree=ast.parse(path.read_bytes());main=next(n for n in tree.body if isinstance(n,ast.FunctionDef) and n.name=='main')
    ns=prior.actual_owner(model);events=[]
    class Lock:
        def __enter__(self):return self
        def __exit__(self,*args):return self.close()
        def close(self):
            events.append('lock-close')
            if close_error is not None:raise close_error
    class LockPath:
        def __truediv__(self,name):return self
        def open(self,mode):
            if mode!='a':raise AssertionError('original lock open mode')
            events.append('lock-open')
            if open_error is not None:raise open_error
            return Lock()
    def working_body():
        events.append('body')
        ns['run_group_capture'](['/inert/time'],'/inert/stage',{},model.folder(),0,1000,0)
        return {'status':status}
    ns.update(HERE=LockPath(),working_body=working_body)
    assignment=ast.parse('receipt = working_body()').body[0]
    if old:
        owner=copy.deepcopy(next(n for n in main.body if isinstance(n,ast.With)))
        owner.body=[assignment,copy.deepcopy(owner.body[-1])]
        body=[owner]
    else:
        start=next(i for i,n in enumerate(main.body) if isinstance(n,ast.If) and
                   any(isinstance(x,ast.Name) and x.id=='_LOCK_CUSTODY' for x in ast.walk(n)))
        body=copy.deepcopy(main.body[start:]);owner=next(n for n in body if isinstance(n,ast.Try))
        owner.body=owner.body[:4]+[assignment,owner.body[-1]]
        cell=next(n for n in tree.body if isinstance(n,ast.Assign) and
                  any(isinstance(x,ast.Name) and x.id=='_LOCK_CUSTODY' for x in n.targets))
        exec(compile(ast.Module(body=[copy.deepcopy(cell)],type_ignores=[]),str(path),'exec'),ns)
    fn=ast.FunctionDef(name='actual_main_lock_projection',args=ast.arguments(posonlyargs=[],args=[],vararg=None,kwonlyargs=[],kw_defaults=[],kwarg=None,defaults=[]),body=body,decorator_list=[],returns=None,type_comment=None)
    exec(compile(ast.fix_missing_locations(ast.Module(body=[fn],type_ignores=[])),str(path),'exec'),ns)
    error=None
    try:ns['actual_main_lock_projection']()
    except BaseException as caught:error=caught
    return ns,events,error


class LockAndBindings(unittest.TestCase):
    def test_actual_old_lock_masks_primary_and_corrected_main_retains_all_six_errors(self):
        def faults():return {k:RuntimeError(k) for k in ('rss','signal','wait','close-stderr.txt','close-stdout.txt')}
        f=faults();lock=OSError('lock-close');ns,events,error=lock_projection(prior.Model(failures=f,observations=[None]),lock,old=True)
        self.assertIs(error,lock);self.assertEqual(events,['lock-open','body','lock-close'])
        f=faults();lock=OSError('lock-close');ns,events,error=lock_projection(prior.Model(failures=f,observations=[None]),lock)
        self.assertIs(error,f['rss']);self.assertEqual(events,['lock-open','body','lock-close'])
        self.assertEqual([row[0] for row in ns['_GROUP_CUSTODY'][0]['errors']],list(f.values()))
        state=ns['_LOCK_CUSTODY'][0]
        self.assertIs(state['errors'][0][0],f['rss']);self.assertIs(state['errors'][1][0],lock)
        self.assertTrue(state['unknown']);self.assertEqual(state['close'],'UNKNOWN')
        for error,tb in state['errors']:
            self.assertIsNotNone(tb);self.assertIsNotNone(tb.tb_next)

    def test_clean_main_and_expected_exit_follow_known_lock_retirement(self):
        ns,events,error=lock_projection(prior.Model())
        self.assertIsNone(error);self.assertEqual(ns['_LOCK_CUSTODY'][0]['close'],'CLOSED')
        ns,events,error=lock_projection(prior.Model(code=1),status='STOP')
        self.assertIs(type(error),SystemExit);self.assertEqual(error.code,1)
        self.assertEqual(events[-1],'lock-close');self.assertEqual(ns['_LOCK_CUSTODY'][0]['close'],'CLOSED')

    def test_lock_error_is_not_hidden_by_expected_exit_and_actual_queue_rejects_stderr(self):
        lock=OSError('lock-close');ns,events,error=lock_projection(prior.Model(code=1),lock,status='STOP')
        self.assertIs(error,lock);self.assertNotIsInstance(error,SystemExit)
        q=NEW/'implementation/source/composed_queue.py'
        tree=ast.parse(q.read_bytes())
        call=next(n for n in ast.walk(tree) if isinstance(n,ast.Call) and
                  any(isinstance(a,ast.Constant) and type(a.value)is str and a.value.startswith('actual common runner terminal code differs: ') for a in ast.walk(n)))
        unit=compile(ast.Expression(body=copy.deepcopy(call)),str(q),'eval')
        env={'need':need,'case':{'job':'optimized','label':'inert'},'result':SimpleNamespace(returncode=1,stderr=b'')}
        self.assertIsNone(eval(unit,env))
        env['result']=SimpleNamespace(returncode=1,stderr=b'Traceback: actual lock-close error\n')
        with self.assertRaises(ValueError):eval(unit,env)

    def test_unknown_lock_acquisition_and_working_cancellation_preserve_errors_once(self):
        primary=OSError('open');ns,events,error=lock_projection(prior.Model(),open_error=primary)
        self.assertIs(error,primary);self.assertEqual(events,['lock-open'])
        self.assertEqual(ns['_LOCK_CUSTODY'][0]['acquire'],'UNKNOWN')
        primary=KeyboardInterrupt('cancel');lock=OSError('close')
        ns,events,error=lock_projection(prior.Model(failures={'rss':primary},observations=[None]),lock)
        self.assertIs(error,primary);self.assertEqual(events.count('lock-close'),1)
        self.assertIs(ns['_LOCK_CUSTODY'][0]['errors'][1][0],lock)

    def test_later_wait_observation_failure_clears_previous_ownership(self):
        m=prior.Model(observations=[None]);original=m.peek;calls=[];error=ChildProcessError('late lost child')
        def peek(*args):
            calls.append(1)
            if len(calls)==2:raise error
            return original(*args)
        m.peek=peek;ns,result,caught=prior.invoke(m)
        self.assertIs(caught,error);self.assertIsNone(result)
        self.assertEqual(len(calls),2);self.assertNotIn('signal',m.events);self.assertNotIn('wait',m.events)

    def test_stale_runner_and_each_stale_stage_are_rejected_by_actual_binding_predicates(self):
        m=json.loads((NEW/'implementation/bindings/SOURCE-MANIFEST-v2.json').read_bytes())

        unit=compile(ast.Expression(body=copy.deepcopy(call)),str(sup),'eval');env={'need':need,'Path':Path,'RUNNER_PATH':Path(p['path']),'RUNNER_SHA':p['sha256'],'common_runner_pin':dict(p)}
        self.assertIsNone(eval(unit,env))

            env['common_runner_pin']={**p,**change}
