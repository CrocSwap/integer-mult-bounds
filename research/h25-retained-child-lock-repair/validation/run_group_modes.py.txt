"""Only the distinct changed retirement/binding suite, within one outer guard."""
import json
from pathlib import Path
import subprocess
import sys

D = Path(__file__).resolve().parent
rows = []
for mode in ('normal', 'optimized'):
    argv = [sys.executable, *(['-O'] if mode == 'optimized' else []),
            '-S', '-s', '-P', '-B', str(D/'test_group_retirement.py')]
    result = subprocess.run(argv, check=False, timeout=8)
    rows.append(dict(mode=mode, argv=argv, returncode=result.returncode))
print(json.dumps(rows, indent=2))
sys.exit(0 if all(row['returncode'] == 0 for row in rows) else 1)
