class AdapterStop(RuntimeError):

def remember(error):
    if FIRST[0] is None:
        FIRST[0] = error
    elif error is not FIRST[0]:
        for index in range(32):
            if SECONDARY[index] is error:
                break
            if SECONDARY[index] is None:
                SECONDARY[index] = error
                break
        else:
            IO_UNKNOWN[0] = True
    return FIRST[0]


def need(value, message):
    if not value:
        raise AdapterStop(message)


def identity(info):
    return (info.st_dev, info.st_ino, info.st_mode, info.st_nlink,
            info.st_uid, info.st_size, info.st_mtime_ns, info.st_ctime_ns)


def canonical(value):
    return (json.dumps(value, sort_keys=True, separators=(',', ':'),
                       ensure_ascii=True, allow_nan=False) + '\n').encode('ascii')


def sha_value(value):

def error_data():
    def item(error):
        if error is None:
            return None
        frames, tb = [], error.__traceback__
        while tb is not None and len(frames) < 32:
            frames.append({'file': tb.tb_frame.f_code.co_filename, 'line': tb.tb_lineno,
                           'function': tb.tb_frame.f_code.co_name})
            tb = tb.tb_next
        return {'type': type(error).__name__, 'frames': frames}
    return {'first': item(FIRST[0]), 'secondary': [item(e) for e in SECONDARY if e is not None],
            'IO_unknown': IO_UNKNOWN[0], 'actual_exception_objects_retained_until_exit': True}


def acquire(path, flags=0, directory=False):
    need(not IO_UNKNOWN[0], 'unknown IO blocks new acquisition')
    slot = next(s for s in SLOTS if s[1] in ('FREE','CLOSED'))
    slot[:] = [None, 'PENDING', str(path), directory]
    try:
        slot[0] = os.scandir(path) if directory else os.open(path, flags | os.O_NOFOLLOW | os.O_CLOEXEC, 0o600)
        slot[1] = 'OWNED'
        return slot
    except BaseException as error:
        IO_UNKNOWN[0] = True
        remember(error)
        raise


def close(slot):
    need(slot[1] == 'OWNED', 'known owned close only; no retry')
    slot[1] = 'CLOSING'
    try:
        slot[0].close() if slot[3] else os.close(slot[0])
    except BaseException as error:
        slot[1] = 'CLOSE_UNKNOWN'
        IO_UNKNOWN[0] = True
        remember(error)
        raise
    slot[1] = 'CLOSED'

    need(not IO_UNKNOWN[0], 'directory close unknown')

def read_fd(slot, cap, proc=False):
    before = os.fstat(slot[0])
    if not proc:
        need(stat.S_ISREG(before.st_mode) and before.st_uid in (0,os.getuid()) and
             before.st_nlink == 1 and before.st_size <= cap, 'stable regular owner/link/size')
        need(identity(Path(slot[2]).lstat()) == identity(before), 'descriptor/path differs')
        os.lseek(slot[0], 0, os.SEEK_SET)
    chunks, count = [], 0
    while True:
        block = os.read(slot[0], min(65536,cap + 1 - count))
        if not block:
            break
        chunks.append(block)
        count += len(block)
        need(count <= cap, 'read cap')
    raw = b''.join(chunks)
    if not proc:
        need(count == before.st_size and identity(os.fstat(slot[0])) == identity(before) ==
             identity(Path(slot[2]).lstat()), 'stable descriptor/path changed')
    return raw


def stable_source(path, expected):
    need(path.is_absolute() and path.resolve(strict=True) == path, 'canonical source path')
    before = path.lstat()
    need(stat.S_IMODE(before.st_mode) == 0o444, 'frozen source mode')
    slot = acquire(path, os.O_RDONLY | os.O_NONBLOCK)
    try:
        need(identity(before) == identity(os.fstat(slot[0])), 'source opened identity')
        raw = read_fd(slot, expected['bytes'])
        need(len(raw) == expected['bytes'] and hashlib.sha256(raw).hexdigest() == expected['sha256'], 'source pin')
    except BaseException as error:
        remember(error)
        raise
    finally:
        try:
            close(slot)
        except BaseException as secondary:
            remember(secondary)
    need(not IO_UNKNOWN[0], 'source close unknown')
    return raw


def metadata(path, cap=8192):

        raise
    finally:
        try:
            close(slot)
        except BaseException as secondary:
            remember(secondary)
    need(not IO_UNKNOWN[0], 'metadata close unknown')
    return raw


def process(pid):
    raw = metadata('/proc/' + str(pid) + '/stat')
    end = raw.rfind(b') ')
    need(end >= 0 and raw.split(b' ',1)[0] == str(pid).encode('ascii'), 'process stat syntax/pid')
    fields = raw[end + 2:].split()
    return {'pid':pid,'ppid':int(fields[1]),'pgid':int(fields[2]),'sid':int(fields[3]),
            'start_ticks':int(fields[19]),'rss_bytes':int(fields[21])*os.sysconf('SC_PAGE_SIZE')}

def lower_limit(kind, ceiling):
    if kind == resource.RLIMIT_FSIZE:
        old_soft, old_hard = resource.getrlimit(kind)
        need(old_hard == resource.RLIM_INFINITY or old_hard >= 536870912, "selected FSIZE hard H unavailable")
        resource.setrlimit(kind, (2097152, 536870912))
        return
    soft, hard = resource.getrlimit(kind)
    bound = min(ceiling,hard) if hard != resource.RLIM_INFINITY else ceiling
    if soft != resource.RLIM_INFINITY:
        bound = min(bound,soft)
    resource.setrlimit(kind,(bound,bound))


def strict_json(raw):
    def pairs(rows):
        value = {}
        for key,item in rows:
            need(key not in value, 'duplicate JSON key')
            value[key] = item
        return value
    def constant(value):
        raise AdapterStop('nonfinite JSON')
    return json.loads(raw.decode('ascii'),object_pairs_hook=pairs,parse_constant=constant)

        at += count
    os.fsync(slot[0])

def _outer_actor_data(value):
    """Bounded exact builtin DATA before key hashing, equality or JSON copy.

    Exact dict/list iteration reads native entries only. Keys are first checked
    by type before length or any use as hashed keys. Parent cycles use is-only
    identity scans, without id/audit calls, object equality or foreign methods.
    Ordinary selected-pristine builtin/OS and single-role stability premises
    remain external; this is not hostile concurrent mutation protection.
    """
    nodes, text, parents = [0], [0], []

    def node_charge():
        nodes[0] += 1
        need(nodes[0] <= 256, 'outer actor DATA node bound')

    def text_charge(count):
        text[0] += count
        need(text[0] <= 4096, 'outer actor DATA total text bound')

    def visit(node, depth):
        need(depth <= 16, 'outer actor DATA depth bound')
        node_charge()
        kind = type(node)
        if kind is dict or kind is list:
            # No `in`, equality, hashing, repr, or id on caller containers.
            for parent in parents:
                need(node is not parent, 'outer actor DATA cycle')
            if kind is dict:
                need(len(node) <= 32, 'outer actor DATA dict bound')
                # Complete key-type pass before recursive values or schema sets.
                # Iteration of an exact dict never rehashes its stored keys.
                for key in node:
                    need(type(key) is str, 'outer actor DATA exact string key')
                    need(0 < len(key) <= 128, 'outer actor DATA key length')
                    node_charge()
                    text_charge(len(key))
            else:
                need(len(node) <= 64, 'outer actor DATA list bound')
            parents.append(node)
            try:
                if kind is dict:
                    for child in dict.values(node):
                        visit(child, depth + 1)
                else:
                    for child in node:
                        visit(child, depth + 1)
            finally:
                parents.pop()
        elif kind is str:
            need(len(node) <= 4096, 'outer actor DATA string bound')
            text_charge(len(node))
        elif kind is int:
            need(-(1 << 63) <= node < (1 << 63), 'outer actor DATA integer bound')
        elif kind is bool or node is None:
            pass
        else:
            # Floats/nonfinite, subclasses, functions, bytes, tuples and all
            # unknown objects are rejected without asking them to convert.
            need(False, 'outer actor DATA exact scalar/container type')

    visit(value, 0)

def _ready_pipe():
    # Both acquisition slots predate the syscall. No retry on ambiguous return.
    a = next(s for s in SLOTS if s[1] in ('FREE','CLOSED'))
    a[:] = [None,'PENDING','control-ready-read',False]
    b = next(s for s in SLOTS if s[1] in ('FREE','CLOSED'))
    b[:] = [None,'PENDING','control-ready-write',False]
    try:
        pair = os.pipe2(os.O_CLOEXEC | os.O_NONBLOCK)
        a[0], b[0] = pair
        a[1] = b[1] = 'OWNED'
        need(199 not in pair, 'reserved readiness descriptor collision')
        # This fixed single-thread carrier owns FD199 before Popen and never
        # issues another channel receive until child has replaced its channel.
        target = next(s for s in SLOTS if s[1] in ('FREE','CLOSED'))
        target[:] = [None,'PENDING','fixed-ready-199',False]
        target[0] = os.dup2(b[0],199,inheritable=True)
        target[1] = 'OWNED'
        close(b)
        return a,target
    except BaseException as error:
        IO_UNKNOWN[0] = True
        remember(error)
        raise


def _await_ready(slot, deadline):
    try:
        while True:
            left = (deadline-time.monotonic_ns())/1e9
            need(left > 0, 'fixed child registration deadline')
            ready,unused,failed = select.select([slot[0]],[],[slot[0]],min(left,0.025))
            need(not failed, 'readiness pipe failed')
            if ready:
                need(os.read(slot[0],2) == b'R', 'exact child private registration acknowledgement')
                break
    except BaseException as error:
        remember(error)
        raise
    finally:
        close(slot)
