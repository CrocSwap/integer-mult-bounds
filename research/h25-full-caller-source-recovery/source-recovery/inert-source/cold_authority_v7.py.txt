                      cell_get, error0, value_error0, tuple0, dict0, str0,

            # This pretarget block uses captured C primitives only. In particular
            # it does not call need(), fresh_function(), or any other validator.

                raise error0(label + ': operation index differs')

                    if id0(getter(function, function_type)) != id0(expected):
                        raise error0(label + ': target function environment differs')
                keywords = record[5]

                    if type0(keywords) is not dict0 or dict_len(keywords) != len0(record[7]):
                        raise error0(label + ': target keyword census differs')
                    iterator = current = None
                    valid_domain = True
                    try:
                        iterator = dict_iter(keywords)
                        while True:
                            current = next0(iterator, missing)
                            if current is missing:
                                break

                    finally:
