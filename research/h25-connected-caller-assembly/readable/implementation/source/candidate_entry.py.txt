def need(value,message):
    if not value:
        raise RuntimeError(message)

def digest(raw):
    return hashlib.sha256(raw).hexdigest()

def remember(error):
    if STATE['first'] is None:
        STATE['first'],STATE['traceback']=error,BaseException.__dict__['__traceback__'].__get__(error,BaseException)
    if not any(item is error for item in STATE['errors']):
        if len(STATE['errors']) < 16: STATE['errors'].append(error)
        else: STATE['overflow']=error

class LocalOwner:
    """Runner-local raw owners; their terminal projection remains DATA to parent."""
    def __init__(self, state, remember):
        self.state, self.remember_error = state, remember
    def pending(self, role, path=None):
        need(len(self.state['fds']) < 64 and self.state['overflow'] is None and
             not any(r['state'] == 'UNKNOWN' for r in self.state['fds']),
             'runner descriptor custody unavailable')
        row = {'role':role, 'path':path, 'object':None, 'state':'PENDING',
               'primary':None, 'secondary':None, 'close_attempted':False}
        self.state['fds'].append(row)
        return row
    def bind(self,relay):
        self.state['relay']=relay
    def retired(self,known):
        pass  # Actual descriptor rows remain in the runner's strong state.
    def acquired(self, row, fd):
        previous=row['state']
        row['object'],row['state']=fd,'OWNED'  # Actual native return before validator.
        need(previous=='PENDING' and type(fd) is int and fd>=0,'one actual runner descriptor return')
    def failed(self, row, error):
        row['primary'], row['state'] = error, 'UNKNOWN'
        self.remember(error)
    def close(self, row):
        need(row['state'] == 'OWNED' and type(row['object']) is int,
             'one known runner descriptor close')
        fd, row['object'], row['state'], row['close_attempted'] = row['object'], None, 'CLOSING', True
        try:
            os.close(fd)
        except BaseException as error:
            row['secondary'], row['state'] = error, 'UNKNOWN'
            self.remember(error)
            raise
        row['state'] = 'CLOSED'
    def remember(self, error):
        self.remember_error(error)

def read_waiting_configuration(path, expected_sha):
    """Read parent-created DATA, never restore its capability or release a child."""
    path = Path(path)
    if (not path.is_absolute() or path.resolve(strict=True) != path or
            type(expected_sha) is not str or len(expected_sha) != 64 or
            any(c not in '0123456789abcdef' for c in expected_sha)):
        raise ValueError('Exact canonical waiting configuration and digest required')
    identity = lambda q: (q.st_dev,q.st_ino,q.st_mode,q.st_uid,q.st_nlink,q.st_size,q.st_mtime_ns,q.st_ctime_ns)
    before = path.lstat()
    if (not stat.S_ISREG(before.st_mode) or stat.S_IMODE(before.st_mode) != 0o444 or
            before.st_uid != os.getuid() or before.st_nlink != 1 or not 0 < before.st_size <= 16384):
        raise ValueError('Owned immutable bounded waiting configuration required')
    fd = None
    primary = secondary = None
    raw = None
    try:
        fd = os.open(path,os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK|os.O_CLOEXEC)
        if identity(os.fstat(fd)) != identity(before):
            raise ValueError('Waiting configuration opened identity')
        blocks, count = [], 0
        while True:
            block = os.read(fd,min(65536,before.st_size+1-count))
            if not block:
                break
            blocks.append(block); count += len(block)
            if count > before.st_size:
                raise ValueError('Waiting configuration extra byte')
        raw = b''.join(blocks)
        if (len(raw) != before.st_size or digest(raw) != expected_sha or
                identity(os.fstat(fd)) != identity(before) or identity(path.lstat()) != identity(before)):
            raise ValueError('Waiting configuration exact stable readback')
    except BaseException as error:
        primary = error
    finally:
        if fd is not None:
            owned, fd = fd, None
            try:
                os.close(owned)
            except BaseException as error:
                secondary = error
    if primary is not None:
        if secondary is not None:
            raise primary from secondary
        raise primary
    if secondary is not None:
        raise secondary
    def pairs(items):
        result = {}
        for key,value in items:
            if key in result:
                raise ValueError('Duplicate waiting configuration key')
            result[key] = value
        return result
    def no_number(value):
        raise ValueError('Waiting configuration forbids floating/nonfinite numbers')
    value = json.loads(raw.decode('ascii'),object_pairs_hook=pairs,parse_float=no_number,parse_constant=no_number)
    if type(value) is not dict or value.get('schema') != 'h25-waiting-entry-config-v1':
        raise ValueError('Exact waiting configuration schema')
    return value


OWNER = LocalOwner(STATE, remember)

def opened(path,flags,role,mode=0o600):
    row=OWNER.pending(role,path)
    try:
        fd=os.open(path,flags|os.O_NOFOLLOW|os.O_CLOEXEC,mode)
    except BaseException as error:
        OWNER.failed(row,error)
        raise
    OWNER.acquired(row,fd)
    return row

def identity(st):
    return (st.st_dev,st.st_ino,st.st_mode,st.st_uid,st.st_nlink,st.st_size,st.st_mtime_ns,st.st_ctime_ns)

def close_preserving(row,first):
    if row is not None and row['state']=='OWNED':
        try: OWNER.close(row)
        except BaseException as error:
            remember(error)
            if first is None: first=error
    return first

def read_stable(path,cap,expected=None,mode=None):
    path=Path(path);before=path.lstat()
    need(path.is_absolute() and path.resolve(strict=True)==path and stat.S_ISREG(before.st_mode)
         and before.st_uid in (0,os.getuid()) and before.st_nlink==1 and 0<=before.st_size<=cap
         and (mode is None or stat.S_IMODE(before.st_mode)==mode),'bounded canonical selected file')
    row=None;first=None;raw=None
    try:
        row=opened(path,os.O_RDONLY|os.O_NONBLOCK,'entry-read')
        need(identity(os.fstat(row['object']))==identity(before),'opened selected inode')
        parts=[];count=0
        while True:
            block=os.read(row['object'],min(65536,cap+1-count))
            if not block: break
            parts.append(block);count+=len(block)
            need(count<=cap,'complete bounded file read')
        raw=b''.join(parts)
        need(identity(os.fstat(row['object']))==identity(path.lstat())==identity(before),
             'stable complete descriptor and path read')
        if expected is not None:
            need(len(raw)==expected['bytes'] and digest(raw)==expected['sha256'],'exact selected bytes')
    except BaseException as error:
        first=error;remember(error)
    finally: first=close_preserving(row,first)
    if first is not None: raise first
    return raw

def json_read(path,cap=16384,expected=None,mode=0o444):
    raw=read_stable(path,cap,expected,mode)
    def pairs(items):
        result={}
        for key,value in items:
            need(key not in result,'duplicate record key');result[key]=value
        return result
    def number(unused): raise ValueError('record floats/nonfinite forbidden')
    value=json.loads(raw.decode('ascii'),object_pairs_hook=pairs,parse_float=number,parse_constant=number)
    need(type(value) is dict,'exact record object')
    return value,raw

def immutable_write(path,value):
    raw=(json.dumps(value,sort_keys=True,separators=(',',':'),allow_nan=False)+'\n').encode('ascii')
    need(len(raw)<=16384,'entry record bound')
    row=None;first=None
    try:
        row=opened(path,os.O_WRONLY|os.O_CREAT|os.O_EXCL,'entry-record')
        view=memoryview(raw)
        while view:
            count=os.write(row['object'],view);need(type(count) is int and count>0,'record write progress');view=view[count:]
        os.fchmod(row['object'],0o444);os.fsync(row['object'])
    except BaseException as error:
        first=error;remember(error)
    finally: first=close_preserving(row,first)
    if first is not None: raise first
    directory=None
    try:
        directory=opened(Path(path).parent,os.O_RDONLY|os.O_DIRECTORY,'entry-record-directory')
        os.fsync(directory['object'])
    except BaseException as error:
        first=error;remember(error)
    finally: first=close_preserving(directory,first)
    if first is not None: raise first
    return {'path':str(path),'bytes':len(raw),'sha256':digest(raw)}

def process_row(pid):
    row=opened(Path('/proc')/str(pid)/'stat',os.O_RDONLY,'entry-proc-read');first=None;raw=None
    try:
        raw=os.read(row['object'],8193);need(len(raw)<=8192 and b') ' in raw,'bounded actual generation')
    except BaseException as error: first=error;remember(error)
    finally: first=close_preserving(row,first)
    if first is not None: raise first
    fields=raw[raw.rfind(b') ')+2:].split()
    return {'pid':pid,'ppid':int(fields[1]),'start_ticks':int(fields[19]),'state':fields[0].decode('ascii')}

def load_source(pin):
    need(type(pin) is dict and set(pin)=={'path','bytes','sha256','mode'} and pin['mode']==0o444,
         'selected source module pin')

def inherited_holders(folder):
    """Snapshot only stdio and the possible GNU-time usage alias; reject others."""
    info=(folder/'usage.json').stat();usage_key=(info.st_dev,info.st_ino)
    rows=[];directory_key=(Path('/proc/self/fd').stat().st_dev,Path('/proc/self/fd').stat().st_ino)
    iterator=os.scandir('/proc/self/fd');first=None;scan_holders=0
    try:
        names=set()
        for item in iterator:
            if not item.name.isdecimal(): continue
            fd=int(item.name);names.add(fd)
            need(len(names)<=16,'finite exact inherited descriptor table')
            # Inspect while scandir still owns its FD. Exhausting it first
            # would close that very descriptor before its fstat.
            st=os.fstat(fd)
            if stat.S_ISDIR(st.st_mode) and (st.st_dev,st.st_ino)==directory_key:
                scan_holders+=1;continue
            if fd==0:
                null=os.stat('/dev/null');need(stat.S_ISCHR(st.st_mode) and st.st_rdev==null.st_rdev,'admitted inherited stdin')
            elif fd in (1,2):
                path=folder/('stdout.txt' if fd==1 else 'stderr.txt');named=path.lstat()
                need(stat.S_ISREG(st.st_mode) and identity(st)==identity(named) and st.st_size==0
                     and stat.S_IMODE(st.st_mode)==0o600,'actual empty inherited common capture')
            else:
                need(stat.S_ISREG(st.st_mode) and (st.st_dev,st.st_ino)==usage_key,
                     'only outer GNU-time usage alias may also survive exec')
            ticket=OWNER.pending('entry-inherited-'+str(fd));OWNER.acquired(ticket,fd)
            rows.append(ticket)
        need(scan_holders==1 and {0,1,2}.issubset({r['object'] for r in rows}),'one known fd-census iterator')
    except BaseException as error: first=error;remember(error)
    finally:
        try: iterator.close()
        except BaseException as error:
            remember(error);first=error if first is None else first
    if first is not None: raise first
    return rows

def _poll_child(process):
    state = _WRITER_STATE
    if state['wait_state']=='COMMITTED':
        return process.returncode
    if state['wait_state']=='UNRESOLVED' or state['wait_calls']>=65536:
        raise RuntimeError('runner native wait unresolved; no retry')
    state['wait_state']='UNRESOLVED';state['wait_calls']+=1
    try:
        actual=os.waitpid(process.pid,os.WNOHANG)
        state['raw_wait']=actual  # Actual tuple BEFORE comparison/decoder.
        if type(actual) is not tuple or len(actual)!=2 or any(type(v) is not int for v in actual):
            raise RuntimeError('runner native wait tuple')
        if actual==(0,0):
            state['wait_state']='SAFE_NONTERMINAL'
            return None
        if actual[0]!=process.pid:
            raise RuntimeError('runner waited actual child differs')
        state['wait_state']='CONSUMED'
        code=os.waitstatus_to_exitcode(actual[1])
        if type(code) is not int:
            raise RuntimeError('runner wait decoder')
        process.returncode=code
        state['wait_known']=True;state['wait_state']='COMMITTED'
        return code
    except BaseException as error:
        state['wait_error']=error
        state['wait_state']='UNRESOLVED'
        _remember_runner(error)
        raise

def _wait_child(process,relay=None,deadline=None):
    if deadline is None:
        deadline=time.monotonic()+2.0
    while True:
        code=_poll_child(process)
        if code is not None:
            return code
        if time.monotonic()>=deadline:
            raise RuntimeError('runner reserved native cleanup wait exhausted')
        if relay is not None and relay.primary is None:
            relay.pump(0.01)
        else:
            time.sleep(0.01)
