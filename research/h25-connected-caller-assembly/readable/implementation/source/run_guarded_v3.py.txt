def require_trusted_launch():
    if not (sys.flags.no_site == 1 and sys.flags.no_user_site == 1 and
            sys.flags.safe_path and sys.flags.dont_write_bytecode == 1 and
            sys.flags.ignore_environment == 0 and sys.flags.isolated == 0 and
            sys.flags.hash_randomization == 0):
        raise ValueError('Explicit no-site/safe-path parent launch required')
    if dict(os.environ) != LAUNCH_ENV:
        raise ValueError('Exact clean parent environment required')
    if sys.executable != '/usr/bin/python3.12':
        raise ValueError('Absolute admitted interpreter required')
    if tuple(sys.path) != ('/usr/lib/python312.zip', '/usr/lib/python3.12',
                           '/usr/lib/python3.12/lib-dynload'):
        raise ValueError('Unadmitted ordered initial Python search path')
    return {'contract': 'trusted-clean-nosite-safe-path-fixedseed-v1',
            'no_site': sys.flags.no_site, 'no_user_site': sys.flags.no_user_site,
            'safe_path': bool(sys.flags.safe_path),
            'ignore_environment': sys.flags.ignore_environment,
            'isolated': sys.flags.isolated, 'hash_seed_environment': '0',
            'hash_randomization': sys.flags.hash_randomization,
            'initial_sys_path': list(sys.path), 'environment': dict(os.environ),
            'trust_boundary': 'Authenticated parent launcher, OS, interpreter and system stdlib before this module; no Python self-authentication claim'}


def digest(raw):
    return hashlib.sha256(raw).hexdigest()


def inventory(stage):
    rows = {}
    for path in sorted(stage.rglob('*')):
        if path.is_symlink():
            raise ValueError('Symlink in immutable stage')
        if path.is_file() and path.name != 'STAGE.json':
            rows[str(path.relative_to(stage))] = {
                'sha256': digest(path.read_bytes()),
                'mode': path.stat().st_mode & 0o777,
            }
    return rows


def group_rss(group):
    total = 0
    for path in Path('/proc').iterdir():
        if not path.name.isdigit():
            continue
        try:
            raw = (path / 'stat').read_text()
            fields = raw[raw.rfind(')') + 2:].split()
            if int(fields[2]) == group:
                total += int(fields[21]) * os.sysconf('SC_PAGE_SIZE')
        except (FileNotFoundError, ProcessLookupError, PermissionError):
            pass
    return total


def log_identity(value):
    """Exclude atime: our own read may change it."""
    return (value.st_dev, value.st_ino, value.st_mode, value.st_size,
            value.st_mtime_ns, value.st_ctime_ns)


def stream_log(path):
    """Hash a finite snapshot in bounded memory, failing closed on file drift.

    A stable record covers every byte of the retained file. An unstable record
    describes only the bytes actually hashed and cannot support a PASS. Reading
    at most the initial length also prevents a growing file prolonging hashing.
    """
    path_before = path.stat(follow_symlinks=False)
    fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK)
    with os.fdopen(fd, 'rb', buffering=0) as stream:
        before = os.fstat(stream.fileno())
        if not stat.S_ISREG(before.st_mode):
            raise ValueError('Retained log must be a regular file')
        hasher = hashlib.sha256()
        count = 0
        while count < before.st_size:
            block = stream.read(min(LOG_CHUNK_BYTES, before.st_size - count))
            if not block:
                break
            hasher.update(block)
            count += len(block)
        after = os.fstat(stream.fileno())
    path_after = path.stat(follow_symlinks=False)
    stable = (log_identity(path_before) == log_identity(before) ==
              log_identity(after) == log_identity(path_after) and
              count == before.st_size)
    return {'sha256': hasher.hexdigest(), 'bytes': count,
            'snapshot_bytes': before.st_size, 'final_bytes': path_after.st_size,
            'stable': stable}


def finalize_logs(folder, stop):
    """Check sizes before any log read, stream complete stable evidence, recheck."""
    paths = [folder / 'stdout.txt', folder / 'stderr.txt']
    before = [path.stat(follow_symlinks=False) for path in paths]
    cap_seen = 'process_monitor' if stop == 'output_cap' else None
    if max(value.st_size for value in before) > OUTPUT_CAP:
        stop = stop or 'output_cap'
        cap_seen = cap_seen or 'before_stream_hash'
    records = [stream_log(path) for path in paths]
    final = [path.stat(follow_symlinks=False) for path in paths]
    if max(value.st_size for value in final) > OUTPUT_CAP:
        stop = stop or 'output_cap'
        cap_seen = cap_seen or 'after_stream_hash'
    for initial, record, ending in zip(before, records, final):
        record['final_bytes'] = ending.st_size
        record['stable'] = (record['stable'] and
                            log_identity(initial) == log_identity(ending) and
                            record['bytes'] == ending.st_size)
    if not all(record['stable'] for record in records):
        stop = stop or 'output_changed_during_hash'
    return records[0], records[1], stop, cap_seen


def read_usage(path):
    """The time tool's small diagnostic must not reintroduce an unbounded read."""
    if not path.exists():
        return None, False
    size = path.stat().st_size
    if size > USAGE_METADATA_CAP:
        return {'raw_omitted': True, 'bytes': size}, True
    with path.open('rb') as stream:
        raw = stream.read(USAGE_METADATA_CAP + 1)
    if len(raw) > USAGE_METADATA_CAP or path.stat().st_size > USAGE_METADATA_CAP:
        return {'raw_omitted': True, 'bytes': path.stat().st_size}, True
    try:
        return json.loads(raw), False
    except (json.JSONDecodeError, UnicodeDecodeError):
        return {'raw': raw.decode('utf-8', errors='replace')}, False


def read_waiting_configuration(path, expected_sha):
    """Read parent-created DATA, never restore its capability or release a child."""
    path = Path(path)
    if (not path.is_absolute() or path.resolve(strict=True) != path or
            type(expected_sha) is not str or len(expected_sha) != 64 or
            any(c not in '0123456789abcdef' for c in expected_sha)):
        raise ValueError('Exact canonical waiting configuration and digest required')
    identity = lambda q: (q.st_dev,q.st_ino,q.st_mode,q.st_uid,q.st_nlink,q.st_size,q.st_mtime_ns,q.st_ctime_ns)
    before = path.lstat()
    if (not stat.S_ISREG(before.st_mode) or stat.S_IMODE(before.st_mode) != 0o444 or
            before.st_uid != os.getuid() or before.st_nlink != 1 or not 0 < before.st_size <= 16384):
        raise ValueError('Owned immutable bounded waiting configuration required')
    fd = None
    primary = secondary = None
    raw = None
    try:
        fd = os.open(path,os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK|os.O_CLOEXEC)
        if identity(os.fstat(fd)) != identity(before):
            raise ValueError('Waiting configuration opened identity')
        blocks, count = [], 0
        while True:
            block = os.read(fd,min(65536,before.st_size+1-count))
            if not block:
                break
            blocks.append(block); count += len(block)
            if count > before.st_size:
                raise ValueError('Waiting configuration extra byte')
        raw = b''.join(blocks)
        if (len(raw) != before.st_size or digest(raw) != expected_sha or
                identity(os.fstat(fd)) != identity(before) or identity(path.lstat()) != identity(before)):
            raise ValueError('Waiting configuration exact stable readback')
    except BaseException as error:
        primary = error
    finally:
        if fd is not None:
            owned, fd = fd, None
            try:
                os.close(owned)
            except BaseException as error:
                secondary = error
    if primary is not None:
        if secondary is not None:
            raise primary from secondary
        raise primary
    if secondary is not None:
        raise secondary
    def pairs(items):
        result = {}
        for key,value in items:
            if key in result:
                raise ValueError('Duplicate waiting configuration key')
            result[key] = value
        return result
    def no_number(value):
        raise ValueError('Waiting configuration forbids floating/nonfinite numbers')
    value = json.loads(raw.decode('ascii'),object_pairs_hook=pairs,parse_float=no_number,parse_constant=no_number)
    if type(value) is not dict or value.get('schema') != 'h25-waiting-entry-config-v1':
        raise ValueError('Exact waiting configuration schema')
    return value
