def _need(condition, message):
    if not condition:
        raise _ERROR(message)


def _fingerprint(info):
    return (info.st_dev, info.st_ino, info.st_size, info.st_mtime_ns, info.st_ctime_ns)


def _hash_regular(path, limit, materialize):
    """Bounded hashing; /proc/self/exe is the one intentional kernel symlink."""
    _need(_TYPE(path) is _STR and path.startswith("/"), "Probe path must be absolute exact str")
    descriptor = None
    try:
        descriptor = _OPEN(path, _PROC_FLAGS if path == "/proc/self/exe" else _FLAGS)
        before = _FSTAT(descriptor)
        _need((before.st_mode & 0o170000) == 0o100000 and 0 <= before.st_size <= limit,
              "Probe authentication file type/size differs")
        digest, count, parts = _SHA256(), 0, []
        while True:
            block = _READ(descriptor, min(1024 * 1024, limit + 1 - count))
            if not block:
                break
            count += _LEN(block)
            _need(count <= limit, "Probe authentication file grew beyond bound")
            digest.update(block)
            if materialize:
                parts.append(block)
        after = _FSTAT(descriptor)
        _need(_fingerprint(before) == _fingerprint(after) and count == before.st_size,
              "Probe authentication file changed during read")
        return digest.hexdigest(), count, _fingerprint(after), b"".join(parts) if materialize else None
    except _OSERROR as error:
        raise _ERROR("Probe authentication file could not be read: " + path) from error
    finally:
        if descriptor is not None:
            _CLOSE(descriptor)


def _profile_files():
    namespace = _GLOBALS()
    _ROOT_VALIDATE(_BOOTSTRAP_CAPABILITY, namespace)
    source_path = namespace.get("__file__")
    _need(_TYPE(source_path) is _STR and source_path.startswith("/"),
          "Probe provider requires its authenticated absolute __file__")
    path = source_path.rsplit("/", 1)[0] + "/" + _PROFILE_BASENAME
    digest, count, fingerprint, raw = _hash_regular(path, 65536, False)
    _need(digest == _PROFILE_SHA256, "Probe runtime profile source differs")
    # These immutable triples are source-bound byte-for-byte to that profile.
    # No JSON decoder or mutable cached decoded profile participates in admission.
    return _RUNTIME_FILES + ((path, digest, count),)


def _audit_files(files):
    for path, expected, size in files:
        digest, count, _, _ = _hash_regular(path, max(size, 1), False)
        _need(digest == expected and count == size, "Probe runtime/header/binary custody differs: " + path)


def _code_names(code):
    names = set(code.co_names)
    for constant in code.co_consts:
        if _TYPE(constant) is CodeType:
            names.update(_code_names(constant))
    return names


def _function_record(function):
    namespace, builtins_owner = function.__globals__, function.__builtins__
    closure = function.__closure__
    cells = () if closure is None else _TUPLE((cell, cell.cell_contents) for cell in closure)
    defaults, keywords = function.__defaults__, function.__kwdefaults__
    keyword_items = () if keywords is None else _TUPLE(keywords.items())
    lookups = []
    for name in _code_names(function.__code__):
        lookups.append((name, namespace.get(name, _MISSING), builtins_owner.get(name, _MISSING)))
    return (function, function.__code__, namespace, builtins_owner, defaults, keywords,
            keyword_items, closure, cells, _TUPLE(lookups))


def _function_fresh(record, observations):
    function, code, namespace, builtins_owner, defaults, keywords, keyword_items, closure, cells, lookups = record
    field, length, get_id, cell_info = observations[:4]
    if not (_TYPE(function) is FunctionType and
            field(function, '__code__') == _ID(code) and
            field(function, '__globals__') == _ID(namespace) and
            field(function, '__builtins__') == _ID(builtins_owner) and
            field(function, '__defaults__') == _ID(defaults) and
            field(function, '__kwdefaults__') == _ID(keywords) and
            field(function, '__closure__') == _ID(closure)):
        return False
    if keywords is not None:
        if length(keywords) != _LEN(keyword_items):
            return False
        for name, value in keyword_items:
            if get_id(keywords, name, _MISSING) != _ID(value):
                return False
    for cell, expected in cells:
        present, observed_id = cell_info(cell)
        if not present or observed_id != _ID(expected):
            return False
    for name, global_value, builtin_value in lookups:
        if (get_id(namespace, name, _MISSING) != _ID(global_value) or
                get_id(builtins_owner, name, _MISSING) != _ID(builtin_value)):
            return False
    return True

            for name, value in class_items:
                if get_id(class_proxy, name, _MISSING) != id0(value):
                    raise error('Probe binding class member changed: ' + name)
            for record in records:
                if not fresh(record, observations):
                    raise error('Probe provider function calling environment changed')
            if (type0(native) is not module_type or
                    field(native, '__dict__') != id0(native_namespace) or
                    length(native_namespace) != len0(native_items)):
                raise error('Probe native module changed')
            for name, value in native_items:
                if get_id(native_namespace, name, _MISSING) != id0(value):
                    raise error('Probe native module member changed: ' + name)
            for function in entries:
                if type0(function) is not builtin_type or field(function, '__self__') != id0(native):
                    raise error('Probe native entry owner changed')
        finally:
            observations[6]()
        # Native ABI check and ensuing inspect occur after all observations have
        # been discarded. Both native obligations remain literally present.
        native_check()
    return guard


def _state(binding):
    _need(_TYPE(binding) is ProbeBinding, "Probe binding class differs")
    state = _OWNERS.get(_ID(binding))
    _need(state is not None and state[0] is binding, "Probe binding is closed or was not authenticated")
    return state


class ProbeBinding:
    __slots__ = ()

    def __new__(cls):
        raise _ERROR("ProbeBinding must be created by authenticated load_probe")

    def check(self):
        _state(self)[3]()

    def inspect(self, mapping, key):
        state = _state(self)
        state[3]()
        result = state[2][0](mapping, key)
        _need(result is True or result is NotImplemented, "Probe returned non-metadata result")
        return result

    def counters(self):
        state = _state(self)
        state[3]()
        result = state[2][3]()
        _need(_TYPE(result) is _DICT, "Probe counters type differs")
        return result

    def close(self):
        state = _state(self)
        try:
            state[3]()
            _audit_files(state[4])
        finally:
            # Original captured entry is used even if the public module drifted.
            try:
                state[2][5]()
            finally:
                del _OWNERS[_ID(self)]


def validate_binding(binding):
    """Exact registered ownership and fresh runtime/provider check; no inspection."""
    _state(binding)[3]()
    return None



def preflight_runtime():
    """Fresh public file/profile check before any scientific/native build."""
    _ROOT_VALIDATE(_BOOTSTRAP_CAPABILITY, _GLOBALS())
    files = _profile_files()
    _audit_files(files)
    return {"schema": "h25-public-runtime-preflight-v2", "authenticated_files": _LEN(files),
            "native_private_access": False, "cached_admission": False}


def load_probe(extension_path, expected_extension_sha256):
    """Authenticate actual runtime/header/binary bytes, then load and activate."""
    _ROOT_VALIDATE(_BOOTSTRAP_CAPABILITY, _GLOBALS())
    _need(_TYPE(extension_path) is _STR and extension_path.startswith("/"),
          "Probe extension path must be an absolute exact str")
    _need(_TYPE(expected_extension_sha256) is _STR and _LEN(expected_extension_sha256) == 64 and
          all(character in "0123456789abcdef" for character in expected_extension_sha256),
          "Probe requires an independently expected lowercase SHA256")
    files = _profile_files()
    _audit_files(files)  # actual /proc/self/exe and complete pinned headers first
    before_hash, binary_size, before_identity, _ = _hash_regular(extension_path, _MAX_BINARY, False)
    _need(before_hash == expected_extension_sha256, "Probe extension bytes differ before load")
    _ROOT_VALIDATE(_BOOTSTRAP_CAPABILITY, _GLOBALS())
    native = _ROOT_CREATE(_BOOTSTRAP_CAPABILITY, extension_path)
    _need(_TYPE(native) is ModuleType, "Probe loader did not return an exact module")
    _ROOT_EXEC(_BOOTSTRAP_CAPABILITY, native)
    _ROOT_VALIDATE(_BOOTSTRAP_CAPABILITY, _GLOBALS())
    after_hash, after_size, after_identity, _ = _hash_regular(extension_path, _MAX_BINARY, False)
    _need((after_hash, after_size, after_identity) == (before_hash, binary_size, before_identity),
          "Probe extension changed across loading")
    namespace = native.__dict__
    _need(namespace.get("__name__") == _NATIVE_NAME and namespace.get("__file__") == extension_path,
          "Probe loaded module origin differs")
    entries = _TUPLE(namespace.get(name) for name in _NATIVE_NAMES)
    for name, entry in zip(_NATIVE_NAMES, entries):
        _need(_TYPE(entry) is BuiltinFunctionType and entry.__self__ is native and entry.__name__ == name,
              "Probe native entry is not an owned builtin: " + name)
    public_profile = entries[2]()
    _need(_TYPE(public_profile) is _TUPLE and
          all(_TYPE(value) is _INT for value in public_profile) and public_profile == _EXPECTED_PROFILE,
          "Probe compiled/public ABI profile differs")
    _ROOT_VALIDATE(_BOOTSTRAP_CAPABILITY, _GLOBALS())
    _ROOT_ACTIVATE(_BOOTSTRAP_CAPABILITY, entries[4])  # final root guard precedes activation
    binding = None
    try:
        binding = _OBJECT.__new__(ProbeBinding)
        guard = _make_guard(native, entries)
        owner_files = files + ((extension_path, expected_extension_sha256, binary_size),)
        _OWNERS[_ID(binding)] = (binding, native, entries, guard, owner_files)
        guard()
    except BaseException:
        entries[5]()
        if binding is not None:
            _OWNERS.pop(_ID(binding), None)
        raise
    return binding
