def need(value, reason):
    if not value:
        raise ValueError(reason)

def canonical(value):
    value = plain_copy(value)
    return (json.dumps(value, sort_keys=True, ensure_ascii=True, separators=(',', ':')) + '\n').encode('ascii')

def parse(raw):
    need(type(raw) in (bytes, str) and len(raw) <= MAX_METADATA, 'exact bounded JSON source bytes')

    def pairs(rows):
        result = {}
        for key, value in rows:
            need(key not in result, 'duplicate metadata key')
            result[key] = value
        return result
    return json.loads(raw, object_pairs_hook=pairs, parse_constant=lambda value: need(False, 'nonfinite metadata'))

def exact(value, fields, reason):
    value = plain_copy(value)
    need(type(value) is dict and set(value) == set(fields), reason)

def digest(value):
    return type(value) is str and len(value) == 64 and (set(value) <= set('0123456789abcdef'))

def normalized(value):
    need(type(value) is str and value.startswith('/') and ('\x00' not in value) and (os.path.normpath(value) == value), 'normalized absolute path required')
    return value

def signature(info):
    return (info.st_dev, info.st_ino, info.st_mode, info.st_size, info.st_uid, info.st_gid, info.st_mtime_ns, info.st_ctime_ns, info.st_nlink)

def _function_state(function, depth=0, seen=None):
    need(depth <= 16, 'bounded tool closure')
    seen = set() if seen is None else seen
    if id(function) in seen:
        return (function, None)
    seen.add(id(function))
    cells = ()
    if function.__closure__ is not None:
        cells = tuple(((cell, cell.cell_contents, _function_state(cell.cell_contents, depth + 1, seen) if type(cell.cell_contents) is type(_function_state) else None) for cell in function.__closure__))
    return (function, (function.__code__, function.__globals__, function.__defaults__, function.__kwdefaults__, cells))

def _check_function_state(state):
    function, details = state
    if details is None:
        return
    code, namespace, defaults, kwdefaults, cells = details
    need(function.__code__ is code and function.__globals__ is namespace and (function.__defaults__ is defaults) and (function.__kwdefaults__ is kwdefaults), 'checked function/code/globals changed')
    closure = () if function.__closure__ is None else function.__closure__
    need(len(closure) == len(cells), 'checked function closure changed')
    for cell, (original, contents, nested) in zip(closure, cells):
        need(cell is original and cell.cell_contents is contents, 'checked closure cell changed')
        if nested is not None:
            _check_function_state(nested)

def _module_state(module):
    namespace = module.__dict__
    functions = []
    constants = {}
    identities = {}
    for name, value in namespace.items():
        if type(value) is type(_module_state):
            functions.append((name, _function_state(value)))
        elif name in ('LIMITS', 'DEFAULTS'):
            constants[name] = plain_copy(value)
        elif name == 'HEX':
            constants[name] = sorted(value)
        else:
            identities[name] = value
    return (namespace, tuple(functions), constants, identities)

def _check_module_state(module, state):
    namespace, functions, constants, identities = state
    need(module.__dict__ is namespace, 'same checked source module globals')
    need(set(namespace) == set(constants) | set(identities) | {name for name, state in functions}, 'checked tool globals membership changed')
    for name, function_state in functions:
        need(namespace.get(name) is function_state[0], 'checked tool function identity changed')
        _check_function_state(function_state)
    for name, value in constants.items():
        actual = sorted(namespace[name]) if name == 'HEX' else plain_copy(namespace[name])
        need(actual == value, 'checked tool mutable constants changed')
    for name, value in identities.items():
        need(namespace.get(name) is value, 'checked tool global identity changed')

def modules(tools_root):

    global _TOOL_CACHE, _TOOL_CACHE_OWNER
    normalized(tools_root)
    raws = {name: read_pinned(tools_root + '/' + name, pin, 1048576) for name, pin in PINNED.items()}
    if _TOOL_CACHE is not None:
        root, pins, archive, oracle, archive_state, oracle_state = _TOOL_CACHE
        need(tools_root == root and pins == PINNED, 'fixed parent preloaded tools binding')
        _check_module_state(archive, archive_state)
        _check_module_state(oracle, oracle_state)
        return (archive, oracle)
    need(not hasattr(parent_authority, 'is_current_borrower') or not parent_authority.is_current_borrower(), 'borrower tools must be preloaded by genuine parent')
    old = sys.dont_write_bytecode
    sys.dont_write_bytecode = True
    fault = native_io._ERRORS.RawSlot()
    fault.kind = 'PARENT-TOOLS-LOAD'
    try:
        loaded = []
        for name in ('archive.py', 'oracle.py'):
            path = tools_root + '/' + name

            module.__dict__['__builtins__'] = __builtins__

        _TOOL_CACHE_OWNER = os.getpid()

        fault.traceback = BaseException.__traceback__.__get__(error, BaseException)

@native_io.owning_leaf
def lease_check(cfg, lease_fd, capability, custody=None):
    fault = native_io._ERRORS.RawSlot()
    fault.kind = 'JOINED-LEASE-OWNER'
    fd_number(lease_fd)
    cfg = validate_config(cfg)
    ctx = native_io.new_context('consumer-lease-preflight', custody)
    try:
        binding = parent_authority.authenticate(capability, cfg, lease_fd, ctx.errors)
        need(parent_authority.is_joined_binding(binding), 'fixed genuine joined owner/borrower only')
        held = parent_authority.verify_lease_history(binding, lease_fd, ctx.errors)
        given = native_io.call_native(ctx, 'JOINED-LEASE-FSTAT', os.fstat, lease_fd)
        need(stat.S_ISREG(given.st_mode) and given.st_nlink == 1 and (given.st_uid == os.geteuid()) and (stat.S_IMODE(given.st_mode) == 384) and (not native_io.call_native(ctx, 'JOINED-LEASE-XATTR', os.listxattr, lease_fd)), 'joined original EX owner/mode/link/xattrs')
        need((given.st_dev, given.st_ino) == (cfg['writer_lease']['device'], cfg['writer_lease']['inode']), 'joined original EX inode')
        need(parent_authority.verify_lease_history(binding, lease_fd, ctx.errors) == held, 'joined EX history changed')
        return signature(given)
    except BaseException as error:
        fault.actual = error
        fault.traceback = BaseException.__traceback__.__get__(error, BaseException)
        native_io._commit(ctx, fault)
        raise BaseException.with_traceback(ctx.primary, ctx.primary_tb)
    finally:
        native_io.finish(ctx)

def validate_domain(cfg, inventory, domain, closure):
    """Versioned complete producer union; missing canonical receipts stay absent."""
    cfg = validate_config(cfg)
    inventory, domain, closure = (plain_copy(value) for value in (inventory, domain, closure))
    exact(domain, ('schema', 'source_root', 'scope', 'root_files', 'runs'), 'complete domain fields')
    need(domain['schema'] == 'h25-completed-source-domain-v2' and domain['source_root'] == cfg['source_root'] and (domain['scope'] == 'complete-actual-attempt-union'), 'prospective complete case grammar')
    entries = {row['path']: row for row in inventory['entries']}
    direct_dirs = sorted((row['path'] for row in inventory['entries'] if row['kind'] == 'dir' and '/' not in row['path']))
    direct_files = sorted((row['path'] for row in inventory['entries'] if row['kind'] == 'file' and '/' not in row['path']))
    need(type(domain['runs']) is list and domain['runs'] and (len(domain['runs']) <= cfg['limits']['max_runs']) and (type(domain['root_files']) is list), 'nonempty actual domain tables')
    fixed_layout = {name: name for name in ('attempt', 'runner', 'intent', 'supervision', 'queue', 'validation', 'serialized-DATA')}
    names = []
    for row in domain['runs']:
        exact(row, ('run', 'terminal_status', 'terminal_receipts', 'layout', 'attempt_state', 'canonical_receipt'), 'case domain row')
        run = row['run']
        need(type(run) is str and run not in ('', '.', '..') and ('/' not in run) and ('\\' not in run), 'case identity')
        need(row['layout'] == fixed_layout and row['attempt_state'] == 'TERMINAL', 'fixed producer layout and genuine terminal')
        need(row['terminal_status'] in ('PASS', 'STOP', 'NONZERO', 'INTERRUPTED'), 'honest terminal status')
        names.append(run)
        need(type(row['terminal_receipts']) is list and row['terminal_receipts'] and (row['terminal_receipts'] == sorted(set(row['terminal_receipts']))), 'actual terminal list')
        need('attempt/TERMINAL.json' in row['terminal_receipts'], 'actual owner terminal required')
        for receipt in row['terminal_receipts']:
            need(type(receipt) is str and (not receipt.startswith('/')) and ('..' not in receipt.split('/')), 'relative terminal path')
            entry = entries.get(run + '/' + receipt)

            need(row['canonical_receipt'] == 'runner/receipt.json' and canonical_path in entries and (entries[canonical_path]['kind'] == 'file'), 'canonical receipt observation mismatch')

            if path.startswith(run + '/'):

    exact(closure, ('schema', 'source_root', 'inventory_sha256', 'domain_sha256', 'lease_protocol_sha256', 'all_producers_enrolled', 'inventory_collected_under_held_lease', 'all_measured_processes_reaped', 'all_terminal_receipts_final', 'all_raw_streams_closed', 'no_future_writes_under_lease', 'purpose', 'attempt_ids'), 'closure fields')
    need(closure['schema'] == 'h25-parent-run-closure-v1' and closure['source_root'] == cfg['source_root'] and (closure['inventory_sha256'] == cfg['inventory']['sha256']) and (closure['domain_sha256'] == cfg['domain']['sha256']) and (closure['lease_protocol_sha256'] == cfg['writer_lease']['protocol_sha256']) and (closure['purpose'] == cfg['purpose']), 'closure exact external bindings')

def footprint(archive, root, cfg):
    cap = cfg['limits']
    census = archive.scan(root, {**cap, 'max_entries': cap['max_entries'] * 5 + 256}, True)
    rows = list(census['entries'].values())
    logical = sum((row['size'] for row in rows if row['kind'] == 'file'))
    need(logical <= cfg['resources']['retained_bytes_cap'], 'retained regular-byte cap')
    return dict(logical_bytes=logical, stat_allocated_bytes=sum((row['allocated'] for row in rows if row['kind'] == 'file')), files=sum((row['kind'] == 'file' for row in rows)), directories=1 + sum((row['kind'] == 'dir' for row in rows)))

def _nonalias(before, restored):
    original = {(row['origin']['device'], row['origin']['inode']) for row in before['entries']}
    restored_ids = {(row['origin']['device'], row['origin']['inode']) for row in restored['entries']}
    original.add((before['root_origin']['device'], before['root_origin']['inode']))
    restored_ids.add((restored['root_origin']['device'], restored['root_origin']['inode']))
    need(not original.intersection(restored_ids), 'complete restored/original inode nonalias')

def run(cfg, lease_fd, expected_authority_sha256, capability=None, custody=None):
    _owned_fault_16 = native_io._ERRORS.RawSlot()
    _owned_fault_16.kind = 'LEXICAL-OWNER:run'
    _owned_fault_17 = native_io._ERRORS.RawSlot()
    _owned_fault_17.kind = 'LEXICAL-OWNER:run'
    custody = error_custody.new_custody() if custody is None else error_custody.owned(custody)
    fault = error_custody.reserve(custody, 'CONSUMER-ENTRY')
    try:
        try:
            return _run_attempt(cfg, lease_fd, expected_authority_sha256, capability, custody)
        except BaseException as error:
            _owned_fault_16.actual = error
            _owned_fault_16.traceback = BaseException.__traceback__.__get__(error, BaseException)
            native_io._commit(native_io.process_context(), _owned_fault_16)
            fault.actual = error

            if custody.primary is None:
                custody.primary = error
                custody.primary_tb = fault.traceback
            error_custody.commit(custody, fault)
            raise BaseException.with_traceback(custody.primary, custody.primary_tb)
    except BaseException as _owned_error_17:
        _owned_fault_17.actual = _owned_error_17
        _owned_fault_17.traceback = BaseException.__traceback__.__get__(_owned_error_17, BaseException)
        native_io._commit(native_io.process_context(), _owned_fault_17)
        raise
    finally:
        error_custody.release(custody, fault)
