/* Reserved before the first acquisition. Pre-journal native failures remain
 * first in this fixed storage even if formatting or the fallback relay fails. */
static struct pq_native_error early_errors[PQ_ERROR_MAX];

static void store_early(unsigned operation, int raw_errno, unsigned detail, uint64_t auxiliary) {
    if (early_count < PQ_ERROR_MAX)
        early_errors[early_count++] = (struct pq_native_error){phase, operation, raw_errno, detail, auxiliary};
    else { early_overflow = true; cleanup_uncertain = true; }
}
/* The raw errno is captured by callers before any fallible bookkeeping. */
static void error_event(unsigned operation, int raw_errno, unsigned detail, uint64_t auxiliary) {
    failed_state = true;
    if (journal) {
        uint32_t i = atomic_fetch_add_explicit(&journal->used, 1, memory_order_relaxed);
        if (i < PQ_ERROR_MAX) {
            journal->entries[i] = (struct pq_native_error){phase, operation, raw_errno, detail, auxiliary};
            uint32_t expected = atomic_load_explicit(&journal->primary_index, memory_order_relaxed);
            /* Monotone minimum preserves the first reserved native slot even
             * if the child and observer finish their stores out of order. */
            while (i < expected &&
                   !atomic_compare_exchange_strong_explicit(&journal->primary_index, &expected, i,
                                                            memory_order_relaxed, memory_order_relaxed)) { }

            atomic_store_explicit(&journal->ready[i], 1, memory_order_release);
        } else {
            atomic_store_explicit(&journal->overflow, 1, memory_order_release);
            cleanup_uncertain = true;
        }
    } else {
        /* Before journal acquisition the genuine outer relay owns this raw
         * native event; missing journal can never become qualified evidence. */
        store_early(operation, raw_errno, detail, auxiliary);
        if (!fallback_relay_ready) { cleanup_uncertain = true; return; }
        char b[160];
        int n = snprintf(b, sizeof b, "PQOBS_NATIVE_ERROR %u %d %u %" PRIu64 "\n",
                         operation, raw_errno, detail, auxiliary);
        if (n < 0 || (size_t)n >= sizeof b) {
            int saved = n < 0 ? errno : 0;
            store_early(E_WRITE, saved, 4, 0); cleanup_uncertain = true;
        } else {
            size_t at = 0; unsigned interrupted = 0;
            while (at < (size_t)n) {
                ssize_t k = output_write(2, b + at, (size_t)n - at);
                if (k < 0) {
                    int saved = errno;
                    if (saved == EINTR && interrupted++ < 8) continue;
                    store_early(E_WRITE, saved, 5, at); cleanup_uncertain = true; break;
                }
                if (!k) { store_early(E_WRITE, 0, 6, at); cleanup_uncertain = true; break; }
                at += (size_t)k;
            }
        }
    }
}
static bool refuse(unsigned operation, unsigned detail, uint64_t auxiliary) {
    error_event(operation, 0, detail, auxiliary); return false;
}
static bool os_error(unsigned operation, int saved, uint64_t auxiliary) {
    error_event(operation, saved, 0, auxiliary); return false;
}
static uint64_t clock_ms(void) {
    struct timespec t;
    if (clock_gettime(CLOCK_MONOTONIC, &t) < 0) {
        int saved = errno; os_error(E_CLOCK, saved, 0); return UINT64_MAX;
    }
    if (t.tv_sec < 0 || (uint64_t)t.tv_sec > (UINT64_MAX - 1000) / 1000)
        return refuse(E_CLOCK, 1, 0), UINT64_MAX;
    return (uint64_t)t.tv_sec * 1000 + (unsigned long)t.tv_nsec / 1000000;
}
static bool close_owned(int fd) {
    if (close(fd) < 0) {
        int saved = errno; cleanup_uncertain = true;
        return os_error(E_CLOSE, saved, (unsigned)fd);
    }
    return true;
}

static bool write_all(int fd, const void *data, size_t n, uint64_t *written) {
    const unsigned char *p = data;
    for (size_t at = 0; at < n;) {
        ssize_t k = output_write(fd, p + at, n - at);
        if (k < 0) {
            int saved = errno;
            if (saved == EINTR) continue;
            return os_error(E_WRITE, saved, (unsigned)fd);
        }
        if (k == 0) return refuse(E_WRITE, 1, (unsigned)fd);
        at += (size_t)k;
        if (written) *written += (uint64_t)k;
    }
    return true;
}
static bool pread_exact(int fd, void *data, size_t n, off_t off) {
    unsigned char *p = data;
    for (size_t at = 0; at < n;) {
        ssize_t k = pread(fd, p + at, n - at, off + (off_t)at);
        if (k < 0) { int saved = errno; if (saved == EINTR) continue; return os_error(E_READBACK, saved, (unsigned)fd); }
        if (!k) return refuse(E_READBACK, 1, (unsigned)fd);
        at += (size_t)k;
    }
    return true;
}
static bool decimal_fd(const char *s, int *out) {
    unsigned long v = 0;
    if (!s || !*s || (s[0] == '0' && s[1])) return false;
    for (; *s; s++) {
        if (*s < '0' || *s > '9' || v > ((unsigned long)INT_MAX - (unsigned)(*s-'0')) / 10) return false;
        v = v * 10 + (unsigned)(*s-'0');
    }
    if (v < 3) return false;
    *out = (int)v; return true;
}
static bool parse_fds(int argc, char **argv) {
    static const char *names[NFD] = {"--request-fd", "--snapshot-fd", "--stdout-fd", "--stderr-fd", "--terminal-fd", "--journal-fd", "--exec-fd", "--cwd-fd", "--stdin-fd"};
    if (argc != 19) return refuse(E_ARGUMENT, 1, (unsigned)argc);
    for (unsigned i = 0; i < NFD; i++) {
        if (strcmp(argv[1+2*i], names[i]) || !decimal_fd(argv[2+2*i], &fds[i])) return refuse(E_ARGUMENT, 2, i);
        for (unsigned k = 0; k < i; k++) if (fds[k] == fds[i]) return refuse(E_ARGUMENT, 3, i);
        fd_live[i] = true;
    }
    return true;
}
static bool initialize_journal(void) {
    struct stat s;
    if (fstat(fds[JR], &s) < 0) { int saved = errno; return os_error(E_FD, saved, JR); }
    int flags = fcntl(fds[JR], F_GETFL);
    if (flags < 0) { int saved = errno; return os_error(E_FD, saved, JR); }
    if (!S_ISREG(s.st_mode) || s.st_size != PQ_JOURNAL_BYTES || s.st_nlink != 1 ||
        (flags & O_ACCMODE) != O_RDWR || (flags & O_APPEND)) return refuse(E_FD, 1, JR);
    unsigned char zero[PQ_JOURNAL_BYTES];
    if (!pread_exact(fds[JR], zero, sizeof zero, 0)) return false;
    for (size_t i = 0; i < sizeof zero; i++) if (zero[i]) return refuse(E_FD, 2, JR);
    void *p = mmap(NULL, PQ_JOURNAL_BYTES, PROT_READ|PROT_WRITE, MAP_SHARED, fds[JR], 0);
    if (p == MAP_FAILED) { int saved = errno; return os_error(E_FD, saved, JR); }
    journal = p;
    memcpy(journal->magic, PQ_JOURNAL_MAGIC, 8); journal->version = 1;
    atomic_store(&journal->primary_index, UINT32_MAX);
    return true;
}
static bool validate_fds(void) {
    struct stat identities[NFD];
    for (unsigned i = 0; i < NFD; i++) {
        if (fstat(fds[i], &identities[i]) < 0) { int saved = errno; return os_error(E_FD, saved, i); }
        int flags = fcntl(fds[i], F_GETFL);
        if (flags < 0) { int saved = errno; return os_error(E_FD, saved, i); }
        if (i <= TM && i != RQ) {
            if (!S_ISREG(identities[i].st_mode) || identities[i].st_size || identities[i].st_nlink != 1 ||
                (flags & O_ACCMODE) != O_RDWR || (flags & O_APPEND)) return refuse(E_FD, 3, i);
            off_t at = lseek(fds[i], 0, SEEK_CUR);
            if (at < 0) { int saved = errno; return os_error(E_FD, saved, i); }
            if (at) return refuse(E_FD, 4, i);
        }
        if (i == RQ && (!S_ISREG(identities[i].st_mode) || identities[i].st_size < 40 ||
            identities[i].st_size > PQ_REQUEST_MAX || (flags & O_ACCMODE) != O_RDONLY)) return refuse(E_FD, 5, i);
        if (i == EX && !S_ISREG(identities[i].st_mode)) return refuse(E_FD, 6, i);
        if (i == CW && !S_ISDIR(identities[i].st_mode)) return refuse(E_FD, 7, i);
        if (i == IN && (!S_ISCHR(identities[i].st_mode) || major(identities[i].st_rdev) != 1 ||
            minor(identities[i].st_rdev) != 3 || (flags & O_ACCMODE) != O_RDONLY)) return refuse(E_FD, 8, i);
    }
    for (unsigned i = RQ; i <= JR; i++) for (unsigned k = RQ; k < i; k++)
        if (identities[i].st_dev == identities[k].st_dev && identities[i].st_ino == identities[k].st_ino)
            return refuse(E_FD, 9, i);
    return true;
}
static uint32_t le32(const unsigned char *p) {
    return (uint32_t)p[0] | (uint32_t)p[1]<<8 | (uint32_t)p[2]<<16 | (uint32_t)p[3]<<24;
}
static bool parse_request(void) {
    struct stat s;
    if (fstat(fds[RQ], &s) < 0) { int saved = errno; return os_error(E_REQUEST, saved, 0); }
    if (s.st_size < 40 || s.st_size > PQ_REQUEST_MAX) return refuse(E_REQUEST, 10, (uint64_t)s.st_size);
    size_t n = (size_t)s.st_size;
    if (!pread_exact(fds[RQ], req.raw, n, 0)) return false;
    memcpy(&req.h, req.raw, sizeof req.h);
    if (memcmp(req.h.magic, PQ_REQUEST_MAGIC, 8) || req.h.query_id < 1 || req.h.query_id > 17 ||
        !req.h.argc || req.h.argc > PQ_ARG_MAX || !req.h.envc || req.h.envc > PQ_ENV_MAX ||
        !req.h.stdout_cap || req.h.stdout_cap > PQ_STREAM_MAX || !req.h.stderr_cap || req.h.stderr_cap > PQ_STREAM_MAX ||
        !req.h.syscall_cap || req.h.syscall_cap > PQ_SYSCALL_MAX || !req.h.write_cap || req.h.write_cap > PQ_WRITE_MAX ||
        !req.h.timeout_ms || req.h.timeout_ms > PQ_TIMEOUT_MAX_MS) return refuse(E_REQUEST, 1, n);
    size_t at = 40, used = 0; bool locale = false, temporary = false;
    for (unsigned i = 0; i < req.h.argc + req.h.envc; i++) {
        if (at > n || n-at < 4) return refuse(E_REQUEST, 2, i);
        uint32_t len = le32(req.raw+at); at += 4;
        if (!len || len > n-at || memchr(req.raw+at, 0, len) || used+len+1 > sizeof req.strings) return refuse(E_REQUEST, 3, i);
        char *p = req.strings+used; memcpy(p, req.raw+at, len); p[len] = 0;
        at += len; used += len+1;
        if (i < req.h.argc) req.argv[i] = p;
        else {
            unsigned k = i-req.h.argc; req.env[k] = p;
            char *eq = strchr(p, '=');
            if (!eq || eq == p || (k && strcmp(req.env[k-1], p) >= 0)) return refuse(E_REQUEST, 4, i);
            for (char *c = p; c < eq; c++) if (!((*c >= 'A' && *c <= 'Z') || (*c >= 'a' && *c <= 'z') || (*c >= '0' && *c <= '9') || *c == '_')) return refuse(E_REQUEST, 5, i);
            if (k) { char *old = strchr(req.env[k-1], '='); if ((size_t)(old-req.env[k-1]) == (size_t)(eq-p) && !memcmp(req.env[k-1], p, (size_t)(eq-p))) return refuse(E_REQUEST, 6, i); }
            if (!strcmp(p, "LC_ALL=C")) locale = true;
            if (!strncmp(p, "TMPDIR=/", 8)) temporary = true;
            if (!strncmp(p, "LD_", 3) || !strncmp(p, "GLIBC_TUNABLES=", 15) || !strncmp(p, "TMP=", 4) || !strncmp(p, "TEMP=", 5)) return refuse(E_REQUEST, 7, i);
        }
    }
    if (at != n || !locale || !temporary) return refuse(E_REQUEST, 8, at);
    if (req.argv[0][0] != '/') for (const unsigned char *p=(unsigned char *)req.argv[0]; *p; p++)
        if (!( (*p>='A'&&*p<='Z') || (*p>='a'&&*p<='z') || (*p>='0'&&*p<='9') || strchr("_.+-", *p))) return refuse(E_REQUEST, 9, *p);
    return true;
}
struct pq_dirent { uint64_t ino; int64_t off; unsigned short reclen; unsigned char type; char name[]; };
static bool audit_directory(const char *path, bool task_directory) {
    int fd = open(path, O_RDONLY|O_DIRECTORY|O_CLOEXEC|O_NOFOLLOW);
    if (fd < 0) { int saved = errno; return os_error(E_AUDIT, saved, 0); }
    bool okay = true, complete = false; unsigned seen = 0;
    unsigned char b[8192] __attribute__((aligned(8)));
    for (unsigned reads=0; okay && reads<4; reads++) {
        long n = syscall(SYS_getdents64, fd, b, sizeof b);
        if (n < 0) { int saved = errno; okay = os_error(E_AUDIT, saved, 1); break; }
        if (!n) { complete = true; break; }
        for (size_t at=0; okay && at<(size_t)n;) {
            struct pq_dirent *d=(struct pq_dirent *)(void *)(b+at);
            if ((size_t)n-at<20 || d->reclen<20 || d->reclen>(size_t)n-at || !memchr(d->name,0,d->reclen-19)) { okay=refuse(E_AUDIT,2,at); break; }
            if (strcmp(d->name,".") && strcmp(d->name,"..")) {
                uint64_t v=0; const char *p=d->name;
                if (!*p) okay=false;
                for (; *p && okay; p++) { if (*p<'0'||*p>'9'||v>((uint64_t)INT_MAX-(unsigned)(*p-'0'))/10) okay=false; else v=v*10+(unsigned)(*p-'0'); }
                bool allowed=task_directory ? v==(uint64_t)getpid() : (v<=2 || v==(unsigned)fd);
                if (!task_directory) for(unsigned k=0;k<NFD;k++) if(v==(unsigned)fds[k]) allowed=true;
                if (!okay || !allowed || ++seen>32) okay=refuse(E_AUDIT,3,v);
            }
            at+=d->reclen;
        }
    }
    if (okay && (!complete || (task_directory && seen!=1))) okay=refuse(E_AUDIT,4,seen);
    bool closed=close_owned(fd); return okay && closed;
}
static bool audit_start(void) {
    if (!audit_directory("/proc/self/fd",false) || !audit_directory("/proc/self/task",true)) return false;
    char path[96]; int n=snprintf(path,sizeof path,"/proc/self/task/%ld/children",(long)getpid());
    if(n<=0||(size_t)n>=sizeof path) return refuse(E_AUDIT,5,0);
    int fd=open(path,O_RDONLY|O_CLOEXEC|O_NOFOLLOW);
    if(fd<0){int saved=errno;return os_error(E_AUDIT,saved,2);}
    char c;ssize_t got=read(fd,&c,1);bool okay;
    if(got<0){int saved=errno;okay=os_error(E_AUDIT,saved,3);}else okay=got==0 || refuse(E_AUDIT,6,(uint64_t)got);
    bool closed=close_owned(fd);return okay&&closed;
}

}
static bool probe_fd(int fd, int expected_pipe) {
#ifdef PQ_TEST
    extern int pq_mock_probe(int,int);
    int result=pq_mock_probe(fd,expected_pipe);
    return result==0 || refuse(E_FD,10,(unsigned)fd);
#else
    char path[96];int n=snprintf(path,sizeof path,"/proc/%ld/fd/%d",(long)child_pid,fd);
    if(n<=0||(size_t)n>=sizeof path)return refuse(E_FD,11,(unsigned)fd);
    int held=open(path,O_PATH|O_CLOEXEC);if(held<0){int saved=errno;return os_error(E_FD,saved,(unsigned)fd);}
    struct stat s;bool okay=true;
    if(fstat(held,&s)<0){int saved=errno;okay=os_error(E_FD,saved,(unsigned)fd);}
    if(okay) {
        bool match[2];for(unsigned i=0;i<2;i++)match[i]=s.st_dev==pipe_identity[i].st_dev&&s.st_ino==pipe_identity[i].st_ino;
        if(expected_pipe>=0)okay=match[expected_pipe] || refuse(E_FD,12,(unsigned)fd);
        else okay=(!match[0]&&!match[1]) || refuse(E_FD,13,(unsigned)fd);
    }
    bool closed=close_owned(held);return okay&&closed;
#endif
}
/* A finite whitelist: unknown/new output or FD-transfer mechanisms stop the
 * report before they run. New query paths require source review, never fallback. */
static bool syscall_allowed(uint64_t nr, const uint64_t a[6]) {
    if(nr==SYS_write||nr==SYS_writev)return true;
    if(nr==SYS_execveat)return exec_events==0 && a[0]==3 && a[4]==AT_EMPTY_PATH;
    uint32_t fd = (uint32_t)a[0], other = (uint32_t)a[1];
    if(nr==SYS_close)return fd>2 && fd<=INT_MAX;
    if(nr==SYS_close_range)return fd>2 && other>=fd && a[2]==0;
    if(nr==SYS_dup||nr==SYS_dup2||nr==SYS_dup3) return fd>2 && fd<=INT_MAX && (nr==SYS_dup || (other>2 && other<=INT_MAX));
    if(nr==SYS_fcntl) {
        unsigned cmd=(unsigned)a[1];
        if(fd<=2)return cmd==F_GETFD||cmd==F_GETFL||cmd==F_GETPIPE_SZ;
        return cmd==F_GETFD||cmd==F_SETFD||cmd==F_GETFL||cmd==F_SETFL||cmd==F_DUPFD||cmd==F_DUPFD_CLOEXEC||cmd==F_GETLK||cmd==F_SETLK||cmd==F_SETLKW;
    }
    if(nr==SYS_ioctl)return a[1]==0x5401U||a[1]==0x5413U||a[1]==0x541bU;
    if(nr==SYS_pwrite64||nr==SYS_pwritev||nr==SYS_pwritev2||nr==SYS_ftruncate||nr==SYS_fchmod||nr==SYS_fchown||nr==SYS_fsync||nr==SYS_fdatasync)return fd>2 && fd<=INT_MAX;
    if(nr==SYS_mmap)return (a[3]&MAP_ANONYMOUS) || (uint32_t)a[4]>2;
    if(nr==SYS_prlimit64)return (uint32_t)a[0]==0 && !((uint32_t)a[1]==RLIMIT_AS && a[2]!=0);
    static const unsigned allowed[]={
        SYS_read,SYS_pread64,SYS_readv,SYS_preadv,SYS_preadv2,SYS_open,SYS_openat,SYS_openat2,SYS_creat,
        SYS_fstat,SYS_stat,SYS_lstat,SYS_newfstatat,SYS_statx,SYS_access,SYS_faccessat,SYS_faccessat2,

static bool capture_output(void) {
    uint32_t actual_fd=(uint32_t)pending.args[0];
    if(actual_fd>INT_MAX)return refuse(E_FD,15,actual_fd);
    if(actual_fd!=1 && actual_fd!=2)return probe_fd((int)actual_fd,-1);
    unsigned which=actual_fd-1;uint64_t amount=0;
    if(write_entries==req.h.write_cap)return refuse(E_CAPACITY,1,write_entries);
    if(!probe_fd((int)actual_fd,(int)which))return false;
    size_t cap=which?req.h.stderr_cap:req.h.stdout_cap;
    struct iovec vec[PQ_IOV_MAX];unsigned count=1;
    if(pending.nr==SYS_write){vec[0]=(struct iovec){(void *)(uintptr_t)pending.args[1],(size_t)pending.args[2]};}
    else {
        if(pending.args[2]>PQ_IOV_MAX)return refuse(E_CAPACITY,2,pending.args[2]);
        count=(unsigned)pending.args[2];
        if(!remote_read(pending.args[1],vec,count*sizeof vec[0]))return false;
    }
    for(unsigned i=0;i<count;i++) {
        if(vec[i].iov_len>cap-snapshot_len[which]-amount)return refuse(E_CAPACITY,3,vec[i].iov_len);
        amount+=vec[i].iov_len;
    }
    size_t at=snapshot_len[which];
    for(unsigned i=0;i<count;i++) {
        if(!remote_read((uint64_t)(uintptr_t)vec[i].iov_base,snapshot[which]+at,vec[i].iov_len))return false;
        at+=vec[i].iov_len;
    }
    pending.output=true;pending.fd=actual_fd;pending.requested=amount;pending.seq=++write_entries;
    return snapshot_record(1,0,snapshot[which]+snapshot_len[which],(size_t)amount);
}
static bool syscall_entry(const struct pq_sysinfo *info) {
    if(pending.inside)return refuse(E_PROTOCOL,1,syscall_entries);
    if(syscall_entries+terminal_exit_entries==req.h.syscall_cap)return refuse(E_CAPACITY,4,syscall_entries);
    uint64_t nr=info->u.entry.nr;
    if(info->arch!=AUDIT_ARCH_X86_64 || nr>INT_MAX || (nr&0x40000000U))return refuse(E_ABI,1,nr);
    if(!syscall_allowed(nr,info->u.entry.args))return refuse(E_PROTOCOL,2,nr);
    pending=(struct pq_pending){.inside=true,.nr=nr};memcpy(pending.args,info->u.entry.args,sizeof pending.args);
    if (nr == SYS_exit || nr == SYS_exit_group) terminal_exit_entries++;
    else syscall_entries++;
    if(nr==SYS_write||nr==SYS_writev)return capture_output();
    return true;
}
static bool syscall_exit(const struct pq_sysinfo *info) {
    if(!pending.inside||info->arch!=AUDIT_ARCH_X86_64||pending.nr==SYS_exit||pending.nr==SYS_exit_group)return refuse(E_PROTOCOL,3,syscall_exits);
    int64_t result=info->u.exit.rval;syscall_exits++;
    if(pending.output) {
        if(!snapshot_record(2,result,NULL,0))return false;
        write_exits++;
        if(info->u.exit.is_error || result<0) {
            int raw=(result<0 && result>=-4095)?(int)-result:0;
            error_event(E_WRITE,raw,2,(uint64_t)result);return false;
        }
        if((uint64_t)result!=pending.requested)return refuse(E_WRITE,3,(uint64_t)result);
        snapshot_len[pending.fd-1]+=(size_t)pending.requested;
    }
    if(!info->u.exit.is_error && result>=0 && fd_returning(pending.nr,pending.args)) {
        if(result>INT_MAX||result<=2)return refuse(E_FD,14,(uint64_t)result);
        if(!probe_fd((int)result,-1))return false;
    }
    pending.inside=false;pending.output=false;return true;
}
/* A nonreturning exit entry has an exit event and a terminal wait, never a

        (pending.nr != SYS_exit && pending.nr != SYS_exit_group) ||
        terminal_exit_entries != 1 || syscall_entries != syscall_exits)

}

static void process_wait(pid_t pid,int status) {
    struct pq_task *t=task(pid,false);
    if(!t){t=task(pid,true);refuse(E_DESCENDANT,1,(unsigned)pid);}
    if(!t){if(kill(pid,SIGKILL)<0){int saved=errno;os_error(E_DESCENDANT,saved,(unsigned)pid);}return;}
    if(WIFEXITED(status)||WIFSIGNALED(status)) {
        if(!t->live){refuse(E_PROTOCOL,4,(unsigned)pid);return;}
        t->live=false;t->stopped=false;retired_tasks++;
        if(pid==child_pid){terminal_wait_seen=1;exit_code=WIFEXITED(status)?WEXITSTATUS(status):-1;term_signal=WIFSIGNALED(status)?(unsigned)WTERMSIG(status):0;phase=6;
            if(!exit_event_seen||pending.output||(!WIFEXITED(status)||exit_code!=0))refuse(E_PROTOCOL,5,(unsigned)status);}
        return;
    }
    if(!WIFSTOPPED(status)){refuse(E_WAIT,1,(unsigned)status);return;}
    t->stopped=true;
    if(failed_state)return;
    if(pid!=child_pid){refuse(E_DESCENDANT,2,(unsigned)pid);return;}
    unsigned event=(unsigned)status>>16;int signal=WSTOPSIG(status);
    if(phase==1 && signal==SIGSTOP && !event) {
        if(ptrace(PTRACE_SETOPTIONS,pid,NULL,(void *)(uintptr_t)PQ_OPTIONS)<0){int saved=errno;os_error(E_PTRACE,saved,1);return;}
        phase=2;(void)resume_task(t,false);return;
    }
    if(event) {
        if(signal!=SIGTRAP){refuse(E_PROTOCOL,6,(unsigned)status);return;}
        if(event==PTRACE_EVENT_EXEC) {
            if(exec_events||!pending.inside||pending.nr!=SYS_execveat){refuse(E_PROTOCOL,7,event);return;}
            exec_events++;phase=3;
            if(!probe_fd(1,0)||!probe_fd(2,1))return;
            (void)resume_task(t,false);return;
        }
        if(event==PTRACE_EVENT_EXIT) {
            unsigned long message=0;
            if(ptrace(PTRACE_GETEVENTMSG,pid,NULL,(void *)&message)<0){int saved=errno;os_error(E_PTRACE,saved,event);return;}
            if(!accept_exit_event())return;
            (void)resume_task(t,true);return;
        }
        if(event==PTRACE_EVENT_FORK||event==PTRACE_EVENT_VFORK||event==PTRACE_EVENT_CLONE) {
            unsigned long descendant=0;
            if(ptrace(PTRACE_GETEVENTMSG,pid,NULL,&descendant)<0){int saved=errno;cleanup_uncertain=true;os_error(E_PTRACE,saved,event);}
            else if(descendant>0&&descendant<=INT_MAX)(void)task((pid_t)descendant,true);
            else cleanup_uncertain=true;
        }
        refuse(E_DESCENDANT,3,event);return;
    }
    if(signal!=(SIGTRAP|0x80)){refuse(E_PROTOCOL,10,(unsigned)signal);return;}
    struct pq_sysinfo info;memset(&info,0,sizeof info);
    long n=ptrace(PTRACE_GET_SYSCALL_INFO,pid,(void *)sizeof info,(void *)&info);
    if(n<0){int saved=errno;os_error(E_PTRACE,saved,2);return;}
    if(info.op==1 && n>=80){if(!syscall_entry(&info))return;}
    else if(info.op==2 && n>=33){if(!syscall_exit(&info))return;}
    else {refuse(E_PROTOCOL,11,(unsigned)n);return;}
    if(exec_events)phase=4;
    (void)resume_task(t,false);
}

int main(int argc,char **argv) {

     * SIGPIPE before its genuine trace stop and exact exec. */
    struct sigaction ignore; memset(&ignore, 0, sizeof ignore);

         * fixed and reserved before fork; no later dynamic payload allocation. */
        memset(snapshot,0,sizeof snapshot);memset(drained,0,sizeof drained);
        for(unsigned i=0;i<2&&okay;i++) {
            if(pipe2(pipes_fd[i],O_CLOEXEC|O_NONBLOCK)<0){int saved=errno;okay=os_error(E_PIPE,saved,i);break;}

    }

    }
    if(okay&&!failed_state) {
        if(!exec_events||!exit_event_seen||!terminal_wait_seen||exit_code||term_signal||live_tasks()||!eof_seen[0]||!eof_seen[1]||write_entries!=write_exits||snapshot_records!=2*write_entries||terminal_exit_entries!=1||syscall_entries!=syscall_exits||cleanup_uncertain)
            refuse(E_PROTOCOL,12,phase);
        else { phase=7;if(stream_readback(0)&&stream_readback(1)&&snapshot_readback())phase=8; }
    }
    for(unsigned p=0;p<2;p++)for(unsigned e=0;e<2;e++)close_pipe(p,e);
    /* Close every evidence/input owner except terminal; journal mapping stays
     * alive through all final closes, and is retired by actual process death. */
    for(unsigned i=0;i<NFD;i++)if(i!=TM)close_input(i);
    if(fd_live[TM]){(void)emit_terminal();close_input(TM);}
    if(early_count||early_overflow||(journal&&(atomic_load(&journal->used)||atomic_load(&journal->overflow))))failed_state=true;
    _exit(!failed_state&&phase==8?0:78);
}
