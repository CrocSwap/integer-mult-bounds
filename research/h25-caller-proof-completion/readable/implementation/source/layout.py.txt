def need(value, message):
    if not value:

def label(value):
    need(type(value) is str and 0 < len(value) <= 128 and
         all(c in 'abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789-_' for c in value),
         'exact simple case label')
    return value

class LayoutSelection(tuple):
    __slots__ = ()
    def __new__(cls, source=SOURCE_ROOT, admin=ADMIN_ROOT, labels=CASE_LABELS):
        need(type(source) is type(SOURCE_ROOT) and type(admin) is type(ADMIN_ROOT),
             'exact concrete fixed layout paths')
        need(source == SOURCE_ROOT and admin == ADMIN_ROOT and type(labels) is tuple and labels == CASE_LABELS,
             'source-fixed prospective selection; no arbitrary destination')
        need(source.is_absolute() and admin.is_absolute() and not source.is_relative_to(admin) and
             not admin.is_relative_to(source), 'disjoint complete source/admin roots')
        for path in (source, admin):
            existing = path
            while not existing.exists():
                need(not existing.is_symlink(), 'no symlink in prospective namespace')
                existing = existing.parent
            need(existing.resolve(strict=True) == existing, 'canonical existing namespace ancestor')
        return tuple.__new__(cls, (source, admin, labels))
    source = property(lambda self: self[0])
    admin = property(lambda self: self[1])
    labels = property(lambda self: self[2])
    def case(self, case_label):
        need(label(case_label) in self.labels, 'fixed selected case')
        return CaseLayout(self.source, case_label)
    def admin_role(self, role):
        need(role in ('G', 'M', 'K', 'C2', 'queue', 'control', 'retention'), 'fixed administrative role')
        return self.admin / role
