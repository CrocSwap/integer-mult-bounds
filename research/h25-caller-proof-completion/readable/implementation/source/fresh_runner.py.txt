def digest(raw):
    return hashlib.sha256(raw).hexdigest()


def inventory(stage):
    rows = {}
    for path in sorted(stage.rglob('*')):
        if path.is_symlink():
            raise ValueError('Symlink in immutable stage')
        if path.is_file() and path.name != 'STAGE.json':

def group_rss(group):
    total = 0
    for path in Path('/proc').iterdir():
        if not path.name.isdigit():
            continue
        try:
            raw = (path / 'stat').read_text()
            fields = raw[raw.rfind(')') + 2:].split()
            if int(fields[2]) == group:
                total += int(fields[21]) * os.sysconf('SC_PAGE_SIZE')
        except (FileNotFoundError, ProcessLookupError, PermissionError):
            pass
    return total

def _raw_open(path, flags, role):
    state=_WRITER_STATE
    if len(state['fds'])>=64 or state['overflow'] is not None or any(r['state']=='UNKNOWN' for r in state['fds']):
        raise RuntimeError('runner raw descriptor custody unavailable')
    row={'role':role,'path':path,'object':None,'state':'PENDING','primary':None,'secondary':None,'close_attempted':False}
    state['fds'].append(row)
    try:
        fd=os.open(path,flags)
    except BaseException as error:
        row['primary'],row['state']=error,'UNKNOWN'
        _remember_runner(error)
        raise
    row['object'],row['state']=fd,'OWNED'
    return row


def _raw_close(row):
    if row['state']!='OWNED':
        raise RuntimeError('one known runner raw descriptor retirement')
    fd,row['object'],row['state'],row['close_attempted']=row['object'],None,'CLOSING',True
    try:
        os.close(fd)
    except BaseException as error:
        row['secondary'],row['state']=error,'UNKNOWN'
        _remember_runner(error)
        raise
    row['state']='CLOSED'


def stream_log(path):
    """Original bounded complete snapshot; safe known-FD error ordering."""
    path_before=path.stat(follow_symlinks=False)
    fd,primary,secondary,traceback=None,None,None,None
    ticket=None
    hasher,count=hashlib.sha256(),0
    before=after=None
    try:
        ticket=_raw_open(path,os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK|os.O_CLOEXEC,'retained-log-hash')
        fd=ticket['object']
        before=os.fstat(fd)
        if not stat.S_ISREG(before.st_mode):
            raise ValueError('Retained log must be a regular file')
        while count<before.st_size:
            block=os.read(fd,min(LOG_CHUNK_BYTES,before.st_size-count))
            if not block:
                break
            hasher.update(block);count+=len(block)
        after=os.fstat(fd)
    except BaseException as error:
        primary=error
        traceback=BaseException.__dict__['__traceback__'].__get__(error,BaseException)
        _remember_runner(error)
    finally:
        if fd is not None:
            closing_fd,fd=fd,None
            try:
                _raw_close(ticket)
            except BaseException as error:
                secondary=error
                _remember_runner(error)
                _WRITER_STATE['overflow']=error
    if primary is not None:
        raise BaseException.with_traceback(primary,traceback)
    if secondary is not None:
        raise secondary
    path_after=path.stat(follow_symlinks=False)
    stable=(log_identity(path_before)==log_identity(before)==log_identity(after)==log_identity(path_after) and count==before.st_size)
    return {'sha256':hasher.hexdigest(),'bytes':count,'snapshot_bytes':before.st_size,
            'final_bytes':path_after.st_size,'stable':stable}


def finalize_logs(folder, stop):
    """Check sizes before any log read, stream complete stable evidence, recheck."""
    paths = [folder / 'stdout.txt', folder / 'stderr.txt']
    before = [path.stat(follow_symlinks=False) for path in paths]
    cap_seen = 'process_monitor' if stop == 'output_cap' else None
    if max(value.st_size for value in before) > OUTPUT_CAP:
        stop = stop or 'output_cap'
        cap_seen = cap_seen or 'before_stream_hash'
    records = [stream_log(path) for path in paths]
    final = [path.stat(follow_symlinks=False) for path in paths]
    if max(value.st_size for value in final) > OUTPUT_CAP:
        stop = stop or 'output_cap'
        cap_seen = cap_seen or 'after_stream_hash'
    for initial, record, ending in zip(before, records, final):
        record['final_bytes'] = ending.st_size
        record['stable'] = (record['stable'] and
                            log_identity(initial) == log_identity(ending) and
                            record['bytes'] == ending.st_size)
    if not all(record['stable'] for record in records):
        stop = stop or 'output_changed_during_hash'
    return records[0], records[1], stop, cap_seen


def read_usage(path):
    """The time tool's small diagnostic must not reintroduce an unbounded read."""
    if not path.exists():
        return None, False
    size = path.stat().st_size
    if size > USAGE_METADATA_CAP:
        return {'raw_omitted': True, 'bytes': size}, True
    with _owned_stream(path,'rb') as stream:
        raw = stream.read(USAGE_METADATA_CAP + 1)
    if len(raw) > USAGE_METADATA_CAP or path.stat().st_size > USAGE_METADATA_CAP:
        return {'raw_omitted': True, 'bytes': path.stat().st_size}, True
    try:
        return json.loads(raw), False
    except (json.JSONDecodeError, UnicodeDecodeError):
        return {'raw': raw.decode('utf-8', errors='replace')}, False


def nested_plain(value, depth=0):
    if depth > 16:
        raise ValueError('nested intent depth')
    kind = type(value)
    if kind is dict:
        if len(value) > 64:
            raise ValueError('nested intent object cap')
        for key,item in value.items():
            if type(key) is not str or len(key) > 128:
                raise ValueError('nested intent exact string key')
            nested_plain(item,depth+1)
    elif kind is list:
        if len(value) > 128:
            raise ValueError('nested intent list cap')
        for item in value:

        if len(value) > 2048:
            raise ValueError('nested intent string cap')
    elif kind is int:
        if not 0 <= value <= 9223372036854775807:
            raise ValueError('nested intent integer bound')
    elif value is not None and kind is not bool:
        raise ValueError('nested intent exact builtin DATA')

def read_pinned(path, size, sha, mode=None):
    before = path.lstat()
    if (path.resolve(strict=True) != path or not stat.S_ISREG(before.st_mode) or
            before.st_uid not in (0,os.getuid()) or before.st_nlink != 1 or before.st_size != size or
            (mode is not None and stat.S_IMODE(before.st_mode) != mode)):
        raise ValueError('canonical selected current-owner native/source inode')
    with _owned_stream(path,'rb') as stream:
        if log_identity(os.fstat(stream.fileno())) != log_identity(before):
            raise ValueError('opened joined source/image identity')
        raw = stream.read(size+1)
        if (len(raw) != size or digest(raw) != sha or
                log_identity(os.fstat(stream.fileno())) != log_identity(before) or
                log_identity(path.lstat()) != log_identity(before)):
            raise ValueError('complete stable selected source/native bytes')
    return raw

def _proc_read(path):
    """One pre-registered reusable read slot; only known-close permits reuse."""
    row=_WRITER_STATE['proc_slot']
    if row['state'] not in ('FREE','CLOSED','NOT_ENTERED'):
        raise RuntimeError('proc census descriptor unknown; no retry')
    row['path'],row['state']=path,'PENDING'
    primary,secondary,traceback,raw=None,None,None,None
    try:
        try:
            fd=os.open(path,os.O_RDONLY|os.O_CLOEXEC|os.O_NOFOLLOW|os.O_NONBLOCK)
        except (FileNotFoundError,ProcessLookupError):
            row['state']='NOT_ENTERED'
            return None
        row['object'],row['state']=fd,'OWNED'
        try:
            raw=os.read(fd,8193)
        except (FileNotFoundError,ProcessLookupError):
            raw=None  # Disappeared proc generation; known descriptor still retires below.
    except BaseException as error:
        primary=error
        traceback=BaseException.__dict__['__traceback__'].__get__(error,BaseException)
        row['primary']=error
        if row['state']=='PENDING':
            row['state']='UNKNOWN'
        _remember_runner(error)
    finally:
        if row['state']=='OWNED':
            try:
                _raw_close(row)
            except BaseException as error:
                secondary=error
                _remember_runner(error)
    if primary is not None:
        raise BaseException.with_traceback(primary,traceback)
    if secondary is not None:
        raise secondary
    return raw

            positive_receipt_present is False and stop_present is False and
            capture_complete is True)


def main():

            raise ValueError('Actual parent-created active nested namespace required')
        if 'entry' in job:

        usage_info = os.fstat(usage_fd)

            raise ValueError('exclusive owned empty regular0600 usage inode')

                                           start_new_session=True,close_fds=True)

                _WRITER_STATE['launch_returned'] = True
            except BaseException as error:

                launch_traceback=BaseException.__dict__['__traceback__'].__get__(error,BaseException)

                    except BaseException as error:

                            launch_error=error
                            launch_traceback=BaseException.__dict__['__traceback__'].__get__(error,BaseException)
            if launch_error is not None:
                raise BaseException.with_traceback(launch_error,launch_traceback)
            while _poll_child(process) is None:
                relay.pump(0.01)
                peak = max(peak,group_rss(process.pid))
                if peak > RSS_CAP:
                    stop = 'sampled_group_rss_cap'
                elif time.monotonic_ns() > nested_deadline_ns:
                    stop = 'wall_or_aggregate_cap'
                elif relay.overflowed():
                    stop = 'output_cap'
                if stop:
                    break
            if stop:
                try:
                    os.killpg(process.pid,signal.SIGKILL)
                except ProcessLookupError:
                    pass
            cleanup_deadline=min(time.monotonic()+2.0,intent['whole_body_deadline_ns']/1000000000)
            code = _wait_child(process,relay,cleanup_deadline)
        except BaseException as error:
            _remember_runner(error)
            raise
        finally:
            cleanup_first = None
            if process is not None:
                try:
                    if _WRITER_STATE['wait_state'] in ('READY','SAFE_NONTERMINAL'):
                        os.killpg(process.pid,signal.SIGKILL)
                except ProcessLookupError:
                    pass
                except BaseException as error:
                    cleanup_first = error
                    _remember_runner(error)
                try:
                    if cleanup_deadline is None:
                        cleanup_deadline=min(time.monotonic()+2.0,intent['whole_body_deadline_ns']/1000000000)

_WRITER_STATE = {'folder':None,'launch_entered':False,'launch_returned':False,
                 'process':None,'wait_known':False,'streams_closed':False,'first':None,'secondary':None,
                 'errors':[None]*32,'overflow':None,'handles':[],'wait_state':'READY','raw_wait':None,
                 'wait_calls':0,'wait_error':None,'fds':[],'modules':[],'relay':None,'relay_receipt':None,
                 'proc_slot':{'object':None,'state':'FREE','path':None,'primary':None,'secondary':None,'close_attempted':False},
                 'process_group_empty':False,'usage_ticket':None,'usage_identity':None,'usage_inode_retired':False}


def _remember_runner(error):
    if _WRITER_STATE['first'] is None:
        _WRITER_STATE['first'] = error
    for item in _WRITER_STATE['errors']:
        if item is error:
            return
    for index,item in enumerate(_WRITER_STATE['errors']):
        if item is None:
            _WRITER_STATE['errors'][index] = error
            return
    if _WRITER_STATE['overflow'] is None:
        _WRITER_STATE['overflow'] = error


@contextmanager
def _owned_stream(path,mode):
    if len(_WRITER_STATE['handles']) >= 32 or _WRITER_STATE['overflow'] is not None or any(row[1]=='UNKNOWN' for row in _WRITER_STATE['handles']):
        raise RuntimeError('runner IO custody unavailable; no new acquisition')
    row = [None,'PENDING',None,None]
    _WRITER_STATE['handles'].append(row)  # Register BEFORE acquisition.
    primary,secondary,traceback = None,None,None
    try:
        stream = path.open(mode)
        row[0],row[1] = stream,'OWNED'
        yield stream
    except BaseException as error:
        primary = error
        traceback = BaseException.__dict__['__traceback__'].__get__(error,BaseException)
        row[2] = error
        _remember_runner(error)
        if row[0] is None:
            row[1] = 'UNKNOWN'
    finally:
        if row[1]=='OWNED':
            closing,row[0],row[1] = row[0],None,'CLOSING'
            try:
                closing.close()
            except BaseException as error:
                secondary,row[3],row[1] = error,error,'UNKNOWN'
                _remember_runner(error)
            else:
                row[1] = 'CLOSED'
    if primary is not None:
        raise BaseException.with_traceback(primary,traceback)
    if secondary is not None:
        raise secondary


def _runner_write(path,raw):
    if type(raw) is not bytes or len(raw)>4194304:
        raise ValueError('bounded runner metadata bytes')
    with _owned_stream(path,'xb') as stream:
        position = 0
        while position<len(raw):
            count=stream.write(raw[position:position+65536])
            if type(count) is not int or count<=0:
                raise ValueError('runner complete metadata write')
            position+=count
        stream.flush();os.fsync(stream.fileno())


def _poll_child(process):
    state = _WRITER_STATE
    if state['wait_state']=='COMMITTED':
        return process.returncode
    if state['wait_state']=='UNRESOLVED' or state['wait_calls']>=65536:
        raise RuntimeError('runner native wait unresolved; no retry')
    state['wait_state']='UNRESOLVED';state['wait_calls']+=1
    try:
        actual=os.waitpid(process.pid,os.WNOHANG)
        state['raw_wait']=actual  # Actual tuple BEFORE comparison/decoder.
        if type(actual) is not tuple or len(actual)!=2 or any(type(v) is not int for v in actual):
            raise RuntimeError('runner native wait tuple')
        if actual==(0,0):
            state['wait_state']='SAFE_NONTERMINAL'
            return None
        if actual[0]!=process.pid:
            raise RuntimeError('runner waited actual child differs')
        state['wait_state']='CONSUMED'
        code=os.waitstatus_to_exitcode(actual[1])
        if type(code) is not int:
            raise RuntimeError('runner wait decoder')
        process.returncode=code
        state['wait_known']=True;state['wait_state']='COMMITTED'
        return code
    except BaseException as error:
        state['wait_error']=error
        state['wait_state']='UNRESOLVED'
        _remember_runner(error)
        raise


def _wait_child(process,relay=None,deadline=None):
    if deadline is None:
        deadline=time.monotonic()+2.0
    while True:
        code=_poll_child(process)
        if code is not None:
            return code
        if time.monotonic()>=deadline:
            raise RuntimeError('runner reserved native cleanup wait exhausted')
        if relay is not None and relay.primary is None:
            relay.pump(0.01)
        else:
            time.sleep(0.01)

def _owned_main():
    try:
        return main()
    except BaseException as error:
        _remember_runner(error)
        raise
    finally:
        state = _WRITER_STATE
        relay = state['relay']
        # Any source/setup failure still retires every known acquired endpoint once.
        if relay is not None and not relay.retired:
            for action in ('writers','relay'):
                try:
                    if action=='writers':
                        relay.close_writers()
                    else:
                        relay.retire()
                except BaseException as error:
                    _remember_runner(error)
        if relay is not None:
            state['relay_receipt'] = relay.snapshot()
        for ticket in state['fds']:
            if ticket['state'] == 'OWNED':
                try:
                    _raw_close(ticket)
                except BaseException as error:
                    _remember_runner(error)
        folder = state['folder']
        if folder is not None and state['overflow'] is None and all(row[1]=='CLOSED' for row in state['handles']):
            try:
                relay_receipt = state['relay_receipt']
                local_fd_closed = (all(row['state']=='CLOSED' for row in state['fds']) and
                    state['proc_slot']['state'] in ('FREE','CLOSED','NOT_ENTERED'))
                no_birth = not state['launch_entered']
                closed = local_fd_closed and (no_birth or (state['launch_returned'] and state['wait_known'] and
                    state['usage_inode_retired'] and relay_receipt is not None and
                    relay_receipt['all_descriptors_known_closed']))
                value = {'schema':'h25-runner-native-writer-closure-v2',
                    'launch_entered':state['launch_entered'],'launch_returned':state['launch_returned'],
                    'wait_known':state['wait_known'],'stdio_aliases_closed':local_fd_closed,
                    'process_group_empty':state['process_group_empty'],
                    'usage_inode_retired':state['usage_inode_retired'],
                    'local_holders_closed':closed,'domain_reconciliation_required':True,'relay':relay_receipt,
                    'error_type':None if state['first'] is None else 'opaque-caught-exception'}
                _runner_write(folder/'WRITER-CLOSURE.json',(json.dumps(value,indent=2,sort_keys=True)+'\n').encode('ascii'))
            except BaseException as secondary:
                state['secondary'] = secondary
                _remember_runner(secondary)

if __name__ == '__main__':
    _owned_main()
