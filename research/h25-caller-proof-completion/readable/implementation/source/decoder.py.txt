class Rejected(ValueError):
    def __init__(self,code,offset=0):
        self.code,self.offset=code,offset
        super().__init__(code+'@'+str(offset))

def require(condition,code):

class Reader:
    def __init__(self,raw,limit=MAX_WIRE,node_limit=MAX_NODES,depth_limit=MAX_DEPTH):
        require(type(raw) is bytes,'INPUT_TYPE')
        require(type(limit) is int and 0<limit<=MAX_WIRE,'LIMIT_DOMAIN')
        require(type(node_limit) is int and 0<node_limit<=MAX_NODES,'LIMIT_DOMAIN')
        require(type(depth_limit) is int and 0<=depth_limit<=MAX_DEPTH,'LIMIT_DOMAIN')
        require(0<len(raw)<=limit,'WIRE_BOUND')
        self.raw,self.at,self.nodes=raw,0,0
        self.node_limit,self.depth_limit=node_limit,depth_limit
    def fail(self,code):raise Rejected(code,self.at)
    def number(self,end,signed=False):
        start=self.at
        stop=self.raw.find(bytes((end,)),start,min(len(self.raw),start+42))
        if stop<0:self.fail('NUMBER_TERMINATOR')
        token=self.raw[start:stop]
        negative=signed and token[:1]==b'-'
        digits=token[1:] if negative else token
        if not digits or any(c<48 or c>57 for c in digits):self.fail('NUMBER_SYNTAX')
        if len(digits)>1 and digits[0]==48:self.fail('NUMBER_CANONICAL')
        if negative and digits==b'0':self.fail('NUMBER_CANONICAL')
        value=int(token)
        if value.bit_length()>MAX_INT_BITS:self.fail('INTEGER_BOUND')
        self.at=stop+1
        return value
    def item(self,depth):
        if depth>self.depth_limit:self.fail('DEPTH_BOUND')
        self.nodes+=1
        if self.nodes>self.node_limit:self.fail('NODE_BOUND')
        if self.at>=len(self.raw):self.fail('TRUNCATED')
        tag=self.raw[self.at];self.at+=1
        if tag==110:return None
        if tag==121:return True
        if tag==122:return False
        if tag==105:return self.number(59,True)
        if tag in (115,98):
            length=self.number(58)
            # A UTF8 string can occupy four bytes per admitted code point.
            cap=MAX_ATOM*4 if tag==115 else MAX_ATOM
            if length>cap:self.fail('ATOM_BOUND')
            stop=self.at+length
            if stop>len(self.raw):self.fail('TRUNCATED_ATOM')
            value=self.raw[self.at:stop];self.at=stop
            if tag==98:return value
            try:value=value.decode('utf-8','strict')
            except UnicodeDecodeError:self.fail('UTF8')
            if len(value)>MAX_ATOM:self.fail('STRING_BOUND')
            return value
        if tag not in (116,108,100):self.fail('TAG')
        count=self.number(58)
        if count>MAX_CONTAINER:self.fail('CONTAINER_BOUND')
        if tag==100:
            result={};previous=None
            for unused in range(count):
                key=self.item(depth+1)
                if type(key) is not str:self.fail('DICT_KEY_TYPE')
                if previous is not None and key<=previous:self.fail('DICT_KEY_ORDER')
                previous=key
                result[key]=self.item(depth+1)
            return result
        values=[]
        for unused in range(count):values.append(self.item(depth+1))
        return tuple(values) if tag==116 else values
    def read(self):
        result=self.item(0)
        if self.at!=len(self.raw):self.fail('TRAILING')
        return result

def decode(raw,limit=MAX_WIRE,node_limit=MAX_NODES,depth_limit=MAX_DEPTH):
    return Reader(raw,limit,node_limit,depth_limit).read()

def dimensions(value):
    """Independent recursive occurrence census and canonical-length formula.

    This emits no encoded bytes and calls no source encoder. Aliased native
    containers, if supplied by fixture authors, count once per occurrence.
    """
    out=dict(nodes=0,max_depth=0,max_container=0,max_bytes_atom=0,
        max_string_chars=0,string_utf8_bytes=0,integer_occurrences=0,
        max_integer_bits=0,tuple_occurrences=0,list_occurrences=0,
        dict_occurrences=0,canonical_bytes_observed=0)
    active=set()
    def visit(x,depth):
        require(depth<=MAX_DEPTH,'DIM_DEPTH')
        out['nodes']+=1;require(out['nodes']<=MAX_NODES,'DIM_NODES')
        out['max_depth']=max(out['max_depth'],depth);k=type(x)
        if x is None or k is bool:size=1
        elif k is int:
            require(x.bit_length()<=MAX_INT_BITS,'DIM_INTEGER')
            out['integer_occurrences']+=1
            out['max_integer_bits']=max(out['max_integer_bits'],x.bit_length())
            size=len(str(x))+2
        elif k is bytes:
            require(len(x)<=MAX_ATOM,'DIM_BYTES');out['max_bytes_atom']=max(out['max_bytes_atom'],len(x));size=2+len(str(len(x)))+len(x)
        elif k is str:
            require(len(x)<=MAX_ATOM,'DIM_STRING');n=len(x.encode('utf-8'))
            out['max_string_chars']=max(out['max_string_chars'],len(x));out['string_utf8_bytes']+=n;size=2+len(str(n))+n
        else:
            require(k in (tuple,list,dict),'DIM_TYPE');require(len(x)<=MAX_CONTAINER and id(x) not in active,'DIM_CONTAINER')
            active.add(id(x));out['max_container']=max(out['max_container'],len(x))
            out[{tuple:'tuple_occurrences',list:'list_occurrences',dict:'dict_occurrences'}[k]]+=1
            size=2+len(str(len(x)))
            if k is dict:
                require(all(type(key) is str for key in x),'DIM_KEY')
                for key in sorted(x):visit(key,depth+1);visit(x[key],depth+1)
            else:
                for child in x:visit(child,depth+1)
            active.remove(id(x))
        out['canonical_bytes_observed']+=size
        require(out['canonical_bytes_observed']<=MAX_WIRE,'DIM_WIRE')
    visit(value,0)
    return out

def exact(left,right):
    """Type-sensitive comparison; bool/int and tuple/list stay distinct."""
    if type(left) is not type(right):return False
    if type(left) in (tuple,list):return len(left)==len(right) and all(exact(a,b) for a,b in zip(left,right))
    if type(left) is dict:return set(left)==set(right) and all(exact(left[k],right[k]) for k in left)
    return left==right

def canonical_parts(value):
    """Fresh streaming literal grammar construction for hash/control witnesses.

    Not the producer encoder and not the decoder. Literal known-byte controls
    separately pin these formulas; a roundtrip is never the sole oracle.
    """
    nodes=[0];total=[0]
    def walk(x,depth):
        nodes[0]+=1;require(nodes[0]<=200000 and depth<=16,'HASH_BOUND');k=type(x)
        if x is None:yield b'n'
        elif k is bool:yield b'y' if x else b'z'
        elif k is int:
            require(-(2**64-1)<=x<=2**64-1,'HASH_INTEGER');yield b'i'+str(x).encode('ascii')+b';'
        elif k in (str,bytes):
            raw=x.encode('utf8') if k is str else x;require(len(raw)<=65536,'HASH_ATOM');yield (b's' if k is str else b'b')+str(len(raw)).encode('ascii')+b':';yield raw
        elif k in (tuple,list,dict):
            require(len(x)<=(4096 if k is dict else 16384),'HASH_CONTAINER');yield {tuple:b't',list:b'l',dict:b'd'}[k]+str(len(x)).encode('ascii')+b':'
            if k is dict:
                require(all(type(n) is str for n in x),'HASH_KEY')
                for key in sorted(x):yield from walk(key,depth+1);yield from walk(x[key],depth+1)
            else:
                for child in x:yield from walk(child,depth+1)
        else:raise Rejected('HASH_TYPE')
    for piece in walk(value,0):
        total[0]+=len(piece);require(total[0]<=16777216,'HASH_WIRE');yield piece

def canonical_hash(value):
    digest=hashlib.sha256()
    for piece in canonical_parts(value):digest.update(piece)
    return digest.hexdigest()

        elif self.overflow[0] is None:self.overflow[0]=error
    def open(self,slot,path,flags,mode=0,dir_fd=None):
        cell=self.slots[slot]
        require(cell[0] is None and not cell[1] and not cell[2] and not cell[3],'IO_SLOT_NOT_REUSABLE')

        except BaseException as error:

            # Never infer/reconstruct/retry it; parent teardown owns this gap.
            cell[2]=True;self.remember(error,'open');raise

        return cell[0]
    def close(self,slot):
        cell=self.slots[slot];fd=cell[0]

        cell[0]=None;cell[3]=True

    def cleanup(self):
        # No failed close is retried; other known descriptors still get a turn.
        for slot in reversed(range(8)):self.close(slot)
    def raise_first(self):
        if self.first[0] is not None:raise self.first[0]
