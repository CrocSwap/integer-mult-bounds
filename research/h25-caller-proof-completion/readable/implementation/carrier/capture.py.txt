def slot():

def acquire(path, flags, mode=0o600):
    s = slot()
    s[4] = path
    PENDING[0] = True
    try:
        s[0] = os.open(path, flags | os.O_NOFOLLOW | os.O_CLOEXEC, mode)
    except BaseException as error:
        s[1] = error
        UNKNOWN[0] = True
        raise
    PENDING[0] = False
    return s


def close(s):

def read_new(path, cap, expected=None, expected_mode=None):
    s = acquire(path, os.O_RDONLY | os.O_NONBLOCK)
    try:
        before = os.fstat(s[0])
        need(stat.S_ISREG(before.st_mode) and before.st_uid == os.geteuid() and
             before.st_nlink == 1 and before.st_size <= cap, 'new regular retained input')
        need(expected_mode is None or stat.S_IMODE(before.st_mode) == expected_mode, 'expected immutable mode')
        chunks, count = [], 0
        while True:
            need(count <= cap, 'input byte cap')
            block = os.read(s[0], min(65536, cap + 1 - count))
            if not block:
                break
            chunks.append(block)
            count += len(block)
        raw = b''.join(chunks)
        need(count == before.st_size and ident(before) == ident(os.fstat(s[0])) and
             ident(before) == ident(os.stat(path, follow_symlinks=False)), 'current input changed')
        if expected is not None:
            need(hashlib.sha256(raw).hexdigest() == expected, 'source/metadata hash differs')
        return raw
    except BaseException as error:
        s[1] = error
        remember(error)
        raise
    finally:
        close(s)
        if s[1] is None and UNKNOWN[0]:
            raise FIRST[0]


def fresh(path):
    s = acquire(path, os.O_RDWR | os.O_CREAT | os.O_EXCL)
    st = os.fstat(s[0])
    need(stat.S_ISREG(st.st_mode) and st.st_uid == os.geteuid() and
         stat.S_IMODE(st.st_mode) == 0o600 and st.st_nlink == 1 and st.st_size == 0,
         'fresh exclusive private output')
    s[4] = [path, st.st_dev, st.st_ino]
    return s


def append(s, raw):
    at = 0
    while at < len(raw):
        n = os.write(s[0], raw[at:at+65536])
        need(type(n) is int and 0 < n <= min(65536, len(raw)-at), 'progressing metadata write')
        at += n
    os.fsync(s[0])

def seal_capture(s, cap):
    need(not UNKNOWN[0], 'uncertain IO blocks capture processing')
    st = os.fstat(s[0])
    path, device, inode = s[4]
    need(stat.S_ISREG(st.st_mode) and st.st_uid == os.geteuid() and st.st_nlink == 1 and
         (st.st_dev, st.st_ino) == (device, inode) and st.st_size <= cap,
         'actual captured output identity/size')
    os.fsync(s[0])
    os.lseek(s[0], 0, os.SEEK_SET)
    chunks, count = [], 0
    while True:
        need(count <= cap, 'capture cap')
        block = os.read(s[0], min(65536, cap+1-count))
        if not block:
            break
        chunks.append(block)
        count += len(block)
    raw = b''.join(chunks)
    need(count == st.st_size and ident(st) == ident(os.fstat(s[0])) and
         ident(st) == ident(os.stat(path, follow_symlinks=False)), 'actual complete capture changed')
    os.fchmod(s[0], 0o400)
    os.fsync(s[0])
    pin = {'bytes':count, 'sha256':hashlib.sha256(raw).hexdigest(), 'mode':0o400,
           'path':path, 'device':device, 'inode':inode}
    close(s)
    need(not UNKNOWN[0], 'capture close uncertain')
    return raw, pin

def _ready_pipe():
    a,b = slot(),slot()
    # Raw descriptors held before projection; ambiguity forbids seal/retry.
    try:
        pair = os.pipe2(os.O_CLOEXEC | os.O_NONBLOCK)
        a[0], b[0] = pair
        a[4], b[4] = ['ready-read',0,0],['ready-write',0,0]
        need(199 not in pair, 'reserved readiness descriptor collision')
        c = slot()
        c[0] = os.dup2(b[0],199,inheritable=True)
        c[4] = ['ready-199',0,0]
        close(b)
        return a,c
    except BaseException as error:
        UNKNOWN[0] = True
        remember(error)
        raise


def _await_ready(s, deadline):
    try:
        while True:
            left=(deadline-time.monotonic_ns())/NS
            need(left>0, 'fixed child registration deadline')
            ready,unused,failed=select.select([s[0]],[],[s[0]],min(left,0.025))
            need(not failed, 'readiness pipe failure')
            if ready:
                need(os.read(s[0],2)==b'R','exact child private registration acknowledgement')
                break
    except BaseException as error:
        remember(error)
        raise
    finally:
        close(s)

def partial_clock_observation(phase):
    """Local partial window only; neither process authority nor a final endpoint."""
    result = {'clock_observation_status':'UNAVAILABLE',
              'clock_observation_phase':phase,
              'wrapper_start_monotonic_ns':START[0],
              'observation_monotonic_ns':None,
              'wrapper_elapsed_ns_through_observation':None,
              'clock_window_scope':'run_entry_to_local_sample_only',
              'imports_setup_before_run_and_publication_close_exit_excluded':True,
              'complete_wrapper_or_workflow_cost':False}
    if (JOURNAL[0] is None or JOURNAL[0][3] is not False or
            UNKNOWN[0] or PENDING[0]):
        return result
    try:
        observed = time.monotonic_ns()
        need(type(START[0]) is int and type(observed) is int and
             0 <= START[0] <= observed, 'valid local partial clock observation')
        elapsed = observed - START[0]
        result.update(clock_observation_status='OBSERVED_PARTIAL_WINDOW',
                      observation_monotonic_ns=observed,
                      wrapper_elapsed_ns_through_observation=elapsed)
    except BaseException as secondary:
        if FIRST[0] is None:
            # At most one pending error: target/fee success gates settle before
            # another invoke; quarantine samples only after remember(original).
            OBSERVATION_ERROR[0] = secondary
        else:
            remember(secondary)
    return result

        manual_start = F.decode(manual_start_raw)
        need(manual_start.get('schema') == 'fresh-owner-meter-incomplete-v1' and
             type(manual_start.get('start_monotonic_ns')) is int and
             0 <= manual_start['start_monotonic_ns'] <= START[0] and
             manual_start.get('source_sha256') == cfg['source_pins']['outer']['sha256'] and
             manual_start.get('probe_sha256') == cfg['source_pins']['capture_bootstrap']['sha256'],
             'selected same-clock-domain genuine outer START, no arbitrary numeric argv')
        outer_begin_ns = manual_start['start_monotonic_ns']
        # Native/actual authentication of manual pre-entry remains the separately
        # admitted top-owner/fixed-bootstrap premise; DATA is not that authority.
        need(RUNTIME_BOUND[0], 'fixed prospective admin paths required')
        markers_clear('PREPARE')
        os.mkdir(CAPTURE,0o700)
        JOURNAL[0] = fresh(CAPTURE+'/CAPTURE.jsonl')
        sync_parent(CAPTURE); sync_parent(CAPTURE+'/CAPTURE.jsonl')
        event({'phase':'CAPTURE_PREPARATION_STARTED','current_preentry':before,
            'capture_local_start_ns':START[0],'clock_scope':'NESTED_CAPTURE_DIAGNOSTIC_NOT_ADDITIVE',
            'complete_outer_body_observed_here':False,'old_ledger_or_frontier_writes':0})
        own = cfg['source_pins']['capture']
        need(own['path'] == os.path.abspath(__file__) and own['sha256'] == source_sha and own['mode'] == 0o444,
             'actual capture source/pin/frozen mode')
        need(len(read_new(own['path'],131072,source_sha,0o444)) == own['bytes'],'capture source byte count')
        # Source/helper bytes retained before the target lease lifetime. There
        # are no history/source reloads after controller release.
        for key in ('publisher','publisher_bootstrap','C','F','A'):
            row = cfg['source_pins'][key]
            need(len(read_new(row['path'],131072,row['sha256'],row['mode'])) == row['bytes'],
                 'selected source closure byte count')
        row = cfg['source_pins']['fee']
        fee_source = read_new(row['path'],65536,row['sha256'],row['mode']).decode('utf-8')
        need(len(fee_source.encode('utf-8')) == row['bytes'],'exact retained fee source')
        grant = F.decode(read_new(GRANT,131072,grant_sha,0o444))
        F.exact(grant,('schema','name','candidate_document_sha256','capture_source_sha256',
            'current_profile_sha256','source_policy_sha256','resource_policy_sha256',
            'body_reservation_ns','complete_delivery_scope_sha256','allow_one_attempt'))
        fixed = {'schema':'fresh-composed-one-use-capture-grant-v1','name':NAME,
            'candidate_document_sha256':cfg['candidate_document_sha256'],'capture_source_sha256':source_sha,
            'current_profile_sha256':cfg['current_profile_pin']['sha256'],
            'source_policy_sha256':cfg['source_policy_pin']['sha256'],
            'resource_policy_sha256':cfg['resource_policy_pin']['sha256'],
            'body_reservation_ns':cfg['body_reservation_ns'],
            'complete_delivery_scope_sha256':cfg['complete_delivery_scope_sha256'],'allow_one_attempt':True}
        need(all(type(grant[k]) is type(v) and grant[k] == v for k,v in fixed.items()),
             'separate genuine current one-use exact command/funding grant, never self-issued')
        captures = [fresh(OUT),fresh(ERR)]
        remaining_ns = cfg['body_reservation_ns']-(time.monotonic_ns()-outer_begin_ns)
        need(remaining_ns > cfg['cleanup_reservation_ns'],'no nested dispatch after reserved body exhaustion')
        target = invoke(list(cfg['publisher_argv']),captures,(remaining_ns-cfg['cleanup_reservation_ns'])/NS)
        out,out_pin = seal_capture(captures[0],2097152)
        err,err_pin = seal_capture(captures[1],2097152)
        sync_parent(OUT)
        event({'phase':'TARGET_CAPTURE_RETAINED','native_returncode':target[3],
            'timed_out':target[4],'captures':[out_pin,err_pin],
            'native_wait_not_full_descendant_cleanup':True})
        need(target[3] == 0 and target[4] is False,'original command failed/late; no new helper after failure')
        settle_observation_error(True)
        markers_clear('POST_TARGET')
        # New controller metadata only; no history/source/original ledger reopen.
        accounting = read_new(ACCOUNTING,4194304)
        metadata = json.loads(accounting,parse_float=Decimal,
            parse_constant=lambda text: (_ for _ in ()).throw(ValueError('nonfinite controller metadata')))
        need(metadata.get('schema') == 'fresh-composed-controller-observation-v1' and
             metadata.get('all_costs_settled') is False,'exact source-bound controller observation')
        diagnostic = {'schema':'fresh-composed-capture-retained-diagnostic-v1','name':NAME,
            'candidate_document_sha256':cfg['candidate_document_sha256'],'native_target_returncode':target[3],
            'target_wait_returned':target[3] is not None,'capture_local_start_ns':START[0],
            'target_local_observed_ns':time.monotonic_ns(),'body_reservation_ns':cfg['body_reservation_ns'],
            'administrative_policy':ADMIN_POLICY,'administrative_debit_ns':ADMIN_DEBIT_NS,
            'nested_windows_additive':False,'whole_outer_body_elapsed_ns':None,
            'physical_final_publication':'OPEN_NOT_ZERO','all_costs_settled':False}
        os.mkdir(RETAINED,0o700)
        pins = {}
        for name,raw in zip(('CONTROLLER.json','CAPTURE-DIAGNOSTIC.json','TARGET-STDOUT.raw','TARGET-STDERR.raw'),
                            (accounting,canonical(diagnostic),out,err)):
            p = publish(RETAINED+'/'+name,raw)
            pins[name] = {'bytes':p['bytes'],'sha256':p['sha256']}
        fee_caps = [fresh(CAPTURE+'/FEE.stdout'),fresh(CAPTURE+'/FEE.stderr')]
        fee_argv = ['/usr/bin/python3.12','-S','-s','-P','-B','-c',fee_source,
            '--retained',RETAINED,'--input-pins',canonical(pins).decode('ascii').strip(),
            '--invocation',NAME,'--command-admission-sha256',grant_sha]
        remaining_ns = cfg['body_reservation_ns']-(time.monotonic_ns()-outer_begin_ns)
        need(remaining_ns > cfg['cleanup_reservation_ns'],'fee denied when actual remaining local reserve exhausted')
        fee = invoke(fee_argv,fee_caps,(remaining_ns-cfg['cleanup_reservation_ns'])/NS)
        fee_outputs = [seal_capture(s,2097152)[1] for s in fee_caps]
        sync_parent(CAPTURE+'/FEE.stdout')
        need(fee[3] == 0 and fee[4] is False,'fee helper failure/overrun remains debt')
        settle_observation_error(True)
        candidate = read_new(RETAINED+'/PROVISIONAL-RETAINED.json',65536)
        candidate_pin = {'bytes':len(candidate),'sha256':hashlib.sha256(candidate).hexdigest()}
        need(F.decode(candidate).get('schema') == 'fresh-composed-provisional-retained-DATA-v1',
             'only provisional DATA, no body receipt/CLOSE generated by nested fee')
        event({'phase':'CAPTURE_AND_RETAINED_DATA_READY','candidate_pin':candidate_pin,
            'fee_captures':fee_outputs,'one_enclosing_outer_body_not_additive':True,'all_costs_settled':False})
        status = 'CAPTURE_AND_RETAINED_DATA_READY'
    except BaseException as error:
        remember(error)
        settle_observation_error(False)
        try:
            event({'phase':'CAPTURE_QUARANTINED','first_type':type(FIRST[0]).__name__,
                'unknown_IO_or_wait':UNKNOWN[0] or PENDING[0],
                'actual_remaining_descendant_state':'UNRESOLVED_UNLESS_REAL_EXTERNAL_TEARDOWN',
                'future_operation_blocked':True,'no_automatic_retry':True,'all_costs_settled':False,
                **partial_clock_observation('CAPTURE_QUARANTINED')})
        except BaseException as secondary:
            remember(secondary)
    finally:
        # All known capture descriptors are attempted before the local endpoint.
        for s in SLOTS:
            close(s)
        try:
            usage_after = (resource.getrusage(resource.RUSAGE_SELF),resource.getrusage(resource.RUSAGE_CHILDREN))
        except BaseException as secondary:
            remember(secondary)
        try:
            end_ns = time.monotonic_ns()
            need(type(end_ns) is int and end_ns >= START[0],'genuine local capture endpoints')
        except BaseException as secondary:
            remember(secondary)
    if FIRST[0] is not None or UNKNOWN[0] or PENDING[0]:
        status = 'STOP'
    record = {'schema':'fresh-composed-capture-terminal-v1','body_status':status,
        'capture_local_begin_ns':START[0],'capture_local_end_after_cleanup_ns':end_ns,
        'actual_outer_START_from_original_manual_pin_ns':outer_begin_ns,
        'capture_local_elapsed_ns':end_ns-START[0] if type(end_ns) is int and end_ns >= START[0] else None,
        'one_enclosing_outer_body_not_additive':True,'complete_outer_body_observed_here':False,
        'local_SELF_and_reaped_CHILDREN_rusage_before_after': [
            [[u.ru_utime,u.ru_stime,u.ru_maxrss] for u in values] if values is not None else None
            for values in (usage_before,usage_after)],
        'rusage_is_not_whole_forest_CPU_or_simultaneous_RSS':True,
        'native_waits_all_returned':bool(CHILDREN) and all(r[0] is not None and not r[1] and
            type(r[3]) is int and r[0].returncode == r[3] for r in CHILDREN),
        'native_waits_do_not_prove_full_descendant_disposal':True,
        'unknown_IO_or_wait':UNKNOWN[0] or PENDING[0],
        'first_error_present':FIRST[0] is not None,'candidate_pin':candidate_pin,
        'physical_final_publication':'OPEN_NOT_ZERO','all_costs_settled':False,
        'whole_body_receipt_or_CLOSE_created':False,'gain':False}
    # This inner record/readback/close/exit is INSIDE the actual outer owner wait,
    # never a second administrative debit or a fabricated outer final endpoint.
    try:
        need(stdout_valid,'no diagnostic write without actual inherited capture validation')
        raw = canonical(record)
        offset = 0
        while offset < len(raw):
            written = os.write(1,raw[offset:])
            need(type(written) is int and 0 < written <= len(raw)-offset,'inherited outer capture progress')
            offset += written
        os.fsync(1)
    except BaseException as secondary:
        remember(secondary)
    if GUARDIAN[0] is not None and status == 'CAPTURE_AND_RETAINED_DATA_READY' and FIRST[0] is None and not UNKNOWN[0]:
        try:
            GUARDIAN[0].phase(9,flags=2,start_ns=os.getpid())
        except BaseException as error:
            remember(error)
    return 0 if status == 'CAPTURE_AND_RETAINED_DATA_READY' and FIRST[0] is None and not UNKNOWN[0] else 78






def main():
    p = argparse.ArgumentParser(allow_abbrev=False)
    p.add_argument('--source-sha256',required=True)
    p.add_argument('--grant-sha256',required=True)
    p.add_argument('--before-json',required=True)
    a = p.parse_args()
    need(len(a.before_json) <= 4096,'BEFORE argument cap')
    return run(digest(a.source_sha256), digest(a.grant_sha256), json.loads(a.before_json))


if __name__ == '__main__':
    sys.exit(main())
