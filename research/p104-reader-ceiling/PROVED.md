## CongestionReaderCore: `Tied`, `rho`, `deltaF` (a tied register has residual at least `deltaF`: `tied_rho`, proved)
```lean
def Tied {ι : Type*} (n : ι → ℕ) (d : ι → ℕ → ℕ) (r : ι) : Prop :=
  (d r 0 = 0 ∧ d r (n r) = 8 ∨ d r 0 = 1 ∧ d r (n r) = 9) ∧ ∃ j ≤ n r, 2 ≤ d r j ∧ d r j ≤ 7

noncomputable def rho {ι : Type*} (s : ℝ) (n : ι → ℕ) (d : ι → ℕ → ℕ) (r : ι) : ℝ :=
  ex (gs s) (lev d r) (n r) - ∑ i ∈ range (n r), c2 (gs s) (lev d r i) (lev d r (i + 1))

noncomputable def deltaF (s : ℝ) : ℝ := (7 : ℝ) ^ s + (2 : ℝ) ^ s - (8 : ℝ) ^ s - 1
```

```lean
def lineOKR (rA rB : Row) (capI dI lam gA gB : ℕ) : Bool :=
  (List.range 49).all (fun x => (List.range 49).all (fun r =>
    decide (lam * min 48 (x + r) ≤ gv rA r + x * capI + dI + gA) &&
      decide (lam * min 48 (x + r) ≤ gv rB r + x * capI + dI + gB)))
  && decide (gB ≤ gA) && decide (lam ≤ capI)
```
