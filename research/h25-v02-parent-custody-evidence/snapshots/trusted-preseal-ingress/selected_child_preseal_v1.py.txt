The outer caller must authenticate this immutable source and the supplied
manifest BEFORE import/entry (inherited C04); no hash supplied here grants that
trust. This module has no loader, exec, callback, source mutation, Root API write,
stage write or target launch. It verifies exact already imported source objects,
then installs only a fixed enum of literal gate bindings once, before capture.
"""

class BindingStop(RuntimeError):
    pass


def _need(value, message):
    if not value:
        raise BindingStop(message)


def _exact(value, keys):
    _need(_TYPE(value) is dict and all(_TYPE(k) is str for k in value) and set(value) == set(keys), 'binding exact keys differ')


def _identity(info):
    return {'device': info.st_dev, 'inode': info.st_ino, 'mode': info.st_mode, 'links': info.st_nlink,
            'uid': info.st_uid, 'gid': info.st_gid, 'size': info.st_size,
            'mtime_ns': info.st_mtime_ns, 'ctime_ns': info.st_ctime_ns}


def _entry(context):
    _need(_TYPE(context) is object, 'exact original installation context required')
    for token, state in _INSTALLATIONS:
        if token is context:
            return state
    raise BindingStop('unregistered pre-capture installation context')


def _primitive_bindings_match_v3(values):
    _need(_TYPE(values) is tuple and _LEN(values) == 5, 'exact primitive binding tuple required')
    return all(actual is expected for actual, expected in zip(values, (os.open, os.read, os.close, os.fstat, os.getpid)))


def _fresh(state):
    owner = state['owner']
    _need(parent._UNKNOWN_OWNER is owner and _GETPID() == state['pid'] and state['phase'] != 'STOP' and
          owner['primary'][0] is None and owner['quarantine'][0] is False and
          owner['pending'][0] is False and owner['iterator_uncertain'][0] is False,
          'original installation owner requires teardown')
    _need(_primitive_bindings_match_v3((_OPEN, _READ, _CLOSE, _FSTAT, _GETPID)) and
          _TYPE is builtins.type and _LEN is builtins.len,
          'original installation primitive binding changed')


def _read_pin(state, pin, limit=MAX_IMAGE):
    """Nofollow component descent; known tokens close once, uncertainty latches."""
    _fresh(state)
    _exact(pin, ('path', 'sha256', 'bytes', 'identity'))
    path = pin['path']
    _need(_TYPE(path) is str and path.startswith('/') and len(path) <= 4096 and
          all(c not in ('', '.', '..') for c in path.split('/')[1:]), 'binding pin path differs')
    _digest(pin['sha256'])
    _need(_TYPE(pin['bytes']) is int and 0 <= pin['bytes'] <= limit, 'binding pin byte bound')
    _exact(pin['identity'], ('device', 'inode', 'mode', 'links', 'uid', 'gid', 'size', 'mtime_ns', 'ctime_ns'))
    _need(all(_TYPE(x) is int and x >= 0 for x in pin['identity'].values()) and
          pin['identity']['links'] == 1 and pin['identity']['size'] == pin['bytes'] and
          stat.S_ISREG(pin['identity']['mode']) and not pin['identity']['mode'] & 0o222,
          'immutable binding source pin required')
    primary = close_primary = None
    parts = path.split('/')[1:]
    _need(len(parts) <= 2048, 'source path component bound')
    record = {'fds': [None, None], 'close_fds': [None, None],
              'close_entered': [False, False], 'close_errors': [None, None],
              'walk_fds': [None] * len(parts), 'walk_entered': [False] * len(parts),
              'walk_errors': [None] * len(parts), 'primary': [None]}
    # Record allocation/retention precedes every acquisition owned by this read.
    state['read_records'].append(record)
    state['fds'] = record['fds']
    try:
        state['owner']['pending'][0] = True
        state['fds'][0] = _OPEN('/', os.O_RDONLY | os.O_DIRECTORY | os.O_CLOEXEC | os.O_NOFOLLOW)
        state['owner']['pending'][0] = False
        for walk_index, component in enumerate(parts[:-1]):
            state['owner']['pending'][0] = True
            state['fds'][1] = _OPEN(component, os.O_RDONLY | os.O_DIRECTORY | os.O_CLOEXEC | os.O_NOFOLLOW,
                                   dir_fd=state['fds'][0])
            state['owner']['pending'][0] = False
            old = state['fds'][0]
            record['walk_fds'][walk_index] = old
            record['walk_entered'][walk_index] = True
            state['fds'][0] = None
            try:
                _CLOSE(old)
            except BaseException as error:
                record['walk_errors'][walk_index] = error
                raise
            state['fds'][0] = state['fds'][1]
            state['fds'][1] = None
        state['owner']['pending'][0] = True
        state['fds'][1] = _OPEN(parts[-1], os.O_RDONLY | os.O_CLOEXEC | os.O_NOFOLLOW, dir_fd=state['fds'][0])
        state['owner']['pending'][0] = False
        _need(_identity(_FSTAT(state['fds'][1])) == pin['identity'], 'actual binding file identity differs')
        chunks, count = [], 0
        while True:
            raw = _READ(state['fds'][1], min(65536, pin['bytes'] + 1 - count))
            _need(_TYPE(raw) is bytes, 'native binding read type differs')
            if not raw:
                break
            count += len(raw); _need(count <= pin['bytes'], 'binding source grew')
            chunks.append(raw)
        raw = b''.join(chunks)
        _need(len(raw) == pin['bytes'] and hashlib.sha256(raw).hexdigest() == pin['sha256'] and
              _identity(_FSTAT(state['fds'][1])) == pin['identity'], 'binding source custody changed')
        return raw
    except BaseException as error:
        primary = error
        record['primary'][0] = error
        if state['owner']['primary'][0] is None:
            state['owner']['primary'][0] = error
        state['owner']['quarantine'][0] = True
        state['phase'] = 'STOP'
        raise
    finally:
        descriptor = state['fds'][1]
        record['close_fds'][1] = descriptor
        if descriptor is not None:
            record['close_entered'][1] = True
            state['fds'][1] = None
            try:
                _CLOSE(descriptor)
            except BaseException as error:
                record['close_errors'][1] = error
                if close_primary is None:
                    close_primary = error
                if state['owner']['primary'][0] is None:
                    state['owner']['primary'][0] = error
                state['owner']['quarantine'][0] = True
                state['phase'] = 'STOP'
        descriptor = state['fds'][0]
        record['close_fds'][0] = descriptor
        if descriptor is not None:
            record['close_entered'][0] = True
            state['fds'][0] = None
            try:
                _CLOSE(descriptor)
            except BaseException as error:
                record['close_errors'][0] = error
                if close_primary is None:
                    close_primary = error
                if state['owner']['primary'][0] is None:
                    state['owner']['primary'][0] = error
                state['owner']['quarantine'][0] = True
                state['phase'] = 'STOP'
        if primary is None and close_primary is not None:
            raise close_primary


def _pairs(items):
    _need(len(items) <= 8192, 'binding mapping bound')
    result = {}
    for key, value in items:
        _need(_TYPE(key) is str and len(key) <= 1048576 and key not in result, 'binding duplicate/foreign key')
        result[key] = value
    return result


def _reject(value):
    raise BindingStop('binding noninteger/nonfinite literal rejected')


def _document(raw):
    value = json.loads(raw.decode('ascii'), object_pairs_hook=_pairs, parse_float=_reject, parse_constant=_reject)
    _literal(value, [0])
    _need(_TYPE(value) is dict, 'binding top-level document differs')
    return value


def _namespace(module):
    _need(_TYPE(module) is types.ModuleType, 'exact installed module type required')
    namespace = _MODULE_DICT(module, types.ModuleType)
    _need(_TYPE(namespace) is dict and _LEN(namespace) <= _MAX_GLOBALS, 'bounded exact installed namespace required')
    # Complete key walk BEFORE any membership, lookup, set, hash or equality.
    for key in namespace:
        _need(_TYPE(key) is str and _LEN(key) <= 256, 'foreign installed namespace key')
    for key in ('__name__', '__file__'):
        _need(key in namespace and _TYPE(namespace[key]) is str and 0 < _LEN(namespace[key]) <= 4096,
              'exact installed module metadata required')
    return namespace


def _startup_helpers_unchanged():
    for module, items in _STARTUP_HELPERS:
        namespace = _namespace(module)
        _need(_LEN(namespace) == _LEN(items), 'startup helper namespace census changed')
        for key, expected in items:
            _need(_TYPE(key) is str and namespace.get(key) is expected, 'startup helper binding identity changed')



def _function_record(function):
    _need(_TYPE(function) is types.FunctionType, 'exact source function required')
    values = tuple(getter(function, types.FunctionType) for getter in _FUNCTION_GETTERS)
    closure = values[5]
    cells = () if closure is None else tuple((cell, _CELL_GET(cell, types.CellType)) for cell in closure)
    _need(_LEN(cells) <= 64, 'source function closure bound exceeded')
    return function, values, cells


def _function_check(record):
    function, values, cells = record
    _need(_TYPE(function) is types.FunctionType, 'source function identity changed')
    for getter, expected in zip(_FUNCTION_GETTERS, values):
        _need(getter(function, types.FunctionType) is expected,
                    'source function code/globals/builtins/defaults/closure/dict changed')
    for cell, expected in cells:
        _need(_CELL_GET(cell, types.CellType) is expected, 'source function closure contents changed')


def _codes(code, result):
    result.setdefault(code.co_qualname, []).append(code)
    for child in code.co_consts:
        if _TYPE(child) is types.CodeType:
            _codes(child, result)


def _data(value, depth=0):
    """Snapshot literal data only; no foreign repr/hash/eq/iteration protocols."""
    _need(depth <= 32, 'literal snapshot depth exceeded')
    kind = _TYPE(value)
    if value is None or (kind is str or kind is int or kind is bool or kind is bytes or kind is float):
        return (kind, value)
    if (kind is list or kind is tuple):
        _need(_LEN(value) <= 65536, 'literal sequence bound exceeded')
        return (kind, tuple(_data(item, depth + 1) for item in value))
    if kind is dict:
        _need(_LEN(value) <= 8192, 'literal mapping bound exceeded')
        for key in value:
            _need((_TYPE(key) is str or _TYPE(key) is int), 'nonliteral mapping key')
        return (kind, tuple((_data(key, depth + 1), _data(item, depth + 1))
                            for key, item in value.items()))
    if (kind is set or kind is frozenset):
        _need(_LEN(value) <= 8192 and all((_TYPE(item) is str or _TYPE(item) is int) for item in value),
                    'nonliteral set value')
        return (kind, frozenset(value))
    raise BindingStop('not literal data')


def _validate_code(function, compiled_codes, namespace, source_role, forest):
    """Collect one source-fixed role; only the complete forest can admit code."""
    _need(_TYPE(function) is types.FunctionType, 'exact installed source function required')
    values = tuple(getter(function, types.FunctionType) for getter in _FUNCTION_GETTERS)
    _need(values[1] is namespace and values[2] is builtins.__dict__,
          'source function original namespace/builtins differs')
    _need(_TYPE(source_role) is str, 'exact source code role required')
    candidates = compiled_codes.get(source_role, ())
    _need(_LEN(candidates) == 1, 'source code role absent or ambiguous')
    forest.append((values[0], candidates[0]))
    return None


def _digest(value):
    _need(type(value) is str and len(value) == 64 and all(x in '0123456789abcdef' for x in value),
          'exact configuration digest required')


def _literal(value, budget, depth=0):
    budget[0] += 1
    _need(budget[0] <= 200000 and depth <= 32, 'finite configured literal tree bound')
    kind = type(value)
    if value is None or kind is bool:
        return
    if kind is int:
        _need(value.bit_length() <= 127, 'configuration integer bound')
        return
    if kind is str or kind is bytes:
        _need(len(value) <= 1048576, 'configuration atom bound')
        return
    _need((kind is tuple or kind is list or kind is dict) and len(value) <= 8192, 'configuration exact literal type')
    if kind is dict:
        for key in value:
            _need(type(key) is str, 'configuration dictionary key type')
    for item in (value.values() if kind is dict else value):
        _literal(item, budget, depth + 1)



def _source_value(node, namespace, definitions, active=()):
    """Finite source expression interpreter; never eval or candidate callbacks."""
    if type(node) is ast.Name:
        _need(node.id in definitions and node.id not in active and len(active) < 32,
              'source default name is not a finite constant')
        value = _source_value(definitions[node.id], namespace, definitions, active + (node.id,))
        _need(_data(namespace[node.id]) == _data(value), 'source default constant changed')
        return value
    if type(node) is ast.Attribute and type(node.value) is ast.Name:
        module = namespace[node.value.id]
        _need(type(module) is types.ModuleType, 'source default owner is not a module')
        value = _MODULE_DICT(module, types.ModuleType)[node.attr]
        _data(value)
        return value
    if type(node) is ast.BinOp:
        left = _source_value(node.left, namespace, definitions, active)
        right = _source_value(node.right, namespace, definitions, active)
        _need(type(left) is int and type(right) is int, 'exact source integer default required')
        if type(node.op) is ast.Add: value = left + right
        elif type(node.op) is ast.Sub: value = left - right
        elif type(node.op) is ast.Mult: value = left * right
        elif type(node.op) is ast.LShift:
            _need(0 <= right <= 127, 'source default shift bound')
            value = left << right
        elif type(node.op) is ast.Pow:
            _need(0 <= right <= 127 and left.bit_length() * right <= 128, 'source default power bound')
            value = left ** right
        else: raise BindingStop('unsupported source default arithmetic')
        _need(value.bit_length() <= 128, 'source default integer bound')
        return value
    value = ast.literal_eval(node)
    _data(value)
    return value


def _defined_function(function, node, codes, namespace, definitions, forest, cls=None):
    source_role = node.name if cls is None else cls.__qualname__ + '.' + node.name
    _validate_code(function, codes, namespace, source_role, forest)
    values = tuple(getter(function, types.FunctionType) for getter in _FUNCTION_GETTERS)
    expected = tuple(_source_value(v, namespace, definitions) for v in node.args.defaults) or None
    keywords = {a.arg: _source_value(v, namespace, definitions)
        for a, v in zip(node.args.kwonlyargs, node.args.kw_defaults) if v is not None} or None
    _need(_data(values[3]) == _data(expected) and _data(values[4]) == _data(keywords),
          'initial function defaults differ from actual selected source')
    if cls is None or not function.__code__.co_freevars:
        _need(values[5] is None, 'unexpected source function closure')
    else:
        _need(function.__code__.co_freevars == ('__class__',) and len(values[5]) == 1 and
              _CELL_GET(values[5][0], types.CellType) is cls, 'class closure does not own actual class')
    return _function_record(function)


def _source_objects(module, source_pin, raw, exact_modules):
    # Original authenticated-source startup seal, never a caller record.
    # These native checks occur inline before new oracle callbacks.
    _need(_MODULE_DICT(builtins, types.ModuleType) is _CODE_ORACLE_BUILTINS,
          'source oracle builtin owner changed')
    _need(_LEN(_CODE_ORACLE_BUILTINS) == _LEN(_CODE_ORACLE_BUILTIN_ITEMS),
          'source oracle builtin census changed')
    for builtin_key in _CODE_ORACLE_BUILTINS:
        _need(_TYPE(builtin_key) is str, 'source oracle foreign builtin key')
    for builtin_key, builtin_value in _CODE_ORACLE_BUILTIN_ITEMS:
        _need(_CODE_ORACLE_BUILTINS[builtin_key] is builtin_value,
              'source oracle builtin binding changed')
    for oracle_name, oracle_value, oracle_record in _CODE_ORACLE_STARTUP_SEAL:
        _need(_CODE_ORACLE_NAMESPACE[oracle_name] is oracle_value,
              'source oracle alias identity changed')
        if oracle_record is not None:
            for getter, expected in zip(_FUNCTION_GETTERS, oracle_record[1]):
                _need(getter(oracle_value, types.FunctionType) is expected,
                      'source oracle function descriptor changed')
            _need(not oracle_record[1][6], 'source oracle function dictionary changed')
            for cell, expected in oracle_record[2]:
                _need(_CELL_GET(cell, types.CellType) is expected, 'source oracle cell changed')
    for oracle_class, class_items, class_mro in _CODE_ORACLE_CLASS_SEAL:
        actual_items = _CLASS_DICT(oracle_class, type)
        _need(_LEN(actual_items) == _LEN(class_items), 'source oracle class census changed')
        for class_key in actual_items:
            _need(_TYPE(class_key) is str, 'source oracle foreign class key')
        for class_key, class_value in class_items:
            _need(actual_items[class_key] is class_value, 'source oracle class binding changed')
        _need(_CODE_ORACLE_MRO_GET(oracle_class, type) is class_mro,
              'source oracle class MRO changed')
    namespace = _namespace(module)
    _need(namespace.get('__file__') == source_pin['path'] and
          sys.modules.get(namespace['__name__']) is module, 'installed actual source module differs')
    tree = ast.parse(raw, filename=source_pin['path'])
    code = compile(raw, source_pin['path'], 'exec', dont_inherit=True, optimize=sys.flags.optimize)
    codes = {}; _codes(code, codes)
    definitions = {n.targets[0].id: n.value for n in tree.body
        if type(n) is ast.Assign and len(n.targets) == 1 and type(n.targets[0]) is ast.Name}
    records, classes, forest = [], [], []
    for node in tree.body:
        if type(node) is ast.FunctionDef:
            _need(not node.decorator_list, 'top-level decorated selected function unsupported')
            records.append(_defined_function(namespace[node.name], node, codes, namespace, definitions, forest))
        elif type(node) is ast.ClassDef:
            cls = namespace[node.name]
            _need(type(cls) is type and not node.keywords and
                  cls.__module__ == namespace['__name__'] and cls.__name__ == node.name and
                  cls.__qualname__ == node.name, 'source class/metaclass/name owner differs')
            bases = []
            for base in node.bases:
                _need(type(base) is ast.Name, 'unreviewed computed class base')
                bases.append(namespace.get(base.id, builtins.__dict__.get(base.id)))
            _need(cls.__bases__ == tuple(bases or [object]), 'actual source class bases differ')
            dataclass_owned = False
            if node.decorator_list:
                _need(len(node.decorator_list) == 1, 'unreviewed source class decorator composition')
                decorator = node.decorator_list[0]
                _need(type(decorator) is ast.Call and type(decorator.func) is ast.Name and
                      namespace.get(decorator.func.id) is dataclasses.dataclass and not decorator.args,
                      'only actual original dataclass factory admitted')
                options = {kw.arg: ast.literal_eval(kw.value) for kw in decorator.keywords}
                _need(options == {'frozen': True, 'slots': True} and cls.__dataclass_params__.frozen is True,
                      'original dataclass constructor options differ')
                fields = tuple(n.target.id for n in node.body
                    if type(n) is ast.AnnAssign and type(n.target) is ast.Name)
                _need(tuple(cls.__dataclass_fields__) == fields and tuple(cls.__slots__) == fields,
                      'actual generated dataclass fields/slots differ')
                dataclass_owned = True
            own = _CLASS_DICT(cls, type)
            declared = set()
            for item in node.body:
                if type(item) is ast.FunctionDef:
                    method = own[item.name]; declared.add(item.name)
                    decorators = tuple(d.id if type(d) is ast.Name else None for d in item.decorator_list)
                    if decorators:
                        _need(decorators in (('staticmethod',), ('classmethod',)),
                              'unreviewed explicit method decorator')
                        descriptor = staticmethod if decorators == ('staticmethod',) else classmethod
                        _need(namespace.get(decorators[0], builtins.__dict__[decorators[0]]) is descriptor and
                              type(method) is descriptor, 'actual method wrapper owner differs')
                        method = (_STATIC_FUNCTION(method, staticmethod) if descriptor is staticmethod
                                  else _CLASS_FUNCTION(method, classmethod))
                    elif item.name == '__new__' and type(method) is staticmethod:
                        method = _STATIC_FUNCTION(method, staticmethod)
                    elif item.name in ('__init_subclass__', '__class_getitem__') and type(method) is classmethod:
                        method = _CLASS_FUNCTION(method, classmethod)
                    records.append(_defined_function(method, item, codes, namespace, definitions, forest, cls))
            for name, method in own.items():
                if type(method) is types.FunctionType and name not in declared:
                    _need(dataclass_owned and name in ('__init__', '__repr__', '__eq__', '__hash__',
                        '__setattr__', '__delattr__', '__getstate__', '__setstate__'),
                        'extra generated source method lacks the original fixed factory')
                    records.append(_function_record(method))
            classes.append((cls, tuple(own.items())))
    # Original authenticated-source startup seal, never a caller record.
    # These native checks occur inline before new oracle callbacks.
    _need(_MODULE_DICT(builtins, types.ModuleType) is _CODE_ORACLE_BUILTINS,
          'source oracle builtin owner changed')
    _need(_LEN(_CODE_ORACLE_BUILTINS) == _LEN(_CODE_ORACLE_BUILTIN_ITEMS),
          'source oracle builtin census changed')
    for builtin_key in _CODE_ORACLE_BUILTINS:
        _need(_TYPE(builtin_key) is str, 'source oracle foreign builtin key')
    for builtin_key, builtin_value in _CODE_ORACLE_BUILTIN_ITEMS:
        _need(_CODE_ORACLE_BUILTINS[builtin_key] is builtin_value,
              'source oracle builtin binding changed')
    for oracle_name, oracle_value, oracle_record in _CODE_ORACLE_STARTUP_SEAL:
        _need(_CODE_ORACLE_NAMESPACE[oracle_name] is oracle_value,
              'source oracle alias identity changed')
        if oracle_record is not None:
            for getter, expected in zip(_FUNCTION_GETTERS, oracle_record[1]):
                _need(getter(oracle_value, types.FunctionType) is expected,
                      'source oracle function descriptor changed')
            _need(not oracle_record[1][6], 'source oracle function dictionary changed')
            for cell, expected in oracle_record[2]:
                _need(_CELL_GET(cell, types.CellType) is expected, 'source oracle cell changed')
    for oracle_class, class_items, class_mro in _CODE_ORACLE_CLASS_SEAL:
        actual_items = _CLASS_DICT(oracle_class, type)
        _need(_LEN(actual_items) == _LEN(class_items), 'source oracle class census changed')
        for class_key in actual_items:
            _need(_TYPE(class_key) is str, 'source oracle foreign class key')
        for class_key, class_value in class_items:
            _need(actual_items[class_key] is class_value, 'source oracle class binding changed')
        _need(_CODE_ORACLE_MRO_GET(oracle_class, type) is class_mro,
              'source oracle class MRO changed')
    if forest:
        _code_oracle_compare_code_forest(tuple(row[0] for row in forest), tuple(row[1] for row in forest))
    return tree, tuple(records), tuple(classes), _import_bindings(tree, namespace, exact_modules)


def _import_namespace(module):
    _need(_TYPE(module) is types.ModuleType, 'exact imported module required')
    namespace = _MODULE_DICT(module, types.ModuleType)
    _need(_TYPE(namespace) is dict and _LEN(namespace) <= _MAX_GLOBALS,
          'bounded imported namespace required')
    for key in namespace:
        _need(_TYPE(key) is str and _LEN(key) <= 256, 'foreign import namespace key')
    _need(_TYPE(namespace.get('__name__')) is str, 'import module metadata differs')
    return namespace


def _import_bindings(tree, namespace, exact_modules):
    """Exact Python import semantics, no import hook or dynamic attribute lookup.

    The caller authenticates source modules before this function is invoked.
    Stdlib modules are the admitted startup objects; every selected object is
    also the actual sys.modules object. Parent package edges are checked and
    retained, rather than treating an import alias as a name-map convention.
    """
    _need(_TYPE(exact_modules) is dict and _LEN(exact_modules) <= 4096,
          'bounded exact module inventory required')
    for key, value in exact_modules.items():
        _need(_TYPE(key) is str and _TYPE(value) is types.ModuleType and
              sys.modules.get(key) is value, 'invented/substituted module inventory')
    records, seen = [], set()
    def selected(name):
        _need(_TYPE(name) is str and 0 < _LEN(name) <= 256 and
              all(part.isidentifier() for part in name.split('.')), 'exact absolute import name')
        parts = name.split('.')
        _need(_LEN(parts) <= 32, 'import package depth exceeded')
        prior = None
        for index in range(1, _LEN(parts) + 1):
            full = '.'.join(parts[:index])
            expected = exact_modules.get(full, _STARTUP_MODULES.get(full))
            _need(expected is not None and _TYPE(expected) is types.ModuleType and
                  sys.modules.get(full) is expected, 'import module missing/substituted')
            startup = _STARTUP_MODULES.get(full)
            _need(startup is None or startup is expected, 'admitted startup import identity changed')
            own = _import_namespace(expected)
            _need(own['__name__'] == full or
                  (_STARTUP_MODULES.get(own['__name__']) is expected and
                   sys.modules.get(own['__name__']) is expected), 'import module canonical/real alias identity differs')
            if prior is not None:
                _need(prior.get(parts[index - 1]) is expected, 'package submodule edge changed')
            if full not in seen:
                seen.add(full)
                records.append((full, expected, own, tuple(own.items())))
            prior = own
        return expected
    for node in tree.body:
        if _TYPE(node) is ast.Import:
            for alias in node.names:
                submodule = selected(alias.name)
                if alias.asname is None:
                    bound_name = alias.name.split('.')[0]
                    expected = selected(bound_name)
                else:
                    bound_name, expected = alias.asname, submodule
                _need(namespace.get(bound_name) is expected, 'installed exact import alias differs')
        elif _TYPE(node) is ast.ImportFrom:
            _need(node.level == 0 and _TYPE(node.module) is str,
                  'relative import requires separately reviewed package context')
            module = selected(node.module)
            imported = _import_namespace(module)
            for alias in node.names:
                _need(alias.name != '*' and alias.name.isidentifier(), 'explicit imported symbol required')
                _need(alias.name in imported, 'imported symbol absent from real namespace')
                expected = imported[alias.name]
                if _TYPE(expected) is types.ModuleType:
                    _need(selected(node.module + '.' + alias.name) is expected,
                          'from-import submodule identity differs')
                _need(namespace.get(alias.asname or alias.name) is expected,
                      'installed from-import symbol/alias differs')
    return tuple(records)


def _imports_unchanged(records, installed=None):
    for name, module, namespace, items in records:
        _need(sys.modules.get(name) is module and _import_namespace(module) is namespace and
              _LEN(namespace) == _LEN(items), 'import namespace identity/census changed')
        for key, value in items:
            expected = value if installed is None else installed.get((name, key), value)
            _need(namespace.get(key) is expected, 'import package/submodule/symbol binding changed')

def _v02_mapping(mapping, key_type=None):
    if key_type is None:
        key_type = str
    _need(_TYPE(mapping) is dict and _LEN(mapping) <= 512, 'V02 finite exact mapping required')
    for key in mapping:
        _need(_TYPE(key) is key_type, 'V02 foreign mapping key')
    return mapping, key_type, tuple(mapping.items())


def _v02_mapping_check(record):
    mapping, key_type, items = record
    _need(_TYPE(mapping) is dict and _LEN(mapping) == _LEN(items), 'V02 mapping census changed')
    for key in mapping:
        _need(_TYPE(key) is key_type, 'V02 foreign mapping key')

def _v02_cell(function, name):
    _need(_TYPE(function) is types.FunctionType, 'V02 exact function required')
    code = _FUNCTION_GETTERS[0](function, types.FunctionType)
    closure = _FUNCTION_GETTERS[5](function, types.FunctionType)
    _need(_TYPE(closure) is tuple and name in code.co_freevars, 'V02 named closure edge absent')
    cell = closure[code.co_freevars.index(name)]
    return cell, _CELL_GET(cell, types.CellType)


def _v02_modules():

          'V02 exact bounded module registry required')
    for key in modules:

    return modules

# Adapt PR6368 _plain with a full primitive-key census before dictionary lookup.
# _same is unchanged except the function prefix.
def _v02_plain(value):
    budget = [0, 0]

    def walk(item, depth):
        budget[0] += 1
        if depth > 24 or budget[0] > 65536:
            raise ValueError('gate tree bound')
        kind = type(item)
        if item is None or kind is bool:
            return
        if kind is int:
            if not -(1 << 64) <= item < (1 << 64):
                raise ValueError('gate integer bound')
            return
        if kind is str:
            budget[1] += len(item)
            if len(item) > 8192 or budget[1] > 524288:
                raise ValueError('gate text bound')
            return
        if (kind is not list and kind is not dict) or len(item) > 1024:
            raise ValueError('gate exact JSON domain')
        if kind is dict:
            for key in item:
                if type(key) is not str:
                    raise ValueError('gate exact string key required')
            for key in item:
                walk(key, depth + 1)
                walk(item[key], depth + 1)
        else:
            for child in item:
                walk(child, depth + 1)

    walk(value, 0)

def _v02_same(left, right):
    if type(left) is not type(right):
        return False
    if type(left) is dict:
        return (set(left) == set(right) and
                all(_v02_same(left[k], right[k]) for k in left))
    if type(left) is list:
        return (len(left) == len(right) and
                all(_v02_same(a, b) for a, b in zip(left, right)))
    return left == right

def _v02_pin_shape(pin, limit):
    # Same complete immutable regular-file pin contract as the owning reader.
    _exact(pin, ('path', 'sha256', 'bytes', 'identity'))
    _need(_TYPE(pin['path']) is str and pin['path'].startswith('/') and
          _LEN(pin['path']) <= 4096 and
          all(c not in ('', '.', '..') for c in pin['path'].split('/')[1:]),
          'V02 complete source path differs')
    _digest(pin['sha256'])
    _need(_TYPE(pin['bytes']) is int and 0 < pin['bytes'] <= limit,
          'V02 complete source byte bound')
    _exact(pin['identity'], ('device', 'inode', 'mode', 'links', 'uid', 'gid', 'size', 'mtime_ns', 'ctime_ns'))
    _need(all(_TYPE(x) is int and x >= 0 for x in pin['identity'].values()) and
          pin['identity']['links'] == 1 and pin['identity']['size'] == pin['bytes'] and
          stat.S_ISREG(pin['identity']['mode']) and not pin['identity']['mode'] & 0o222,
          'V02 immutable source identity differs')



def _v02_preseal_native_guard():
    """Startup-native ownership check; this does not authenticate a pipe writer."""
    registry = _v02_modules()
    _need(_TYPE(_V02_PRESEAL_NATIVE) is tuple and _LEN(_V02_PRESEAL_NATIVE) == 2,
          'V02 preseal native census differs')
    values = []
    for row, expected in zip(_V02_PRESEAL_NATIVE, (('select', 'select'), ('time', 'monotonic_ns'))):
        _need(_TYPE(row) is tuple and _LEN(row) == 5, 'V02 preseal native row differs')
        name, module, namespace, function, value = row
        _need(_TYPE(name) is str and _TYPE(function) is str and (name, function) == expected and
              _TYPE(module) is types.ModuleType and registry.get(name) is module and
              _MODULE_DICT(module, types.ModuleType) is namespace,
              'V02 preseal native module owner differs')
        _v02_mapping(namespace)
        _need(namespace.get(function) is value and _TYPE(value) is types.BuiltinFunctionType and
              value.__self__ is module, 'V02 preseal native function owner differs')
        values.append(value)
    return tuple(values)


def _v02_preseal_binding_check(state, record):
    _need(state['v02_preseal_binding'] is record and _TYPE(record) is tuple and _LEN(record) == 6,

def _construct_v02_issued_input(context):
    """Prospective owning constructor; ordinary ingress has no admission issuer.

    The private record is never decoded from P/E or provided through a setter.
    Tests may model it only as an explicitly inert pre-admitted owning seam.
    Its absence remains the original refusal. Representation and source custody
    checks below cannot confer the missing parent decisions or launch authority.
    """
    state = _entry(context)
    try:
        _fresh(state)
        _need(state['phase'] == 'PREPARED' and state['consumed'] is True and
              state['v02_installing'] is True and state['v02_typed_attempted'] is False and
              state['v02_issued_input'] is None and state['v02_attempted'] is False,
              'V02 typed construction is one-shot while PREPARED')
        state['v02_typed_attempted'] = True
        handoff = _check_v02_driver_handoff(context)
        admission = state['v02_pre_admitted_input']
        _need(_TYPE(admission) is tuple and _LEN(admission) == 5 and admission[0] is context,
              'V02 concrete authenticated issued input absent')
        retained = state['v02_preflight_package_data']
        _need(_TYPE(retained) is tuple and _LEN(retained) == 3 and admission[1] is retained,
              'V02 admission retained package owner differs')
        pin, snapshot, raw = retained
        _v02_pin_shape(pin, 1048576)
        _need(_data(pin) == snapshot and _TYPE(raw) is bytes and _LEN(raw) == pin['bytes'] and
              hashlib.sha256(raw).hexdigest() == pin['sha256'], 'V02 retained package custody differs')
        package, gates, admitted_snapshot = admission[2:]
        _v02_plain(package)
        decoded = _document(raw)
        _v02_plain(decoded)
        _need(_v02_same(package, decoded) and _data(package) == admitted_snapshot,
              'V02 admitted package contents differ')
        _exact(package, ('schema', 'producer_subject', 'original_source', 'root_source', 'provider_source', 'decisions'))
        _need(package['schema'] == 'h25-v02-typed-package-v1', 'V02 typed package schema differs')
        subject = package['producer_subject']
        _exact(subject, ('source_sha256', 'mode', 'presentation'))
        _need(subject['source_sha256'] == _V02_ORIGINAL_SHA and subject['mode'] == 'normal' and
              subject['presentation'] == 'nonconsuming-v02-omission', 'V02 early producer subject differs')
        decisions = package['decisions']
        _need(_TYPE(decisions) is list and _LEN(decisions) == 8 and
              _TYPE(gates) is tuple and _LEN(gates) == 8, 'V02 eight typed decisions required')
        for index, name in enumerate(_V02_GATE_NAMES):
            decision = decisions[index]
            _exact(decision, ('name', 'source', 'value'))
            _need(decision['name'] == name and decision['value'] is gates[index] and
                  gates[index] is not None, 'V02 genuine decision identity/order differs')

            _v02_pin_shape(package[field], limit)
            _need(package[field]['sha256'] == digest, 'V02 fixed source subject differs')
        root, capability, api = handoff[4:7]
        namespace = _namespace(root)
        _need(namespace['__file__'] == package['root_source']['path'], 'V02 Root source path differs')
        registry = _v02_mapping(namespace['_CAPABILITIES'], int)[0]
        _need(_LEN(registry) == 1 and id(capability) in registry, 'V02 capability custody differs')
        row = registry[id(capability)]
        _need(_TYPE(row) is tuple and _LEN(row) == 7 and row[0] is capability,
              'V02 capability row custody differs')
        provider = _namespace(row[1])
        _need(row[6] is provider and row[4] is provider['__file__'] and
              provider['__file__'] == package['provider_source']['path'] and
              _TYPE(row[3]) is str and row[3] == _V02_PROVIDER_SHA and
              provider['_BOOTSTRAP_CAPABILITY'] is capability, 'V02 provider source custody differs')
        metadata = _v02_mapping(namespace['_ROOT_METADATA'])[0]
        _need(_TYPE(metadata.get('source_sha256')) is str and metadata['source_sha256'] == _V02_ROOT_SHA and
              _TYPE(metadata.get('source_bytes')) is int and metadata['source_bytes'] == package['root_source']['bytes'],
              'V02 initialized Root source custody differs')
        root_source = _read_pin(state, package['root_source'], 262144)
        provider_source = _read_pin(state, package['provider_source'], 262144)
        _need(_data(package) == admitted_snapshot and _data(pin) == snapshot,
              'V02 package changed during source custody read')
        _need(state['v02_pre_admitted_input'] is admission and
              state['v02_preflight_package_data'] is retained,
              'V02 retained admission changed during source custody read')
        _need(all(package['decisions'][index]['value'] is gates[index] for index in range(8)),
              'V02 genuine decision identity changed during source custody read')
        _check_v02_driver_handoff(context)
        _fresh(state)
        configuration = dict(schema='h25-live-preflight-context-v1',
            unit_id='variant/V02/python-preflight-v1', original_source=package['original_source'],
            bootstrap_root=root, bootstrap_capability=capability, bootstrap_api=api,
            genuine_gate_values=gates)
        state['v02_issued_input'] = (context, state['modules']['component_control_root_v1'],

    except BaseException as error:
        _v02_stop(state, error)
        raise

def _v02_check_anchor(context, state, anchor, module, value):
    try:
        _fresh(state)
        _check_v02_driver_handoff(context)
        _need(state['v02_busy'] is False, 'V02 verification reentry requires teardown')
        state['v02_busy'] = True
        _need(_TYPE(anchor) is tuple and _LEN(anchor) == 14 and
              state['v02_anchor'] is anchor and anchor[1] is context and anchor[3] is state,
              'V02 original immutable token anchor differs')
        rows = tuple(row for row in _INSTALLATIONS if row[0] is context)
        _need(_LEN(rows) == 1 and rows[0] is anchor[2] and rows[0][1] is state,
              'V02 original token registry row differs')
        _need(module is anchor[4] and value is anchor[5] and
              _namespace(module)['PREFLIGHT_CONFIGURATION'] is value,
              'V02 actual installed live configuration differs')
        _exact(value, _V02_CONFIG_KEYS)
        for key, expected in anchor[6]:
            _need(value[key] is expected, 'V02 current installed field identity differs')
        actual_gates = value['genuine_gate_values']
        _need(actual_gates is anchor[7] and _TYPE(actual_gates) is tuple and _LEN(actual_gates) == 8,
              'V02 original gate tuple differs')
        for gate, expected in zip(actual_gates, anchor[8]):
            _need(_data(gate) == expected, 'V02 genuine gate contents changed')
        _need(_data(value['original_source']) == anchor[13], 'V02 original source pin changed')
        # Pass actual current fields, never tautological anchor-owned arguments.
        _v02_check_live_profile(anchor[9], value['bootstrap_root'],
            value['bootstrap_capability'], value['bootstrap_api'])
        _fresh(state)  # a suppressed inner failure cannot restore success
        _need(state['v02_busy'] is True, 'V02 verification state changed')
        state['v02_busy'] = False
        return None
    except BaseException as error:
        _v02_stop(state, error)
        raise


def _capture_installed_gate(context, state, name, module, key, value):
    if name == 'component_control_root_v1' and key == 'PREFLIGHT_CONFIGURATION':
        anchor = state['v02_anchor']
        _v02_check_anchor(context, state, anchor, module, value)
        return anchor
    return _data(value)


def installed_gate_snapshot(context, name, module, key, value):
    state = _entry(context)
    _fresh(state)
    _need(state['phase'] == 'SEALED', 'installed gate snapshot requires sealed token')
    row = state['installed'][(name, key)]
    _need(row[0] is module and row[1] is value, 'installed gate snapshot owner differs')

def check_installed_gate(context, name, module, key, value, snapshot):
    state = _entry(context)
    if name == 'component_control_root_v1' and key == 'PREFLIGHT_CONFIGURATION':
        _v02_check_anchor(context, state, snapshot, module, value)
        return None
    _need(_namespace(module)[key] is value and _data(value) == snapshot,
          'installed gate identity/content changed')
    return None

compares one module's whole matched forest; do not reset aliases per function.
Adaptation of cache_contract_v8._constant_equal/_code_equal exact-type discipline,
with native metadata coverage, finite floats/sets, bounds and forest aliases.
"""
import marshal
import types

_code_oracle_TYPE = type
_code_oracle_LEN = len
_code_oracle_ID = id
_code_oracle_GET = object.__getattribute__
_code_oracle_CODE = types.CodeType
_code_oracle_DUMPS = marshal.dumps
_code_oracle_BIT_LENGTH = int.bit_length
_code_oracle_LOCAL_NAME = types.CodeType.__dict__['_varname_from_oparg']
_code_oracle_INDEX_ERROR = IndexError

_code_oracle_MAX_ROOTS = 1024
_code_oracle_MAX_VISITS = 100000
_code_oracle_MAX_DEPTH = 64
_code_oracle_MAX_CONTAINER = 8192
_code_oracle_MAX_ATOM_LENGTH = 1048576
_code_oracle_MAX_INT_BITS = 4096
_code_oracle_MAX_SERIALIZED_BOUND = 8 * 1024 * 1024

_code_oracle_INTEGER_FIELDS = ('co_argcount', 'co_posonlyargcount', 'co_kwonlyargcount',
                   'co_nlocals', 'co_stacksize', 'co_flags', 'co_firstlineno')
_code_oracle_BYTES_FIELDS = ('co_code', 'co_linetable', 'co_exceptiontable')
_code_oracle_TEXT_FIELDS = ('co_filename', 'co_name', 'co_qualname')
_code_oracle_NAMES_FIELDS = ('co_names', 'co_varnames', 'co_cellvars', 'co_freevars')


class _code_oracle_CodeOracleStop(RuntimeError):
    pass


def _code_oracle_need(condition, reason):
    if not condition:
        raise _code_oracle_CodeOracleStop(reason)


def _code_oracle_code_children(code):
    """Only exact CodeType/native descriptors; no candidate lookup callbacks."""
    children = []
    for field in _code_oracle_INTEGER_FIELDS:
        value = _code_oracle_GET(code, field)
        _code_oracle_need(_code_oracle_TYPE(value) is int and 0 <= value <= 0x7fffffff,
              'code integer metadata type/range')
        children.append(value)
    for field in _code_oracle_BYTES_FIELDS:
        value = _code_oracle_GET(code, field)
        _code_oracle_need(_code_oracle_TYPE(value) is bytes, 'code byte metadata type')
        children.append(value)
    for field in _code_oracle_TEXT_FIELDS:
        value = _code_oracle_GET(code, field)
        _code_oracle_need(_code_oracle_TYPE(value) is str, 'code text metadata type')
        children.append(value)
    for field in _code_oracle_NAMES_FIELDS:
        value = _code_oracle_GET(code, field)
        _code_oracle_need(_code_oracle_TYPE(value) is tuple and _code_oracle_LEN(value) <= _code_oracle_MAX_CONTAINER,
              'code names metadata shape')
        for name in value:
            _code_oracle_need(_code_oracle_TYPE(name) is str, 'code name exact text type')
        children.append(value)
    constants = _code_oracle_GET(code, 'co_consts')
    _code_oracle_need(_code_oracle_TYPE(constants) is tuple, 'code constants exact tuple')
    children.append(constants)
    # Public name tuples omit hidden-only localsplus entries. Enumerate the
    # pinned native backing tuple before marshal so its size cannot evade the
    # allocation bound. The native descriptor uses PyTuple_GetItem, not Python
    # indexing hooks. The same count bounds the internal kinds byte array.
    local_names = []
    for index in range(_code_oracle_MAX_CONTAINER + 1):
        try:
            name = _code_oracle_LOCAL_NAME(code, index)
        except _code_oracle_INDEX_ERROR:
            break
        _code_oracle_need(index < _code_oracle_MAX_CONTAINER and _code_oracle_TYPE(name) is str,
              'internal local name count/type')
        local_names.append(name)
    children.append(tuple(local_names))
    return tuple(children), _code_oracle_LEN(local_names)


def _code_oracle_preflight(roots):
    """Iterative expanded walk; all exact types/bounds precede any marshal.

    Walk shared subgraphs again to bound format-2's actual expansion, rather
    than merely bounding unique nodes in a potentially exponential DAG.
    """
    _code_oracle_need(_code_oracle_TYPE(roots) is tuple and 1 <= _code_oracle_LEN(roots) <= _code_oracle_MAX_ROOTS,
          'exact nonempty bounded root tuple')
    for root in roots:
        _code_oracle_need(_code_oracle_TYPE(root) is _code_oracle_CODE, 'exact code root required')
    stack = [(0, root, 0) for root in reversed(roots)]
    active = set()
    visits = 0
    bound = 0
    peak_depth = 0
    local_slots = 0
    while stack:
        exiting, value, depth = stack.pop()
        if exiting:
            active.remove(_code_oracle_ID(value))
            continue
        visits += 1
        _code_oracle_need(visits <= _code_oracle_MAX_VISITS and depth <= _code_oracle_MAX_DEPTH,
              'expanded graph visit/depth bound')
        peak_depth = max(peak_depth, depth)
        kind = _code_oracle_TYPE(value)
        children = None
        if value is None or value is Ellipsis:
            bound += 1
        elif kind is bool:
            bound += 1
        elif kind is int:
            bits = _code_oracle_BIT_LENGTH(value)
            _code_oracle_need(bits <= _code_oracle_MAX_INT_BITS, 'integer bit bound')
            bound += 8 + 2 * ((bits + 14) // 15)
        elif kind is float:
            # Exact floats only: finite x-x is zero, infinities/NaNs produce
            # NaN and fail. No added math-module/native-owner dependency.
            _code_oracle_need(value - value == 0.0, 'nonfinite float constant')
            bound += 9
        elif kind is str:
            _code_oracle_need(_code_oracle_LEN(value) <= _code_oracle_MAX_ATOM_LENGTH, 'text length bound')
            bound += 5 + 4 * _code_oracle_LEN(value)
        elif kind is bytes:
            _code_oracle_need(_code_oracle_LEN(value) <= _code_oracle_MAX_ATOM_LENGTH, 'bytes length bound')
            bound += 5 + _code_oracle_LEN(value)
        elif kind is tuple or kind is frozenset:
            _code_oracle_need(_code_oracle_LEN(value) <= _code_oracle_MAX_CONTAINER, 'container width bound')
            if kind is frozenset:
                for item in value:
                    _code_oracle_need(_code_oracle_TYPE(item) is str or _code_oracle_TYPE(item) is int,
                          'frozenset restricted exact scalar members')
            bound += 5
            children = value
        elif kind is _code_oracle_CODE:
            children, count = _code_oracle_code_children(value)
            local_slots += count
            # More than native fixed header/kinds overhead; public name tuples
            # plus the native-name enumeration conservatively duplicate names.
            bound += 128 + count
        else:
            raise _code_oracle_CodeOracleStop('unsupported exact graph type')
        _code_oracle_need(bound <= _code_oracle_MAX_SERIALIZED_BOUND, 'expanded serialized-size bound')
        if children is not None:
            identity = _code_oracle_ID(value)
            _code_oracle_need(identity not in active, 'cyclic code constant graph')
            _code_oracle_need(_code_oracle_LEN(stack) + _code_oracle_LEN(children) + 1 <= 2 * _code_oracle_MAX_VISITS,
                  'pending graph work bound')
            active.add(identity)
            stack.append((1, value, depth))
            # Each admitted container is bounded before materializing its
            # native iterator. No foreign iterable reaches this branch.
            for child in children:
                stack.append((0, child, depth + 1))
            _code_oracle_need(_code_oracle_LEN(stack) <= 2 * _code_oracle_MAX_VISITS, 'pending graph work bound')
    return visits, bound, peak_depth, local_slots


def _code_oracle_code_alias_children(code):
    # These include the stable cached var/cell/free tuple views materialized
    # on both sides during preflight. Their aliases with constants and other
    # code roots are observable. Integer descriptors are value fields instead.
    result = [_code_oracle_GET(code, field) for field in (
        'co_code', 'co_consts', 'co_names', 'co_filename', 'co_name',
        'co_qualname', 'co_linetable', 'co_exceptiontable',
        'co_varnames', 'co_cellvars', 'co_freevars')]
    for index in range(_code_oracle_MAX_CONTAINER + 1):
        try:
            name = _code_oracle_LOCAL_NAME(code, index)
        except _code_oracle_INDEX_ERROR:
            break
        _code_oracle_need(index < _code_oracle_MAX_CONTAINER, 'alias internal local bound')
        result.append(name)
    return tuple(result)


def _code_oracle_aliases(actual_roots, expected_roots):
    """Reciprocal constant/code graph aliases across all roots in this call."""
    actual_to_expected = {}
    expected_to_actual = {}
    pending = list(zip(actual_roots, expected_roots))
    visits = 0
    while pending:
        actual, expected = pending.pop()
        visits += 1
        _code_oracle_need(visits <= _code_oracle_MAX_VISITS, 'alias graph visit bound')
        kind = _code_oracle_TYPE(actual)
        _code_oracle_need(kind is _code_oracle_TYPE(expected), 'constant exact type differs')
        ai, ei = _code_oracle_ID(actual), _code_oracle_ID(expected)
        previous_a = expected_to_actual.get(ei)
        previous_e = actual_to_expected.get(ai)
        if previous_a is not None or previous_e is not None:
            _code_oracle_need(previous_a == ai and previous_e == ei,
                  'constant/code graph alias split or merge')
            continue
        expected_to_actual[ei] = ai
        actual_to_expected[ai] = ei
        if kind is _code_oracle_CODE:
            a = _code_oracle_code_alias_children(actual)
            e = _code_oracle_code_alias_children(expected)
            _code_oracle_need(_code_oracle_LEN(a) == _code_oracle_LEN(e), 'code alias field arity differs')
            pending.extend(zip(a, e))
        elif kind is tuple:
            _code_oracle_need(_code_oracle_LEN(actual) == _code_oracle_LEN(expected), 'tuple arity differs')
            pending.extend(zip(actual, expected))
        elif kind is frozenset:
            _code_oracle_need(_code_oracle_LEN(actual) == _code_oracle_LEN(expected), 'frozenset arity differs')
            # Complete forest preflight already proved exact str/int members
            # and bounded lengths/bits. Native format-2 encodings distinguish
            # these types/values and supply deterministic ordering without
            # candidate comparisons or iteration callbacks.
            a = sorted(actual, key=_code_oracle_scalar_key)
            e = sorted(expected, key=_code_oracle_scalar_key)
            pending.extend(zip(a, e))
    return visits, _code_oracle_LEN(actual_to_expected)


def _code_oracle_scalar_key(value):
    return _code_oracle_DUMPS(value, 2)


def _code_oracle_compare_code_forest(actual_roots, expected_roots):
    """Compare one module's source-aligned exact CodeType forest, or STOP.

    Repeated roots are allowed and their aliases are checked. Roots are never
    selected by actual names, and no candidate callback or equality is used.
    Return immutable diagnostics, not an authority token:
      (root_count, actual_preflight, expected_preflight,
       alias_visits, alias_nodes, actual_serialized_bytes, expected_bytes).
    Preflight tuples are (expanded_visits, byte_upper_bound, depth, local_slots).
    """
    expected_counts = _code_oracle_preflight(expected_roots)
    actual_counts = _code_oracle_preflight(actual_roots)
    _code_oracle_need(_code_oracle_LEN(actual_roots) == _code_oracle_LEN(expected_roots), 'source root cardinality differs')
    alias_visits, alias_nodes = _code_oracle_aliases(actual_roots, expected_roots)
    actual_bytes = expected_bytes = 0
    for actual, expected in zip(actual_roots, expected_roots):
        a = _code_oracle_DUMPS(actual, 2)
        e = _code_oracle_DUMPS(expected, 2)
        actual_bytes += _code_oracle_LEN(a)
        expected_bytes += _code_oracle_LEN(e)
        _code_oracle_need(actual_bytes <= actual_counts[1] and expected_bytes <= expected_counts[1],
              'native serialization exceeds preflight upper bound')
        _code_oracle_need(a == e, 'native code metadata or constant value differs')
    return (_code_oracle_LEN(actual_roots), actual_counts, expected_counts,
            alias_visits, alias_nodes, actual_bytes, expected_bytes)
