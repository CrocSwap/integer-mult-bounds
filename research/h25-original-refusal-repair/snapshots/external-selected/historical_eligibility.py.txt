def need(value, text):

def layout_only(active_paths):

    need(not os.path.lexists(OLD_BARRIER), 'unclassified old BARRIER')
    observed = {}
    for name in DIRECTORIES:
        path = Path(name)
        info = path.lstat()
        need(path.resolve(strict=True) == path and stat.S_ISDIR(info.st_mode) and
             info.st_uid == os.getuid() and stat.S_IMODE(info.st_mode) == 0o700,
             'canonical current-owner historical directory0700')

                          info.st_mtime_ns,info.st_ctime_ns)
    for name in active_paths:
        need(not os.path.lexists(name), 'active successor barrier/quarantine still blocks')
    return observed  # Metadata observation only, never historical/source eligibility.
