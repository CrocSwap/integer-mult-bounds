class AdapterStop(RuntimeError):

def remember(error):
    if FIRST[0] is None:
        FIRST[0] = error
    elif error is not FIRST[0]:
        for index in range(32):
            if SECONDARY[index] is error:
                break
            if SECONDARY[index] is None:
                SECONDARY[index] = error
                break
        else:
            IO_UNKNOWN[0] = True
    return FIRST[0]


def checkpoint():
    if IO_UNKNOWN[0]:
        raise AdapterStop('uncertain IO; process teardown required, no new acquisition')
    if SOFT_DEADLINE[0] is not None and time.monotonic() > SOFT_DEADLINE[0]:
        raise AdapterStop('owning-parent soft allowance crossed; cleanup remains paid')
    if ACTIVE_CASE[0] is not None:
        ACTIVE_CASE[0].check()
    if resource.getrusage(resource.RUSAGE_SELF).ru_maxrss * 1024 > 1073741824:
        raise AdapterStop('owning-parent observed RSS cap')


class OwnedIO:
    """Fixed direct-handle slots; ordinary POSIX under selected operating TCB."""
    def __init__(self):
        self.slots = [[None, 'FREE', None] for unused in range(16)]

    def acquire(self, path, flags, mode=0o600, directory=False):
        checkpoint()
        slot = next(row for row in self.slots if row[1] in ('FREE', 'CLOSED'))
        slot[0], slot[2], slot[1] = None, directory, 'PENDING'
        try:
            slot[0] = os.scandir(path) if directory else os.open(path, flags, mode)
            slot[1] = 'OWNED'
            return slot
        except BaseException as error:
            IO_UNKNOWN[0] = True
            remember(error)
            raise

    def close(self, slot):
        if slot[1] != 'OWNED':
            IO_UNKNOWN[0] = True
            raise AdapterStop('handle close not known/replay')
        slot[1] = 'CLOSING'
        try:
            slot[0].close() if slot[2] else os.close(slot[0])
        except BaseException as error:
            IO_UNKNOWN[0] = True
            remember(error)
            raise
        slot[1] = 'CLOSED'

    def names(self, path, cap=8192):
        names, primary, slot = [], None, None
        try:
            slot = self.acquire(path, 0, directory=True)
            for entry in slot[0]:
                checkpoint()
                if len(names) >= cap:
                    raise AdapterStop('directory census cap')
                names.append(entry.name)
        except BaseException as error:
            primary = remember(error)
        finally:
            if slot is not None and slot[1] == 'OWNED':
                try:
                    self.close(slot)
                except BaseException as error:
                    primary = remember(error)
        if primary is not None:
            raise primary
        return names

    def read(self, path, cap, mode=None, expected=None, materialize=False, proc=False):
        checkpoint()
        path = Path(path)
        if not proc:
            need(path.is_absolute() and path.resolve(strict=True) == path, 'canonical source/input/output path')
        before = path.lstat()
        if not proc:
            need(stat.S_ISREG(before.st_mode) and before.st_nlink == 1 and
                 before.st_uid in (0, os.getuid()) and before.st_size <= cap and
                 (mode is None or stat.S_IMODE(before.st_mode) == mode), 'regular stable file precondition')
        pieces, count, hasher, slot, primary = [], 0, hashlib.sha256(), None, None
        try:
            slot = self.acquire(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK | os.O_CLOEXEC)
            if not proc:
                need(identity(os.fstat(slot[0])) == identity(before), 'opened stable file differs')
            while True:
                checkpoint()
                block = os.read(slot[0], min(65536, cap + 1 - count))
                if not block:
                    break
                count += len(block)
                need(count <= cap, 'file read cap')
                hasher.update(block)
                if materialize:
                    pieces.append(block)
            if not proc:
                need(identity(os.fstat(slot[0])) == identity(before) and count == before.st_size,
                     'stable descriptor changed')
        except BaseException as error:
            primary = remember(error)
        finally:
            if slot is not None and slot[1] == 'OWNED':
                try:
                    self.close(slot)
                except BaseException as error:
                    primary = remember(error)
        if primary is not None:
            raise primary
        if not proc:
            need(identity(path.lstat()) == identity(before), 'stable path changed')
        row = {'path': str(path), 'bytes': count, 'sha256': hasher.hexdigest(),
               'mode': stat.S_IMODE(before.st_mode), 'uid': before.st_uid,
               'nlink': before.st_nlink, 'identity': list(identity(before))}
        if expected is not None:
            need(type(expected['bytes']) is int and row['bytes'] == expected['bytes'] and
                 row['sha256'] == expected['sha256'], 'fixed file bytes/hash differs')
        return row, b''.join(pieces) if materialize else None

    def write(self, path, value, cap=4194304):
        checkpoint()
        path = Path(path)
        need(path.is_absolute() and path.parent.resolve(strict=True) == path.parent, 'canonical retained evidence parent')
        raw, slot, primary, offset = canonical(value), None, None, 0
        need(len(raw) <= cap, 'retained JSON evidence cap')
        try:
            slot = self.acquire(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW | os.O_CLOEXEC)
            while offset < len(raw):
                n = os.write(slot[0], raw[offset:offset + 65536])
                need(type(n) is int and 0 < n <= min(65536, len(raw) - offset), 'retained short write')
                offset += n
            os.fsync(slot[0])
        except BaseException as error:
            primary = remember(error)
        finally:
            if slot is not None and slot[1] == 'OWNED':
                try:
                    self.close(slot)
                except BaseException as error:
                    primary = remember(error)
        if primary is not None:
            raise primary
        return self.read(path, cap, 0o600, {'bytes': len(raw), 'sha256': digest(raw)})[0]

def need(value, message):
    if not value:
        raise AdapterStop(message)


def exact(value, keys):
    need(type(value) is dict and set(value) == set(keys), 'exact DATA object keys differ')


def digest(raw):
    return hashlib.sha256(raw).hexdigest()


def sha_value(value):
    need(type(value) is str and len(value) == 64 and
         all(c in '0123456789abcdef' for c in value), 'exact sha256 required')


def canonical(value):
    return (json.dumps(value, sort_keys=True, separators=(',', ':'),
                       ensure_ascii=True, allow_nan=False) + '\n').encode('ascii')


def json_data(raw, cap):
    """Bounded file/line before JSON parsing; post-parse node/type checks.

    json.loads allocations and recursion precede postchecks; outer 1GiB/180s
    supervision remains mandatory. This is not a proved successful-fit bound.
    """
    need(type(raw) is bytes and 0 < len(raw) <= cap, 'JSON byte cap')
    def pairs(rows):
        out = {}
        for key, value in rows:
            checkpoint()
            need(type(key) is str and key not in out, 'duplicate or foreign JSON key')
            out[key] = value
        return out
    def constant(value):
        raise AdapterStop('nonfinite JSON constant')
    checkpoint()
    value = json.loads(raw.decode('ascii'), object_pairs_hook=pairs, parse_constant=constant)
    checkpoint()
    nodes = [0]
    def walk(item, depth):
        checkpoint()
        nodes[0] += 1
        need(nodes[0] <= 200000 and depth <= 32, 'JSON nodes/depth')
        kind = type(item)
        if item is None or kind is bool:
            return
        if kind is int:
            need(item.bit_length() <= 128, 'JSON integer bound')
        elif kind is float:
            need(math.isfinite(item), 'JSON finite float')
        elif kind is str:
            need(len(item) <= 1048576, 'JSON string bound')
        elif kind in (dict, list):
            need(len(item) <= 16384, 'JSON collection bound')
            if kind is dict:
                for key, child in item.items():
                    checkpoint()
                    need(type(key) is str and len(key) <= 65536, 'JSON key bound')
                    walk(child, depth + 1)
            else:
                for child in item:
                    checkpoint()
                    walk(child, depth + 1)
        else:
            raise AdapterStop('foreign JSON type')
    walk(value, 0)
    return value


def identity(info):
    return (info.st_dev, info.st_ino, info.st_mode, info.st_nlink,
            info.st_uid, info.st_size, info.st_mtime_ns, info.st_ctime_ns)


def verify_journal(raw, inputs, inventory, mode, outcome):
    """DATA journal only; never decode the typed input or invoke an oracle again."""
    need(raw.endswith(b'\n'), 'journal lacks final newline')
    lines = raw.splitlines()
    need(len(lines) == len(INPUT_ORDER) + 2 * len(inventory) + 1, 'complete journal inventory differs')
    position = 0
    for name in INPUT_ORDER:
        checkpoint()
        observed = json_data(lines[position], 1048576)
        position += 1
        exact(observed, ('kind', 'name', 'bytes', 'sha256', 'mode'))
        expected = inputs[name]
        need(observed['kind'] == 'INPUT_RAW_BEFORE_ORACLE' and observed['name'] == name and
             type(observed['bytes']) is int and observed['bytes'] == expected['bytes'] and
             observed['sha256'] == expected['sha256'] and observed['mode'] == mode,
             'input RAW-before-oracle journal differs')
    for control in inventory:
        checkpoint()
        first = json_data(lines[position], 1048576)
        second = json_data(lines[position + 1], 1048576)
        position += 2
        exact(first, ('kind', 'name', 'scope', 'raw_hex', 'raw_bytes', 'raw_sha256', 'expected_code'))
        exact(second, ('kind', 'name', 'code'))
        need(first['kind'] == 'CONTROL_RAW_BEFORE_ORACLE' and first['name'] == control['name'] and
             first['scope'] == control['scope'] and first['expected_code'] == control['expected_code'] and
             second['kind'] == 'CONTROL_OBSERVED' and second['name'] == control['name'] and
             second['code'] == control['expected_code'], 'literal RAW/order/observed code differs')
        hx = first['raw_hex']
        need(type(first['raw_bytes']) is int and 0 <= first['raw_bytes'] <= 131072 and
             type(hx) is str and len(hx) == first['raw_bytes'] * 2 and
             all(c in '0123456789abcdef' for c in hx), 'observed RAW hex type/length')
        sha_value(first['raw_sha256'])
        # Observed transport consistency only. Fixture identity is source/body-pinned;
        # expected code/value checks stay independently fixed, never learned here.
        need(digest(bytes.fromhex(hx)) == first['raw_sha256'], 'observed RAW transport digest differs')
    final = json_data(lines[position], 1048576)
    exact(final, ('kind', 'outcome'))
    need(final['kind'] == 'SEMANTIC_OBSERVATION' and canonical(final['outcome']) == canonical(outcome),
         'complete semantic observation cross-binding differs')

def metadata(path, cap=65536):

def process(pid):
    raw = metadata('/proc/' + str(pid) + '/stat', 8192)
    end = raw.rfind(b') ')
    need(end >= 0, 'process stat syntax')
    fields = raw[end + 2:].split()
    return {'pid': pid, 'ppid': int(fields[1]), 'pgid': int(fields[2]), 'sid': int(fields[3]),
            'start_ticks': int(fields[19]), 'rss_bytes': int(fields[21]) * os.sysconf('SC_PAGE_SIZE')}


class GlobalLease:
    """This process's real FD, never supplied in JSON or passed to children."""
    def __init__(self):
        self.fd, self.phase, self.birth, self.lock_info = None, 'NEW', None, None

    def acquire(self):
        self.phase = 'OPEN_PENDING'
        try:
            self.fd = os.open(HISTORICAL_LOCK, os.O_RDWR | os.O_NOFOLLOW | os.O_CLOEXEC)
            self.phase = 'OPEN_RETURNED'
            fcntl.flock(self.fd, fcntl.LOCK_EX | fcntl.LOCK_NB)
            self.phase = 'EX_HELD'
            self.birth = process(os.getpid())
            self.lock_info = self.verify()
        except BaseException as error:
            remember(error)
            raise
        return self

    def verify(self):
        need(self.phase == 'EX_HELD' and type(self.fd) is int, 'actual global FD is not held')
        actual = os.fstat(self.fd)
        need(stat.S_ISREG(actual.st_mode) and actual.st_uid == os.getuid() and actual.st_nlink == 1 and
             HISTORICAL_LOCK.resolve(strict=True) == HISTORICAL_LOCK and
             identity(HISTORICAL_LOCK.lstat()) == identity(actual), 'global lock canonical inode/owner differs')
        need(fcntl.fcntl(self.fd, fcntl.F_GETFD) & fcntl.FD_CLOEXEC, 'global FD must close on exec')
        info = metadata('/proc/self/fdinfo/' + str(self.fd), 16384).decode('ascii')
        fields = dict(line.split(':', 1) for line in info.splitlines() if ':' in line)
        need(int(fields['ino'].strip()) == actual.st_ino and
             any('FLOCK' in line and 'WRITE' in line and (' ' + str(os.getpid()) + ' ') in line
                 for line in info.splitlines() if line.startswith('lock:')), 'actual held EX flock fdinfo missing')
        current = process(os.getpid())
        if self.birth is not None:
            need(current['pid'] == self.birth['pid'] and current['start_ticks'] == self.birth['start_ticks'],
                 'global lease process generation changed')
        return {'path': str(HISTORICAL_LOCK), 'fd_number_observed': self.fd,
                'fstat': list(identity(actual)), 'fdinfo': info, 'parent': current,
                'source_acquired_here': True, 'FD_CLOEXEC': True,
                'not_JSON_or_native_memory_authority': True}

    def close(self):
        if self.fd is None:
            return
        self.phase = 'CLOSE_ENTERED'
        try:
            os.close(self.fd)  # Only release point; no unlock/duplicate/SCM transfer.
        except BaseException as error:
            self.phase = 'CLOSE_UNKNOWN'
            IO_UNKNOWN[0] = True
            remember(error)
            raise
        self.phase = 'CLOSE_RETURNED'

    sys.modules[module_name] = module
    exec(compile(raw, str(path), 'exec', dont_inherit=True), module.__dict__)
    RETAINED_MODULES.append(module)
    return module, row


def clean_launch():
    need(sys.executable == '/usr/bin/python3.12' and sys.implementation.name == 'cpython' and
         sys.version_info[:2] == (3, 12) and sys.base_prefix == '/usr', 'actual selected interpreter/prefix')
    need(sys.flags.optimize == 0 and sys.flags.no_site == sys.flags.no_user_site ==
         sys.flags.dont_write_bytecode == 1 and sys.flags.safe_path and
         sys.flags.ignore_environment == sys.flags.isolated == sys.flags.hash_randomization == 0 and
         tuple(sys.path) == INITIAL_PATH and dict(os.environ) == ENV, 'clean owning-parent launch differs')
    need(os.getcwd() == str(W), 'exact parent working directory')
    need(len(IO.names('/proc/self/task', 2)) == 1 and
         signal.getitimer(signal.ITIMER_REAL) == (0.0, 0.0) and
         signal.getsignal(signal.SIGALRM) == signal.SIG_DFL,
         'single-task/no prior alarm or incompatible SIGALRM owner required')
    status = metadata('/proc/self/status').decode('ascii')
    values = dict(line.split(':', 1) for line in status.splitlines() if ':' in line)
    need(int(values['Umask'].strip(), 8) == 0o077, 'actual inherited umask must be0077')
    streams = {}
    for fd in (1, 2):
        info = os.fstat(fd)
        need(stat.S_ISREG(info.st_mode) and stat.S_IMODE(info.st_mode) == 0o600 and
             info.st_uid == os.getuid() and info.st_nlink == 1, 'captured regular0600 stdout/stderr')
        streams[str(fd)] = list(identity(info))
    return {'actual_parent': process(os.getpid()), 'sys_base_prefix': sys.base_prefix,
            'environment': dict(os.environ), 'initial_path': list(sys.path), 'stdio': streams,
            'umask_observed': 0o077, 'prior_SIGALRM_handler': 'SIG_DFL',
            'operating_TCB_not_loaded_image_proof': True}

def snapshot_domain(pins):
    result = {}
    for path, expectations in pins.items():
        checkpoint()

def writer_probe(pins):
    """Actual bounded observations under cooperative closed-world lease premise.

    Not an atomic all-host proof. Same-UID churn/permission/unknown IO fails closed;
    root/admin/hostile actors and mount/namespace replacement need external scope.
    """
    protected = {(row['identity'][0], row['identity'][1]) for row in pins.values()}
    observations, aliases = [], []
    for name in IO.names('/proc', 65536):
        checkpoint()
        if not name.isdecimal():
            continue
        path = Path('/proc') / name
        info = path.stat()
        if info.st_uid != os.getuid():
            continue
        before = process(int(name))
        fd_hash, fd_count = hashlib.sha256(), 0
        for descriptor in IO.names(path / 'fd', 4096):
            checkpoint()
            fdpath = path / 'fd' / descriptor
            try:
                st = fdpath.stat()
            except FileNotFoundError:
                # No descriptor was acquired by stat. In particular our closed
                # census/metadata FDs can disappear; this is not release proof.
                fd_hash.update(canonical({'fd': descriptor, 'observation': 'FD_LINK_ABSENT_AT_STAT'}))
                continue
            data = metadata(path / 'fdinfo' / descriptor, 16384).decode('ascii')
            fields = dict(line.split(':', 1) for line in data.splitlines() if ':' in line)
            flags = int(fields['flags'].strip(), 8)
            row = {'fd': descriptor, 'dev': st.st_dev, 'ino': st.st_ino, 'flags': flags}
            fd_hash.update(canonical(row));fd_count += 1
            if (st.st_dev, st.st_ino) in protected and flags & os.O_ACCMODE != os.O_RDONLY:
                aliases.append({'pid': int(name), 'birth': before['start_ticks'], 'fd': descriptor, 'inode': row})
        maps = metadata(path / 'maps', 2097152).decode('utf-8', 'strict')
        for line in maps.splitlines():
            checkpoint()
            fields = line.split(None, 5)
            if len(fields) < 5 or fields[4] == '0':
                continue
            devparts = fields[3].split(':')
            dev = os.makedev(int(devparts[0], 16), int(devparts[1], 16))
            if 'w' in fields[1] and 's' in fields[1] and (dev, int(fields[4])) in protected:
                aliases.append({'pid': int(name), 'birth': before['start_ticks'], 'writable_shared_map': line[:4096]})
        after = process(int(name))
        need(before['start_ticks'] == after['start_ticks'], 'process generation churn during writer probe')
        observations.append({'before': before, 'after': after, 'FD_count': fd_count,
                             'FD_census_sha256': fd_hash.hexdigest(), 'maps_sha256': digest(maps.encode())})
        need(len(observations) <= 8192, 'same-UID process census cap')
    need(not aliases, 'writable original/source-domain alias found')
    return {'processes': observations, 'writable_aliases': aliases,
            'atomic_or_hostile_writer_exclusion_proved': False,
            'requires_reviewed_registered_writer_and_admin_exclusion_premise': True}

class CaseBudget:
    """Original case_begin through retained validation; no hard-fault theorem."""
    def __init__(self):
        self.begin, self.deadline, self.case = None, None, None
        self.active = False

    def start(self, begin, case):
        need(not self.active and type(begin) is float and math.isfinite(begin) and
             0 <= time.monotonic() - begin < 180, 'actual original case start absent/expired')
        need(signal.getitimer(signal.ITIMER_REAL) == (0.0, 0.0) and
             signal.getsignal(signal.SIGALRM) == signal.SIG_DFL,
             'case cannot borrow alarm or replace incompatible signal owner')
        self.begin, self.deadline, self.case = begin, begin + 180, case
        self.active = True
        ACTIVE_CASE[0] = self
        self.check()

    def check(self):
        need(self.active and time.monotonic() < self.deadline, 'complete case180 exhausted')

    def close(self):
        if self.active:
            self.active = False
            ACTIVE_CASE[0] = None

    def original_metric(self, case):
        # Separate paid read of original measurement, never cached validate PASS.
        path = ROOT / 'validation/attempts' / NAME / (case['label'] + '.gate.json')
        row, pin = read_json(path, 4194304, 0o600)
        seconds = row['parent_command_wall_seconds_through_validation']
        need(type(seconds) in (int, float) and math.isfinite(seconds) and 0 <= seconds <= 180,
             'actual original complete-case measurement exceeds180; no next Root')
        return {'case': case['label'], 'original_complete_seconds': seconds, 'gate': pin}


class ActualRootResults:
    """Retain genuine run_root return objects, not restored Root capabilities."""
    def __init__(self, held_run_root, clock, case_bill):
        self.original, self.clock, self.bill = held_run_root, clock, case_bill
        self.results, self.entered, self.next = [None, None], [False, False], 0

    def run_root(self, root, case, evidence, seconds, shared_deadline,
                 as_bytes=201326592, rss_bytes=201326592):
        index = self.next
        need(index < 2 and case['job'] == ('normal', 'optimized')[index] and
             self.clock.active and self.clock.case == case, 'real root call order/clock')
        self.clock.check()
        deadline = min(shared_deadline, self.clock.deadline)
        row = self.bill[case['job']]
        validation_tail = (row['validation_ns']+row['closing_ns'])/1000000000
        allowance = min(seconds,row['root_ns']/1000000000,deadline-time.monotonic()-2-validation_tail)
        need(0 < allowance <= 180, 'no actual case allowance with reserved cleanup')
        self.next += 1
        self.entered[index] = True
        try:
            self.clock.check()
            self.results[index] = self.original(root, case, evidence, allowance, deadline,
                                                as_bytes, rss_bytes)
        except BaseException as error:
            remember(error)
            raise
        # Retain actual return BEFORE cooperative post-return deadline check.
        self.clock.check()
        return self.results[index]  # Same actual object returned unchanged.


class StrictQueueFacade:
    """Explicit transport; held derivative validator executes once per actual case."""
    def __init__(self, queue, proxy, lease, supervisor, pins, before, inputs, spec, binding, evidence, clock):
        self.queue, self.proxy, self.lease, self.s = queue, proxy, lease, supervisor
        self.pins, self.before, self.inputs, self.spec = pins, before, inputs, spec
        self.binding, self.evidence, self.clock = binding, evidence, clock
        self.original_validate, self.original_main_code = queue.validate, queue.main.__code__
        self.nested_root = proxy.evidence
        self.validated, self.observations, self.case_metrics = [], [], []

    def run(self, command, **kwargs):
        frame, local = None, None
        try:
            # Explicit read-only CPython frame boundary, no JSON clock/capability.
            frame = sys._getframe(1)
            need(frame.f_code is self.original_main_code, 'caller is not held original queue.main')
            local = frame.f_locals
            case = local['case']
            need(local['command'] is command and case == self.proxy.cases[self.proxy.next],
                 'actual original caller case/command differs')
            begin = local['case_begin']
            frame, local = None, None  # Release transient references before any IO/Root.
            self.clock.start(begin, case)
            if self.validated:
                self.case_metrics.append(self.clock.original_metric(self.proxy.cases[self.proxy.next - 1]))
            self.lease.verify()
            checkpoint()
            need(self.proxy.next == len(self.validated), 'previous real Root has not passed strict validation')
            need(snapshot_domain(self.pins) == self.before, 'before-Root original/source custody changed')
            writer_probe(self.before)
            checkpoint()
            result = self.proxy.run(command, **kwargs)
            checkpoint()  # Opaque original proxy has returned; no parent alarm.
            return result
        except BaseException as error:
            remember(error)  # BEFORE unchanged queue.main finally may fail later.
            try:
                self.clock.close()
            except BaseException as secondary:
                remember(secondary)
            raise
        finally:
            frame, local = None, None

    def validate(self, case):
        try:
            # Original queue already rejected nonzero/stderr before this call.
            checkpoint()  # Immediately before opaque held original validator.
            row = self.original_validate(case,self.nested_root,self.s.RUNNER_PATH)  # Sole actual validator call.
            checkpoint()
            self.lease.verify()
            folder = self.proxy.evidence / (case['label']+'-supervision')
            supervision, unused = read_json(folder / 'SUPERVISION.json', 4194304, 0o600)
            cleanup = supervision['cleanup']
            need(supervision['returncode'] == 0 and supervision['failure'] is None and
                 cleanup['complete'] is True and cleanup['group_empty_proved'] is True and
                 cleanup['remaining'] == [] and cleanup['observed_after_root_exit'] == [] and
                 cleanup['reaped_adopted'] == [] and not self.s.children_now(), 'actual owned cleanup not complete')
            data = strict_data(case, self.inputs, self.spec, self.binding, self.nested_root)
            need(snapshot_domain(self.pins) == self.before, 'post-Root original/source custody changed')
            writers = writer_probe(self.before)
            self.lease.verify()
            checkpoint()
            record = {'queue_original_row': row, 'strict_DATA': data, 'writers': writers,
                      'global_lease': self.lease.lock_info, 'cleanup': cleanup,
                      'actual_original_case_begin': self.clock.begin, 'case_deadline': self.clock.deadline}
            pin = IO.write(self.evidence / (case['job'] + '-STRICT.json'), record)
            checkpoint()  # Full strict record retained before success.
            self.observations.append(record)
            self.validated.append(case['job'])
            row['strict_parent_DATA_receipt'] = pin
            return row
        except BaseException as error:
            remember(error)  # Includes original_validate and each strict check.
            raise
        finally:
            try:
                self.clock.close()
            except BaseException as secondary:
                remember(secondary)
                raise


def retain_queue_failure(error, queue):
    """Keep actual native context; classify only held queue body traceback.

    No ambient sys.exception or trace hook. No universal recovery claim for
    underlying caught errors. Existing facade first object always keeps priority.
    """
    chain, item = [], error
    while item is not None and all(item is not held for held in chain) and len(chain) < 32:
        chain.append(item)
        item = item.__context__
    if item is not None:
        IO_UNKNOWN[0] = True
    if FIRST[0] is None:
        for candidate in reversed(chain):
            tb = candidate.__traceback__
            while tb is not None:
                # Exact pinned original queue body's try/except, not its finally.
                if tb.tb_frame.f_code is queue.main.__code__ and 124 <= tb.tb_lineno <= 157:
                    remember(candidate)
                    break
                tb = tb.tb_next
            if FIRST[0] is not None:
                break
    remember(error)
    for candidate in chain:
        remember(candidate)
        if candidate.__cause__ is not None:
            remember(candidate.__cause__)

def error_data():
    def item(error):
        if error is None:
            return None
        frames, tb = [], error.__traceback__
        while tb is not None and len(frames) < 32:

                           'function': tb.tb_frame.f_code.co_name})
            tb = tb.tb_next
        return {'type': type(error).__name__, 'frames': frames}
    return {'first': item(FIRST[0]), 'secondary': [item(e) for e in SECONDARY if e is not None],
            'IO_unknown': IO_UNKNOWN[0], 'actual_exception_objects_retained_until_exit': True}

class ExecutionBarrier:
    """Persistent closed-one-parent barrier; never a claim of orphan disposal."""
    def __init__(self):
        self.slot, self.phase = None, 'NEW'
        self.parent_slots = [None, None]

    def prepare(self, lease):
        private_directory(QUARANTINE.parent)  # Admitted initializer owns fresh sibling namespace.
        successor_guard(lease)  # Exact history rechecked under this actual EX before NEW marker.
        need(not os.path.lexists(ROOT/'validation'/(NAME+'-SEALED-ONE-PARENT')), 'own sealed oneuse marker already exists')
        self.phase = 'PREPARING'
        need(ROOT.resolve(strict=True) == ROOT and (ROOT / 'validation').resolve(strict=True) == ROOT / 'validation' and
             QUARANTINE.parent.resolve(strict=True) == QUARANTINE.parent, 'canonical retained namespaces')
        for index, directory in enumerate((BARRIER.parent, QUARANTINE.parent)):
            self.parent_slots[index] = IO.acquire(directory, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW | os.O_CLOEXEC)
        BARRIER.mkdir(mode=0o700)
        IO.write(BARRIER / 'START.json', {'kind': 'persistent_execution_quarantine',
                 'source_owner': lease.verify(), 'future_actual_operation_prohibited_until_resolution': True,
                 'not_an_attempt_PASS_or_runtime_qualification': True})
        self.slot = IO.acquire(BARRIER / 'details.jsonl', os.O_WRONLY | os.O_CREAT | os.O_EXCL |
                               os.O_NOFOLLOW | os.O_CLOEXEC)
        for parent_slot in self.parent_slots:
            os.fsync(parent_slot[0])
        self.phase = 'PREPARED'

    def quarantine(self):
        # No new FD or original-domain read is needed. Keep first evidence if
        # allocation/write/fsync/rename fails; PREPARED is itself a blocking state.
        self.phase = 'QUARANTINED'
        if self.slot is not None and self.slot[1] == 'OWNED':
            try:
                raw = canonical({'status': 'EXECUTION_QUARANTINED', 'errors': error_data(),
                      'future_actual_operation_prohibited': True,
                      'unknown_remaining_writers_not_disposed_by_lock_release': True,
                      'external_actual_teardown_and_ownership_resolution_required': True})
                offset = 0
                while offset < len(raw):
                    n = os.write(self.slot[0], raw[offset:offset + 65536])
                    need(type(n) is int and n > 0, 'quarantine detail write')
                    offset += n
                os.fsync(self.slot[0])
            except BaseException as error:
                remember(error)
        try:
            if not QUARANTINE.exists():
                source = BARRIER if BARRIER.exists() else ROOT / 'validation' / (NAME + '-SEALED-ONE-PARENT')
                os.rename(source, QUARANTINE)  # Unclosed run: unchanged budget/runner refuse future roots.
            for parent_slot in self.parent_slots:
                if parent_slot is not None and parent_slot[1] == 'OWNED':
                    os.fsync(parent_slot[0])
        except BaseException as error:
            remember(error)  # Remaining canonical BARRIER still blocks registered parent admission.
        self.close()

    def seal_success(self):
        need(not IO_UNKNOWN[0], 'barrier ownership unknown; cannot seal success')
        os.rename(BARRIER, ROOT / 'validation' / (NAME + '-SEALED-ONE-PARENT'))
        for parent_slot in self.parent_slots:
            if parent_slot is not None and parent_slot[1] == 'OWNED':
                os.fsync(parent_slot[0])
        self.close()
        need(not IO_UNKNOWN[0], 'barrier close unknown; cannot seal success')
        self.phase = 'SEALED_SUCCESS'

    def close(self):
        for slot in (self.slot, *self.parent_slots):
            if slot is not None and slot[1] == 'OWNED':
                try:
                    IO.close(slot)
                except BaseException as error:
                    remember(error)

            need(type(row[key]) is int and 0 <= row[key] <= 180000000000,'exact nonoverlapping case bill term')
            total += row[key]
        need(row['root_ns'] > 0 and row['validation_ns'] > 0 and row['cleanup_ns'] >= 2000000000,
             'full actual root+validation+cleanup pricing required')
    need(type(value['whole_command_ns']) is int and total == value['whole_command_ns'] <= request['body_reservation_ns'] <= 180000000000,
         'complete declared command, not two180 roots or old420 arithmetic')
    # Every case term is INSIDE queue.main, including its prologue/footer
    # allocated to case guard/closing terms. All nonqueue work is below.
    outside_declared_ns = value['startup_preflight_ns']+value['controller_final_ns']+value['outer_post_controller_ns']
    need(outside_declared_ns <= 30000000000, 'declared complete nonqueue/outside30 scope exceeded')
    need(total-outside_declared_ns <= 390000000000, 'declared queue390 scope exceeded')

    value['queue_declared_ns'] = total-outside_declared_ns
    value['remaining_before_queue_ns'] = total-value['startup_preflight_ns']
    return value


def actual_context_ports(active_module,ctx):
    """Fixed safe dispatch consistency; loaded source authentication is external."""
    cls = type(ctx)
    need(type(cls) is type, 'ordinary exact private context class')
    mro = type.__getattribute__(cls,'__mro__')
    need(len(mro) == 2 and mro[0] is cls and mro[1] is object, 'direct private context class')
    namespace = type.__getattribute__(cls,'__dict__')
    keys = tuple(namespace)
    need(all(type(k) is str for k in keys), 'exact context class keys')
    image = {k:namespace[k] for k in keys}
    methods = ('initialize','refresh_opening','current_binding','begin','native_lease_for_controller',
               'check','release_for_exec','adopt_handoff','publish_reviewed_close','release')
    need(set(image) == {'__module__','__doc__','__slots__','__new__'} | set(methods) and
         type(image.get('__slots__')) is tuple and image['__slots__'] == () and
         type(image['__new__']) is staticmethod and type(image['__new__'].__func__) is FunctionType and
         image['__new__'].__func__.__globals__ is active_module.__dict__, 'fixed empty-slotted A2 context layout')
    need(all(type(image[name]) is FunctionType and image[name].__globals__ is active_module.__dict__
             for name in methods), 'source-private actual context methods')
    def call(name,*args):
        current = type.__getattribute__(cls,'__dict__');current_keys = tuple(current)
        need(all(type(k) is str for k in current_keys) and set(current_keys) == set(image) and
             all(current[k] is image[k] for k in current_keys), 'no context method/special-hook substitution')
        now = type.__getattribute__(cls,'__mro__')
        need(len(now) == 2 and now[0] is cls and now[1] is object, 'unchanged context ancestry')
        return image[name](ctx,*args)  # Actual A2 rec() still enforces private identity/role.
    return call
