class FundingError(ValueError):
    """Stop; preserve evidence. A missing number never becomes a zero receipt."""


def need(condition, message):
    if condition is not True:
        raise FundingError(message)


def plain(value, depth=0):
    need(depth <= 16, 'metadata depth cap')
    t = type(value)
    if t is dict:
        need(len(value) <= 2 * MAX_ENTRIES, 'object cap')
        for k, v in value.items():
            need(type(k) is str and len(k) <= 128, 'exact string key')
            plain(v, depth + 1)
    elif t is list:
        need(len(value) <= 2 * MAX_ENTRIES, 'array cap')
        for v in value:
            plain(v, depth + 1)
    elif t is str:
        need(len(value) <= META_CAP, 'bounded metadata wire/string cap')
    elif t is int:
        need(0 <= value <= MAX_VALUE, 'bounded exact integer data')
    else:
        need(value is None or t is bool, 'only exact JSON integer/bool/null data')


def exact(value, keys):
    need(type(value) is dict and set(value) == set(keys), 'exact object fields')


def uint(value):
    need(type(value) is int and 0 <= value <= MAX_VALUE, 'bounded exact integer, not bool/float')
    return value


def digest(value):
    need(type(value) is str and len(value) == 64 and all(c in '0123456789abcdef' for c in value), 'sha256')
    return value


def label(value):
    need(type(value) is str and 1 <= len(value) <= 64 and
         all(c in 'abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789-_' for c in value), 'simple label')
    return value


def pin(value):
    plain(value)
    exact(value, ('path', 'bytes', 'sha256', 'mode'))
    need(type(value['path']) is str and Path(value['path']).is_absolute(), 'absolute pin path')
    uint(value['bytes']); digest(value['sha256'])
    need(type(value['mode']) is int and value['mode'] in (0o444, 0o600), 'immutable control or private record mode')
    need(value['bytes'] <= META_CAP, 'metadata byte cap')
    return value


def pairs(items):
    result = {}
    for k, v in items:
        need(k not in result, 'duplicate JSON key')
        result[k] = v
    return result


def forbidden_number(unused):
    raise FundingError('JSON floating/nonfinite number forbidden')


def decode(raw):
    need(type(raw) is bytes and 0 < len(raw) <= META_CAP, 'bounded exact bytes')
    obj = json.loads(raw.decode('utf-8'), object_pairs_hook=pairs,
                     parse_float=forbidden_number, parse_constant=forbidden_number)
    plain(obj)
    return obj


def registry(raw):

    exact(obj, ('schema', 'round_id', 'authorization_sha256', 'opening_pin',
                'predecessor_sha256', 'phase', 'phase_release_sha256', 'policies', 'events'))
    need(obj['schema'] == 'fresh-round-registry-v2' and obj['round_id'] == ROUND_ID and
         obj['authorization_sha256'] == AUTHORIZATION_SHA256, 'selected new round only')
    pin(obj['opening_pin']); need(obj['opening_pin']['mode'] == 0o444, 'immutable selected opening')
    if obj['predecessor_sha256'] is not None:
        digest(obj['predecessor_sha256'])
    need(obj['phase'] in ('INITIAL', 'FULL'), 'phase')
    need((obj['phase'] == 'INITIAL' and obj['phase_release_sha256'] is None) or
         (obj['phase'] == 'FULL' and type(obj['phase_release_sha256']) is str), 'explicit phase release')
    if obj['phase'] == 'FULL':
        digest(obj['phase_release_sha256'])
    need(type(obj['policies']) is list and type(obj['events']) is list, 'bounded arrays')
    need(len(obj['events']) <= 2 * MAX_ENTRIES, 'event cap')

        need(Path(p['path']).parent == CONTROL and p['mode'] == 0o444 and
             p['sha256'] not in policies, 'selected distinct policy pin')
        policies[p['sha256']] = p

        name = label(event['label']); p = pin(event['pin'])
        need(p['mode'] == 0o600, 'private START/receipt')
        if event['kind'] == 'REGISTER':
            need(name not in starts and len(starts) < MAX_ENTRIES, 'one register, no reused/seed label')
            need(p['path'] == str(DOMAIN / name / 'START.json'), 'only new START')
            starts[name] = p

            need(name in starts and name not in closes, 'close once after register')
            need(p['path'] == str(DOMAIN / name / 'receipt.json'), 'only new receipt')
            closes[name] = p
        else:
            raise FundingError('unexpected event')
    return obj, policies, starts, closes

    need(old['phase'] == new['phase'] or (old['phase'] == 'INITIAL' and new['phase'] == 'FULL'), 'no phase rollback')

        need(old['phase_release_sha256'] == new['phase_release_sha256'], 'phase authority unchanged')
    if old['opening_pin'] != new['opening_pin']:

             old['phase'] == new['phase'] and old['phase_release_sha256'] == new['phase_release_sha256'],

        _, _, starts, closes = registry(previous_raw)
        need(set(starts) == set(closes) == set(opening_transition['records']), 'all formal records CLOSED before refresh')

        validate_opening_partition(opening_transition['current'], old, opening_transition['records'])

def directory_identity(io, path, owner_uid):
    need(path.is_absolute() and path.resolve(strict=True) == path, 'canonical new directory')
    st = path.lstat()
    need(stat.S_ISDIR(st.st_mode) and st.st_uid == owner_uid and
         stat.S_IMODE(st.st_mode) == 0o700, 'private current-owner directory')
    return (st.st_dev, st.st_ino, st.st_mode, st.st_nlink, st.st_uid,
            st.st_mtime_ns, st.st_ctime_ns)


def read_pinned(io, expected, owner_uid):
    pin(expected)
    row, raw = io.read(Path(expected['path']), META_CAP, mode=expected['mode'],
                       expected={'bytes': expected['bytes'], 'sha256': expected['sha256']}, materialize=True)
    need(type(raw) is bytes and row['uid'] == owner_uid and row['nlink'] == 1 and
         row['mode'] == expected['mode'] and row['bytes'] == expected['bytes'] and
         row['sha256'] == expected['sha256'] and len(raw) == expected['bytes'] and
         hashlib.sha256(raw).hexdigest() == expected['sha256'], 'actual owned stable file differs')
    return row, raw


def snapshot_round(io, registry_pin, owner_uid):
    """Future actual-IO operation; caller owns native lease, not any JSON flag.

    Failures propagate: do not retry, silently skip entries, or create receipts.
    io.names and io.read must be the reviewed actual OwnedIO methods; read must
    retain canonical/lstat/open/fstat/post-path and known-close guarantees.
    """
    # The caller authenticates exact type(io), original source hash and module
    # instance; this function does not turn an arbitrary reader into authority.
    # It audits NEW admission only. It grants no in-flight/pending exemption.
    uint(owner_uid); plain(registry_pin); pin(registry_pin)
    rp = Path(registry_pin['path'])
    middle = rp.name[len('REGISTRY-'):-len('.json')]
    need(rp.parent == CONTROL and rp.name.startswith('REGISTRY-') and rp.name.endswith('.json') and
         len(middle) == 6 and all(c in '0123456789' for c in middle) and
         registry_pin['mode'] == 0o444, 'explicit new hash-selected registry file')
    seen = []
    def take(p):
        row, raw = read_pinned(io, p, owner_uid)
        seen.append((p, row, raw))
        return raw
    raw_registry = take(registry_pin)
    reg, policy_pins, starts, closes = registry(raw_registry)
    opening_snapshot = read_opening(io, reg['opening_pin'], owner_uid)
    before = directory_identity(io, DOMAIN, owner_uid)
    names = io.names(DOMAIN, MAX_ENTRIES + 1)
    need(type(names) is list and all(type(n) is str for n in names) and
         len(names) == len(set(names)) and set(names) == set(starts), 'every new entry exactly registered')
    records, directories = {}, {}
    for name in sorted(starts):
        folder = DOMAIN / name
        identity = directory_identity(io, folder, owner_uid)
        members = io.names(folder, 3)
        expected_members = {'START.json', 'receipt.json'} if name in closes else {'START.json'}
        need(type(members) is list and len(members) == len(set(members)) and
             set(members) == expected_members, 'exact new entry completeness; no hidden/missing receipt')
        start = decode(take(starts[name]))
        receipt = decode(take(closes[name])) if name in closes else None
        directories[name] = (identity, expected_members)
        records[name] = {'start': start, 'receipt': receipt}
    policies = {k: decode(take(p)) for k, p in policy_pins.items()}
    for p, row, raw in seen:
        after_row, after_raw = read_pinned(io, p, owner_uid)
        need(after_raw == raw and after_row == row, 'full pre/post immutable snapshot')
    for name, (identity, members) in directories.items():
        need(directory_identity(io, DOMAIN / name, owner_uid) == identity and
             set(io.names(DOMAIN / name, 3)) == members, 'new entry directory/census changed')
    need(directory_identity(io, DOMAIN, owner_uid) == before and
         set(io.names(DOMAIN, MAX_ENTRIES + 1)) == set(names), 'new round census changed')
    opening_after = read_opening(io, reg['opening_pin'], owner_uid)
    need(opening_after == opening_snapshot, 'whole snapshot opening remains current and unchanged')
    validate_opening_partition(opening_snapshot, reg, records)
    return {'schema': 'fresh-round-owned-snapshot-v2', 'registry': reg,
            'registry_pin': registry_pin, 'records': records, 'policies': policies,
            'opening_snapshot': opening_snapshot}

    """Pure DATA assessment. Eligibility is NOT permission/native qualification."""
    plain(snapshot); plain(request)
    exact(snapshot, ('schema', 'registry', 'registry_pin', 'records', 'policies',
                     'opening_snapshot'))
    exact(request, ('schema', 'round_id', 'label', 'command_sha256',
                    'body_reservation_ns', 'policy_sha256', 'phase_release_sha256'))
    need(snapshot['schema'] == 'fresh-round-owned-snapshot-v2' and
         request['schema'] == 'fresh-round-request-v1' and request['round_id'] == ROUND_ID,
         'new round data only')
    opening = opening_evidence(snapshot['opening_snapshot'])
    need(snapshot['opening_snapshot']['opening_pin'] == snapshot['registry']['opening_pin'], 'selected opening snapshot')
    reg = snapshot['registry']
    # Revalidate exact registry shape and event semantics for direct DATA callers.
    _, policy_pins, starts, closes = registry((json.dumps(reg, separators=(',', ':')) + '\n').encode('utf-8'))
    need(set(snapshot['records']) == set(starts) and set(snapshot['policies']) == set(policy_pins), 'snapshot closed domain')
    need(request['phase_release_sha256'] == reg['phase_release_sha256'], 'separately authorized phase binding')
    cap = INITIAL_NS if reg['phase'] == 'INITIAL' else TOTAL_NS
    policies = snapshot['policies']
    for key, p in policies.items():
        exact(p, ('schema', 'round_id', 'command_sha256', 'administrative_debit_ns',
                  'scope', 'adoption_sha256', 'claims_physical_upper_bound'))
        need(p['schema'] == 'fresh-round-admin-policy-v1' and p['round_id'] == ROUND_ID and
             p['scope'] == 'STARTUP_AND_FINAL_METADATA_PUBLICATION_ONLY' and
             p['claims_physical_upper_bound'] is False, 'admin convention never body/cleanup cover')
        digest(p['command_sha256']); digest(p['adoption_sha256']); uint(p['administrative_debit_ns'])
    opening = validate_opening_partition(snapshot['opening_snapshot'], reg, snapshot['records'])
    body, admin, reserved, unknown, blockers, commands, open_tails = (
        opening['manual_body_ns'], opening['manual_admin_ns'], 0, [], list(opening['blockers']),
        set(opening['manual_command_sha256s']), [])
    for name in starts:
        bundle = snapshot['records'][name]
        exact(bundle, ('start', 'receipt'))
        s, r = bundle['start'], bundle['receipt']
        exact(s, ('schema', 'round_id', 'label', 'command_sha256', 'body_reservation_ns', 'policy_sha256', 'pre_entry_provenance'))
        validate_pre_entry_provenance(s['pre_entry_provenance'], name, s['command_sha256'], s['body_reservation_ns'])
        need(s['schema'] == 'fresh-round-start-v1' and s['round_id'] == ROUND_ID and s['label'] == name, 'START binding')
        cmd = digest(s['command_sha256']); digest(s['policy_sha256'])
        need(name not in opening['manual_entry_ids'] and cmd not in commands and
             s['policy_sha256'] in policies, 'no imported/formal duplicate/reroll or undeclared policy')
        commands.add(cmd); p = policies[s['policy_sha256']]
        need(p['command_sha256'] == cmd and 0 < uint(s['body_reservation_ns']) <= BODY_LIMIT_NS, 'per-command body reservation')
        fee = uint(p['administrative_debit_ns'])
        if r is None:
            need(name not in closes, 'CLOSE without receipt')
            reserved += s['body_reservation_ns'] + fee
            unknown.append(name); blockers.append(name + ':MISSING_NEW_RECEIPT')
            continue
        need(name in closes, 'unregistered new receipt')
        exact(r, ('schema', 'round_id', 'label', 'start_sha256', 'outcome', 'body_elapsed_ns',
                  'administrative_debit_ns', 'body_begin_ns', 'body_end_ns', 'body_fully_observed', 'cleanup_known', 'custody_complete',
                  'nested_entries_complete', 'uncertainty', 'peak_sampled_rss_bytes', 'tail_state',
                  'independent_outcome_review_sha256'))
        need(r['schema'] == 'fresh-round-receipt-v1' and r['round_id'] == ROUND_ID and
             r['label'] == name and r['start_sha256'] == starts[name]['sha256'] and
             r['outcome'] in ('PASS', 'STOP') and uint(r['administrative_debit_ns']) == fee, 'receipt identity/fee')
        for k in ('body_begin_ns', 'body_end_ns'):
            if r[k] is not None:
                uint(r[k])
        admin += fee  # Failure fees and measured failure bodies are retained in full.
        if r['body_elapsed_ns'] is None:
            unknown.append(name); blockers.append(name + ':UNPRICED_BODY')
            reserved += s['body_reservation_ns']
        else:
            elapsed = uint(r['body_elapsed_ns'])
            need(type(r['body_begin_ns']) is int and type(r['body_end_ns']) is int and
                 r['body_end_ns'] >= r['body_begin_ns'] and
                 r['body_end_ns'] - r['body_begin_ns'] == elapsed, 'exact own body endpoints/difference')
            body += elapsed
            if elapsed > BODY_LIMIT_NS or elapsed > s['body_reservation_ns']:
                blockers.append(name + ':BODY_RESERVATION_OR180_EXCEEDED')
        for k in ('body_fully_observed', 'cleanup_known', 'custody_complete', 'nested_entries_complete'):
            need(type(r[k]) is bool, 'exact boolean qualification')
            if r[k] is not True:
                blockers.append(name + ':' + k.upper())
        need(type(r['uncertainty']) is list and all(type(x) is str for x in r['uncertainty']), 'uncertainty list')
        if r['uncertainty']:
            blockers.append(name + ':UNRESOLVED_UNCERTAINTY')
        if r['peak_sampled_rss_bytes'] is None:
            blockers.append(name + ':UNKNOWN_SAMPLED_RSS')
        elif uint(r['peak_sampled_rss_bytes']) > RSS_LIMIT:
            blockers.append(name + ':SAMPLED_RSS_EXCEEDED')
        need(r['tail_state'] in ('COMPLETE', 'ADMIN_ONLY_OPEN', 'UNKNOWN'), 'tail state')
        if r['tail_state'] == 'UNKNOWN':
            blockers.append(name + ':UNKNOWN_NONADMIN_TAIL')
        elif r['tail_state'] == 'ADMIN_ONLY_OPEN':
            open_tails.append(name)
        digest(r['independent_outcome_review_sha256'])
    label(request['label']); cmd = digest(request['command_sha256'])
    need(request['label'] not in starts and request['label'] not in opening['manual_entry_ids'] and cmd not in commands, 'one use, no automatic retry')
    need(request['policy_sha256'] in policies, 'request separately adopted policy absent')
    policy = policies[request['policy_sha256']]
    need(policy['command_sha256'] == cmd, 'request exact policy/command')
    wanted_body = uint(request['body_reservation_ns'])
    need(0 < wanted_body <= BODY_LIMIT_NS, 'requested body <=180s; actual fit remains separate')
    wanted = wanted_body + policy['administrative_debit_ns']
    charged = body + admin
    remaining = cap - charged - reserved  # No clipping, no unknown-body zero receipt.
    if remaining < wanted:
        blockers.append('INSUFFICIENT_POLICY_ALLOWANCE')
    return {'schema': 'fresh-round-funding-assessment-v1', 'round_id': ROUND_ID,
            'funding_data_eligible_only': not blockers, 'blockers': blockers,
            'cap_ns': cap, 'known_observed_body_ns': body, 'administrative_policy_ns': admin,
            'known_policy_charged_ns': charged, 'outstanding_reservation_ns': reserved,
            'unpriced_new_entries': unknown, 'remaining_policy_ns': remaining,
            'requested_reservation_ns': wanted, 'new_admin_only_open_tail_entries': open_tails,
            'opening_physical_tail_open': True, 'complete_physical_total_ns': None,
            'old_historical_total_ns': None, 'all_costs_settled': False,
            'native_ownership_or_cleanup_authenticated': False, 'execution_permission': False,
            'command_resource_profile_or_economic_fit_proved': False}

def metadata_decode(raw):
    need(type(raw) is bytes and 0 < len(raw) <= META_CAP, 'bounded exact metadata bytes')
    # Legacy diagnostic floats are retained as lexical strings, NEVER charges.
    value = json.loads(raw.decode('utf-8'), object_pairs_hook=pairs,

    metadata_plain(value)
    return value


def metadata_plain(value, depth=0):
    need(depth <= 16, 'metadata depth')
    t = type(value)
    if t is dict:
        need(len(value) <= 256, 'metadata object bound')
        for k, v in value.items():
            need(type(k) is str and len(k) <= 128, 'metadata key')
            metadata_plain(v, depth + 1)
    elif t is list:
        need(len(value) <= 256, 'metadata array bound')
        for v in value:
            metadata_plain(v, depth + 1)
    elif t is str:
        need(len(value) <= META_CAP, 'metadata string bound')
    elif t is int:
        need(-MAX_VALUE <= value <= MAX_VALUE, 'bounded reported metadata integer')
    else:
        need(value is None or t is bool, 'exact metadata builtin types')


def select(value, path):
    need(type(path) is list and 1 <= len(path) <= 8 and all(type(k) is str for k in path), 'literal field path')
    for key in path:
        need(type(value) is dict and key in value, 'original metadata field absent')
        value = value[key]
    return value


def seconds_ns(value):
    need(type(value) is str and value.count('.') <= 1, 'canonical decimal seconds string')
    parts = value.split('.')
    whole = parts[0]
    need(whole == '0' or (whole and whole[0] in '123456789' and all(c in '0123456789' for c in whole)), 'canonical whole seconds')
    fraction = parts[1] if len(parts) == 2 else ''
    need(len(fraction) <= 9 and all(c in '0123456789' for c in fraction), 'integer-ns decimal precision')
    return uint(int(whole) * NS + (int(fraction.ljust(9, '0')) if fraction else 0))

def review_pin_rows(review):
    """Closed native-tool family review inputs; still retained DATA, not authority."""
    values = review['inputs']
    need(type(values) is dict and 1 <= len(values) <= 256, 'typed closed review input pin map')
    result = []
    for key, value in values.items():
        need(type(key) is str, 'review pin key')
        exact(value, ('path', 'bytes', 'sha256', 'mode'))
        need(type(value['path']) is str and Path(value['path']).is_absolute(), 'review original absolute pin')
        uint(value['bytes']); digest(value['sha256']); uint(value['mode'])
        need(value['mode'] <= 0o777, 'review original mode metadata')
        result.append(value)
    return result

    raw = snapshot['descriptor_wire'].encode('utf-8')
    need(len(raw) == op['bytes'] and hashlib.sha256(raw).hexdigest() == op['sha256'], 'opening wire pin')

         desc['physical_final_boundary'] == 'OPEN_NOT_ZERO', 'selected opening scope')

            if type(actual) is not type(check['expected']) or actual != check['expected']:
                blockers.append(name + ':IMPORTED_UNCLOSED_CONDITION')
        pair = {'cost': origin_ref(desc['files'][row['cost']]), 'independent_outcome_review': origin_ref(desc['files'][row['review']])}
        pairs_seen.append(pair)
        state_pairs_seen.append({'cost': pair['cost'], 'independent_review': pair['independent_outcome_review']} if parent_family else pair)
        regs.append(origin_ref(desc['files'][row['registration']])); natives.append(origin_ref(desc['files'][row['native']]))
        views.append({'kind': row['kind'], 'entry_id': name, 'command_sha256': command, 'exact_shell_sha256': row['exact_shell_sha256'], 'body_ns': elapsed,
                      'admin_ns': debit, 'blockers': blockers, 'formal_start_sha256': row['formal_start_sha256'], 'formal_receipt_sha256': row['formal_receipt_sha256']})
        body += elapsed; admin += debit
    if parent_family_seen:
        need(state['schema'] == 'h25-fresh-metered-round-parent-state-v5' and
             selected['schema'] == 'h25-current-bootstrap-opening-source-inputs-v2', 'closed parent state/selection schema variants')
    need(state['accepted_attempts'] == state_pairs_seen and pairs_seen == selected['observations'] and
         selected['pre_entry_registrations'] == regs and selected['genuine_native_results'] == natives and
         selected['current_parent_state'] == origin_ref(desc['files'][desc['state_file']]), 'no accepted operation/provenance omitted')
    need(type(state['guarded_attempts']) is int and state['guarded_attempts'] == len(views) and
         state['observed_body_elapsed_ns_total'] == body and type(state['observed_body_elapsed_ns_total']) is int and
         type(state['administrative_policy_debit_seconds_total']) is int and
         state['administrative_policy_debit_seconds_total'] * NS == admin and
         seconds_ns(state['policy_charged_seconds_total']) == body + admin and
         selected['body_ns'] == body and selected['administrative_policy_ns'] == admin and
         selected['policy_total_ns'] == body + admin and
         state['complete_new_physical_total_seconds'] is None and state['old_complete_total_seconds'] is None and
         state['physical_publication_boundary'] == 'OPEN_NOT_ZERO' and state['all_costs_settled'] is False,
         'current complete selected corpus totals, never physical total')
    exact(desc['expected_totals'], ('body_ns', 'administrative_ns', 'policy_ns', 'operations'))
    for amount in desc['expected_totals'].values():
        uint(amount)
    need(desc['expected_totals'] == {'body_ns': body, 'administrative_ns': admin, 'policy_ns': body + admin, 'operations': len(views)}, 'independent literal expected totals')
    manual = [v for v in views if v['kind'] == 'ORIGINAL_MANUAL_OPERATION']
    return {'descriptor': desc, 'state_sha256': desc['files'][desc['state_file']]['sha256'], 'views': views,
            'manual_body_ns': sum(v['body_ns'] for v in manual), 'manual_admin_ns': sum(v['admin_ns'] for v in manual),
            'manual_entry_ids': [v['entry_id'] for v in manual], 'manual_command_sha256s': [v['command_sha256'] for v in manual],
            'registry_rows': [v for v in views if v['kind'] == 'ALREADY_REGISTERED_OPERATION'],
            'parent_body_ns': body, 'parent_admin_ns': admin,
            'blockers': [reason for v in views for reason in v['blockers']]}


def read_opening(io, opening_pin, owner_uid):
    """Actual source-authenticated OwnedIO boundary, not arbitrary reader authority."""
    pin(opening_pin); need(opening_pin['mode'] == 0o444, 'frozen selected opening')
    row, raw = read_pinned(io, opening_pin, owner_uid)
    desc = decode(raw); folder = Path(opening_pin['path']).parent / 'corpus'
    need(folder.resolve(strict=True) == folder, 'canonical corpus directory')
    before = folder.lstat()
    need(stat.S_ISDIR(before.st_mode) and before.st_uid == owner_uid and stat.S_IMODE(before.st_mode) == 0o555, 'immutable current-owner corpus')
    names = io.names(folder, 8 * OPENING_LIMIT + 1)
    need(type(names) is list and all(type(n) is str for n in names) and
         len(names) == len(set(names)) and set(names) == set(desc['files']), 'exact complete corpus census')
    wires, observed = {}, []
    for name in sorted(names):
        item = desc['files'][name]
        need(type(name) is str and '/' not in name and '..' not in name, 'corpus basename')
        p = {'path': str(folder / name), 'bytes': item['bytes'], 'sha256': item['sha256'], 'mode': 0o444}
        actual, data = read_pinned(io, p, owner_uid); wires[name] = data.decode('utf-8'); observed.append((p, actual, data))
    snapshot = {'schema': 'fresh-current-opening-snapshot-v1', 'opening_pin': opening_pin,
                'descriptor_wire': raw.decode('utf-8'), 'corpus_wires': wires}
    opening_evidence(snapshot)
    for p, actual, data in observed:
        after_row, after = read_pinned(io, p, owner_uid)
        need(after_row == actual and after == data, 'immutable corpus pre/post custody')
    after_row, after = read_pinned(io, opening_pin, owner_uid)
    directory_after = folder.lstat()
    need(after_row == row and after == raw and
         (directory_after.st_dev, directory_after.st_ino, directory_after.st_mode, directory_after.st_uid,
          directory_after.st_mtime_ns, directory_after.st_ctime_ns) ==
         (before.st_dev, before.st_ino, before.st_mode, before.st_uid, before.st_mtime_ns, before.st_ctime_ns) and
         set(io.names(folder, 8 * OPENING_LIMIT + 1)) == set(names), 'opening/corpus current custody')
    return snapshot


def require_opening_append(previous, current):
    a, b = opening_evidence(previous), opening_evidence(current)
    old, new = a['descriptor'], b['descriptor']
    need(new['previous_opening_sha256'] == previous['opening_pin']['sha256'] and
         new['observations'][:len(old['observations'])] == old['observations'] and
         all(new['files'].get(k) == v for k, v in old['files'].items()) and
         new['authorization_sha256'] == old['authorization_sha256'], 'opening immutable prefix; no omission/reprice/reclassification')
    return True


def validate_opening_partition(opening, reg, records):
    evidence = opening_evidence(opening)
    _, _, starts, closes = registry((json.dumps(reg, separators=(',', ':')) + '\n').encode('utf-8'))
    formal_commands = set()
    for name, bundle in records.items():
        s = bundle['start']; command = digest(s['command_sha256']); formal_commands.add(command)
        need(name not in evidence['manual_entry_ids'] and command not in evidence['manual_command_sha256s'], 'manual/formal charges disjoint')
    for v in evidence['registry_rows']:
        name = v['entry_id']
        need(name in starts and name in closes and starts[name]['sha256'] == v['formal_start_sha256'] and
             closes[name]['sha256'] == v['formal_receipt_sha256'] and records[name]['receipt'] is not None,
             'state formal partition bound to actual registered closed records')
        receipt = records[name]['receipt']
        need(records[name]['start']['command_sha256'] == v['command_sha256'] and
             receipt['body_elapsed_ns'] == v['body_ns'] and receipt['administrative_debit_ns'] == v['admin_ns'], 'formal body charged once by registry')
    return evidence


def validate_opening_binding(io, binding, owner_uid):
    plain(binding)
    exact(binding, ('schema', 'opening_pin', 'adoption_pin', 'funding_source_sha256'))
    need(binding['schema'] == 'fresh-current-opening-binding-v1', 'current opening binding')
    digest(binding['funding_source_sha256']); snapshot = read_opening(io, binding['opening_pin'], owner_uid)
    evidence = opening_evidence(snapshot); _, raw = read_pinned(io, binding['adoption_pin'], owner_uid)
    adoption = decode(raw)
    exact(adoption, ('schema', 'opening_sha256', 'state_sha256', 'manual_body_ns', 'manual_admin_ns',
                     'operations', 'authorization_sha256', 'funding_source_sha256', 'authority_scope'))
    for key in ('manual_body_ns', 'manual_admin_ns', 'operations'):
        uint(adoption[key])
    need(adoption == {'schema': 'fresh-current-opening-adoption-v1', 'opening_sha256': binding['opening_pin']['sha256'],
         'state_sha256': evidence['state_sha256'], 'manual_body_ns': evidence['manual_body_ns'],
         'manual_admin_ns': evidence['manual_admin_ns'], 'operations': len(evidence['views']),
         'authorization_sha256': AUTHORIZATION_SHA256, 'funding_source_sha256': binding['funding_source_sha256'],
         'authority_scope': 'OPENING_DATA_SELECTION_ONLY_NOT_OPERATIONAL_GRANT'}, 'selected current opening adoption; not native grant')
    return snapshot


def initial_registry(opening_pin, policy_pins, phase='INITIAL', phase_release_sha256=None):
    plain(opening_pin); plain(policy_pins); plain(phase); plain(phase_release_sha256)
    obj = {'schema': 'fresh-round-registry-v2', 'round_id': ROUND_ID, 'authorization_sha256': AUTHORIZATION_SHA256,
           'opening_pin': opening_pin, 'predecessor_sha256': None, 'phase': phase,
           'phase_release_sha256': phase_release_sha256, 'policies': policy_pins, 'events': []}
    raw = (json.dumps(obj, sort_keys=True, separators=(',', ':')) + '\n').encode('utf-8'); registry(raw)
    return raw



def validate_pre_entry_provenance(value, name, command, reservation_ns):
    plain(value)
    exact(value, ('schema', 'entry_id', 'command_sha256', 'body_reservation_ns', 'registration_pin', 'original_start_pin', 'pending_manifest_pin'))
    need(value['schema'] == 'fresh-original-preentry-provenance-v1' and value['entry_id'] == name and
         value['command_sha256'] == command and type(value['body_reservation_ns']) is int and
         value['body_reservation_ns'] == reservation_ns, 'same genuine enclosing manual/formal operation')
    for key in ('registration_pin', 'original_start_pin', 'pending_manifest_pin'):
        pin(value[key])
    return True
