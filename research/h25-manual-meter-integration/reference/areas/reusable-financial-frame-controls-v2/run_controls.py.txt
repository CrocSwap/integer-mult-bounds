class HarnessFailure(RuntimeError):
    pass


def need(condition, message):
    if condition is not True:
        raise HarnessFailure(message)


def identity(info):
    return (info.st_dev,info.st_ino,info.st_mode,info.st_uid,info.st_nlink,
            info.st_size,info.st_mtime_ns,info.st_ctime_ns)


def read_exact(path, size, sha):
    need(type(size) is int and 0 < size <= 8388608 and type(sha) is str and
         len(sha) == 64 and all(c in '0123456789abcdef' for c in sha), 'fixed read pin')
    need(type(path) is type(FUNDING_PATH) and path.is_absolute() and
         path.resolve(strict=True) == path, 'exact frozen source path')
    before = path.lstat()
    need(stat.S_ISREG(before.st_mode) and stat.S_IMODE(before.st_mode) == 0o444 and
         before.st_uid in (0,os.getuid()) and before.st_nlink == 1 and before.st_size == size,
         'frozen regular owner/link/size')
    fd,primary,secondary,raw = None,None,None,None
    try:
        fd = os.open(path,os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK | os.O_CLOEXEC)
        need(identity(os.fstat(fd)) == identity(before), 'opened fixed source identity')
        parts,count = [],0
        while True:
            chunk = os.read(fd,min(65536,size+1-count))
            if not chunk:
                break
            count += len(chunk)
            need(count <= size, 'fixed source read cap')
            parts.append(chunk)
        raw = b''.join(parts)
        need(count == size and identity(os.fstat(fd)) == identity(before) == identity(path.lstat()) and
             hashlib.sha256(raw).hexdigest() == sha, 'fixed source bytes and final identity')
    except BaseException as error:
        primary = error
    finally:
        if fd is not None:
            try:
                os.close(fd)  # Exactly one attempt; no retry/reconstruction.
            except BaseException as error:
                secondary = error
    if primary is not None:
        raise primary
    if secondary is not None:
        raise secondary
    return raw



def selected_import(name, globals=None, locals=None, fromlist=(), level=0):
    allowed = ('builtins','copy','hashlib','json','os','pathlib','stat','sys','types')
    need(type(name) is str and level == 0 and name in allowed, 'fixed stdlib source imports only')
    need(fromlist in (None,(),('Path',),('ModuleType','FunctionType')), 'fixed stdlib import forms')
    return _IMPORT(name, globals, locals, fromlist, level)


def load_pure_source(pin, name):
    need(type(pin) is dict and set(pin) == {'path','bytes','sha256','mode'} and pin['mode'] == 0o444,
         'fixed pure source pin')
    source = read_exact(Path(pin['path']), pin['bytes'], pin['sha256'])
    module = ModuleType(name)
    b = dict(vars(builtins))
    b['__import__'] = selected_import
    module.__dict__.update(__file__=pin['path'], __package__=None, __spec__=None, __builtins__=b)
    # Future separately admitted pure source loading only. This private module
    # is never passed to A4, OwnedIO, Context, factories or native lease APIs.
    exec(compile(source,pin['path'],'exec',dont_inherit=True,optimize=0),module.__dict__)
    return module


def read_json_pin(pin):
    need(type(pin) is dict and set(pin) == {'path','bytes','sha256','mode'} and pin['mode'] == 0o444,
         'fixed DATA pin')
    return json.loads(read_exact(Path(pin['path']),pin['bytes'],pin['sha256']).decode('ascii'))



def final_observations(rows, cases, hooks):
    # Inputs are the actual returned validation rows, exact executed inventory
    # and the SAME source-private control module's cumulative callback log.
    # No expected golden/exit/native scalar supplies these observations.
    need(type(rows) is list and type(cases) is list and 0 < len(cases) <= 256 and
         type(hooks) is list, 'exact actual final observation inputs')
    unexpected_callbacks = len(hooks)
    all_literal_expectations_checked = (len(rows) == len(cases) and
        all(type(row) is dict and type(case) is dict and
            type(row.get('id')) is str and type(case.get('id')) is str and
            row['id'] == case['id'] and row.get('literal_expectation_checked') is True
            for row, case in zip(rows, cases)))
    return {'unexpected_callbacks': unexpected_callbacks,
            'all_literal_expectations_checked': all_literal_expectations_checked}

              'runtime_or_profile_qualified':False,'science_or_Root_permission':False}
