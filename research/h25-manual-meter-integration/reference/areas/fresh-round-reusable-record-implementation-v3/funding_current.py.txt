class FundingError(ValueError):
    """Stop; preserve evidence. A missing number never becomes a zero receipt."""


def need(condition, message):
    if condition is not True:
        raise FundingError(message)


def plain(value, depth=0):
    need(depth <= 16, 'metadata depth cap')
    t = type(value)
    if t is dict:
        need(len(value) <= 2 * MAX_ENTRIES, 'object cap')
        for k, v in value.items():
            need(type(k) is str and len(k) <= 128, 'exact string key')
            plain(v, depth + 1)
    elif t is list:
        need(len(value) <= 2 * MAX_ENTRIES, 'array cap')
        for v in value:
            plain(v, depth + 1)
    elif t is str:
        need(len(value) <= META_CAP, 'bounded metadata wire/string cap')
    elif t is int:
        need(0 <= value <= MAX_VALUE, 'bounded exact integer data')
    else:
        need(value is None or t is bool, 'only exact JSON integer/bool/null data')


def exact(value, keys):
    need(type(value) is dict and set(value) == set(keys), 'exact object fields')


def uint(value):
    need(type(value) is int and 0 <= value <= MAX_VALUE, 'bounded exact integer, not bool/float')
    return value


def digest(value):
    need(type(value) is str and len(value) == 64 and all(c in '0123456789abcdef' for c in value), 'sha256')
    return value


def label(value):
    need(type(value) is str and 1 <= len(value) <= 64 and
         all(c in 'abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789-_' for c in value), 'simple label')
    return value


def pin(value):
    plain(value)
    exact(value, ('path', 'bytes', 'sha256', 'mode'))
    need(type(value['path']) is str and Path(value['path']).is_absolute(), 'absolute pin path')
    uint(value['bytes']); digest(value['sha256'])
    need(type(value['mode']) is int and value['mode'] in (0o444, 0o600), 'immutable control or private record mode')
    need(value['bytes'] <= META_CAP, 'metadata byte cap')
    return value


def pairs(items):
    result = {}
    for k, v in items:
        need(k not in result, 'duplicate JSON key')
        result[k] = v
    return result


def forbidden_number(unused):
    raise FundingError('JSON floating/nonfinite number forbidden')


def decode(raw):
    need(type(raw) is bytes and 0 < len(raw) <= META_CAP, 'bounded exact bytes')
    obj = json.loads(raw.decode('utf-8'), object_pairs_hook=pairs,
                     parse_float=forbidden_number, parse_constant=forbidden_number)
    plain(obj)
    return obj


def registry(raw):
    obj = decode(raw)
    exact(obj, ('schema', 'round_id', 'authorization_sha256', 'opening_pin',
                'predecessor_sha256', 'phase', 'phase_release_sha256', 'policies', 'events'))
    need(obj['schema'] == 'fresh-round-registry-v2' and obj['round_id'] == ROUND_ID and
         obj['authorization_sha256'] == AUTHORIZATION_SHA256, 'selected new round only')
    pin(obj['opening_pin']); need(obj['opening_pin']['mode'] == 0o444, 'immutable selected opening')
    if obj['predecessor_sha256'] is not None:
        digest(obj['predecessor_sha256'])
    need(obj['phase'] in ('INITIAL', 'FULL'), 'phase')
    need((obj['phase'] == 'INITIAL' and obj['phase_release_sha256'] is None) or
         (obj['phase'] == 'FULL' and type(obj['phase_release_sha256']) is str), 'explicit phase release')
    if obj['phase'] == 'FULL':
        digest(obj['phase_release_sha256'])
    need(type(obj['policies']) is list and type(obj['events']) is list, 'bounded arrays')
    need(len(obj['events']) <= 2 * MAX_ENTRIES, 'event cap')
    policies, starts, closes = {}, {}, {}
    for p in obj['policies']:
        pin(p)
        need(Path(p['path']).parent == CONTROL and p['mode'] == 0o444 and
             p['sha256'] not in policies, 'selected distinct policy pin')
        policies[p['sha256']] = p
    for event in obj['events']:
        exact(event, ('kind', 'label', 'pin'))
        name = label(event['label']); p = pin(event['pin'])
        need(p['mode'] == 0o600, 'private START/receipt')
        if event['kind'] == 'REGISTER':
            need(name not in starts and len(starts) < MAX_ENTRIES, 'one register, no reused/seed label')
            need(p['path'] == str(DOMAIN / name / 'START.json'), 'only new START')
            starts[name] = p

            need(name in starts and name not in closes, 'close once after register')
            need(p['path'] == str(DOMAIN / name / 'receipt.json'), 'only new receipt')
            closes[name] = p
        else:
            raise FundingError('unexpected event')
    return obj, policies, starts, closes

def directory_identity(io, path, owner_uid):
    need(path.is_absolute() and path.resolve(strict=True) == path, 'canonical new directory')
    st = path.lstat()
    need(stat.S_ISDIR(st.st_mode) and st.st_uid == owner_uid and
         stat.S_IMODE(st.st_mode) == 0o700, 'private current-owner directory')
    return (st.st_dev, st.st_ino, st.st_mode, st.st_nlink, st.st_uid,
            st.st_mtime_ns, st.st_ctime_ns)


def read_pinned(io, expected, owner_uid):
    pin(expected)
    row, raw = io.read(Path(expected['path']), META_CAP, mode=expected['mode'],
                       expected={'bytes': expected['bytes'], 'sha256': expected['sha256']}, materialize=True)
    need(type(raw) is bytes and row['uid'] == owner_uid and row['nlink'] == 1 and
         row['mode'] == expected['mode'] and row['bytes'] == expected['bytes'] and
         row['sha256'] == expected['sha256'] and len(raw) == expected['bytes'] and
         hashlib.sha256(raw).hexdigest() == expected['sha256'], 'actual owned stable file differs')
    return row, raw


def snapshot_round(io, registry_pin, owner_uid):
    """Future actual-IO operation; caller owns native lease, not any JSON flag.

    Failures propagate: do not retry, silently skip entries, or create receipts.
    io.names and io.read must be the reviewed actual OwnedIO methods; read must
    retain canonical/lstat/open/fstat/post-path and known-close guarantees.
    """
    # The caller authenticates exact type(io), original source hash and module
    # instance; this function does not turn an arbitrary reader into authority.
    # It audits NEW admission only. It grants no in-flight/pending exemption.
    uint(owner_uid); plain(registry_pin); pin(registry_pin)
    rp = Path(registry_pin['path'])
    middle = rp.name[len('REGISTRY-'):-len('.json')]
    need(rp.parent == CONTROL and rp.name.startswith('REGISTRY-') and rp.name.endswith('.json') and
         len(middle) == 6 and all(c in '0123456789' for c in middle) and
         registry_pin['mode'] == 0o444, 'explicit new hash-selected registry file')
    seen = []
    def take(p):
        row, raw = read_pinned(io, p, owner_uid)
        seen.append((p, row, raw))
        return raw
    raw_registry = take(registry_pin)
    reg, policy_pins, starts, closes = registry(raw_registry)
    opening_snapshot = read_opening(io, reg['opening_pin'], owner_uid)
    before = directory_identity(io, DOMAIN, owner_uid)
    names = io.names(DOMAIN, MAX_ENTRIES + 1)
    need(type(names) is list and all(type(n) is str for n in names) and
         len(names) == len(set(names)) and set(names) == set(starts), 'every new entry exactly registered')
    records, directories = {}, {}
    for name in sorted(starts):
        folder = DOMAIN / name
        identity = directory_identity(io, folder, owner_uid)
        members = io.names(folder, 3)
        expected_members = {'START.json', 'receipt.json'} if name in closes else {'START.json'}
        need(type(members) is list and len(members) == len(set(members)) and
             set(members) == expected_members, 'exact new entry completeness; no hidden/missing receipt')
        start = decode(take(starts[name]))
        receipt = decode(take(closes[name])) if name in closes else None
        directories[name] = (identity, expected_members)
        records[name] = {'start': start, 'receipt': receipt}
    policies = {k: decode(take(p)) for k, p in policy_pins.items()}
    for p, row, raw in seen:
        after_row, after_raw = read_pinned(io, p, owner_uid)
        need(after_raw == raw and after_row == row, 'full pre/post immutable snapshot')
    for name, (identity, members) in directories.items():
        need(directory_identity(io, DOMAIN / name, owner_uid) == identity and
             set(io.names(DOMAIN / name, 3)) == members, 'new entry directory/census changed')
    need(directory_identity(io, DOMAIN, owner_uid) == before and
         set(io.names(DOMAIN, MAX_ENTRIES + 1)) == set(names), 'new round census changed')
    opening_after = read_opening(io, reg['opening_pin'], owner_uid)
    need(opening_after == opening_snapshot, 'whole snapshot opening remains current and unchanged')
    validate_opening_partition(opening_snapshot, reg, records)
    return {'schema': 'fresh-round-owned-snapshot-v2', 'registry': reg,
            'registry_pin': registry_pin, 'records': records, 'policies': policies,
            'opening_snapshot': opening_snapshot}

    opening = validate_opening_partition(snapshot['opening_snapshot'], reg, snapshot['records'])
    body, admin, reserved, unknown, blockers, commands, open_tails = (

    for name in starts:

        cmd = digest(s['command_sha256']); digest(s['policy_sha256'])
        need(name not in opening['manual_entry_ids'] and cmd not in commands and
             s['policy_sha256'] in policies, 'no imported/formal duplicate/reroll or undeclared policy')
        commands.add(cmd); p = policies[s['policy_sha256']]
        need(p['command_sha256'] == cmd and 0 < uint(s['body_reservation_ns']) <= BODY_LIMIT_NS, 'per-command body reservation')
        fee = uint(p['administrative_debit_ns'])
        if r is None:
            need(name not in closes, 'CLOSE without receipt')
            reserved += s['body_reservation_ns'] + fee
            unknown.append(name); blockers.append(name + ':MISSING_NEW_RECEIPT')
            continue
        need(name in closes, 'unregistered new receipt')
        exact(r, ('schema', 'round_id', 'label', 'start_sha256', 'outcome', 'body_elapsed_ns',
                  'administrative_debit_ns', 'body_begin_ns', 'body_end_ns', 'body_fully_observed', 'cleanup_known', 'custody_complete',
                  'nested_entries_complete', 'uncertainty', 'peak_sampled_rss_bytes', 'tail_state',
                  'independent_outcome_review_sha256'))
        need(r['schema'] == 'fresh-round-receipt-v1' and r['round_id'] == ROUND_ID and
             r['label'] == name and r['start_sha256'] == starts[name]['sha256'] and
             r['outcome'] in ('PASS', 'STOP') and uint(r['administrative_debit_ns']) == fee, 'receipt identity/fee')
        for k in ('body_begin_ns', 'body_end_ns'):
            if r[k] is not None:
                uint(r[k])
        admin += fee  # Failure fees and measured failure bodies are retained in full.
        if r['body_elapsed_ns'] is None:
            unknown.append(name); blockers.append(name + ':UNPRICED_BODY')
            reserved += s['body_reservation_ns']
        else:
            elapsed = uint(r['body_elapsed_ns'])
            need(type(r['body_begin_ns']) is int and type(r['body_end_ns']) is int and
                 r['body_end_ns'] >= r['body_begin_ns'] and
                 r['body_end_ns'] - r['body_begin_ns'] == elapsed, 'exact own body endpoints/difference')

            if elapsed > BODY_LIMIT_NS or elapsed > s['body_reservation_ns']:

        for k in ('body_fully_observed', 'cleanup_known', 'custody_complete', 'nested_entries_complete'):
            need(type(r[k]) is bool, 'exact boolean qualification')
            if r[k] is not True:
                blockers.append(name + ':' + k.upper())
        need(type(r['uncertainty']) is list and all(type(x) is str for x in r['uncertainty']), 'uncertainty list')
        if r['uncertainty']:
            blockers.append(name + ':UNRESOLVED_UNCERTAINTY')
        if r['peak_sampled_rss_bytes'] is None:
            blockers.append(name + ':UNKNOWN_SAMPLED_RSS')
        elif uint(r['peak_sampled_rss_bytes']) > RSS_LIMIT:
            blockers.append(name + ':SAMPLED_RSS_EXCEEDED')
        need(r['tail_state'] in ('COMPLETE', 'ADMIN_ONLY_OPEN', 'UNKNOWN'), 'tail state')
        if r['tail_state'] == 'UNKNOWN':
            blockers.append(name + ':UNKNOWN_NONADMIN_TAIL')
        elif r['tail_state'] == 'ADMIN_ONLY_OPEN':
            open_tails.append(name)
        digest(r['independent_outcome_review_sha256'])
    label(request['label']); cmd = digest(request['command_sha256'])
    need(request['label'] not in starts and request['label'] not in opening['manual_entry_ids'] and cmd not in commands, 'one use, no automatic retry')
    need(request['policy_sha256'] in policies, 'request separately adopted policy absent')
    policy = policies[request['policy_sha256']]
    need(policy['command_sha256'] == cmd, 'request exact policy/command')
    wanted_body = uint(request['body_reservation_ns'])
    need(0 < wanted_body <= BODY_LIMIT_NS, 'requested body <=180s; actual fit remains separate')
    wanted = wanted_body + policy['administrative_debit_ns']
    charged = body + admin
    remaining = cap - charged - reserved  # No clipping, no unknown-body zero receipt.
    if remaining < wanted:
        blockers.append('INSUFFICIENT_POLICY_ALLOWANCE')
    return {'schema': 'fresh-round-funding-assessment-v1', 'round_id': ROUND_ID,
            'funding_data_eligible_only': not blockers, 'blockers': blockers,
            'cap_ns': cap, 'known_observed_body_ns': body, 'administrative_policy_ns': admin,

            'requested_reservation_ns': wanted, 'new_admin_only_open_tail_entries': open_tails,
            'opening_physical_tail_open': True, 'complete_physical_total_ns': None,
            'old_historical_total_ns': None, 'all_costs_settled': False,
            'native_ownership_or_cleanup_authenticated': False, 'execution_permission': False,
            'command_resource_profile_or_economic_fit_proved': False}

def metadata_decode(raw):
    need(type(raw) is bytes and 0 < len(raw) <= META_CAP, 'bounded exact metadata bytes')
    # Legacy diagnostic floats are retained as lexical strings, NEVER charges.
    value = json.loads(raw.decode('utf-8'), object_pairs_hook=pairs,
                       parse_float=lambda text: text, parse_constant=forbidden_number)
    metadata_plain(value)
    return value


def metadata_plain(value, depth=0):
    need(depth <= 16, 'metadata depth')
    t = type(value)
    if t is dict:
        need(len(value) <= 256, 'metadata object bound')
        for k, v in value.items():
            need(type(k) is str and len(k) <= 128, 'metadata key')
            metadata_plain(v, depth + 1)
    elif t is list:
        need(len(value) <= 256, 'metadata array bound')
        for v in value:
            metadata_plain(v, depth + 1)
    elif t is str:
        need(len(value) <= META_CAP, 'metadata string bound')
    elif t is int:
        need(-MAX_VALUE <= value <= MAX_VALUE, 'bounded reported metadata integer')
    else:
        need(value is None or t is bool, 'exact metadata builtin types')


def select(value, path):
    need(type(path) is list and 1 <= len(path) <= 8 and all(type(k) is str for k in path), 'literal field path')
    for key in path:
        need(type(value) is dict and key in value, 'original metadata field absent')
        value = value[key]
    return value


def seconds_ns(value):
    need(type(value) is str and value.count('.') <= 1, 'canonical decimal seconds string')
    parts = value.split('.')
    whole = parts[0]
    need(whole == '0' or (whole and whole[0] in '123456789' and all(c in '0123456789' for c in whole)), 'canonical whole seconds')
    fraction = parts[1] if len(parts) == 2 else ''
    need(len(fraction) <= 9 and all(c in '0123456789' for c in fraction), 'integer-ns decimal precision')
    return uint(int(whole) * NS + (int(fraction.ljust(9, '0')) if fraction else 0))

def review_pin_rows(review):
    """Closed native-tool family review inputs; still retained DATA, not authority."""
    values = review['inputs']
    need(type(values) is dict and 1 <= len(values) <= 256, 'typed closed review input pin map')
    result = []
    for key, value in values.items():
        need(type(key) is str, 'review pin key')
        exact(value, ('path', 'bytes', 'sha256', 'mode'))
        need(type(value['path']) is str and Path(value['path']).is_absolute(), 'review original absolute pin')
        uint(value['bytes']); digest(value['sha256']); uint(value['mode'])
        need(value['mode'] <= 0o777, 'review original mode metadata')
        result.append(value)
    return result

def _read_opening_f3(io, opening_pin, owner_uid):
    """Actual source-authenticated OwnedIO boundary, not arbitrary reader authority."""
    pin(opening_pin); need(opening_pin['mode'] == 0o444, 'frozen selected opening')
    row, raw = read_pinned(io, opening_pin, owner_uid)
    desc = decode(raw); folder = Path(opening_pin['path']).parent / 'corpus'
    need(folder.resolve(strict=True) == folder, 'canonical corpus directory')
    before = folder.lstat()
    need(stat.S_ISDIR(before.st_mode) and before.st_uid == owner_uid and stat.S_IMODE(before.st_mode) == 0o555, 'immutable current-owner corpus')
    names = io.names(folder, 8 * OPENING_LIMIT + 1)
    need(type(names) is list and all(type(n) is str for n in names) and
         len(names) == len(set(names)) and set(names) == set(desc['files']), 'exact complete corpus census')
    wires, observed = {}, []
    for name in sorted(names):
        item = desc['files'][name]
        need(type(name) is str and '/' not in name and '..' not in name, 'corpus basename')
        p = {'path': str(folder / name), 'bytes': item['bytes'], 'sha256': item['sha256'], 'mode': 0o444}
        actual, data = read_pinned(io, p, owner_uid); wires[name] = data.decode('utf-8'); observed.append((p, actual, data))
    snapshot = {'schema': 'fresh-current-opening-snapshot-v1', 'opening_pin': opening_pin,
                'descriptor_wire': raw.decode('utf-8'), 'corpus_wires': wires}
    opening_evidence(snapshot)
    for p, actual, data in observed:
        after_row, after = read_pinned(io, p, owner_uid)
        need(after_row == actual and after == data, 'immutable corpus pre/post custody')
    after_row, after = read_pinned(io, opening_pin, owner_uid)
    directory_after = folder.lstat()
    need(after_row == row and after == raw and
         (directory_after.st_dev, directory_after.st_ino, directory_after.st_mode, directory_after.st_uid,
          directory_after.st_mtime_ns, directory_after.st_ctime_ns) ==
         (before.st_dev, before.st_ino, before.st_mode, before.st_uid, before.st_mtime_ns, before.st_ctime_ns) and
         set(io.names(folder, 8 * OPENING_LIMIT + 1)) == set(names), 'opening/corpus current custody')
    return snapshot

def frame_need(condition, code):
    if condition is not True:
        raise FundingError('FRAME:' + code)


def financial_plain(value, counter=None, depth=0):
    if counter is None:
        counter = [0]
    counter[0] += 1
    frame_need(counter[0] <= FIN_MAX_NODES and depth <= 16, 'TYPE')
    t = type(value)
    if t is dict:
        keys = tuple(dict.__iter__(value))
        frame_need(len(keys) <= FIN_MAX_FILES and all(type(k) is str and len(k) <= 4096 for k in keys), 'TYPE')
        for k in keys:
            financial_plain(value[k], counter, depth + 1)
    elif t is list:
        frame_need(len(value) <= FIN_MAX_FILES, 'TYPE')
        for item in value:
            financial_plain(item, counter, depth + 1)
    elif t is str:
        frame_need(len(value) <= META_CAP, 'TYPE')
    elif t is int:
        frame_need(-MAX_VALUE <= value <= MAX_VALUE, 'TYPE')
    else:
        frame_need(value is None or t is bool, 'TYPE')


def fx(value, keys):
    frame_need(type(value) is dict, 'TYPE')
    actual=tuple(dict.__iter__(value))
    frame_need(all(type(k) is str for k in actual), 'TYPE')
    frame_need(set(actual) == set(keys), 'SCHEMA')


def fu(value):
    frame_need(type(value) is int and 0 <= value <= MAX_VALUE, 'TYPE')
    return value


def fsigned(value):
    frame_need(type(value) is int and -MAX_VALUE <= value <= MAX_VALUE, 'TYPE')
    return value


def fdigest(value):
    frame_need(type(value) is str and len(value) == 64 and all(c in '0123456789abcdef' for c in value), 'TYPE')

def flabel(value):

               all(c in 'abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789-_' for c in value), 'TYPE')
    return value

def _tool_request(obj, tool, command, session):
    financial_plain(obj)
    if tool == 'exec_command':
        allowed=('cmd','workdir','login','shell','tty','yield_time_ms','max_output_tokens','sandbox_permissions','justification','prefix_rule')
        frame_need(type(obj) is dict and set(obj) <= set(allowed) and 'cmd' in obj and 'workdir' in obj, 'SCHEMA')
        frame_need(type(obj['cmd']) is str and obj['cmd'] == command['exact_shell_command'] and obj['workdir'] == str(W), 'CORRELATION')
        for k in ('login','tty'):
            if k in obj: frame_need(type(obj[k]) is bool, 'TYPE')
        for k in ('yield_time_ms','max_output_tokens'):
            if k in obj: fu(obj[k])
        for k in ('shell','sandbox_permissions','justification'):
            if k in obj: frame_need(type(obj[k]) is str, 'TYPE')
        if 'prefix_rule' in obj: frame_need(type(obj['prefix_rule']) is list and len(obj['prefix_rule']) <= 16 and all(type(x) is str for x in obj['prefix_rule']), 'TYPE')
    else:
        frame_need(type(obj) is dict and set(obj) <= {'session_id','chars','yield_time_ms','max_output_tokens'} and 'session_id' in obj, 'SCHEMA')
        frame_need(type(obj['session_id']) is int and obj['session_id'] == session, 'CORRELATION')
        if 'chars' in obj: frame_need(type(obj['chars']) is str and len(obj['chars']) <= 4096, 'TYPE')
        for k in ('yield_time_ms','max_output_tokens'):
            if k in obj: fu(obj[k])


def _native_result(obj, session):
    # Unknown API shapes stay unresolved. Original raw Tool is never changed.
    if type(obj) is not dict or not {'output','wall_time_seconds'} <= set(obj) or not set(obj) <= {'output','wall_time_seconds','chunk_id','original_token_count','session_id','exit_code'}:
        return 'UNKNOWN', session, None
    if type(obj['output']) is not str: return 'UNKNOWN', session, None
    wait=obj['wall_time_seconds']
    if not ((type(wait) is int and wait >= 0) or (type(wait) is str and 0 < len(wait) <= 64 and wait.count('.') <= 1 and all(c in '0123456789.' for c in wait) and any(c in '0123456789' for c in wait))):
        return 'UNKNOWN', session, None
    if 'chunk_id' in obj and (type(obj['chunk_id']) is not str or not 1 <= len(obj['chunk_id']) <= 64): return 'UNKNOWN', session, None
    if 'original_token_count' in obj and (type(obj['original_token_count']) is not int or obj['original_token_count'] < 0): return 'UNKNOWN', session, None
    if obj.get('exit_code') is not None:
        if type(obj['exit_code']) is not int or not -MAX_VALUE <= obj['exit_code'] <= MAX_VALUE: return 'UNKNOWN', session, None
        if 'session_id' in obj:
            if type(obj['session_id']) is not int or obj['session_id'] < 0 or (session is not None and obj['session_id'] != session): return 'UNKNOWN', session, None
            session=obj['session_id']
        return 'TERMINAL', session, obj['exit_code']
    if type(obj.get('session_id')) is not int or obj['session_id'] < 0: return 'UNKNOWN', session, None
    if session is not None and obj['session_id'] != session: return 'UNKNOWN', session, None
    return 'RUNNING', obj['session_id'], None


def _wire_pin(path, wire, mode):
    raw=wire.encode('utf-8')
    return {'path':str(path),'bytes':len(raw),'sha256':hashlib.sha256(raw).hexdigest(),'mode':mode}

        frame_need(a['financial_adjudication_pin']==b['financial_adjudication_pin'] and a['native_chain_pin']==b['native_chain_pin'], 'APPEND')

    frame_need(a['phase'] != 'FORMAL_AWAIT_CLOSE' or b['phase'] in ('FORMAL_AWAIT_CLOSE','CLOSED_KNOWN','OPEN_UNCERTAIN'), 'APPEND')


def _financial_docs(snapshot):
    financial_plain(snapshot)
    fx(snapshot, ('schema','opening_pin','opening_wire','legacy_snapshot','wires','head_pin','head_wire','census'))
    frame_need(snapshot['schema'] == 'fresh-financial-snapshot-v1', 'SCHEMA')
    fp(snapshot['opening_pin']); fp(snapshot['head_pin'])
    frame_need(type(snapshot['wires']) is dict and len(snapshot['wires']) <= FIN_MAX_FILES and type(snapshot['census']) is dict, 'TYPE')
    total=0; docs={}; rawwires=snapshot['wires']; parsed_nodes=[0]
    for path,wire in rawwires.items():
        frame_need(type(path) is str and type(wire) is str, 'TYPE')
        raw=wire.encode('utf-8'); total+=len(raw)
        frame_need(len(raw) <= META_CAP and total <= FIN_MAX_BYTES, 'TYPE')
        value=financial_decode(raw) if raw else None
        financial_plain(value, parsed_nodes); docs[path]=value
    used=set()
    def take(p):
        fp(p, True); frame_need(p['bytes'] <= META_CAP, 'PIN'); path=p['path']; frame_need(path in rawwires, 'PIN')
        wire=rawwires[path]; frame_need(_wire_pin(path,wire,p['mode']) == p, 'PIN'); used.add(path)
        return docs[path]
    frame_need(snapshot['opening_wire'] == rawwires.get(snapshot['opening_pin']['path']) and
               snapshot['head_wire'] == rawwires.get(snapshot['head_pin']['path']), 'PIN')
    take.__dict__['wires']=rawwires
    take.__dict__['used']=used
    return docs,take,used

def opening_evidence(snapshot):
    # Exact builtin gate before selecting a parser; no caller callbacks.
    frame_need(type(snapshot) is dict, 'TYPE')
    keys=tuple(dict.__iter__(snapshot)); frame_need(all(type(k) is str for k in keys), 'TYPE')
    if type(snapshot.get('schema')) is str and snapshot['schema']=='fresh-financial-snapshot-v1':
        return financial_evidence(snapshot)
    if type(snapshot.get('schema')) is str and snapshot['schema']=='fresh-financial-history-view-v1':
        financial_plain(snapshot); fx(snapshot,('schema','opening_pin','complete_snapshot'))
        return _financial_history_evidence(snapshot['complete_snapshot'],snapshot['opening_pin'])
    return _opening_evidence_f3(snapshot)



def _financial_schema(value):
    # Reject non-string dictionary keys before lookup can hash/compare them.
    if type(value) is not dict:
        return None
    keys=tuple(dict.__iter__(value))
    frame_need(all(type(k) is str for k in keys), 'TYPE')
    schema=dict.get(value,'schema')
    return schema if type(schema) is str else None


def require_opening_append(previous,current):
    a=_financial_schema(previous) in ('fresh-financial-snapshot-v1','fresh-financial-history-view-v1')
    b=_financial_schema(current)=='fresh-financial-snapshot-v1'
    if not a and not b:
        return _require_opening_append_f3(previous,current)
    financial_plain(previous); financial_plain(current)
    if not a and b:
        frame_need(previous['opening_pin']==LEGACY_BASE_PIN and current['legacy_snapshot']==previous, 'APPEND')
        _opening_evidence_f3(previous); financial_evidence(current); return True
    frame_need(a and b, 'APPEND')
    if previous.get('schema')=='fresh-financial-history-view-v1':
        prior=_financial_history_evidence(current,previous['opening_pin'])
        frame_need(prior['descriptor']['legacy_base_pin']==LEGACY_BASE_PIN, 'APPEND')
        return True
    return require_financial_state_append(previous,current)


def validate_opening_partition(opening,reg,records):
    if _financial_schema(opening) not in ('fresh-financial-snapshot-v1','fresh-financial-history-view-v1'):
        return _validate_opening_partition_f3(opening,reg,records)
    e=opening_evidence(opening); financial_plain(reg); financial_plain(records)
    _,_,starts,closes=registry(canonical_financial(reg))
    manual_ids=set(e['manual_entry_ids']); manual_commands=set(e['manual_command_sha256s'])
    expected_formal={}
    for row in e['registry_rows']:
        name=row['entry_id']; frame_need(name not in expected_formal, 'CORRELATION')
        if row['formal_start_sha256'] is None:
            # Legitimate prelude is visible and pending, but not yet registered.
            frame_need(row['phase'] in ('RESERVED','STARTED') and row['meter_body'] is None and
                       row['adjudication'] is None and row['formal_receipt_sha256'] is None and
                       name not in starts and name not in records, 'CORRELATION')
        else:
            expected_formal[name]=row
    frame_need(set(starts)==set(records)==set(expected_formal), 'CORRELATION')
    complete=opening['complete_snapshot'] if opening['schema']=='fresh-financial-history-view-v1' else opening
    for name,bundle in records.items():
        fx(bundle,('start','receipt')); row=expected_formal[name]
        manifest=financial_decode(complete['wires'][row['manifest_pin']['path']].encode())
        original_start=financial_decode(complete['wires'][manifest['formal_start_pin']['path']].encode())
        frame_need(bundle['start']==original_start and name not in manual_ids and
                   bundle['start']['command_sha256'] not in manual_commands, 'CORRELATION')
        if row['phase']=='CLOSED_KNOWN':
            original_receipt=financial_decode(complete['wires'][manifest['formal_receipt_pin']['path']].encode())
            frame_need(bundle['receipt']==original_receipt, 'CORRELATION')
        else:
            frame_need(bundle['receipt'] is None, 'CORRELATION')
    for row in e['registry_rows']:
        name=row['entry_id']
        if row['phase']=='CLOSED_KNOWN':
            frame_need(name in starts and name in closes and name in records and records[name]['receipt'] is not None and
                       starts[name]['sha256']==row['formal_start_sha256'] and closes[name]['sha256']==row['formal_receipt_sha256'], 'CORRELATION')
            receipt=records[name]['receipt']; body=row['meter_body']['body']
            _formal_receipt(receipt,starts[name],name,row['body_ns'],body['begin_ns'],body['end_ns'])
        elif row['formal_start_sha256'] is not None:
            frame_need(name in starts and starts[name]['sha256']==row['formal_start_sha256'] and name not in closes, 'CORRELATION')
    return e


def canonical_financial(value):
    financial_plain(value)
    return (json.dumps(value,sort_keys=True,separators=(',',':'),allow_nan=False)+'\n').encode('utf-8')


def assess_round(snapshot,request):
    if type(snapshot) is dict:
        keys=tuple(dict.__iter__(snapshot)); frame_need(all(type(k) is str for k in keys), 'TYPE')
    selected_schema=_financial_schema(dict.get(snapshot,'opening_snapshot')) if type(snapshot) is dict else None
    if selected_schema=='fresh-financial-history-view-v1':
        financial_plain(snapshot); financial_plain(request)
        return {'schema':'fresh-round-funding-assessment-v2','funding_data_eligible_only':False,'blockers':['STALE_FINANCIAL_OPENING'],'remaining_policy_ns':None,'execution_permission':False,'all_costs_settled':False}
    if selected_schema!='fresh-financial-snapshot-v1':
        return _assess_round_f3(snapshot,request)
    financial_plain(snapshot); financial_plain(request)
    fx(snapshot,('schema','registry','registry_pin','records','policies','opening_snapshot'))
    fx(request,('schema','round_id','label','command_sha256','body_reservation_ns','policy_sha256','phase_release_sha256'))
    frame_need(snapshot['schema']=='fresh-round-owned-snapshot-v2' and request['schema']=='fresh-round-request-v1' and request['round_id']==ROUND_ID, 'SCHEMA')
    e=validate_opening_partition(snapshot['opening_snapshot'],snapshot['registry'],snapshot['records'])
    reg,policy_pins,starts,closes=registry(canonical_financial(snapshot['registry']))
    frame_need(request['phase_release_sha256']==reg['phase_release_sha256'] and snapshot['opening_snapshot']['opening_pin']==reg['opening_pin'] and
               set(snapshot['policies'])==set(policy_pins) and set(snapshot['records'])==set(starts), 'CORRELATION')
    cap=INITIAL_NS if reg['phase']=='INITIAL' else TOTAL_NS
    flabel(request['label']); fdigest(request['command_sha256']); fdigest(request['policy_sha256'])
    all_ids={v['entry_id'] for v in e['views']}; all_cmds={v['command_sha256'] for v in e['views']}
    frame_need(request['label'] not in all_ids and request['label'] not in starts and request['command_sha256'] not in all_cmds and
               all(request['command_sha256']!=x['start']['command_sha256'] for x in snapshot['records'].values()), 'CORRELATION')
    frame_need(request['policy_sha256'] in snapshot['policies'], 'CORRELATION')
    for key,p in snapshot['policies'].items():
        fx(p,('schema','round_id','command_sha256','administrative_debit_ns','scope','adoption_sha256','claims_physical_upper_bound'))
        frame_need(p['schema']=='fresh-round-admin-policy-v1' and p['round_id']==ROUND_ID and p['scope']=='STARTUP_AND_FINAL_METADATA_PUBLICATION_ONLY' and
                   p['claims_physical_upper_bound'] is False and fu(p['administrative_debit_ns'])==ADMIN_NS, 'CORRELATION')
        fdigest(p['command_sha256']); fdigest(p['adoption_sha256'])
    policy=snapshot['policies'][request['policy_sha256']]
    frame_need(policy['command_sha256']==request['command_sha256'] and 0<fu(request['body_reservation_ns'])<=BODY_LIMIT_NS, 'CORRELATION')
    blockers=list(e['blockers']); reserved=0; body=e['manual_body_ns']; admin=e['manual_admin_ns']
    formal_commands=set(e['manual_command_sha256s'])
    for name,bundle in snapshot['records'].items():
        fx(bundle,('start','receipt')); st=bundle['start']
        fx(st,('schema','round_id','label','command_sha256','body_reservation_ns','policy_sha256','pre_entry_provenance'))
        frame_need(st['schema']=='fresh-round-start-v1' and st['round_id']==ROUND_ID and st['label']==name, 'CORRELATION')
        fdigest(st['command_sha256']); fdigest(st['policy_sha256'])
        frame_need(st['command_sha256'] not in formal_commands and st['policy_sha256'] in snapshot['policies'], 'CORRELATION')
        formal_commands.add(st['command_sha256']); p=snapshot['policies'][st['policy_sha256']]
        frame_need(p['command_sha256']==st['command_sha256'] and 0<fu(st['body_reservation_ns'])<=BODY_LIMIT_NS, 'CORRELATION')
        validate_pre_entry_provenance(st['pre_entry_provenance'],name,st['command_sha256'],st['body_reservation_ns'])
        rc=bundle['receipt']
        if rc is None:
            frame_need(name not in closes, 'CORRELATION')
            blockers.append(name+':MISSING_NEW_RECEIPT'); reserved+=fu(st['body_reservation_ns'])+ADMIN_NS
        else:
            frame_need(name in closes, 'CORRELATION')
            _formal_receipt(rc,starts[name],name,fu(rc['body_elapsed_ns']),fu(rc['body_begin_ns']),fu(rc['body_end_ns']))
            body+=rc['body_elapsed_ns']; admin+=ADMIN_NS
            if rc['body_elapsed_ns']>st['body_reservation_ns'] or rc['body_elapsed_ns']>BODY_LIMIT_NS: blockers.append(name+':BODY180_EXCEEDED')
    requested=fu(request['body_reservation_ns'])+ADMIN_NS
    remaining=cap-body-admin-reserved
    if remaining<requested: blockers.append('INSUFFICIENT_POLICY_ALLOWANCE')
    if e['pending'] or e['unknown']: remaining=None
    return {'schema':'fresh-round-funding-assessment-v2','round_id':ROUND_ID,'funding_data_eligible_only':not blockers,
            'blockers':list(dict.fromkeys(blockers)),'cap_ns':cap,'known_observed_body_ns':body,'administrative_policy_ns':admin,
            'known_policy_charged_ns':body+admin,'outstanding_reservation_ns':reserved,
            'unpriced_new_entries':list(e['pending']),'remaining_policy_ns':remaining,'requested_reservation_ns':requested,
            'new_admin_only_open_tail_entries':[],'opening_physical_tail_open':True,'complete_physical_total_ns':None,
            'old_historical_total_ns':None,'all_costs_settled':False,'native_ownership_or_cleanup_authenticated':False,
            'execution_permission':False,'command_resource_profile_or_economic_fit_proved':False}


def _financial_directory(path,owner_uid):
    frame_need(path.is_absolute() and path.resolve(strict=True)==path, 'PIN')
    st=path.lstat()
    frame_need(stat.S_ISDIR(st.st_mode) and st.st_uid==owner_uid and stat.S_IMODE(st.st_mode) in (0o700,0o555), 'PIN')
    return (st.st_dev,st.st_ino,st.st_mode,st.st_uid,st.st_mtime_ns,st.st_ctime_ns)


def read_financial_opening(io,opening_pin,owner_uid):
    # Caller authenticates actual original OwnedIO and holds genuine EX. This
    # method does not turn a duck reader/module/hash into that authority.
    fp(opening_pin); frame_need(type(owner_uid) is int, 'TYPE')
    wires={}; observed={}; modes={}; census={}; dirs={}; total=[0]
    def grab(path, expected=None, mode=None):
        path=Path(path); frame_need(path.is_relative_to(R) and '..' not in path.parts, 'PIN')
        if str(path) in wires:
            if expected is not None: frame_need(_wire_pin(path,wires[str(path)],expected['mode'])==expected and modes[str(path)]==expected['mode'], 'PIN')
            return financial_decode(wires[str(path)].encode())
        row,raw=io.read(path,META_CAP,mode,expected,True)
        frame_need(row['uid']==owner_uid and row['mode'] in (0o400,0o444,0o600), 'PIN')
        total[0]+=len(raw); frame_need(len(wires)<FIN_MAX_FILES and total[0]<=FIN_MAX_BYTES, 'TYPE')
        wire=raw.decode('utf-8'); wires[str(path)]=wire; observed[str(path)]=(row,raw); modes[str(path)]=row['mode']
        if expected is not None: frame_need(_wire_pin(path,wire,expected['mode'])==expected and row['mode']==expected['mode'], 'PIN')
        return financial_decode(raw) if raw else None
    def directory(path):
        path=Path(path); before=_financial_directory(path,owner_uid); names=io.names(path,FIN_MAX_FILES+1)
        frame_need(type(names) is list and all(type(n) is str for n in names) and len(names)==len(set(names)), 'CENSUS')
        dirs[str(path)]=before; census[str(path)]=sorted(names); return sorted(names)
    rootnames=directory(FIN_ROOT); names=directory(FIN_ENTRIES)
    for n in rootnames:
        if n=='entries': continue
        frame_need('/' not in n and n.endswith('.json'), 'CENSUS'); grab(FIN_ROOT/n,mode=0o600 if n=='HEAD.json' or n.startswith('HEAD-') else 0o444)
    for name in names:
        flabel(name); members=directory(FIN_ENTRIES/name)
        for n in members:
            frame_need('/' not in n and n.endswith('.json'), 'CENSUS')
            grab(FIN_ENTRIES/name/n,mode=0o444 if n.startswith('ENTRY-') else None)
    requested_pin=opening_pin
    actual_head=financial_decode(wires[str(FIN_HEAD)].encode())
    fx(actual_head,('schema','generation','state_pin','opening_pin','previous_head_sha256'))
    opening_pin=actual_head['opening_pin']
    op=grab(opening_pin['path'],opening_pin)
    validate_original_record('OPENING',op)
    grab(op['frame_contract_pin']['path'],op['frame_contract_pin'])
    # Follow only source-fixed financial roles, never opaque payload/source files.
    changed=True
    while changed:
        changed=False
        for path,wire in list(wires.items()):
            obj=financial_decode(wire.encode()) if wire else None
            if type(obj) is not dict: continue
            targets=[]; schema=obj.get('schema')
            if schema==FRAME_RECORDS['ENTRY_MANIFEST']['schema']:
                targets=[obj[k] for k in ('command_pin','reservation_pin','pre_entry_pin','meter_start_pin','meter_body_pin','native_chain_pin','financial_adjudication_pin','formal_start_pin','formal_receipt_pin') if obj[k] is not None]
            elif schema==FRAME_RECORDS['COMMAND']['schema']:
                targets=[obj['permission_pin']]
                if obj['metadata_close_intent'] is not None: targets.append(obj['metadata_close_intent']['receipt_pin'])
            elif schema==FRAME_RECORDS['PRE_ENTRY']['schema']:
                targets=[obj['provider_request_pin']]
            elif schema==FRAME_RECORDS['STATE']['schema']:
                targets=[pp for e in obj['metadata_close_events'] for k in ('intent_pin','formal_head_pin','financial_state_pin','receipt_pin') for pp in (e[k],) if pp is not None]
            elif schema=='fresh-active-head-v1':
                targets=[obj['registry_pin']]
            elif schema==FRAME_RECORDS['NATIVE_CHAIN']['schema']:
                targets=[v for c in obj['calls'] for k in ('request_pin','original_tool_result_pin') for v in (c[k],) if v is not None and v['bytes']<=META_CAP]
            for pp in targets:
                fp(pp,True)
                if pp['path'] not in wires:
                    grab(pp['path'],pp); changed=True
                else:
                    frame_need(modes[pp['path']]==pp['mode'], 'PIN')
    legacy=_read_opening_f3(io,LEGACY_BASE_PIN,owner_uid)
    frame_need(str(FIN_HEAD) in wires, 'PIN')
    head_pin=_wire_pin(FIN_HEAD,wires[str(FIN_HEAD)],0o600)
    result={'schema':'fresh-financial-snapshot-v1','opening_pin':opening_pin,'opening_wire':wires[opening_pin['path']],
            'legacy_snapshot':legacy,'wires':wires,'head_pin':head_pin,'head_wire':wires[str(FIN_HEAD)],'census':census}
    financial_evidence(result)
    for path,(row,raw) in observed.items():
        after,again=io.read(Path(path),META_CAP,row['mode'],{'bytes':row['bytes'],'sha256':row['sha256']},True)
        frame_need(after==row and again==raw, 'PIN')
    for path,before in dirs.items():
        frame_need(_financial_directory(Path(path),owner_uid)==before and sorted(io.names(Path(path),FIN_MAX_FILES+1))==census[path], 'CENSUS')
    frame_need(_read_opening_f3(io,LEGACY_BASE_PIN,owner_uid)==legacy, 'PIN')
    if requested_pin != opening_pin:
        _financial_history_evidence(result,requested_pin)
        return {'schema':'fresh-financial-history-view-v1','opening_pin':requested_pin,'complete_snapshot':result}
    return result


def read_opening(io,opening_pin,owner_uid):
    # Added dispatch read is paid; original legacy reader body stays byte exact.
    row,raw=read_pinned(io,opening_pin,owner_uid)
    desc=decode(raw)
    if desc.get('schema')=='fresh-current-financial-opening-v1':
        return read_financial_opening(io,opening_pin,owner_uid)
    return _read_opening_f3(io,opening_pin,owner_uid)

               _wire_pin(requested_pin['path'],current_snapshot['wires'][requested_pin['path']],requested_pin['mode'])==requested_pin, 'PIN')
    descriptor=validate_original_record('OPENING',financial_decode(current_snapshot['wires'][requested_pin['path']].encode()))
    docs,take,used=_financial_docs(current_snapshot); state=take(descriptor['suffix_state_pin'])
    legacy=_opening_evidence_f3(current_snapshot['legacy_snapshot']); entries={}
    for mp in state['ordered_entries']:
        manifest=take(mp); view=_entry_financial_view(mp,manifest,take,legacy); entries[view['entry_id']]=view
    manual=[v for v in entries.values() if v['phase']=='CLOSED_KNOWN' and v['settlement_route']=='MANUAL_ONLY']
    result=dict(current)
    result.update({'descriptor':descriptor,'state_sha256':descriptor['suffix_state_pin']['sha256'],'state_pin':descriptor['suffix_state_pin'],
                   'opening_pin':requested_pin,'views':legacy['views']+[{k:x for k,x in v.items() if k!='_unknown'} for v in entries.values()],
                   'manual_body_ns':legacy['manual_body_ns']+sum(v['body_ns'] for v in manual),
                   'manual_admin_ns':legacy['manual_admin_ns']+len(manual)*ADMIN_NS,
                   'manual_entry_ids':legacy['manual_entry_ids']+[v['entry_id'] for v in manual],
                   'manual_command_sha256s':legacy['manual_command_sha256s']+[v['command_sha256'] for v in manual],
                   'registry_rows':legacy['registry_rows']+[{k:x for k,x in v.items() if k!='_unknown'} for v in entries.values() if v['settlement_route']=='FORMAL_OUTER'],
                   'entries':{n:{k:x for k,x in v.items() if k!='_unknown'} for n,v in entries.items()},
                   'blockers':list(current['blockers'])+['STALE_FINANCIAL_OPENING']})
    return result


def require_append_only(previous_raw,current_raw,opening_transition=None):
    if type(opening_transition) is dict:
        keys=tuple(dict.__iter__(opening_transition)); frame_need(all(type(k) is str for k in keys), 'TYPE')
    selected_schema=_financial_schema(dict.get(opening_transition,'current')) if type(opening_transition) is dict else None
    if selected_schema!='fresh-financial-snapshot-v1':
        return _require_append_only_f3(previous_raw,current_raw,opening_transition)
    financial_plain(opening_transition); fx(opening_transition,('previous','current','records'))
    old,_,starts,closes=registry(previous_raw); new,_,_,_=registry(current_raw)
    frame_need(new['predecessor_sha256']==hashlib.sha256(previous_raw).hexdigest() and old['events']==new['events'] and
               old['policies']==new['policies'] and old['phase']==new['phase'] and old['phase_release_sha256']==new['phase_release_sha256'], 'APPEND')
    frame_need(set(starts)==set(closes)==set(opening_transition['records']) and
               opening_transition['previous']['opening_pin']==old['opening_pin'] and opening_transition['current']['opening_pin']==new['opening_pin'], 'APPEND')
    require_opening_append(opening_transition['previous'],opening_transition['current'])
    validate_opening_partition(opening_transition['current'],old,opening_transition['records'])
    return True


def verify_financial_candidate(io,candidate,owner_uid,expected_head_pin=None):
    """Verify all existing originals before actual owner-only publication.

    Candidate control images are proposed DATA, not current head observations.
    Current entry/source custody and actual native EX must be supplied by A4.
    Only the fixed next generation control files may be absent physically.
    """
    e=financial_evidence(candidate); financial_plain(expected_head_pin)
    head=financial_decode(candidate['head_wire'].encode()); g=head['generation']
    staged={str(FIN_HEAD),str(FIN_ROOT/('HEAD-%06d.json'%g)),str(FIN_ROOT/('STATE-%06d.json'%g)),str(FIN_ROOT/('OPENING-%06d.json'%g))}
    if expected_head_pin is None:
        frame_need(g==0 and not FIN_HEAD.exists() and not FIN_HEAD.is_symlink(), 'APPEND')
    else:
        fp(expected_head_pin); row,raw=read_pinned(io,expected_head_pin,owner_uid)
        before=financial_decode(raw)
        frame_need(expected_head_pin['path']==str(FIN_HEAD) and g==before['generation']+1 and head['previous_head_sha256']==expected_head_pin['sha256'], 'APPEND')
        prior=_financial_history_evidence(candidate,before['opening_pin'])
        frame_need(candidate['wires'].get(str(FIN_ROOT/('HEAD-%06d.json'%before['generation'])))==raw.decode('utf-8'), 'APPEND')
    observed=[]; mode_index={}
    def collect(obj):
        if type(obj) is dict:
            if set(obj)=={'path','bytes','sha256','mode'}:
                mode_index.setdefault(obj['path'],[]).append(obj['mode'])
            for value in obj.values(): collect(value)
        elif type(obj) is list:
            for value in obj: collect(value)
    for text in candidate['wires'].values():
        if text: collect(financial_decode(text.encode()))
    for path,wire in candidate['wires'].items():
        if path in staged: continue
        mode=0o600 if not wire or (Path(path).parent==FIN_ROOT and Path(path).name.startswith('HEAD-')) else 0o444
        wanted_modes=mode_index.get(path,[])
        if wanted_modes:
            frame_need(all(x==wanted_modes[0] for x in wanted_modes), 'PIN'); mode=wanted_modes[0]
        p=_wire_pin(path,wire,mode); row,raw=_read_frame_pin(io,p,owner_uid)
        frame_need(raw.decode('utf-8')==wire, 'PIN'); observed.append((p,row,raw))
    for directory,names in candidate['census'].items():
        path=Path(directory)
        if path.exists(): _financial_directory(path,owner_uid)
        actual=sorted(io.names(path,FIN_MAX_FILES+1)) if path.exists() else []
        expected=sorted(n for n in names if (str(path/n) not in staged or (path==FIN_ROOT and n=='HEAD.json' and expected_head_pin is not None)) and not (path==FIN_ROOT and n=='entries' and not FIN_ENTRIES.exists()))
        frame_need(actual==expected, 'CENSUS')
    frame_need(_read_opening_f3(io,LEGACY_BASE_PIN,owner_uid)==candidate['legacy_snapshot'], 'PIN')
    for p,row,raw in observed:
        again,bytes_again=_read_frame_pin(io,p,owner_uid); frame_need(again==row and bytes_again==raw,'PIN')
    return e

def _read_frame_pin(io,p,owner_uid):
    fp(p,True); frame_need(p['bytes']<=META_CAP, 'PIN')
    row,raw=io.read(Path(p['path']),META_CAP,p['mode'],{'bytes':p['bytes'],'sha256':p['sha256']},True)
    frame_need(type(raw) is bytes and row['uid']==owner_uid and row['nlink']==1 and row['mode']==p['mode'] and
               len(raw)==p['bytes'] and hashlib.sha256(raw).hexdigest()==p['sha256'], 'PIN')
    return row,raw


def financial_commit_candidate(snapshot,own_id,target_id,receipt_pin,formal_head_pin,intent_pin):
    """Pure staged DATA only. Actual A4 owns all validation/publication authority."""
    flabel(own_id); flabel(target_id)  # Exact builtin gates before hashing/equality/path use.
    e=financial_evidence(snapshot); frame_need(own_id in e['entries'] and target_id in e['entries'] and own_id!=target_id,'CORRELATION')
    own,target=e['entries'][own_id],e['entries'][target_id]
    frame_need(own['command']['operation_class']=='METADATA_CLOSER' and own['command']['metadata_close_intent']['target_id']==target_id and
               target['phase']=='FORMAL_AWAIT_CLOSE' and target['known_charge_ns'] is not None,'CORRELATION')
    for pp in (receipt_pin,formal_head_pin,intent_pin): fp(pp)
    result=json.loads(json.dumps(snapshot))
    financial_plain(result)
    wires=result['wires']; old_head=financial_decode(result['head_wire'].encode()); g=old_head['generation']+1
    old_manifest=financial_decode(wires[target['manifest_pin']['path']].encode())

    manifest=dict(old_manifest,phase='CLOSED_KNOWN',formal_receipt_pin=receipt_pin)

    wires[str(mp_path)]=mp_wire; result['census'][str(mp_path.parent)].append(mp_path.name); result['census'][str(mp_path.parent)].sort()
    state=financial_decode(wires[e['state_pin']['path']].encode())
    state['generation']=g; state['previous_state_sha256']=e['state_pin']['sha256']
    state['ordered_entries']=[mp if pp==target['manifest_pin'] else pp for pp in state['ordered_entries']]
    state['unsettled_entries']=[n for n in state['unsettled_entries'] if n!=target_id]
    state['metadata_close_events'].append({'closer_id':own_id,'target_id':target_id,'phase':'FINANCIAL_COMMITTED','intent_pin':intent_pin,
        'formal_head_pin':formal_head_pin,'financial_state_pin':e['state_pin'],'receipt_pin':receipt_pin})
    spath=FIN_ROOT/('STATE-%06d.json'%g); swire=canonical_financial(state).decode(); sp=_wire_pin(spath,swire,0o444); wires[str(spath)]=swire
    opening=dict(e['descriptor'],suffix_state_pin=sp,previous_opening_sha256=e['opening_pin']['sha256'])
