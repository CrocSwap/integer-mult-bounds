DIR = os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW | os.O_CLOEXEC

def need(ok, message):
    if ok is not True:
        raise ValueError(message)


def identity(s):
    return (s.st_dev, s.st_ino, s.st_mode, s.st_nlink, s.st_size, s.st_mtime_ns, s.st_ctime_ns)


def absolute(path):
    need(type(path) is str and (path == W or path.startswith(W + '/')) and os.path.normpath(path) == path
         and '//' not in path and all(32 <= ord(c) < 127 for c in path), 'fixed ordinary absolute path')
    return path


def relative(path):
    need(type(path) is str and 0 < len(path) <= 8192, 'relative path')
    parts = path.split('/')
    need(all(p not in ('', '.', '..') and len(p) <= 255 for p in parts), 'relative components')
    return parts


def pin_tuple(pin):
    need(type(pin) is dict and {'path', 'bytes', 'mode', 'sha256'} <= set(pin), 'ordinary source pin fields')
    absolute(pin['path'])
    need(type(pin['bytes']) is int and 0 <= pin['bytes'] <= FILE_CAP
         and type(pin['mode']) is int and 0 <= pin['mode'] <= 0o777, 'ordinary source pin bounds')
    need(type(pin['sha256']) is str and len(pin['sha256']) == 64
         and all(c in '0123456789abcdef' for c in pin['sha256']), 'ordinary source pin SHA256')
    return tuple(pin[k] for k in ('path', 'bytes', 'mode', 'sha256'))


class State:
    def __init__(self):
        self.primary = None
        self.secondaries = []
        self.overflow = False
        self.known_fds = 0
        self.peak_fds = 0
        self.uncertain = False
    def error(self, error, stage):
        if self.primary is None:
            self.primary = error
        elif error is not self.primary:
            if len(self.secondaries) < 8192:
                self.secondaries.append((stage, error))
            else:
                self.overflow = True


class Owner:
    """Known descriptors retired before close; first/secondary errors kept raw."""
    def __init__(self, state, label):
        self.state, self.label = state, label
        self.slots, self.chain = [], []
        self.pending = False
    def __enter__(self):
        return self
    def open(self, name, flags, mode=0, parent=None):
        need(self.state.known_fds < 192, 'reader known descriptor bound')
        slot = {'fd': None, 'close_attempted': False}
        self.slots.append(slot)
        self.pending = True
        fd = os.open(name, flags, mode, dir_fd=parent)
        slot['fd'] = fd
        self.pending = False
        self.state.known_fds += 1
        self.state.peak_fds = max(self.state.peak_fds, self.state.known_fds)
        return fd
    def directory(self, path):
        absolute(path)
        fd = self.open('/', DIR)
        for name in path[1:].split('/'):
            parent = fd
            fd = self.open(name, DIR, parent=parent)
            s = os.fstat(fd)
            need(stat.S_ISDIR(s.st_mode), 'ordinary ancestor directory')
            self.chain.append((parent, name, fd, (s.st_dev, s.st_ino)))
        return fd
    def stable(self):
        for parent, name, fd, key in self.chain:
            held, named = os.fstat(fd), os.stat(name, dir_fd=parent, follow_symlinks=False)
            need(stat.S_ISDIR(named.st_mode) and (held.st_dev, held.st_ino) == key
                 and (named.st_dev, named.st_ino) == key, 'reader ancestor replaced')
    def retire(self, fd):
        slot = next(s for s in self.slots if s['fd'] == fd)
        slot['fd'] = None
        slot['close_attempted'] = True
        self.state.known_fds -= 1
        try:
            os.close(fd)
        except BaseException as error:
            self.state.error(error, self.label + ':close')
            raise
    def __exit__(self, kind, error, traceback):
        first = error
        if error is not None:
            self.state.error(error, self.label)
        if self.pending:
            self.state.uncertain = True
        for slot in reversed(self.slots):
            fd = slot['fd']
            slot['fd'] = None
            if fd is None:
                continue
            slot['close_attempted'] = True
            self.state.known_fds -= 1
            try:
                os.close(fd)
            except BaseException as close_error:
                self.state.error(close_error, self.label + ':close')
                if first is None:
                    first = close_error
        if error is None and first is not None:
            raise first
        return False


def json_data(raw):
    def unique(pairs):
        result = {}
        for key, value in pairs:
            need(key not in result, 'duplicate JSON field')
            result[key] = value
        return result
    return json.loads(raw, object_pairs_hook=unique,
        parse_constant=lambda value: (_ for _ in ()).throw(ValueError('nonfinite JSON')))


def read_fd(fd, parent, name, pin, retain=False):
    first = os.fstat(fd)
    need(stat.S_ISREG(first.st_mode) and first.st_nlink == 1, 'ordinary single-link file')
    need(0 <= first.st_size <= FILE_CAP and stat.S_IMODE(first.st_mode) == pin['mode'], 'file size/mode')
    if 'bytes' in pin:
        need(first.st_size == pin['bytes'], 'exact file bytes')
    need(not retain or first.st_size <= JSON_CAP, 'retained JSON cap')
    before = identity(first)
    need(identity(os.stat(name, dir_fd=parent, follow_symlinks=False)) == before, 'initial named identity')
    digest, count, chunks = hashlib.sha256(), 0, []
    while count < first.st_size:
        block = os.read(fd, min(1024**2, first.st_size - count))
        need(type(block) is bytes and 0 < len(block) <= first.st_size - count, 'read progress')
        count += len(block)
        digest.update(block)
        if retain:
            chunks.append(block)
    need(os.read(fd, 1) == b'', 'file grew during read')
    need(identity(os.fstat(fd)) == before
         and identity(os.stat(name, dir_fd=parent, follow_symlinks=False)) == before, 'stable full named/FD identity')
    need(digest.hexdigest() == pin['sha256'], 'exact file SHA256')
    return before, b''.join(chunks) if retain else None


def source(state, pin, retain=False):
    path = absolute(pin['path'])
    with Owner(state, 'source:' + path) as owner:
        parent = owner.directory(os.path.dirname(path))
        fd = owner.open(os.path.basename(path), READ, parent=parent)
        result = read_fd(fd, parent, os.path.basename(path), pin, retain)
        owner.stable()
    return {'path': path, 'identity': result[0], 'sha256': pin['sha256']}, result[1]

    dirs = {''}

    need(not (set(files) & dirs) and len(dirs) <= 128 and len(files) <= 8192, 'reader expected tree bounds')
    return files, dirs


def scan_names(state, fd):
    names, scan, primary = [], os.scandir(fd), None
    try:
        for entry in scan:
            names.append(entry.name)
            need(len(names) <= 8192, 'directory entry bound')
    except BaseException as error:
        primary = error
        state.error(error, 'directory_scan')
    try:
        scan.close()
    except BaseException as error:
        if primary is None:
            primary = error
        state.error(error, 'directory_scan_close')
    if primary is not None:
        raise primary
    return names


def inspect_role(state, role, files, dirs, inodes, previous=None):
    root = absolute(role['runtime_stage_root'])
    need(root == STAGES + role['runtime_stage_name'], 'original exact stage root')
    observed, directory_ids = {}, {}
    with Owner(state, 'payload:' + role['role']) as owner:
        root_fd = owner.directory(root)
        held = {'': root_fd}
        for path in sorted(dirs, key=lambda p: (p.count('/'), p)):
            if path:
                parent, _, name = path.rpartition('/')
                held[path] = owner.open(name, DIR, parent=held[parent])
            fd = held[path]
            info = os.fstat(fd)
            need(stat.S_ISDIR(info.st_mode) and stat.S_IMODE(info.st_mode) == 0o555, 'exact directory mode')
            directory_ids[path] = identity(info)
            if previous is not None:
                need(directory_ids[path] == tuple(previous['directories'][path]), 'directory changed after full read')
            expected_names = {p[len(path) + 1:] if path else p for p in set(files) | dirs
                              if p and p.rpartition('/')[0] == path}
            need(set(scan_names(state, fd)) == expected_names, 'exact directory entries')
        for path, pin in sorted(files.items()):
            parent, _, name = path.rpartition('/')
            fd = owner.open(name, READ, parent=held[parent])
            if previous is None:
                info, _ = read_fd(fd, held[parent], name, pin)
                key = (info[0], info[1])
                need(key not in inodes, 'distinct inode across every installed occurrence')
                inodes.add(key)
            else:
                info = identity(os.fstat(fd))
                need(stat.S_ISREG(info[2]) and info[3] == 1
                     and info == tuple(previous['files'][path]['identity'])
                     and identity(os.stat(name, dir_fd=held[parent], follow_symlinks=False)) == info,
                     'global final named/FD identity unchanged')
            observed[path] = {'bytes': pin['bytes'], 'sha256': pin['sha256'], 'identity': info}
            owner.retire(fd)
        for path, fd in held.items():
            need(identity(os.fstat(fd)) == directory_ids[path], 'directory changed during role read')
            if path:
                parent, _, name = path.rpartition('/')
                need(identity(os.stat(name, dir_fd=held[parent], follow_symlinks=False)) == directory_ids[path],
                     'named directory identity changed')
        owner.stable()
    return {'role': role['role'], 'root': root, 'directories': directory_ids, 'files': observed,
            'file_count': len(files), 'logical_bytes': sum(pin['bytes'] for pin in files.values())}


def save(state, output, report):
    raw = (json.dumps(report, sort_keys=True, indent=2, allow_nan=False) + '\n').encode()
    need(len(raw) < EVIDENCE_CAP, 'READBACK evidence cap')
    with Owner(state, 'evidence') as owner:
        parent = owner.directory(os.path.dirname(output))
        name = os.path.basename(output)
        os.mkdir(name, 0o700, dir_fd=parent)
        directory = owner.open(name, DIR, parent=parent)
        fd = owner.open('READBACK.json', os.O_RDWR | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW | os.O_CLOEXEC,
                        0o600, directory)
        count, calls = 0, 0
        while count < len(raw):
            calls += 1
            need(calls <= 8192, 'evidence write call bound')
            n = os.write(fd, raw[count:])
            need(type(n) is int and 0 < n <= len(raw) - count, 'evidence write progress')
            count += n
        os.fchmod(fd, 0o444)
        os.fsync(fd)
        os.lseek(fd, 0, os.SEEK_SET)
        read_fd(fd, directory, 'READBACK.json', {'bytes': len(raw), 'mode': 0o444,
                'sha256': hashlib.sha256(raw).hexdigest()})
        os.fchmod(directory, 0o555)
        os.fsync(directory)
        os.fsync(parent)
        owner.stable()
    return {'path': output + '/READBACK.json', 'bytes': len(raw), 'sha256': hashlib.sha256(raw).hexdigest()}


def error_data(error):
    return None if error is None else {'type': type(error).__name__, 'message': str(error)[:4096]}
