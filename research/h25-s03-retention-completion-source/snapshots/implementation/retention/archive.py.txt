class Rejected(ValueError):
    pass

def need(value, reason):
    if not value:
        raise Rejected(reason)

def canonical(value):
    return (json.dumps(value, sort_keys=True, ensure_ascii=True, separators=(',', ':')) + '\n').encode('ascii')

def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def is_hash(value):
    return type(value) is str and len(value) == 64 and (set(value) <= HEX)

def exact(obj, fields, name):
    need(type(obj) is dict and set(obj) == set(fields), name + ' fields')

def number(value, maximum, name):
    need(type(value) is int and 0 <= value <= maximum, name + ' bounds/type')

def caps(overrides=None):
    out = dict(LIMITS)
    if overrides:
        need(type(overrides) is dict and set(overrides) <= set(out), 'unknown limits')
        out.update(overrides)
    need(all((type(v) is int and v > 0 for v in out.values())), 'positive integer limits')
    need(out['chunk_bytes'] <= 1048576, 'stream buffer cap')
    return out

def normalized(path):
    value = os.fspath(path)
    need(type(value) is str and value.startswith('/') and (os.path.normpath(value) == value) and ('\x00' not in value), 'normalized absolute path required')
    return value

def relative(value, limits):
    need(type(value) is str and value and (not value.startswith('/')) and ('\\' not in value) and ('\x00' not in value) and (':' not in value), 'unsafe relative path')
    parts = value.split('/')
    need(all((part not in ('', '.', '..') for part in parts)), 'noncanonical path')
    need(len(parts) <= limits['max_depth'] and len(value.encode('utf-8', 'strict')) <= limits['max_path_bytes'], 'path bounds')
    return parts

def signature(info):
    return (info.st_dev, info.st_ino, info.st_mode, info.st_size, info.st_uid, info.st_gid, info.st_mtime_ns, info.st_ctime_ns, info.st_nlink)

def origin(info):
    return dict(device=info.st_dev, inode=info.st_ino, uid=info.st_uid, gid=info.st_gid, mtime_ns=info.st_mtime_ns, ctime_ns=info.st_ctime_ns, nlink=info.st_nlink)

@native_io.owning_leaf
def disjoint(*paths):

    def ancestry(path):
        _owned_fault_5 = native_io._ERRORS.RawSlot()
        _owned_fault_5.kind = 'LEXICAL-OWNER:ancestry'
        descriptor = native_io.open_fd('/', os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
        info = native_io.native_effect('fstat', descriptor)
        chain = {(info.st_dev, info.st_ino)}
        endpoint = (info.st_dev, info.st_ino)
        try:
            for part in path.split('/')[1:]:
                if not part:
                    continue
                try:
                    child = native_io.open_optional_fd(part, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=descriptor)
                except FileNotFoundError:
                    return (chain, None)
                native_io.close_fd(descriptor)
                descriptor = child
                info = native_io.native_effect('fstat', descriptor)
                endpoint = (info.st_dev, info.st_ino)
                chain.add(endpoint)
            return (chain, endpoint)
        except BaseException as _owned_error_5:
            _owned_fault_5.actual = _owned_error_5
            _owned_fault_5.traceback = BaseException.__traceback__.__get__(_owned_error_5, BaseException)
            native_io._commit(native_io.process_context(), _owned_fault_5)
            raise
        finally:
            native_io.close_fd(descriptor)
    identities = [ancestry(path) for path in paths]
    for i, first in enumerate(paths):
        for j, second in enumerate(paths[i + 1:], i + 1):
            need(os.path.commonpath((first, second)) not in (first, second), 'overlapping source/package/output/attempt paths')
            first_chain, first_end = identities[i]
            second_chain, second_end = identities[j]
            need(not (first_end in second_chain or second_end in first_chain), 'filesystem alias overlaps source/package/output/attempt')

def parse(raw):

    def pairs(items):
        result = {}
        for key, value in items:
            need(key not in result, 'duplicate JSON key')
            result[key] = value
        return result
    obj = json.loads(raw, object_pairs_hook=pairs, parse_constant=lambda value: (_ for _ in ()).throw(Rejected('nonfinite JSON')))
    need(canonical(obj) == raw, 'noncanonical JSON bytes')
    return obj

@native_io.owning_leaf
def read_small(root, name, limit, counters=None):
    with opened(root, name) as fd:
        info = native_io.native_effect('fstat', fd)
        need(stat.S_ISREG(info.st_mode) and info.st_size <= limit, 'metadata size/type cap')
        raw = bytearray()
        while len(raw) <= limit:
            block = native_io.read_fd(fd, min(65536, limit + 1 - len(raw)))
            if not block:
                break
            raw.extend(block)
            if counters is not None:
                counters['metadata_read_bytes'] += len(block)
        need(len(raw) == info.st_size and len(raw) <= limit, 'metadata length drift/cap')
        need(signature(info) == signature(native_io.native_effect('fstat', fd)), 'metadata changed during read')
        return bytes(raw)

@native_io.owning_leaf
def stream(root, name, limits, counters, category, expected=None, target=None):
    with opened(root, name) as fd:
        info = native_io.native_effect('fstat', fd)
        need(stat.S_ISREG(info.st_mode) and info.st_size <= limits['max_file_bytes'], 'file type/size')
        need(not native_io.native_effect('listxattr', fd), 'unsupported xattrs/ACLs')
        h, size = (hashlib.sha256(), 0)
        while True:
            block = native_io.read_fd(fd, limits['chunk_bytes'])
            if not block:
                break
            size += len(block)
            counters[category] += len(block)
            need(size <= info.st_size, 'file grew during read')
            h.update(block)
            if target is not None:
                view = memoryview(block)
                while view:
                    count = native_io.write_fd(target, view)
                    need(count > 0, 'zero-length stream write')
                    counters['payload_write_bytes'] += count
                    view = view[count:]
        need(size == info.st_size and signature(info) == signature(native_io.native_effect('fstat', fd)), 'file changed during stream')
        result = dict(size=size, sha256=h.hexdigest())
        if expected:
            need(result == {k: expected[k] for k in ('size', 'sha256')}, 'payload hash/length mismatch')
        return result

@native_io.owning_leaf
def scan(root, limits, single_links=False):
    """Metadata-only bounded tree census, independent of the oracle code."""
    _owned_fault_11 = native_io._ERRORS.RawSlot()
    _owned_fault_11.kind = 'LEXICAL-OWNER:scan'
    found, identities = ({}, set())
    fd = dirfd(root)
    try:
        root_info = native_io.native_effect('fstat', fd)
        need(not native_io.native_effect('listxattr', fd), 'unsupported root xattrs/ACLs')
        need(stat.S_IMODE(root_info.st_mode) <= 511, 'unsupported root permission bits')
        need(stat.S_IMODE(root_info.st_mode) & 320 == 320, 'unsupported root owner access mode')

        def descend(parent, prefix, depth):
            _owned_fault_10 = native_io._ERRORS.RawSlot()
            _owned_fault_10.kind = 'LEXICAL-OWNER:descend'
            need(depth <= limits['max_depth'], 'tree depth cap')
            before = native_io.native_effect('fstat', parent)
            names = []
            with native_io.scandir_fd(parent) as items:
                for item in items:
                    names.append(item.name)
                    need(len(names) <= limits['max_entries'], 'directory entry cap')
            for name in sorted(names):
                path = name if not prefix else prefix + '/' + name
                relative(path, limits)
                info = native_io.native_effect('stat', name, dir_fd=parent, follow_symlinks=False)
                need(stat.S_ISDIR(info.st_mode) or stat.S_ISREG(info.st_mode), 'symlink/special file')
                mode = stat.S_IMODE(info.st_mode)
                need(mode <= 511, 'unsupported special mode')
                owner_access = 320 if stat.S_ISDIR(info.st_mode) else 256
                need(mode & owner_access == owner_access, 'unsupported owner access mode')
                flags = os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK
                if stat.S_ISDIR(info.st_mode):
                    flags |= os.O_DIRECTORY
                child = native_io.open_fd(name, flags, dir_fd=parent)
                try:
                    need(signature(info) == signature(native_io.native_effect('fstat', child)), 'entry changed during census')
                    need(not native_io.native_effect('listxattr', child), 'unsupported xattrs/ACLs')
                    need(len(found) < limits['max_entries'], 'total entry cap')
                    row = dict(path=path, kind='dir' if stat.S_ISDIR(info.st_mode) else 'file', mode=mode, origin=origin(info))
                    if row['kind'] == 'file':
                        number(info.st_size, limits['max_file_bytes'], 'file size')
                        row['size'] = info.st_size
                        row['allocated'] = info.st_blocks * 512
                        if single_links:
                            identity = (info.st_dev, info.st_ino)
                            need(info.st_nlink == 1 and identity not in identities, 'file hardlink alias')
                            identities.add(identity)
                    found[path] = row
                    if row['kind'] == 'dir':
                        descend(child, path, depth + 1)
                    need(signature(info) == signature(native_io.native_effect('fstat', child)), 'entry changed during census')
                except BaseException as _owned_error_10:
                    _owned_fault_10.actual = _owned_error_10
                    _owned_fault_10.traceback = BaseException.__traceback__.__get__(_owned_error_10, BaseException)
                    native_io._commit(native_io.process_context(), _owned_fault_10)
                    raise
                finally:
                    native_io.close_fd(child)
            need(signature(before) == signature(native_io.native_effect('fstat', parent)), 'directory changed during census')
        descend(fd, '', 0)
        return dict(root_mode=stat.S_IMODE(root_info.st_mode), root_origin=origin(root_info), entries=found)
    except BaseException as _owned_error_11:
        _owned_fault_11.actual = _owned_error_11
        _owned_fault_11.traceback = BaseException.__traceback__.__get__(_owned_error_11, BaseException)
        native_io._commit(native_io.process_context(), _owned_fault_11)
        raise
    finally:
        native_io.close_fd(fd)

def validate_inventory(value, limits):
    exact(value, ('schema', 'source_root', 'root_mode', 'root_origin', 'entries', 'counts'), 'inventory')
    need(value['schema'] == 'h25-tree-inventory-v1', 'inventory schema')
    normalized(value['source_root'])
    number(value['root_mode'], 511, 'root mode')
    need(value['root_mode'] & 320 == 320, 'unsupported root owner access mode')
    need(type(value['entries']) is list and len(value['entries']) <= limits['max_entries'], 'entry count')
    modes, total, files, dirs, previous = ({}, 0, 0, 1, None)
    for row in value['entries']:
        need(type(row) is dict and row.get('kind') in ('dir', 'file'), 'entry kind')
        fields = ('path', 'kind', 'mode', 'origin')
        if row['kind'] == 'file':
            fields += ('size', 'sha256')
        exact(row, fields, 'entry')
        parts = relative(row['path'], limits)
        need(previous is None or previous < row['path'], 'duplicate/unsorted paths')
        previous = row['path']
        number(row['mode'], 511, 'entry mode')
        owner_access = 320 if row['kind'] == 'dir' else 256
        need(row['mode'] & owner_access == owner_access, 'unsupported owner access mode')
        exact(row['origin'], ('device', 'inode', 'uid', 'gid', 'mtime_ns', 'ctime_ns', 'nlink'), 'origin')
        need(all((type(v) is int for v in row['origin'].values())), 'origin integers')
        if len(parts) > 1:
            need(modes.get('/'.join(parts[:-1])) == 'dir', 'missing directory or file ancestor')
        modes[row['path']] = row['kind']
        if row['kind'] == 'file':
            number(row['size'], limits['max_file_bytes'], 'file bytes')
            need(is_hash(row['sha256']), 'file digest')
            total += row['size']
            need(total <= limits['max_total_bytes'], 'total restored size cap')
            files += 1
        else:
            dirs += 1
    exact(value['root_origin'], ('device', 'inode', 'uid', 'gid', 'mtime_ns', 'ctime_ns', 'nlink'), 'root origin')
    need(all((type(v) is int for v in value['root_origin'].values())), 'root origin integers')
    exact(value['counts'], ('files', 'directories', 'bytes', 'read_bytes', 'allocated_bytes'), 'inventory counts')
    need(all((type(v) is int and v >= 0 for v in value['counts'].values())), 'count types')
    need(value['counts']['files'] == files and value['counts']['directories'] == dirs and (value['counts']['bytes'] == total) and (value['counts']['read_bytes'] == total), 'inventory aggregate mismatch')
    return value

def inventory_file(path, expected, limits, counters):
    path = normalized(path)
    need(is_hash(expected), 'expected inventory hash required')
    raw = read_small(os.path.dirname(path), os.path.basename(path), limits['max_metadata_bytes'], counters)
    need(sha(raw) == expected, 'independent inventory digest mismatch')
    return (validate_inventory(parse(raw), limits), raw)

def same_source(census, inventory):
    need(census['root_mode'] == inventory['root_mode'] and census['root_origin'] == inventory['root_origin'], 'source root changed')
    expected = {r['path']: {k: v for k, v in r.items() if k != 'sha256'} for r in inventory['entries']}
    actual = {p: {k: v for k, v in r.items() if k != 'allocated'} for p, r in census['entries'].items()}
    need(actual == expected, 'source complete census/metadata changed')

@native_io.owning_leaf
def publish(source, target):
    """Publish readonly roots within one parent using no-replace rename only."""
    _owned_fault_12 = native_io._ERRORS.RawSlot()
    _owned_fault_12.kind = 'LEXICAL-OWNER:publish'
    old = dirfd(os.path.dirname(source))
    new = dirfd(os.path.dirname(target))
    try:
        old_info, new_info = (native_io.native_effect('fstat', old), native_io.native_effect('fstat', new))
        need((old_info.st_dev, old_info.st_ino) == (new_info.st_dev, new_info.st_ino), 'publication must stay within one parent')
        library = ctypes.CDLL(None, use_errno=True)
        function = getattr(library, 'renameat2', None)
        need(function is not None, 'renameat2 unavailable')
        function.argtypes = [ctypes.c_int, ctypes.c_char_p, ctypes.c_int, ctypes.c_char_p, ctypes.c_uint]
        function.restype = ctypes.c_int
        result = function(old, os.fsencode(os.path.basename(source)), new, os.fsencode(os.path.basename(target)), 1)
        if result != 0:
            error = ctypes.get_errno()
            raise OSError(error, os.strerror(error), target)
        native_io.native_effect('fsync', old)
        native_io.native_effect('fsync', new)
    except BaseException as _owned_error_12:
        _owned_fault_12.actual = _owned_error_12
        _owned_fault_12.traceback = BaseException.__traceback__.__get__(_owned_error_12, BaseException)
        native_io._commit(native_io.process_context(), _owned_fault_12)
        raise
    finally:
        native_io.close_fd(old)
        native_io.close_fd(new)

def operation(kind, attempt, **context):
    """All body/accounting/receipt errors retain actual identity before projection."""
    _owned_fault_15 = native_io._ERRORS.RawSlot()
    _owned_fault_15.kind = 'LEXICAL-OWNER:operation'
    _owned_fault_16 = native_io._ERRORS.RawSlot()
    _owned_fault_16.kind = 'LEXICAL-OWNER:operation'
    _owned_fault_17 = native_io._ERRORS.RawSlot()
    _owned_fault_17.kind = 'LEXICAL-OWNER:operation'
    _owned_fault_18 = native_io._ERRORS.RawSlot()
    _owned_fault_18.kind = 'LEXICAL-OWNER:operation'
    _owned_fault_19 = native_io._ERRORS.RawSlot()
    _owned_fault_19.kind = 'LEXICAL-OWNER:operation'
    _owned_fault_20 = native_io._ERRORS.RawSlot()
    _owned_fault_20.kind = 'LEXICAL-OWNER:operation'
    ctx = native_io.process_context()
    fault = native_io._slot_for_error(ctx, 'ARCHIVE-OPERATION')
    counters = dict(metadata_read_bytes=0, source_read_bytes=0, blob_read_bytes=0, payload_write_bytes=0, metadata_write_bytes=0, independent_oracle_read_bytes=0)
    result = dict(operation=kind, status='INCOMPLETE', context=context, counters=counters, parent_envelope_required=True, terminal_receipt_complete=False, accounting_exclusions=['RESULT and final closes are physically included by the genuine enclosing parent bill'])
    before = start = None
    attempt_ready = False
    body_complete = False
    try:
        try:
            attempt = normalized(attempt)
            new_dir(attempt)
            attempt_ready = True
            before = resource.getrusage(resource.RUSAGE_SELF)
            start = time.monotonic()
            write_new(attempt + '/START.json', canonical(dict(operation=kind, context=context)), counters=counters)
            yield result
            body_complete = True
        except BaseException as error:
            _owned_fault_15.actual = error
            _owned_fault_15.traceback = BaseException.__traceback__.__get__(error, BaseException)
            native_io._commit(native_io.process_context(), _owned_fault_15)
            fault.actual = error
            fault.traceback = BaseException.__traceback__.__get__(error, BaseException)
            native_io._commit(ctx, fault)
    except BaseException as _owned_error_20:
        _owned_fault_20.actual = _owned_error_20
        _owned_fault_20.traceback = BaseException.__traceback__.__get__(_owned_error_20, BaseException)
        native_io._commit(native_io.process_context(), _owned_fault_20)
        raise
    finally:
        after = finish = None
        usage_fault = native_io._slot_for_error(ctx, 'ARCHIVE-FINAL-RUSAGE')
        try:
            after = resource.getrusage(resource.RUSAGE_SELF)
        except BaseException as error:
            _owned_fault_16.actual = error
            _owned_fault_16.traceback = BaseException.__traceback__.__get__(error, BaseException)
            native_io._commit(native_io.process_context(), _owned_fault_16)
            usage_fault.actual = error
            usage_fault.traceback = BaseException.__traceback__.__get__(error, BaseException)
            native_io._commit(ctx, usage_fault)
        else:
            native_io._ERRORS.release(ctx.errors, usage_fault)
        clock_fault = native_io._slot_for_error(ctx, 'ARCHIVE-FINAL-CLOCK')
        try:
            finish = time.monotonic()
        except BaseException as error:
            _owned_fault_17.actual = error
            _owned_fault_17.traceback = BaseException.__traceback__.__get__(error, BaseException)
            native_io._commit(native_io.process_context(), _owned_fault_17)
            clock_fault.actual = error
            clock_fault.traceback = BaseException.__traceback__.__get__(error, BaseException)
            native_io._commit(ctx, clock_fault)
        else:
            native_io._ERRORS.release(ctx.errors, clock_fault)
        projection_fault = native_io._slot_for_error(ctx, 'ARCHIVE-ACCOUNTING-PROJECTION')
        try:
            result['status'] = 'PROVISIONAL_PASS' if body_complete and ctx.primary is None else 'STOP'
            result['raw_error_custody'] = native_io._ERRORS.snapshot(ctx.errors)
            result['accounting_complete'] = before is not None and start is not None and (after is not None) and (finish is not None)
            if result['accounting_complete']:
                result.update(wall_seconds=finish - start, cpu_seconds=after.ru_utime + after.ru_stime - before.ru_utime - before.ru_stime, process_peak_rss_bytes=after.ru_maxrss * 1024)
        except BaseException as error:
            _owned_fault_18.actual = error
            _owned_fault_18.traceback = BaseException.__traceback__.__get__(error, BaseException)
            native_io._commit(native_io.process_context(), _owned_fault_18)
            projection_fault.actual = error
            projection_fault.traceback = BaseException.__traceback__.__get__(error, BaseException)
            native_io._commit(ctx, projection_fault)
        else:
            native_io._ERRORS.release(ctx.errors, projection_fault)
        receipt_fault = native_io._slot_for_error(ctx, 'ARCHIVE-RESULT-WRITE')
        try:
            if attempt_ready:
                write_new(attempt + '/RESULT.json', canonical(result))
                result['terminal_receipt_complete'] = True
        except BaseException as error:
            _owned_fault_19.actual = error
            _owned_fault_19.traceback = BaseException.__traceback__.__get__(error, BaseException)
            native_io._commit(native_io.process_context(), _owned_fault_19)
            receipt_fault.actual = error
            receipt_fault.traceback = BaseException.__traceback__.__get__(error, BaseException)
            native_io._commit(ctx, receipt_fault)
        else:
            native_io._ERRORS.release(ctx.errors, receipt_fault)
        if ctx.primary is not None:
            native_io.cleanup(ctx)
    if ctx.primary is not None:
        raise BaseException.with_traceback(ctx.primary, ctx.primary_tb)
    native_io._ERRORS.release(ctx.errors, fault)
    need(body_complete and result['accounting_complete'] and result['terminal_receipt_complete'], 'archive operation unclosed')
    result['status'] = 'PASS'

def package_metadata(stage, name, raw, descriptors, counters, limits):
    need(len(raw) <= limits['max_manifest_bytes'], 'individual manifest/tool byte cap')
    descriptors[name] = dict(size=len(raw), sha256=sha(raw), mode=292)
    need(sum((row['size'] for row in descriptors.values())) <= limits['max_metadata_bytes'], 'aggregate metadata cap')
    write_new(stage + '/' + name, raw, counters=counters)

def _verify(archive, expected_root, limits, counters):
    need(is_hash(expected_root), 'external expected ROOT hash required')
    raw = read_small(archive, 'ROOT.json', limits['max_root_bytes'], counters)
    need(sha(raw) == expected_root, 'ROOT authentication mismatch')
    root = parse(raw)
    exact(root, ('schema', 'metadata', 'blobs', 'inventory_sha256', 'source_files', 'source_directories', 'source_bytes', 'unique_blob_bytes', 'provenance'), 'ROOT')
    need(root['schema'] == FORMAT and is_hash(root['inventory_sha256']), 'ROOT format/inventory hash')
    need(type(root['metadata']) is dict and type(root['blobs']) is dict, 'ROOT tables')
    need(len(root['metadata']) <= limits['max_entries'] and len(root['blobs']) <= limits['max_entries'], 'ROOT table count cap')
    total_meta = len(raw)
    for name, record in root['metadata'].items():
        relative(name, limits)
        need(name == 'ORIGINAL.json' or name in tuple(('tools/' + tool for tool in TOOL_NAMES)) or (name.startswith('runs/') and name.endswith('.json') and (len(name.split('/')) == 2)), 'unknown metadata path')
        exact(record, ('size', 'sha256', 'mode'), 'metadata record')
        number(record['size'], limits['max_manifest_bytes'], 'manifest bytes')
        need(is_hash(record['sha256']) and record['mode'] == 292, 'metadata digest/mode')
        total_meta += record['size']
    need(total_meta <= limits['max_metadata_bytes'], 'aggregate metadata byte cap')
    required = {'ORIGINAL.json', 'runs/0000.json'} | {'tools/' + tool for tool in TOOL_NAMES}
    need(required <= set(root['metadata']), 'missing required package metadata')
    run_names = sorted((n for n in root['metadata'] if n.startswith('runs/')))
    need(len(run_names) <= limits['max_runs'] + 1 and run_names == ['runs/%04d.json' % n for n in range(len(run_names))], 'run manifest sequence/count')
    blob_total = 0
    for digest, size in root['blobs'].items():
        need(is_hash(digest), 'blob hash name')
        number(size, limits['max_file_bytes'], 'blob size')
        blob_total += size
        need(blob_total <= limits['max_unique_bytes'], 'unique payload byte cap')
    for key, maximum in (('source_files', limits['max_entries']), ('source_directories', limits['max_entries'] + 1), ('source_bytes', limits['max_total_bytes']), ('unique_blob_bytes', limits['max_unique_bytes'])):
        number(root[key], maximum, key)
    need(blob_total == root['unique_blob_bytes'], 'blob total mismatch')
    need(type(root['provenance']) is dict, 'provenance shape')
    complete_raw = read_small(archive, 'COMPLETE.json', 1024, counters)
    complete = parse(complete_raw)
    exact(complete, ('schema', 'root_sha256'), 'completion')
    need(complete == dict(schema=FORMAT, root_sha256=expected_root), 'incomplete package')
    census = scan(archive, {**limits, 'max_entries': limits['max_entries'] * 3 + 32}, True)
    need(census['root_mode'] == 365, 'package root mode')
    wanted_dirs = {'blobs', 'blobs/sha256', 'runs', 'tools'}
    wanted_files = set(root['metadata']) | {'ROOT.json', 'COMPLETE.json'} | {'blobs/sha256/' + digest for digest in root['blobs']}
    need(set(census['entries']) == wanted_dirs | wanted_files, 'package complete membership mismatch')
    for name, row in census['entries'].items():
        need(row['kind'] == ('dir' if name in wanted_dirs else 'file') and row['mode'] == (365 if name in wanted_dirs else 292), 'package entry type/mode')
    metadata = {}
    for name, expected in root['metadata'].items():
        value = read_small(archive, name, limits['max_manifest_bytes'], counters)
        need(len(value) == expected['size'] and sha(value) == expected['sha256'], 'metadata digest mismatch')
        metadata[name] = value
    local_tools = os.path.dirname(os.path.abspath(__file__))
    for tool in TOOL_NAMES:
        local_raw = read_small(local_tools, tool, limits['max_manifest_bytes'], counters)
        need(sha(local_raw) == root['metadata']['tools/' + tool]['sha256'], 'local tool source mismatch')
    need(sha(metadata['ORIGINAL.json']) == root['inventory_sha256'], 'original inventory binding')
    inventory = validate_inventory(parse(metadata['ORIGINAL.json']), limits)
    merged, run_ids = ([], [])
    for name in run_names:
        manifest = parse(metadata[name])
        exact(manifest, ('schema', 'run', 'entries'), 'run manifest')
        need(manifest['schema'] == FORMAT and type(manifest['entries']) is list, 'run manifest shape')
        run = manifest['run']
        if name == 'runs/0000.json':
            need(run is None, 'root manifest identity')
        else:
            need(type(run) is str and len(relative(run, limits)) == 1 and (run not in run_ids), 'run identity')
            run_ids.append(run)
        for row in manifest['entries']:
            need(type(row) is dict and type(row.get('path')) is str, 'run entry shape')
            if run is None:
                need('/' not in row['path'] and row.get('kind') == 'file', 'root manifest entry ownership')
            else:
                need(row['path'] == run or row['path'].startswith(run + '/'), 'run manifest ownership')
        merged.extend(manifest['entries'])
        need(len(merged) <= limits['max_entries'], 'merged manifest count cap')
    need(sorted(merged, key=lambda r: r['path']) == inventory['entries'], 'incomplete/duplicate run union')
    need(set(run_ids) == {r['path'] for r in inventory['entries'] if r['kind'] == 'dir' and '/' not in r['path']}, 'run root partition')
    expected_blobs = {}
    for row in inventory['entries']:
        if row['kind'] == 'file':
            if row['sha256'] in expected_blobs:
                need(expected_blobs[row['sha256']] == row['size'], 'digest/size collision')
            expected_blobs[row['sha256']] = row['size']
    need(root['blobs'] == expected_blobs, 'exact blob closure mismatch')
    need((root['source_files'], root['source_directories'], root['source_bytes']) == tuple((inventory['counts'][k] for k in ('files', 'directories', 'bytes'))), 'source aggregate mismatch')
    for digest, size in root['blobs'].items():
        stream(archive, 'blobs/sha256/' + digest, limits, counters, 'blob_read_bytes', dict(size=size, sha256=digest))
    need(scan(archive, {**limits, 'max_entries': limits['max_entries'] * 3 + 32}, True) == census, 'package changed during verification')
    archive_bytes = sum((r['size'] for r in census['entries'].values() if r['kind'] == 'file'))
    allocated = sum((r['allocated'] for r in census['entries'].values() if r['kind'] == 'file'))
    return (root, inventory, dict(archive_logical_bytes=archive_bytes, archive_allocated_bytes=allocated, source_logical_bytes=root['source_bytes'], unique_blob_bytes=blob_total, derived_bytes_avoided=root['source_bytes'] - archive_bytes))

@native_io.owning_leaf
def create_archive(source_root, original_inventory, expected_inventory_sha256, destination, attempt_dir, provenance=None, limits=None):
    _owned_fault_21 = native_io._ERRORS.RawSlot()
    _owned_fault_21.kind = 'LEXICAL-OWNER:create_archive'
    _owned_fault_22 = native_io._ERRORS.RawSlot()
    _owned_fault_22.kind = 'LEXICAL-OWNER:create_archive'
    limits = caps(limits)
    source_root, destination, attempt_dir = map(normalized, (source_root, destination, attempt_dir))
    stage = sibling_stage(destination)
    disjoint(source_root, destination, attempt_dir, stage)
    with operation('create', attempt_dir, source=source_root, destination=destination, staging=stage, expected_inventory_sha256=expected_inventory_sha256) as result:
        counts = result['counters']
        private_destination_parent(destination)
        absent(destination)
        inventory, original_raw = inventory_file(original_inventory, expected_inventory_sha256, limits, counts)
        need(inventory['source_root'] == source_root, 'inventory source root mismatch')
        same_source(scan(source_root, limits), inventory)
        new_dir(stage)
        for name in ('blobs', 'blobs/sha256', 'runs', 'tools'):
            new_dir(stage + '/' + name)
        descriptors, blobs = ({}, {})
        package_metadata(stage, 'ORIGINAL.json', original_raw, descriptors, counts, limits)
        run_ids = sorted((r['path'] for r in inventory['entries'] if r['kind'] == 'dir' and '/' not in r['path']))
        need(len(run_ids) <= limits['max_runs'], 'run count cap')
        for index, run in enumerate([None] + run_ids):
            rows = [r for r in inventory['entries'] if (r['kind'] == 'file' and '/' not in r['path'] if run is None else r['path'] == run or r['path'].startswith(run + '/'))]
            package_metadata(stage, 'runs/%04d.json' % index, canonical(dict(schema=FORMAT, run=run, entries=rows)), descriptors, counts, limits)
        tool_dir = os.path.dirname(os.path.abspath(__file__))
        for name in TOOL_NAMES:
            raw = read_small(tool_dir, name, limits['max_manifest_bytes'], counts)
            package_metadata(stage, 'tools/' + name, raw, descriptors, counts, limits)
        for row in inventory['entries']:
            if row['kind'] != 'file':
                continue
            digest = row['sha256']
            if digest in blobs:
                need(blobs[digest] == row['size'], 'digest size collision')
                with opened(stage, 'blobs/sha256/' + digest) as fd:
                    need(stat.S_IMODE(native_io.native_effect('fstat', fd).st_mode) == 292 and native_io.native_effect('fstat', fd).st_nlink == 1, 'reused blob mode/alias')
                stream(stage, 'blobs/sha256/' + digest, limits, counts, 'blob_read_bytes', row)
                stream(source_root, row['path'], limits, counts, 'source_read_bytes', row)
            else:
                need(sum(blobs.values()) + row['size'] <= limits['max_unique_bytes'], 'unique byte cap')
                path = stage + '/blobs/sha256/' + digest
                parent = dirfd(os.path.dirname(path))
                try:
                    fd = native_io.open_fd(digest, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 384, dir_fd=parent)
                    try:
                        stream(source_root, row['path'], limits, counts, 'source_read_bytes', row, fd)
                        native_io.native_effect('fchmod', fd, 292)
                        native_io.native_effect('fsync', fd)
                    except BaseException as _owned_error_21:
                        _owned_fault_21.actual = _owned_error_21
                        _owned_fault_21.traceback = BaseException.__traceback__.__get__(_owned_error_21, BaseException)
                        native_io._commit(native_io.process_context(), _owned_fault_21)
                        raise
                    finally:
                        native_io.close_fd(fd)
                    native_io.native_effect('fsync', parent)
                except BaseException as _owned_error_22:
                    _owned_fault_22.actual = _owned_error_22
                    _owned_fault_22.traceback = BaseException.__traceback__.__get__(_owned_error_22, BaseException)
                    native_io._commit(native_io.process_context(), _owned_fault_22)
                    raise
                finally:
                    native_io.close_fd(parent)
                blobs[digest] = row['size']
        same_source(scan(source_root, limits), inventory)
        root = dict(schema=FORMAT, metadata=descriptors, blobs=blobs, inventory_sha256=expected_inventory_sha256, source_files=inventory['counts']['files'], source_directories=inventory['counts']['directories'], source_bytes=inventory['counts']['bytes'], unique_blob_bytes=sum(blobs.values()), provenance=provenance or {})
        raw = canonical(root)
        need(len(raw) <= limits['max_root_bytes'] and len(raw) + sum((r['size'] for r in descriptors.values())) <= limits['max_metadata_bytes'], 'ROOT/metadata byte cap')
        expected_root = sha(raw)
        write_new(stage + '/ROOT.json', raw, counters=counts)
        complete = canonical(dict(schema=FORMAT, root_sha256=expected_root))
        write_new(stage + '/COMPLETE.json', complete, counters=counts)
        freeze_tree(stage)
        _, _, accounting = _verify(stage, expected_root, limits, counts)
        same_source(scan(source_root, limits), inventory)
        publish(stage, destination)
        result.update(root_sha256=expected_root, inventory_sha256=expected_inventory_sha256, accounting=accounting, raw_bytes_removed=0, destination=destination)
    return result

@native_io.owning_leaf
def verify_archive(archive, expected_root_sha256, attempt_dir, limits=None):
    archive, attempt_dir = map(normalized, (archive, attempt_dir))
    disjoint(archive, attempt_dir)
    with operation('verify', attempt_dir, archive=archive, expected_root_sha256=expected_root_sha256) as result:
        _, _, accounting = _verify(archive, expected_root_sha256, caps(limits), result['counters'])
        result.update(root_sha256=expected_root_sha256, accounting=accounting)
    return result
_PRELOADED_ORACLE = None
_PRELOADED_ORACLE_PIN = None
_PRELOADED_ORACLE_STATE = None

def _bind_preloaded_oracle(module, source_sha256):
    """Called only by checked parent tool constructor; grants no authority."""
    global _PRELOADED_ORACLE, _PRELOADED_ORACLE_PIN, _PRELOADED_ORACLE_STATE
    need(_PRELOADED_ORACLE is None and is_hash(source_sha256), 'one exact preloaded oracle')
    namespace = module.__dict__
    functions = tuple(((name, value, value.__code__, value.__globals__, value.__defaults__, value.__kwdefaults__) for name, value in namespace.items() if type(value) is types.FunctionType))
    _PRELOADED_ORACLE = module
    _PRELOADED_ORACLE_PIN = source_sha256
    _PRELOADED_ORACLE_STATE = (namespace, functions, dict(module.DEFAULTS))

def load_oracle(expected_sha256, counters):
    directory = os.path.dirname(os.path.abspath(__file__))
    raw = read_small(directory, 'oracle.py', 1048576, counters)
    need(sha(raw) == expected_sha256, 'independent oracle source changed')
    need(_PRELOADED_ORACLE is not None and _PRELOADED_ORACLE_PIN == expected_sha256, 'genuine parent must preload independent oracle before fork')
    namespace, functions, defaults = _PRELOADED_ORACLE_STATE
    need(_PRELOADED_ORACLE.__dict__ is namespace and _PRELOADED_ORACLE.DEFAULTS == defaults, 'preloaded oracle globals/constants changed')
    for name, function, code, globals_object, args, kwargs in functions:
        need(namespace.get(name) is function and function.__code__ is code and (function.__globals__ is globals_object) and (function.__defaults__ is args) and (function.__kwdefaults__ is kwargs), 'preloaded oracle function changed')
    return _PRELOADED_ORACLE

@native_io.owning_leaf
def restore_archive(archive, expected_root_sha256, destination, attempt_dir, limits=None):
    _owned_fault_23 = native_io._ERRORS.RawSlot()
    _owned_fault_23.kind = 'LEXICAL-OWNER:restore_archive'
    _owned_fault_24 = native_io._ERRORS.RawSlot()
    _owned_fault_24.kind = 'LEXICAL-OWNER:restore_archive'
    _owned_fault_25 = native_io._ERRORS.RawSlot()
    _owned_fault_25.kind = 'LEXICAL-OWNER:restore_archive'
    _owned_fault_26 = native_io._ERRORS.RawSlot()
    _owned_fault_26.kind = 'LEXICAL-OWNER:restore_archive'
    limits = caps(limits)
    archive, destination, attempt_dir = map(normalized, (archive, destination, attempt_dir))
    stage = sibling_stage(destination)
    disjoint(archive, destination, attempt_dir, stage)
    with operation('restore', attempt_dir, archive=archive, destination=destination, staging=stage, expected_root_sha256=expected_root_sha256) as result:
        counters = result['counters']
        private_destination_parent(destination)
        absent(destination)
        root, inventory, accounting = _verify(archive, expected_root_sha256, limits, counters)
        historical = inventory['source_root']
        for live in (archive, destination, attempt_dir):
            need(os.path.commonpath((historical, live)) not in (historical, live), 'historical source path overlaps restoration operation')
        new_dir(stage)
        for row in inventory['entries']:
            path = stage + '/' + row['path']
            if row['kind'] == 'dir':
                new_dir(path)
            else:
                parent = dirfd(os.path.dirname(path))
                try:
                    fd = native_io.open_fd(os.path.basename(path), os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 384, dir_fd=parent)
                    try:
                        stream(archive, 'blobs/sha256/' + row['sha256'], limits, counters, 'blob_read_bytes', row, fd)
                        native_io.native_effect('fchmod', fd, row['mode'])
                        native_io.native_effect('fsync', fd)
                    except BaseException as _owned_error_23:
                        _owned_fault_23.actual = _owned_error_23
                        _owned_fault_23.traceback = BaseException.__traceback__.__get__(_owned_error_23, BaseException)
                        native_io._commit(native_io.process_context(), _owned_fault_23)
                        raise
                    finally:
                        native_io.close_fd(fd)
                    native_io.native_effect('fsync', parent)
                except BaseException as _owned_error_24:
                    _owned_fault_24.actual = _owned_error_24
                    _owned_fault_24.traceback = BaseException.__traceback__.__get__(_owned_error_24, BaseException)
                    native_io._commit(native_io.process_context(), _owned_fault_24)
                    raise
                finally:
                    native_io.close_fd(parent)
        for row in sorted(inventory['entries'], key=lambda r: r['path'].count('/'), reverse=True):
            if row['kind'] == 'dir':
                fd = dirfd(stage + '/' + row['path'])
                try:
                    native_io.native_effect('fchmod', fd, row['mode'])
                    native_io.native_effect('fsync', fd)
                except BaseException as _owned_error_25:
                    _owned_fault_25.actual = _owned_error_25
                    _owned_fault_25.traceback = BaseException.__traceback__.__get__(_owned_error_25, BaseException)
                    native_io._commit(native_io.process_context(), _owned_fault_25)
                    raise
                finally:
                    native_io.close_fd(fd)
        fd = dirfd(stage)
        try:
            native_io.native_effect('fchmod', fd, inventory['root_mode'])
            native_io.native_effect('fsync', fd)
        except BaseException as _owned_error_26:
            _owned_fault_26.actual = _owned_error_26
            _owned_fault_26.traceback = BaseException.__traceback__.__get__(_owned_error_26, BaseException)
            native_io._commit(native_io.process_context(), _owned_fault_26)
            raise
        finally:
            native_io.close_fd(fd)
        oracle = load_oracle(root['metadata']['tools/oracle.py']['sha256'], counters)
        oracle_limits = {k: limits[k] for k in ('chunk_bytes', 'max_entries', 'max_depth', 'max_path_bytes', 'max_file_bytes', 'max_total_bytes')}
        oracle_limits['max_inventory_bytes'] = limits['max_metadata_bytes']
        restored = oracle.collect(stage, oracle_limits, require_single_links=True, io_counters=counters)
        need(oracle.content_projection(restored) == oracle.content_projection(inventory), 'independent restoration oracle mismatch')
        proof_raw = oracle.canonical(restored)
        write_new(attempt_dir + '/RESTORED-INVENTORY.json', proof_raw, counters=counters)
        proof_digest = sha(proof_raw)
        before_commit = scan(stage, limits, True)
        same_source(before_commit, restored)
        _verify(archive, expected_root_sha256, limits, counters)
        need(scan(stage, limits, True) == before_commit, 'restored tree changed before commit')
        publish(stage, destination)
        result.update(root_sha256=expected_root_sha256, inventory_sha256=root['inventory_sha256'], restored_inventory_sha256=proof_digest, accounting=accounting, restored_logical_bytes=restored['counts']['bytes'], restored_allocated_bytes=restored['counts']['allocated_bytes'], independent_oracle_read_bytes=restored['counts']['read_bytes'], destination=destination, exact_restoration=True)
    return result

def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('operation', choices=('create', 'verify', 'restore'))
    parser.add_argument('--source')
    parser.add_argument('--inventory')
    parser.add_argument('--inventory-sha256')
    parser.add_argument('--archive')
    parser.add_argument('--root-sha256')
    parser.add_argument('--destination')
    parser.add_argument('--attempt', required=True)
    args = parser.parse_args()
    if args.operation == 'create':
        result = create_archive(args.source, args.inventory, args.inventory_sha256, args.destination, args.attempt)
    elif args.operation == 'verify':
        result = verify_archive(args.archive, args.root_sha256, args.attempt)
    else:
        result = restore_archive(args.archive, args.root_sha256, args.destination, args.attempt)
    print(json.dumps(result, sort_keys=True))
if __name__ == '__main__':
    main()
