class Refused(RuntimeError):
    pass


def need(value, reason):
    if value is not True:
        raise Refused(reason)


def exact(value, keys):
    need(type(value) is dict and set(value) == set(keys), 'EXACT_FIELDS')


def definition(path, sha256, names, namespace):
    raw = path.read_bytes()
    need(hashlib.sha256(raw).hexdigest() == sha256, 'FROZEN_SOURCE_PIN')
    tree = ast.parse(raw, filename=str(path))
    selected = [node for node in tree.body
                if isinstance(node, (ast.FunctionDef, ast.ClassDef)) and node.name in names]
    need(len(selected) == len(names), 'EXACT_NAMED_DEFINITIONS')
    # Compiling these definitions has no top-level candidate/native execution.
    exec(compile(ast.Module(body=selected, type_ignores=[]), str(path), 'exec'), namespace)
    return namespace
