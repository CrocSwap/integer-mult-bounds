def _check(condition,message='independent fault oracle failed'):
    if not condition:
        raise AssertionError(message)


def _identity(action, expected):
    observed = None
    try:
        action()
    except BaseException as error:
        observed = error
    _check(observed is expected, ('actual outward object differs', id(observed), id(expected)))

def _contains_traceback(current, target):
    while current is not None:
        if current is target:
            return True
        current = current.tb_next
    return False

@contextmanager
def _ports(native_io, errors, **overrides):
    saved = {name: getattr(native_io, name) for name in ('_OPEN', '_CLOSE', '_READ', '_WRITE', '_DUP', '_NATIVE', '_PROCESS')}
    native_io._PROCESS = native_io.new_context('independent-fake-control', errors.new_custody())
    native_io._NATIVE = dict(native_io._NATIVE)
    try:
        for name, value in overrides.items():
            if name in native_io._NATIVE:
                native_io._NATIVE[name] = value
            else:
                setattr(native_io, name, value)
        yield native_io._PROCESS
    finally:
        for name, value in saved.items():
            setattr(native_io, name, value)

def _sequential_open(calls, first=10):

    def acquire(path, flags, mode=511, **kwargs):
        returned = first + len(calls)
        calls.append((path, returned, kwargs.get('dir_fd')))
        return returned
    return acquire

def expected_missing_is_success(archive, oracle, native_io, errors):

def nonnative_body_error_precedes_two_closes(archive, oracle, native_io, errors):
    primary = ValueError('independently selected pure body failure')
    first = OSError(errno.EIO, 'leaf close')
    second = OSError(errno.EIO, 'parent close')
    opened, closed = ([], [])
    old_need = archive.need

    def validation(value, reason):
        if not value:
            raise primary

    def close(fd):
        closed.append(fd)
        if fd == 12:
            raise first
        if fd == 11:
            raise second
    archive.need = validation
    try:
        with _ports(native_io, errors, _OPEN=_sequential_open(opened), _CLOSE=close, _WRITE=lambda fd, raw: 0, fsync=lambda fd: None) as ctx:
            _identity(lambda: archive.write_new('/parent/file', b'x'), primary)
            _check(ctx.primary is primary and _contains_traceback(primary.__traceback__, ctx.primary_tb), 'independent fault oracle failed')
            _check(closed == [10, 12, 11], 'independent fault oracle failed')
            secondary = [row.actual for row in ctx.errors.events if row.actual is first or row.actual is second]
            _check(secondary == [first, second], 'independent fault oracle failed')
            sequence = [row.sequence for row in ctx.errors.events if row.actual is first or row.actual is second]

def optional_journal_failure_does_not_skip_known_closes(archive,oracle,native_io,errors):
    close_fault=OSError(errno.EIO,'known close12')
    second_close_fault=OSError(errno.EIO,'known close11')
    journal_fault=MemoryError('optional event export')
    later_journal_fault=MemoryError('forbidden journal retry')
    opened,closed,journal_calls=[],[],[]
    original=errors.record
    def record(value,slot):
        journal_calls.append(slot)
        raise journal_fault if len(journal_calls)==1 else later_journal_fault
    def close(fd):
        closed.append(fd)
        if fd==12:raise close_fault
        if fd==11:raise second_close_fault
    with _ports(native_io,errors,_OPEN=_sequential_open(opened),_CLOSE=close) as ctx:

        errors.record=record

        _check(ctx.journal_fault.actual is journal_fault and ctx.errors.incomplete,'first journal fault retained')

def operation_body_and_accounting_identity(archive, oracle, native_io, errors):

    primary = BadString('body')

    clock_fault = OSError(errno.EIO, 'clock')

    calls = []

def run_fault_oracles(archive, oracle, native_io, errors):

    outcomes=[]

    actual_tracebacks=[None]*len(cases)

            case(archive,oracle,native_io,errors)
        except BaseException as error:
            actual_failures[index]=error

            outcomes.append(dict(case=case.__name__,status='FAIL',actual_id=id(error)))
        else:
            outcomes.append(dict(case=case.__name__,status='PASS'))
    global _CONTROL_FAILURES,_CONTROL_TRACEBACKS
    _CONTROL_FAILURES=actual_failures
    _CONTROL_TRACEBACKS=actual_tracebacks
    return dict(schema='s03-retention-changed-code-fault-oracles-v1',
                status='PASS' if all(row['status']=='PASS' for row in outcomes) else 'FAIL',
                cases=outcomes,source_port_controls_only=True,kernel_native_currentness_fit=False,
