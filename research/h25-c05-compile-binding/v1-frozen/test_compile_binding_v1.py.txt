def raw(value):
    data = (json.dumps(value,sort_keys=True,separators=(',',':'),ensure_ascii=True)+'\n').encode('ascii')
    return data,hashlib.sha256(data).hexdigest()

class CompileBindingTests(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        cls.corpus = corpus()

    def bind(self,fixture_value=None,source_value=None):
        plan,config,launches = fixture_value or fixture()
        plan_raw,plan_pin = raw(plan)
        cfg_raw,cfg_pin = raw(config)
        return construct_original_compile_pair(source_value or self.corpus,parent_plan_raw=plan_raw,parent_plan_sha256=plan_pin,
            variant_configuration_raw=cfg_raw,variant_configuration_sha256=cfg_pin,compiler_launches=launches)

    def reject(self,value):
        with self.assertRaises(BindingError): self.bind(value)

    def test_original_disjoint_stage_output_cwd(self):
        out = self.bind()
        self.assertEqual(out['stage_root'],'/immutable/stage')
        self.assertEqual(out['producer_output'],'/owned/evidence/failure_variants/producer-output')
        self.assertEqual(out['parent_cwd'],'/independent/work')
        self.assertFalse(out['compile_bindings'][0]['source_path'].startswith(out['parent_cwd']+'/'))
        self.assertFalse(out['compile_bindings'][0]['output_path'].startswith(out['parent_cwd']+'/'))
        self.assertEqual([r['role'] for r in out['compile_bindings']],ROLES)

    def test_helper_inheritance_is_derived_from_actual_Popen(self):
        rules=authenticate_source_rules(self.corpus)
        launch=rules['helper_launch']
        self.assertEqual(launch['cwd_rule'],'inherit_producer_current_cwd')
        self.assertEqual(launch['environment_rule'],'inherit_producer_current_environment')
        self.assertNotIn('cwd=',launch['anchor']['text'])
        self.assertNotIn('env=',launch['anchor']['text'])
        self.assertEqual(launch['anchor']['first_line'],220)

    def test_normal_optimized_do_not_change_gcc_argv(self):
        a,b = self.bind(fixture('normal')),self.bind(fixture('optimized'))
        self.assertEqual([r['command'] for r in a['compile_bindings']],[r['command'] for r in b['compile_bindings']])
        self.assertNotIn('-S',a['compile_bindings'][0]['command'])
        self.assertIn('-O3',a['compile_bindings'][0]['command'])
        self.assertNotIn('-O',b['compile_bindings'][0]['command'])

    def test_wrong_role_label_and_order(self):
        for replacement in ('control','no-op','header','exec_failure'):
            value = fixture();value[1]['roles'][0]['role']=replacement;self.reject(value)
        value = fixture();value[2].reverse();self.reject(value)

    def test_equal_C_bytes_do_not_permit_swapped_paths(self):
        value=fixture()
        self.assertEqual(value[1]['roles'][0]['source'],value[1]['roles'][1]['source'])
        value[2][0]['command'][10]=value[2][1]['command'][10]
        self.reject(value)

    def test_wrong_output_role_or_common_ancestor_rewrite(self):
        for output in ('/owned/evidence/failure_variants/producer-output/exec_failure/_h25_dict_probe.cpython-312-x86_64-linux-gnu.so',
                       '/immutable/stage/build/_h25_dict_probe.cpython-312-x86_64-linux-gnu.so',
                       '/independent/work/out.so'):
            value=fixture();value[2][0]['command'][12]=output;self.reject(value)

    def test_wrong_fixed_options_and_compiler(self):
        for index,replacement in ((0,'/usr/bin/gcc'),(1,'-nostdlib'),(4,'-O2'),(5,'-std=gnu11'),(9,'-I/other'),(11,'--output')):
            value=fixture();value[2][0]['command'][index]=replacement;self.reject(value)
        value=fixture();value[2][0]['command'].append('-S');self.reject(value)

    def test_native_environment_exact_six(self):
        for key,item in (('CPATH','/other'),('LANG','C'),('LD_PRELOAD','/x')):
            value=fixture();value[2][0]['environment'][key]=item;self.reject(value)
        value=fixture();del value[2][0]['environment']['TZ'];self.reject(value)
        value=fixture();value[2][0]['environment'][b'PATH']=b'/usr/bin:/bin';self.reject(value)

        value=fixture()

        self.assertEqual([r['helper_name'] for r in out['compile_bindings']],['control','control'])
        self.assertEqual([r['context']['negative_variant_role'] for r in out['compile_bindings']],ROLES)

    def test_parent_output_origin_and_root_argv_are_bound(self):
        value=fixture();value[0]['producers']['failure_variants']['output']='/elsewhere/out';self.reject(value)
        value=fixture();value[0]['producers']['failure_variants']['command'][0]='/usr/bin/python3';self.reject(value)
        value=fixture();value[0]['producers']['failure_variants']['command'][1]='-I';self.reject(value)
        value=fixture();value[0]['producers']['failure_variants']['command'][5]='/wrong/stage/integration/full_driver.py';self.reject(value)

    def test_plan_and_config_tampering_and_duplicate_keys_reject(self):
        plan,cfg,launches=fixture();p,ph=raw(plan);c,ch=raw(cfg)
        with self.assertRaises(BindingError):
            construct_original_compile_pair(self.corpus,parent_plan_raw=p+b' ',parent_plan_sha256=ph,
                variant_configuration_raw=c,variant_configuration_sha256=ch,compiler_launches=launches)
        duplicate=b'{"mode":"normal","mode":"optimized"}'
        with self.assertRaises(BindingError):
            construct_original_compile_pair(self.corpus,parent_plan_raw=duplicate,parent_plan_sha256=hashlib.sha256(duplicate).hexdigest(),
                variant_configuration_raw=c,variant_configuration_sha256=ch,compiler_launches=launches)

    def test_authenticated_source_tampering_rejects_before_AST_use(self):
        value=dict(self.corpus);value['variant_driver']+=b'\n'
        with self.assertRaises(BindingError): self.bind(source_value=value)
        value=dict(self.corpus);value.pop('cold_build_evidence')
        with self.assertRaises(BindingError): self.bind(source_value=value)

    def test_exact_nine_options_have_source_anchors(self):
        rules=authenticate_source_rules(self.corpus)
        self.assertEqual(rules['fixed_options'],OPTIONS)
        self.assertEqual(len(rules['fixed_options']),9)
        for anchor in rules['anchors']:
            text=self.corpus[anchor['source']].decode().split('\n')
            self.assertEqual('\n'.join(text[anchor['first_line']-1:anchor['last_line']]),anchor['text'])

    def test_result_is_not_admission_or_anycwd_semantics(self):
        out=self.bind()
        for key in ('full_C05','full_V03','execution_ready','effective_GCC_configuration','parent_plan_admitted','namespace_currentness_proven','generic_any_cwd_correctness'):
            self.assertIs(out[key],False)

    def test_path_spellings_with_spaces_are_preserved(self):
        value=fixture(stage='/immutable/stage one',cwd='/independent/work two',session='/owned/evidence three')
        out=self.bind(value)
        self.assertEqual(out['compile_bindings'][0]['command'][10],'/immutable/stage one/areas/field-negative-variants-v1/pyinit_failure/dict_probe_fields_v1.c')

    def test_no_path_normalization(self):
        value=fixture(stage='/immutable/a/../stage',cwd='/independent/a/../work')
        out=self.bind(value)
        self.assertIn('/a/../stage/',out['compile_bindings'][0]['command'][10])
        self.assertEqual(out['parent_cwd'],'/independent/a/../work')
